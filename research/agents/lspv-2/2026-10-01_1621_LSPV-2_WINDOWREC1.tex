\paragraph{Exact ORBITSLICE1 input.}
Fix one legal RMV line and the declared SU05/Hadamard realization. Let
\[
a_t=\kappa(f(A^tv_0))\in\mathbb F_2^k,\qquad t\in\mathbb Z/N\mathbb Z,
\]
and for every cyclic shift \(s\) let
\[
X_s=\begin{bmatrix}a_{s+1}^{T}\\ \vdots\\ a_{s+m}^{T}\end{bmatrix}\in\mathbb F_2^{m\times k},
\qquad S_s=\operatorname{im}X_s\le V:=\mathbb F_2^m.
\]
ORBITSLICE1 proves these are exactly the nonzero-seed legal Hadamard slices for the declared realization on the fixed line.

\paragraph{PROVED/EXACT: simultaneous linear-recurrence criterion.}
Define \(W_L=\operatorname{span}_{s}S_s\). For \(c=(c_1,\ldots,c_m)\in V\),
\[
c\in W_L^\perp
\iff c^TX_s=0\quad\forall s
\iff
\boxed{\sum_{i=1}^{m}c_i a_{s+i}=0\in\mathbb F_2^k\quad\forall s.}
\]
Hence
\[
\boxed{W_L=V\iff (a_t)_t\text{ has no nonzero common cyclic linear recurrence of length }m.}
\]
Writing \(a_t=(a_t^{(1)},\ldots,a_t^{(k)})\), the same coefficient vector must annihilate every coordinate sequence at every cyclic shift. This is construction-specific.

\paragraph{PROVED/EXACT: simultaneous quadratic-correlation criterion.}
Define
\[
U_L=\operatorname{span}_{s}\bigwedge\nolimits^2S_s\le\bigwedge\nolimits^2V.
\]
Identify \(U_L^\perp\) with alternating matrices \(Q\in\mathbb F_2^{m\times m}\). By WEDGE1/POLAR1,
\[
Q\in U_L^\perp\iff X_s^TQX_s=0\quad\forall s.
\]
Expanding the \((r,t)\) entry gives
\[
\boxed{
\sum_{1\le i<j\le m}Q_{ij}
\left(a_{s+i}^{(r)}a_{s+j}^{(t)}+a_{s+i}^{(t)}a_{s+j}^{(r)}\right)=0
}
\]
for every shift \(s\) and every \(1\le r<t\le k\). Therefore
\[
\boxed{
U_L=\bigwedge\nolimits^2V
\iff
\text{no nonzero alternating }Q\text{ satisfies all shifted bilinear correlations.}
}
\]

\paragraph{PROVED/EXACT: direct full-rank falsifier.}
Combining these criteria with WEDGE1,
\[
\boxed{
\begin{array}{c}
\text{no nonzero common length-}m\text{ linear recurrence}\\
\text{and no nonzero common alternating shifted-correlation relation}
\end{array}
\Longrightarrow K_2(H_L)=0,\quad \rho(H_L)=1+m+\binom m2.

\]
Thus the low-product-span route for this fixed-line SAME\_REALIZATION can be falsified by two explicit kernel computations. Conversely, failure returns an explicit recurrence \(c\) or alternating matrix \(Q\), useful algebraic structure rather than evidence of hardness.
