\section{Exact all-input theorem}
\begin{lemma}[Binary-premise source capacity]
Every sound conservative refutation with $q$ charged nodes has $q\ge\mu(F)-1$.
\end{lemma}
\begin{proof}
The set $S$ of original clauses in the backwards ancestry of contradiction is UNSAT: otherwise a satisfying assignment extends through each conservative auxiliary definition, contradicting soundness. Thus $|S|\ge\mu(F)$. The connected undirected ancestry graph has $|S|+q$ vertices, at least $|S|+q-1$ edges, and at most $2q$ dependency edges by the fixed fan-in bound. Therefore $q\ge |S|-1\ge\mu(F)-1$. Reconvergence, state growth, affine steps, and source reuse do not evade this count.
\end{proof}
\begin{theorem}[Exact mixed-DAG minimum on support at most three]
For every finite valid UNSAT general CNF formula $F$ whose source clauses collectively depend on at most three distinct original variables,
\[
\boxed{\WR(F)=\mu(F)-1.}
\]
There is no bound on the number of clauses, repeated literals, repeated clauses, or source-order changes. With larger declared variable indices the theorem applies after authenticated support renaming; the executable standalone optimizer deliberately returns OUT\_OF\_EXACT\_SCOPE for declared $n>3$.
\end{theorem}
\begin{proof}
The source-capacity lemma supplies the lower bound. Choose a minimum-cardinality UNSAT core $C_1,\ldots,C_\mu$. If $\mu=1$, its empty clause is contradiction already. Otherwise recursively construct $D_1=C_1$ and $D_i=\mathrm{AND3}(D_{i-1},C_i)$ for $2\le i\le\mu$. Since the union of all variable supports has size at most three, each eight-row exact conjunction is a legal charged inference. The last truth-table factor is identically false. There are exactly $\mu-1$ derived nodes, and neither search for the minimum core nor a global semantic-oracle step is included in the proof cost.
\end{proof}
\begin{lemma}[Clause-deletion witness injection]
An UNSAT formula minimal by source-clause deletion on $r$ variables has at most $2^r$ source clauses.
\end{lemma}
\begin{proof}
For each indispensable clause $C_i$, choose an assignment $a_i$ satisfying all other clauses and falsifying $C_i$. If $a_i=a_j$ for $i\ne j$, the common assignment would simultaneously satisfy and falsify $C_j$. Thus the deletion-witness map is injective into the $2^r$ assignments.
\end{proof}
\begin{theorem}[Exact global finite extremum, no enumeration cutoff]
For each $r\in\{0,1,2,3\}$,
\[
\boxed{\max_{\substack{F\ \mathrm{UNSAT}\\|\mathrm{support}(F)|\le r}}\WR(F)=2^r-1.}
\]
