\section*{Exact setup}
Fix one SAME\_REALIZATION. Let
\[
E=\{\{i,j\}:1\le i<j\le m\},\qquad
U\le\mathbb F_2^E,\qquad d=\dim U,
\]
and let $G$ be a full-row-rank generator of $U$. For $I\subseteq[m]$, let
\[
E(I)=\{\{i,j\}\in E:i,j\in I\},\qquad
\epsilon(I)=d-\operatorname{rank}G_{E(I)}.
\]
By EXCESSDUAL1,
\[
\epsilon(I)=\dim\{u\in U:u_e=0\ \forall e\in E(I)\}.
\]
For a subspace $W\le U$, define its union-support graph
\[
\Gamma(W)=([m],\operatorname{Supp}(W)),\qquad
\operatorname{Supp}(W)=\bigcup_{u\in W}\operatorname{supp}(u).
\]

\section*{PROVED/EXACT: higher-dimensional support-avoidance equivalence}
For every integer $r\ge1$,
\[
\boxed{
\epsilon(I)\ge r
\iff
\exists\,W\le U,\ \dim W=r,\ \ I\text{ is independent in }\Gamma(W).
}
\]
Indeed, $\epsilon(I)\ge r$ iff the kernel of $\pi_I|_U$ contains an $r$-dimensional subspace $W$. This is equivalent to every $u\in W$ vanishing on every edge of $E(I)$, which is equivalent to
\[
E(I)\cap\operatorname{Supp}(W)=\varnothing.
\]
The latter says exactly that $I$ is independent in $\Gamma(W)$.

For $r=1$ this recovers EXCESSDUAL1. For $r>1$ it identifies high excess with a common independent set in the union-support graph of an entire $r$-dimensional product-span subspace.

\section*{PROVED/EXACT: Tur\'an/generalized-weight obstruction}
Let $t_q(m)$ denote the number of edges in the balanced complete $q$-partite Tur\'an graph $T_q(m)$. Equivalently,
\[
t_q(m)=\max\{|F|:([m],F)\text{ contains no }K_{q+1}\}.
\]
Let the $r$th generalized Hamming weight of $U$ be
\[
d_r(U)=
\min_{\substack{W\le U\\ \dim W=r}}
|\operatorname{Supp}(W)|.
\]

Suppose that for every $I\subseteq[m]$ with $|I|=s$,
\[
\epsilon(I)<r.
\]
By the preceding equivalence, every $r$-dimensional $W\le U$ has
\[
\alpha(\Gamma(W))<s.
\]
Hence the complement graph $\overline{\Gamma(W)}$ is $K_s$-free. Tur\'an's theorem gives
\[
|E(\overline{\Gamma(W)})|\le t_{s-1}(m),
\]
so
\[
|\operatorname{Supp}(W)|
\ge
\binom m2-t_{s-1}(m).
\]
Taking the minimum over $W$ yields
\[
\boxed{
\left[\forall I,\ |I|=s:\epsilon(I)<r\right]
\Longrightarrow
d_r(U)\ge
\binom m2-t_{s-1}(m).
}
\]

The contrapositive is an explicit excess certificate:
\[
\boxed{
d_r(U)<
\binom m2-t_{s-1}(m)
\Longrightarrow
\exists I,\ |I|=s:\epsilon(I)\ge r.
}
\]

For $r=1$,
\[
\boxed{
\left[\forall |I|=s:\epsilon(I)=0\right]
\Longrightarrow
d_{\min}(U)\ge
\binom m2-t_{s-1}(m).
}
\]
Thus universal projection injectivity on all $s$-variable cliques forces every nonzero primal product-span vector to have a dense support graph.

For $s=o(m)$ with $s\to\infty$, the exact Tur\'an expression has scale
\[
\binom m2-t_{s-1}(m)=\Theta(m^2/s).
\]
In particular, at the EXCESS1 threshold scale $s=\Theta(\sqrt m)$, universal zero excess would force
\[
d_{\min}(U)=\Omega(m^{3/2}).
\]
This is a structural code-distance consequence only, not a runtime lower bound.
