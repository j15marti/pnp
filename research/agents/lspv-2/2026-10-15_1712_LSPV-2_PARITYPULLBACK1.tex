\section{CONSTRUCTION OR PROOF ATTEMPT}
Let $O_n=\{x\in\{0,1\}^n:\sum_i x_i\text{ is odd}\}$ for $n\ge2$.
\begin{theorem}[Sharp source-only cut count]
Every conjunction of real-affine inequalities in the $n$ original variables, having exactly $O_n$ as Boolean solutions, contains at least $2^{n-1}$ inequalities. Exactly $2^{n-1}$ suffice.
\end{theorem}
\begin{proof}
For two distinct even-parity words $u,v$ select a differing coordinate $i$ and put $p=u\oplus e_i$, $q=v\oplus e_i$. Both $p,q$ are odd and $u+v=p+q$. Any affine inequality valid at both $p,q$ cannot fail at both $u,v$, since affine values obey the same sum equality. Every even word must be excluded by some inequality, hence at least one per even word. For each even $e$, the Hamming-distance inequality
\[
\sum_{i:e_i=0}x_i+\sum_{i:e_i=1}(1-x_i)\ge1
\]
excludes precisely $e$ from the Boolean cube, proving sharpness.
\end{proof}
\begin{theorem}[Linear proper 3-CNF extension]
There is a proper clause-distinct 3-CNF $C_n$ of exactly $4n$ clauses on $2n+1$ variables projecting existentially onto $O_n$, with four satisfying full extensions per odd input and none per even input.
\end{theorem}
\begin{proof}
Use $n-1$ XOR gates $u_2=x_1\oplus x_2$, $u_i=u_{i-1}\oplus x_i$. Each $z=a\oplus b$ is expressed by four proper clauses
\[
(\neg a\vee\neg b\vee\neg z),\
(a\vee b\vee\neg z),\
(\neg a\vee b\vee z),\
(a\vee\neg b\vee z).
\]
Enforce $u_n=1$ with four proper clauses $(u_n\vee s^{\pm}\vee t^{\pm})$ covering the four sign choices for fresh bits $s,t$. Each XOR output is uniquely forced to prefix parity; $s,t$ remain free precisely on odd inputs. All variables within every clause are distinct. Count $4(n-1)+4=4n$ clauses and $n+(n-1)+2=2n+1$ variables.
\end{proof}
\begin{theorem}[Linear continuous extended formulation]
The polytope $\operatorname{conv}(O_n)$ is the projection of a nonnegative unit-flow polytope with $4n$ arc variables and $O(n)$ equations.
\end{theorem}
\begin{proof}
Use the layered two-state parity automaton, sending a unit flow from parity zero before bit one to parity one after bit $n$. Arc $(i,p,b)$ updates state $p$ to $p\oplus b$. Define $x_i=\sum_p f_{i,p,1}$. Each source-to-sink path spells an odd word; every feasible acyclic unit flow decomposes into a convex combination of these paths. Hence the projection equals the claimed convex hull.
\end{proof}
\begin{theorem}[Exact affine-value pullback classification]
If a Boolean map $g:\{0,1\}^n\to\{0,1\}^r$ pulls back every affine target expression to an equal-valued affine expression on source Boolean points, then each coordinate $g_j$ is a constant or source literal. The converse holds.
\end{theorem}
\begin{proof}
Test the target coordinate functional $L(y)=y_j$. If $g_j(x)=c+\sum_i a_ix_i$ on the cube, a unit bit flip implies $a_i\in\{-1,0,1\}$. The range of this affine expression is $\sum_i|a_i|\le1$, since it is Boolean-valued. Thus at most one coefficient is nonzero, giving a constant or literal. Direct substitution proves sufficiency.
\end{proof}
