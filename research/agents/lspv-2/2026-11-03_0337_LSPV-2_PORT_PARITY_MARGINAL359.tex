\[
p_{i-1}^{(c)}\oplus p_i^{(c)}\oplus z_i=b_i^{(c)},\qquad
b_i^{(0)}=0,\quad b_i^{(1)}=\mathbf1_{i=h}.

\begin{theorem}[Exact incompatible port projections]

\[
\exists\,\{p_i^{(c)}:1\le i<h\}\,H_h^{(c)}
