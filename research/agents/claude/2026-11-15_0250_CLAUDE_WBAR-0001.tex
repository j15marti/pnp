\noindent\textbf{Status summary.} P versus NP remains \textbf{OPEN}. This run proves (PROVED/EXACT, elementary) that the
proposed ``weakest sufficient bridge'' is \emph{not} weaker than $\mathrm{P}\neq\mathrm{NP}$ for any fixed system $R$ in which an
unconditional superpolynomial lower bound is currently provable; it gives a general recognizable-family counterexample to every
pointwise run-to-derivation simulation; and it records the exact known form of the bridge for Extended Frege (KNOWN THEOREM:
holds for PV-provably-correct algorithms). The bridge for unrestricted algorithms and the lower bound for strong systems both remain \textbf{OPEN}.
This report changes no schedules, roles, ownership regions or allocations, and does not edit root \texttt{main.tex}
(this routine has no ownership region there).

\section{Objects}
Let $R$ be a Cook--Reckhow refutation system for CNF: a polynomial-time checker $V_R(F,\pi)$ with $V_R(F,\pi)=1$ only if $F$ is UNSAT,
and complete. Let $\operatorname{cost}_R(\pi)=|\pi|$ (bits), or any cost polynomially related to it. Define
\[
 \overline W_R(N)=\max_{F\in\mathrm{UNSAT},\ |\langle F\rangle|\le N}\ \min_{\pi:\,F\vdash_R\bot}\operatorname{cost}_R(\pi).
\]
The multisign DAG hypergraph reduction of the directive (signed hyperedges $=$ clauses, reductions $=$ sound local merges toward a
minimal reduced state containing $\bot$) is a Cook--Reckhow system as soon as each reduction step is polynomial-time checkable.
If its steps are resolution/DP merges it is p-equivalent to resolution; if it also has GF(2) or local exact-truth-table rules it is
the historical Rmix341 family (LSPV-1/348). Minimizing over all derivations is exactly granting the optimal sequence for free, as the
directive intends: $\overline W_R$ is a \emph{nondeterministic} measure.
\[
 \textsc{Bridge}_R:\qquad \mathrm{SAT}\in\mathrm{P}\ \Longrightarrow\ \overline W_R(N)\le N^{O(1)}.
\]

\section{Results}
\begin{proposition}[KNOWN THEOREM; Cook--Reckhow 1979]\label{p:cr}
Some $R$ has $\overline W_R(N)\le\mathrm{poly}(N)$ iff $\mathrm{NP}=\mathrm{coNP}$. Hence a superpolynomial lower bound for
\emph{every} $R$ is equivalent to $\mathrm{NP}\neq\mathrm{coNP}$, which is strictly stronger than the target unless the two coincide.
\end{proposition}

\begin{proposition}[PROVED/EXACT: bridge collapse for weak systems]\label{p:collapse}
Let $R$ be any fixed system for which $\overline W_R(N)\ne N^{O(1)}$ is an unconditional theorem (resolution: Haken 1985,
Urquhart 1987, Chv\'atal--Szemer\'edi 1988; cutting planes: Pudl\'ak 1997; bounded-depth Frege; polynomial calculus). Then
\[
\textsc{Bridge}_R\iff \mathrm{SAT}\notin\mathrm{P}.
\]
\end{proposition}
\begin{proof}
$\textsc{Bridge}_R$ has the form $\mathrm{SAT}\in\mathrm P\Rightarrow Q$ with $Q$ a known falsehood, so it is equivalent to
$\neg(\mathrm{SAT}\in\mathrm P)$.
\end{proof}
\noindent\emph{Consequence.} For every $R$ where the lower-bound half is achievable today, the whole difficulty of P versus NP sits
in the bridge; the two-step plan does not divide the problem. Proving $\textsc{Bridge}_{\mathrm{Res}}$ \emph{is} proving
$\mathrm P\neq\mathrm{NP}$.

\begin{proposition}[PROVED/EXACT: recognizable hard families defeat pointwise transfer]\label{p:recog}
Let $(F_n)$ be any family of UNSAT CNFs that is polynomial-time recognizable (membership of an input in $\{F_n\}$ is decidable in
polynomial time). Then there is a correct total deterministic SAT decider $A$ with $T_A(F_n)\le\mathrm{poly}(|F_n|)$:
on input $G$, if $G\in\{F_n\}$ reject, else run brute force. Therefore no pointwise inequality
$T_A(F)\ge \operatorname{cost}_R(F)^{\Omega(1)}$, valid for all correct $A$ and all UNSAT $F$, holds for any $R$ with a
superpolynomial lower bound on such a family.
\end{proposition}
\noindent Concrete instances: Tseitin formulas on expanders (exponential for resolution; Gaussian elimination is a natural, not
merely hard-coded, decider) and $\mathrm{PHP}^{n+1}_n$ (exponential for resolution; polynomial for cutting planes/counting). The
historical Rmix341 has GF(2) rules, so the Tseitin example is \emph{not} claimed to be hard for Rmix341; Proposition~\ref{p:recog}
needs only \emph{some} recognizable hard family for the given $R$. \emph{Consequence:} the user's argument ``finding the optimal sequence
can only add work'' is valid only for algorithms whose executions \emph{are} (or efficiently yield) $R$-derivations. A worst-case
lower bound for arbitrary machines must use algorithm-dependent, non-recognizable hard instances, matching the TMR quantifier contract
(\texttt{research/coordination/2026-11-13\_1249\_TM\_RESOURCE\_AGENT\_REPLACEMENT.md}).

\begin{proposition}[PROVED/EXACT: trace-extraction transfer]\label{p:trace}
If a decider $A$ is \emph{$R$-proof-producing with overhead $p$} (from the run of $A$ on each UNSAT $F$ an $R$-refutation is
computable in time $p(T_A(F))$), then $\overline W_R(N)\le p(\max_{|F|\le N}T_A(F))$. Hence $\overline W_R$ superpolynomial excludes
polynomial-time $R$-proof-producing deciders (e.g.\ DPLL/CDCL for $R=$ resolution, Beame--Kautz--Sabharwal; Pipatsrisawat--Darwiche).
\end{proposition}

\begin{proposition}[PROVED/EXACT, elementary; uses Cook--Reckhow]\label{p:opt}
If $R$ is \emph{optimal} (every system $Q$ has $|\pi_R|\le\mathrm{poly}(|\pi_Q|)$ for some $R$-proof of each $Q$-provable
formula), then $\textsc{Bridge}_R$ holds. Proof: $\mathrm{SAT}\in\mathrm P\Rightarrow\mathrm{NP}=\mathrm{coNP}\Rightarrow$ some $Q$ is
polynomially bounded $\Rightarrow R$ is polynomially bounded.
\end{proposition}
\noindent Known context (KNOWN THEOREM, Kraj\'i\v{c}ek--Pudl\'ak 1989): $\mathrm{E}=\mathrm{NE}$ implies p-optimal systems exist, and
$\mathrm{P}=\mathrm{NP}\Rightarrow \mathrm E=\mathrm{NE}$ by padding; so under the bridge hypothesis an optimal system exists, but
not as a \emph{fixed, named} $R$ chosen in advance. Whether EF (or any explicit system) is optimal is OPEN. Note also that an optimal $R$
with $\overline W_R$ superpolynomial gives $\mathrm{NP}\neq\mathrm{coNP}$ directly, so this sufficient route proves more than asked.

\begin{proposition}[KNOWN THEOREM; Cook 1975, Kraj\'i\v{c}ek 1995]\label{p:ef}
If $A$ is a polynomial-time algorithm and the theory PV proves ``$A(F)=0\Rightarrow F$ has no satisfying assignment'', then
$\overline W_{\mathrm{EF}}(N)\le\mathrm{poly}(N)$ provided $A$ decides SAT: EF proves the propositional translation of the PV soundness
statement in polynomial size and verifies the polynomial-length computation $A(F)=0$. Moreover $\mathrm{PV}\vdash\mathrm P=\mathrm{NP}$
iff PV defines a polynomial-time map giving either a satisfying assignment or an EF refutation.
\end{proposition}
\noindent So $\textsc{Bridge}_{\mathrm{EF}}$ restricted to PV-provably-correct algorithms is a theorem; the unrestricted bridge is OPEN
and its gap is exactly \emph{true-but-not-PV-provable} correctness. Combined with an EF lower bound (itself a major OPEN problem)
this yields only that no PV-provably-correct polynomial SAT algorithm exists (consistency of $\mathrm P\neq\mathrm{NP}$ with PV),
not $\mathrm P\neq\mathrm{NP}$.
