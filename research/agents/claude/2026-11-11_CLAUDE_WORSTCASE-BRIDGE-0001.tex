\noindent\textbf{Overall status.} P versus NP remains \textbf{OPEN}. Nothing here proves a superpolynomial lower bound for any system stronger than those already in the literature, and nothing here proves the bridge B3 for any system for which a lower bound is known. The results are of two kinds: an elementary \textbf{PROVED/EXACT} collapse lemma about how the shared target decomposes, and a \textbf{KNOWN THEOREM} map of which literature results bear on the bridge.

\section{Setting (unchanged shared objective)}
Fix a Cook--Reckhow proof system $R$ for UNSAT (polynomial-time verifier, sound, complete) and a cost function with $\mathrm{cost}_R(\pi)\ge 1$ that is polynomially related to proof length. For UNSAT $F$, $W_R(F)=\min_{\pi:F\vdash_R\bot}\mathrm{cost}_R(\pi)$, and
\[
\overline W_R(N)=\max_{F\in\mathrm{UNSAT},\ |\langle F\rangle|\le N} W_R(F).
\]
The minimum ranges over \emph{all} valid derivations, so it already grants free knowledge of the best reduction sequence. That is the user's clarification: discovering the sequence can only add work, so leaving it uncharged is legitimate for a lower bound. In corpus labels: B1 (here, LOWER) means $\overline W_R$ is superpolynomial (P2-B1). B3 (here, BRIDGE) means $\mathrm{SAT}\in P\Rightarrow\overline W_R(N)\le\mathrm{poly}(N)$ for the \emph{same} $R$ (P2-B2).

\section{Collapse lemma (PROVED/EXACT, elementary)}
\begin{theorem}[Collapse]\label{thm:collapse}
For every fixed $R$: if LOWER$(R)$ holds, then BRIDGE$(R)\iff \mathrm{SAT}\notin P$. Equivalently,
$\mathrm{LOWER}(R)\wedge\mathrm{BRIDGE}(R)\iff \mathrm{LOWER}(R)\wedge \mathrm{SAT}\notin P$.
\end{theorem}
\begin{proof}
If $\mathrm{SAT}\notin P$, BRIDGE$(R)$ is vacuously true. If $\mathrm{SAT}\in P$, BRIDGE$(R)$ asserts $\overline W_R(N)\le \mathrm{poly}(N)$, which contradicts LOWER$(R)$; so BRIDGE$(R)$ is false. Hence, under LOWER$(R)$, BRIDGE$(R)$ has exactly the truth value of $\mathrm{SAT}\notin P$.
\end{proof}

\begin{corollary}[Known-lower-bound systems]\label{cor:known}
Let $R$ be any system with a published superpolynomial worst-case lower bound: Resolution (Haken 1985, pigeonhole; Urquhart 1987, Tseitin), tree-like Resolution, Cutting Planes, Polynomial Calculus, bounded-depth Frege, the artificial $R_{\mathrm{clock}}$ of P2-B6. For each of these, BRIDGE$(R)$ is equivalent to $\mathrm{SAT}\notin P$, and therefore (Cook--Levin) to $P\neq NP$.
\end{corollary}

\noindent\textbf{Meaning for the program.} The bridge is the ``weakest sufficient'' condition only in the sense of logical strength over an \emph{unknown} $R$. Once a lower bound for $R$ is in hand, proving BRIDGE$(R)$ is exactly as hard as $P\ne NP$. The decomposition LOWER + BRIDGE makes the problem easier only if BRIDGE$(R)$ is proved by a method that never uses LOWER$(R)$: a \emph{uniform compiler} that turns any correct polynomial-time UNSAT decider into polynomial-size $R$-proofs. That is the noncircular same-$R$ compiler that P2-L5-409 already identified as needed. The present lemma adds a constraint on where such a compiler can exist. It cannot exist, by any argument formalizable without already knowing $P\ne NP$, for any $R$ in Corollary~\ref{cor:known}. Proving that such a compiler exists for Resolution would itself prove $P\neq NP$.

\begin{corollary}[Strength floor]\label{cor:floor}
A noncircular proof of BRIDGE$(R)$ by uniform compilation must target an $R$ for which LOWER$(R)$ is not currently provable. In particular the compiler must handle every polynomial-time algorithm, including algorithms whose correctness uses arbitrary polynomial-time reasoning. The only known general compilers of this kind land in Extended Frege or stronger systems (\S3).
\end{corollary}
\noindent Status of Corollary~\ref{cor:floor}: the first sentence is PROVED/EXACT (it restates Theorem~\ref{thm:collapse}). The second is a KNOWN-THEOREM-based classification, not an impossibility theorem.

\section{Literature check for this run}
\begin{itemize}
\item \textbf{Cook--Reckhow (1979)}, KNOWN THEOREM. Some $R$ has $\overline W_R$ polynomial if and only if $NP=coNP$. So proving LOWER$(R)$ for \emph{all} $R$ is equivalent to $NP\ne coNP$, which is stronger than $P\neq NP$. Free optimal sequences make $\overline W_R$ a \emph{nondeterministic} measure, and that is why it naturally targets $NP$ versus $coNP$.
\item \textbf{Cook (1975), PV $\to$ EF}, KNOWN THEOREM. If $\mathrm{PV}\vdash\forall x\,[A(x)=0\to x\in\mathrm{UNSAT}]$ for a polynomial-time $A$ that is complete on UNSAT, then EF has polynomial-size proofs of the propositional translations, which gives a polynomial $\overline W_{EF}$. This is the canonical \emph{noncircular} BRIDGE, and it is CONDITIONAL on PV-provable correctness. Missing piece: algorithms whose correctness is true but not PV-provable.
\item \textbf{Pich--Santhanam, ``Towards $P\ne NP$ from Extended Frege lower bounds'' (arXiv:2312.08163, 2023)}, KNOWN THEOREM (conditional). If EF is not p-bounded, then $P\neq NP$ follows under any of three $S^1_2$-provability hypotheses: feasible anticheckers for SAT; feasible witnessing of $NP\not\subseteq P/\mathrm{poly}$; or a hardness condition on E combined with an $S^1_2$-provable OWF-to-SAT reduction. This is the closest published approach to BRIDGE(EF), and the hypotheses are OPEN. It is the best-matched literature clue for the shared target: \emph{the bridge for EF reduces to feasible-provability statements in $S^1_2$}.
\item \textbf{Pich--Santhanam, ``Why are proof complexity lower bounds hard?'' (FOCS 2019)}, KNOWN THEOREM. There are unconditional and conditional barriers for EF lower bounds, including a natural-proofs-type barrier under hardness assumptions. LOWER(EF) faces structural obstacles.
\item \textbf{State of LOWER for strong systems.} Per Je\v{r}\'abek (Bull.\ Symb.\ Logic, 2025), the best Frege lower bounds remain linear in lines and quadratic in size. LOWER(Frege) and LOWER(EF) are OPEN and far from current methods.
\item \textbf{Failed transfer flagged.} One secondary search summary asserted that ``$P=NP$ implies EF is polynomially bounded''. That is \textbf{not} a known theorem. $P=NP$ gives \emph{some} polynomially bounded system, not EF. The statement would be BRIDGE(EF) itself, which is open (cf.\ P1-05, P2-L5-409). It was rejected.
\end{itemize}
