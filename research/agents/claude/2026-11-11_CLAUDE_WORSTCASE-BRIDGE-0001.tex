\section{Objective, formalized}

\begin{definition}[Fixed refutation system]
A \emph{refutation system} $R$ is a polynomial-time verifier $V_R(F,\pi)$ that is \emph{sound}
($V_R(F,\pi)=1\Rightarrow F\in\UNSAT$) and \emph{complete} (every $F\in\UNSAT$ has some $\pi$ with
$V_R(F,\pi)=1$) [Cook--Reckhow 1979]. A cost $\cost_R(\pi)$ is \emph{admissible} if
$|\pi|^{1/c}/c\le \cost_R(\pi)\le c\,|\pi|^c$ for a constant $c$. The multisign DAG-hypergraph
reductions of the project (signed hyperedges = clauses, a reduction step = a sound inference
producing a new node, ``minimal reduced logical state'' = the node $\bot$) form such a system once
the rule set and the weights are fixed; R\_mix341 is the intended instance (\texttt{proof2.txt}
[P2-D1]).
\end{definition}

\[
\boxed{\;\Wbar_R(N)=\max_{F\in\UNSAT,\ |\langle F\rangle|\le N}\ \min_{\pi:\,F\vdash_R\bot}\cost_R(\pi)\;}
\qquad
\Br(R):\ \SAT\in\mathrm{P}\Longrightarrow \Wbar_R(N)\le \mathrm{poly}(N),
\qquad
\LB(R):\ \Wbar_R(N)\ne N^{O(1)} .
\]
Then $\Br(R)\wedge\LB(R)\Rightarrow \SAT\notin\mathrm{P}\Rightarrow \mathrm{P}\ne\mathrm{NP}$.

\paragraph{On the free optimal sequence.} The owner's point is right: $\min_\pi$ gives every
derivation the best possible reduction sequence for free, so $\Wbar_R$ is a lower bound on the work
of any procedure \emph{whose execution is an $R$-derivation}. For example, CDCL runs are
resolution derivations [Pipatsrisawat--Darwiche 2011]. Not charging for proof search is sound for a
lower bound, and automatability results [Atserias--M\"uller 2019] need not be paid. The catch is
that an arbitrary polynomial-time algorithm's execution is \emph{not} an $R$-derivation. The whole
question of translating algorithm runs into derivations is contained in $\Br(R)$.

\section{Results of this run}

\begin{proposition}[Cook--Reckhow; \status{KNOWN THEOREM}]\label{p:cr}
Some refutation system $R$ has $\Wbar_R(N)\le\mathrm{poly}(N)$ iff $\mathrm{NP}=\mathrm{coNP}$.
In particular $\SAT\in\mathrm{P}\Rightarrow\exists R:\ \Wbar_R\le\mathrm{poly}$. The bridge for
\emph{some} system is free. The project needs it for one \emph{fixed} $R$ chosen before the
hypothetical algorithm, so the quantifier order is $\exists R\,[\SAT\in\mathrm P\Rightarrow\dots]$,
not $\SAT\in\mathrm P\Rightarrow\exists R[\dots]$.
\end{proposition}

\begin{proposition}[Bridge collapse; \status{PROVED/EXACT}, elementary]\label{p:collapse}
If $\LB(R)$ is an unconditional theorem, then $\Br(R)\iff \SAT\notin\mathrm{P}$.
\end{proposition}
\begin{proof}
($\Rightarrow$) $\Br(R)$ and $\LB(R)$ give $\SAT\notin\mathrm P$ by contraposition.
($\Leftarrow$) If $\SAT\notin\mathrm P$, the implication $\Br(R)$ has a false antecedent and is true.
\end{proof}

\begin{corollary}[\status{PROVED/EXACT} given the cited KNOWN THEOREMS]
For each of the following systems, proving $\Br(R)$ is exactly as hard as proving
$\mathrm{P}\ne\mathrm{NP}$: Resolution (Haken 1985; Urquhart 1987; Ben-Sasson--Wigderson 2001),
Cutting Planes (Pudl\'ak 1997), bounded-depth Frege (Ajtai 1988; Krají\v{c}ek--Pudl\'ak--Woods 1995;
Pitassi--Beame--Impagliazzo 1993), Polynomial Calculus / Nullstellensatz (Razborov 1998;
Ben-Sasson--Impagliazzo 1999), and any R\_mix341 completion in which a superpolynomial lower bound
is ever proved. For these systems the ``weakest sufficient bridge'' is a restatement of the
conjecture, not a weaker sufficient condition.
\end{corollary}

\begin{proposition}[Monotonicity; \status{PROVED/EXACT}]\label{p:mono}
If $R$ p-simulates $R'$, then $\Br(R')\Rightarrow\Br(R)$ and $\LB(R)\Rightarrow\LB(R')$.
So within the p-simulation order, the systems with a provable bridge are closed upward and the
systems with a provable lower bound are closed downward. A proof of P$\ne$NP along this route needs
one system that is in both sets.
\end{proposition}
\begin{proof}
p-simulation gives $\Wbar_R(N)\le \mathrm{poly}(\Wbar_{R'}(\mathrm{poly}(N)))$ (cost admissibility
absorbs the polynomial).
\end{proof}

\begin{proposition}[Weakest known proven bridge; \status{KNOWN THEOREM} + \status{PROVED/EXACT} corollary]\label{p:ef}
(Cook 1975; Krají\v{c}ek--Pudl\'ak 1990.) Extended Frege p-simulates every proof system $Q$ whose
soundness (reflection) statement is provable in Cook's theory $\mathrm{PV}$.
\emph{Corollary.} Suppose a polynomial-time $A$ decides SAT and
$\mathrm{PV}\vdash \forall F\,\forall a\,[A(F)=\mathrm{UNSAT}\rightarrow \neg F(a)]$.
Then $\Wbar_{\mathrm{EF}}(N)\le\mathrm{poly}(N)$.
\end{proposition}
\begin{proof}[Proof of corollary]
Let $Q_A$ accept $(F,\varepsilon)$ iff $A(F)=\mathrm{UNSAT}$. It is a sound, complete,
polynomial-time refutation system with empty proofs. Its reflection statement is the displayed
PV-provable sentence, so EF p-simulates $Q_A$. Every UNSAT $F$ of size $\le N$ therefore has an EF
refutation of size $\mathrm{poly}(N)$.
\end{proof}

\begin{remark}[The gap left by Prop.~\ref{p:ef}; \status{OPEN}]
Prop.~\ref{p:ef} establishes the \emph{feasibly-provable} bridge $\Br^{\mathrm{PV}}(\mathrm{EF})$.
Its contrapositive is: an $\LB(\mathrm{EF})$ would rule out every polynomial-time SAT algorithm whose
correctness PV can prove. That is a major CONDITIONAL-type separation, but not P$\ne$NP. A correct
algorithm whose correctness is unprovable in PV escapes it. Strengthening PV to a theory $T$ (up to
ZFC, via $\mathrm{EF}+\mathrm{RFN}_T$) shrinks the escape set but never removes it, and it makes
$\LB$ harder. The unconditional $\Br(\mathrm{EF})$ is a recognized open problem in proof complexity
(Krají\v{c}ek, \emph{Proof Complexity}, CUP 2019), and no unconditional EF lower bound is known.
Recent unprovability results (Pich--Santhanam STOC 2021; Li--Oliveira 2023) show that PV cannot
prove strong co-nondeterministic lower bounds. This is relevant if the project hopes to prove
$\LB(\mathrm{EF})$ inside a feasible theory.
\end{remark}

\paragraph{Consequence for the five-role program.} The two halves cannot both be easy:
\begin{itemize}
\item \textbf{Weak $R$ (Resolution, CP, PC, AC$^0$-Frege, R\_mix341 fragments):} $\LB(R)$ is known or
within reach (Tseitin/PHP/random 3-CNF), and $\Br(R)\iff\mathrm P\ne\mathrm{NP}$ by
Prop.~\ref{p:collapse}. All the difficulty is in the bridge.
\item \textbf{Strong $R$ (EF and above):} the bridge is proved for feasibly-verified algorithms
(Prop.~\ref{p:ef}). All the difficulty is in $\LB(R)$, which is beyond every current technique
(Natural-proofs-type and feasible-unprovability barriers apply).
\end{itemize}
A hard UNSAT family therefore has to survive \emph{every} system up to the bridging system. In
practice that means EF. Families that are hard for resolution but easy for EF (PHP: Buss 1987;
Tseitin: Gaussian elimination in EF or in a GF(2) rule, the corpus's own $W=4h-1$ for R\_mix341 on
cubic Tseitin, \texttt{proof2.txt} [P2-D1]) cannot close the route. The surviving candidate families
are those conjectured hard for EF: random $k$-CNF at the threshold, Nisan--Wigderson-generator
tautologies $\tau(g)_b$ (Krají\v{c}ek; Alekhnovich--Ben-Sasson--Razborov--Wigderson 2004), and
$\mathrm{Circuit}$-lower-bound tautologies (Razborov 2015).

\section{Barrier-transfer literature preflight}
Obstacle 1: the quantifier swap in Prop.~\ref{p:cr}. Obstacle 2: the provability gap in
Prop.~\ref{p:ef}.
\begin{enumerate}
\item \textbf{Directly transferable theorems:} Cook--Reckhow 1979; Cook 1975 (PV $\leftrightarrow$ EF);
Krají\v{c}ek--Pudl\'ak 1989/1990 (reflection, optimal proof systems: NE$=$coNE $\Rightarrow$ an
optimal system exists; E$=$NE $\Rightarrow$ a p-optimal one exists, both non-explicit). Atserias--M\"uller
2019 (automatability, which is irrelevant to the lower bound). Pich--Santhanam 2021 (feasible unprovability).
\item \textbf{Useful analogy needing a new transfer:} obstruction theory and Lyapunov certificates,
in the form ``a run of $A$ yields a certificate in a fixed calculus''. This is exactly the
reflection-principle bridge and adds nothing beyond Prop.~\ref{p:ef} unless it supplies a
\emph{fixed} calculus that verifies all correct algorithms. Heuristically (\status{CANDIDATE}
obstruction, not a theorem here), any fixed sound r.e.\ theory fails to prove the correctness of
some correct algorithms, by incompleteness. So a provability-based bridge always leaves an escape
set, and an unconditional bridge must use something other than provable correctness.
\item \textbf{Misleading transfer:} reading ``free knowledge of the best sequence'' as making the
bridge free. It makes the \emph{lower-bound side} cheaper to state, but it does not turn an
arbitrary algorithm's run into an $R$-derivation.
\end{enumerate}
\textbf{PQ ledger:} items 1--7 classified FAILED TRANSFER for this target. None of them supplies a
bridge for a fixed $R$.
