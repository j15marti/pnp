\title{Worst-case derivation cost: a dichotomy for the bridge $\SAT\in P\Rightarrow \overline W_R\le\mathrm{poly}$}
\author{Claude (scheduled run, 2026-11-03)}\maketitle
\paragraph{Status.} P versus NP remains \textbf{OPEN}. Nothing below proves a lower bound; it classifies what the proposed bridge can and cannot be.

\section{Setting}
$\overline W_R(N)=\max_{F\in\UNSAT,|F|\le N}\min_{\pi:F\vdash_R\bot}\mathrm{cost}_R(\pi)$. Strategy discovery is not charged, so $\overline W_R$ is a \emph{nondeterministic} measure.
Call $R$ \emph{verifiable} if $\{(F,\pi)\}$ with $\pi:F\vdash_R\bot$ is decidable in time poly$(|F|+|\pi|)$ (Cook--Reckhow system), with $\mathrm{cost}_R=|\pi|$.

\section{Results (all PROVED/EXACT, elementary)}
\begin{prop}[Weak systems: bridge $=$ P$\ne$NP]\label{p1}
Let $R$ be verifiable with an unconditional superpolynomial lower bound (resolution, Haken; cutting planes, Pudl\'ak; bounded-depth Frege, Ajtai/H\aa stad). Then ``$\SAT\in P\Rightarrow \overline W_R\le\mathrm{poly}$'' is equivalent to $\SAT\notin P$.
\end{prop}
\emph{Proof.} The conclusion is false unconditionally, so $A\Rightarrow\bot$ is $\neg A$. \qed

\begin{prop}[Nondeterminism]\label{p2}
If $R$ is verifiable and $\overline W_R\le\mathrm{poly}$ then $\mathrm{NP}=\mathrm{coNP}$ (guess $\pi$, verify). Conversely, by Cook--Reckhow, NP$\ne$coNP iff $\overline W_R$ is superpolynomial for \emph{every} verifiable $R$.
\end{prop}
\noindent Consequence: a lower bound for a \emph{single fixed} verifiable $R$, however strong, proves only ``$R$ is not polynomially bounded''; it implies $\SAT\notin P$ only through the bridge, and the bridge is exactly the missing content.

\begin{prop}[Circularity for unverifiable cost]\label{p3}
If derivations may be arbitrary decider traces (cost = running time, no verifier), the bridge holds trivially and $\overline W_R\ge$ superpoly is equivalent to $\SAT\notin P$ (for deterministic-time cost); the lower bound is then P$\ne$NP restated. Free knowledge of the strategy is harmless there only because the measure has been made deterministic-time; the ``minimum over sequences'' adds nothing.
\end{prop}

\begin{thm}[Conditional bridge for EF via provable rejection-soundness]\label{t4}
Let $M$ decide SAT in time $n^c$ and suppose the $\Pi^b_1$ sentence $\forall F,a\,(M(F)=0\to a\not\models F)$ is provable in PV. Then $\mathrm{EF}$ has polynomial-size refutations of every unsatisfiable $F$.
\end{thm}
\emph{Proof sketch.} The propositional translation of the sentence has poly-size EF proofs (Cook; Krajíček). For unsatisfiable $F$ the computation $M(F)=0$ has a poly-size trace, which EF verifies line by line using extension variables; instantiating the translated sentence gives $\neg(a\models F)$ for the variable vector $a$, i.e.\ a refutation. \qed

\noindent So the bridge for EF holds under ``$\SAT\in P$ via a PV-provably sound-on-rejection algorithm''. Whether $\SAT\in P$ alone suffices is \textbf{OPEN}; Levin search is provably correct on acceptance but not on rejection, which is precisely the gap.

\section{Consequences for the programme}
\begin{enumerate}
\item The weakest sufficient bridge cannot be proved for resolution-like $R$ without proving P$\ne$NP (Prop.~\ref{p1}). Bridge candidates must have $R$ strong (EF/Frege or stronger), where unconditional lower bounds are unknown (best known Frege/EF size bounds are only quadratic; secondary source).
\item For such $R$ the lower bound is at least as hard as showing $R$ not polynomially bounded; the programme's real target is a bridge \emph{plus} an EF-strength lower bound. Pich--Santhanam give a conditional route (EF-unprovability of circuit lower bounds $\Rightarrow$ P$\ne$NP); this is a hint, not a proof.
\item Next targets: (T-A) decide whether ``$\SAT\in P\Rightarrow$ EF p-bounded'' follows from known conservativity results (CANDIDATE, literature check); (T-B) find a hard-instance family whose EF lower bound would follow from a circuit-complexity hypothesis weaker than P$\ne$NP; (T-C) audit the repo's LSPV-1 worst-case-minimum reports against Prop.~\ref{p3}: any ``minimum over solver traces'' with unverified answers is circular.
\end{enumerate}
