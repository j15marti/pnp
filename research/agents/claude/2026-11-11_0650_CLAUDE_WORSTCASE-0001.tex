\section{Objects}

Fix a Cook--Reckhow proof system $R$ for $\UNSAT$ (polynomial-time checkable, sound, complete) with a
nonnegative cost that is polynomially related to encoded proof length. As in the directive,
\[
W_R(F)=\min_{\pi:F\vdash_R\bot}\operatorname{cost}_R(\pi),\qquad
\Wb_R(N)=\max_{F\in\UNSAT,\ |\langle F\rangle|\le N}W_R(F).
\]
The inner minimum grants free selection of the best derivation. This is exactly the proof-complexity
measure of $R$: $\Wb_R\le\poly$ iff $R$ is polynomially bounded. Free selection is legitimate for a
\emph{lower bound on $R$-derivations}; the hardness of \emph{finding} the derivation (e.g.\ Atserias--M\"uller:
automating resolution is NP-hard) is irrelevant to it, as the directive says. \status{PROVED/EXACT}
(definitional).

\[
\mathrm{LB}(R):\ \Wb_R(N)\ne N^{O(1)},\qquad
\mathrm{B3}(R):\ \SAT\in\mathrm P\ \Longrightarrow\ \Wb_R(N)\le N^{O(1)}.
\]
$\mathrm{LB}(R)\wedge\mathrm{B3}(R)\Rightarrow\SAT\notin\mathrm P$ by contraposition.

\section{The seesaw theorem}

\begin{theorem}[B3 collapses on weak systems]\label{thm:collapse}
\status{PROVED/EXACT} If $\mathrm{LB}(R)$ is a theorem, then $\mathrm{B3}(R)$ is logically equivalent to
$\SAT\notin\mathrm P$.
\end{theorem}
\begin{proof}
($\Rightarrow$) $\mathrm{B3}(R)$ and $\mathrm{LB}(R)$ give $\SAT\notin\mathrm P$.
($\Leftarrow$) If $\SAT\notin\mathrm P$ the implication $\mathrm{B3}(R)$ is vacuously true.
\end{proof}

\begin{corollary}\label{cor:weak}
\status{PROVED/EXACT, using KNOWN THEOREMS} For $R\in\{$resolution, regular/tree resolution, Res$(k)$ for
constant $k$, cutting planes, polynomial calculus over any field, bounded-depth Frege$\}$ the bridge
$\mathrm{B3}(R)$ is \emph{literally} the statement $\mathrm P\ne\mathrm{NP}$. Proving it is not a weaker
target than the separation; it is a restatement. (Lower bounds: Haken 1985, Chv\'atal--Szemer\'edi 1988 for
resolution; Pudl\'ak 1997 for CP; Razborov 1998 / Impagliazzo--Pudl\'ak--Sgall for PC; Ajtai 1988,
Beame--Impagliazzo--Pitassi, Kraj\'{\i}\v{c}ek--Pudl\'ak--Woods, H{\aa}stad--Risse 2025 for bounded-depth Frege.)
\end{corollary}

In the classification of the world ledger (\url{research/agents/claude/world-ledger/}), $\mathrm{B3}(R)$ for
such $R$ is a \emph{reformulation}, not progress.

\begin{definition}[theory-indexed system]
For a sound r.e.\ theory $T$ extending $\mathrm{PV}$ (e.g.\ $T=\mathrm{PV}, S^1_2, \mathrm{PA}, \mathrm{ZFC}$),
let $R_T$ accept $(F,\langle A,c,\sigma,1^t,\tau\rangle)$ iff $\sigma$ is a $T$-proof of
$\forall x\,[A(x)=0\to x\notin\SAT]$ for the clocked machine $A$ with clock $|x|^c+c$, $t\ge|F|^c+c$, and
$\tau$ is the halting computation of $A$ on $F$ with output $0$. Check $\sigma$ and replay $\tau$.
\end{definition}

$R_T$ is a Cook--Reckhow system as soon as $T$ is $\Pi_1$-sound. It p-simulates every system whose reflection
principle $T$ proves (Cook 1975; Kraj\'{\i}\v{c}ek--Pudl\'ak 1989; Pudl\'ak 2020). In particular
$R_{\mathrm{PV}}$ is p-equivalent to Extended Frege, up to the usual normalisations. \status{KNOWN THEOREM}

\begin{theorem}[B3 is nearly free on strong systems]\label{thm:strong}
\status{PROVED/EXACT} If $\SAT\in\mathrm P$ is witnessed by an algorithm $A$ (with polynomial clock $c$)
whose soundness $\forall x\,[A(x)=0\to x\notin\SAT]$ is provable in $T$, then $\Wb_{R_T}(N)=O(N^{c'})$.
\end{theorem}
\begin{proof}
Fix the single $T$-proof $\sigma_A$ once; its length is a constant independent of $F$. For $F\in\UNSAT$,
correctness of $A$ gives $A(F)=0$ within $|F|^c+c$ steps. The proof $\langle A,c,\sigma_A,1^t,\tau\rangle$
has length $O(1)+O(|F|^{c})+|\tau|=O(|F|^{2c})$ for a standard trace encoding.
\end{proof}

\begin{corollary}[the exact residual gap of B3]
\status{PROVED/EXACT} For every sound $T$, $\mathrm{B3}(R_T)$ can fail only if $\SAT\in\mathrm P$ and
\emph{no} polynomial-time SAT algorithm has $T$-provable soundness. Equivalently, an unconditional
$\mathrm{LB}(R_{\mathrm{ZFC}})$ already rules out every polynomial-time SAT algorithm whose correctness ZFC
can prove.
\end{corollary}

\begin{proposition}[the price on the other side]
\status{PROVED/EXACT, using KNOWN THEOREMS} $\mathrm{LB}(R_T)\Rightarrow\mathrm{LB}(\mathrm{EF})$ for
$T\supseteq\mathrm{PV}$, since $R_T$ p-simulates EF. Superpolynomial EF lower bounds are a central open
problem; no superpolynomial lower bound is known even for Frege or $\mathrm{AC}^0[p]$-Frege. Lu--Santhanam--Tzameret
(ITCS 2026) give an unconditional infinitely-often bound only for one specific DNF family, and whether that
family is a tautology is itself tied to open conjectures.
\end{proposition}

\paragraph{Seesaw summary.} For any \emph{fixed} $R$ the decomposition $\mathrm{LB}(R)+\mathrm{B3}(R)$
conserves difficulty. When $R$ is weak, LB is known and B3 \emph{is} $\mathrm P\ne\mathrm{NP}$ (Theorem~\ref{thm:collapse}).
When $R$ is strong, B3 is a theorem except for the independence gap (Theorem~\ref{thm:strong}), and LB is at
least the open EF lower-bound problem. No known $R$ has both arrows proved. Any new $R$ must be placed on this
seesaw before work starts. \status{OPEN} remains the status of the combined route.

\paragraph{A remark on fixing $R$.} $\SAT\in\mathrm P\Rightarrow\mathrm E=\mathrm{NE}$ by padding, and
$\mathrm E=\mathrm{NE}$ implies that a p-optimal proof system exists (Kraj\'{\i}\v{c}ek--Pudl\'ak 1989). That
system is built from the hypothetical algorithm, so it is not a preselected $R$. This is exactly why the
directive warns that ``some p-bounded system exists'' does not identify $R$. \status{KNOWN THEOREM}
(citation from the standard literature; the exact hypothesis should be re-checked against KP89 before
anyone relies on it).

\section{Where the multisign DAG-hypergraph calculus sits}

Model the multisign hypergraph state as a set of signed hyperedges (clauses). A reduction step derives a
signed hyperedge $D$ from at most $k$ current hyperedges whenever $\{C_1,\dots,C_k\}\models D$. This is a
semantic, bounded-fan-in rule, and steps are shared in a DAG.

\begin{proposition}[bounded-arity semantic hyperedge rules are resolution]\label{prop:tarsi}
\status{PROVED/EXACT} If $\{C_1,\dots,C_k\}\models D$ (clauses), then resolution plus one weakening derives
$D$ from $C_1,\dots,C_k$ with at most $2^{k-1}-1$ resolution steps. Hence the calculus $\mathrm{MSH}_k$ with
such steps is p-simulated by resolution with a constant factor $2^{k-1}$, and
$\Wb_{\mathrm{MSH}_k}(N)\ge 2^{\Omega(N/\log N)}$ (via random 3-CNF, Chv\'atal--Szemer\'edi).
\end{proposition}
\begin{proof}
Restrict by the assignment $\rho$ falsifying $D$. The clauses $C_i|_\rho$ (those not satisfied) form an
unsatisfiable CNF with $m\le k$ clauses. Take a minimally unsatisfiable subset. By Tarsi's lemma it has more
clauses than variables, so it has at most $k-1$ variables. A complete decision tree on those variables is a
tree-like refutation with at most $2^{k-1}$ leaves, hence at most $2^{k-1}-1$ resolution steps. Re-adding
the literals of $D$ lifts it to a derivation of some $D'\subseteq D$. One weakening gives $D$.
\end{proof}

\status{COMPUTATIONAL} \texttt{research/agents/claude/worstcase/tarsi\_sim\_check.py} exhaustively checks
random samples: 6{,}232 implications ($n=4,k=2$, maximum 1 step), 9{,}175 ($n=4,k=3$, maximum 2 steps,
bound 3) and 3{,}540 ($n=5,k=4$, maximum 2, bound 7). The bound holds in every case. Each invocation finishes
in under 30\,s. Raw output is in \texttt{tarsi\_sim\_results.jsonl}.

\paragraph{Consequence.} Without auxiliary variables, the multisign hypergraph calculus has a
\emph{proved} exponential $\Wb$. By Theorem~\ref{thm:collapse} its bridge is then exactly $\mathrm P\ne\mathrm{NP}$.
The calculus can move to the strong side of the seesaw only by adding extension (fresh signed hyperedges
defined from old ones), algebraic counting or circuit compression. Those additions make it at least
EF-like. Its lower bound then becomes open: PHP has polynomial EF proofs (Cook--Reckhow), and it also has
polynomial cutting-planes proofs. Under adaptive switching, PHP and Tseitin fail the survival test.
Tseitin has polynomial-calculus proofs over $\mathbb F_2$ and quasi-polynomial CP proofs
(Dadush--Tiwari 2020). Random 3-CNF is the surviving candidate against resolution, PC and CP. Its Frege and
EF status is OPEN.

\section{Recommended next targets (strict statuses)}
\begin{enumerate}
\item \status{OPEN} Any route through a fixed $R$ must say which side of the seesaw it uses. Weak-$R$ B3
work should be retired as a ``weaker bridge'', because it is $\mathrm P\ne\mathrm{NP}$ restated.
\item \status{OPEN} The highest-value proof-complexity target compatible with the directive is
$\mathrm{LB}(R)$ for some $R$ that p-simulates $\mathrm{AC}^0[p]$-Frege. Even that would not give B3 without
the provability gap closed.
\item \status{CANDIDATE} Classify every multisign rule extension (auxiliary hyperedges, sign-counting,
shared XOR) by the weakest standard system that p-simulates it. Proposition~\ref{prop:tarsi} settles the
extension-free case.
\end{enumerate}
