\section{Setting (unchanged from the 2026-11-02 directive)}
Fix one sound, complete, effectively specified refutation system $R$ for CNF, with nonnegative weights. For an unsatisfiable $F$ with encoding $\langle F\rangle$, let
\[
W_R(F)=\min_{\pi:F\vdash_R\bot}\operatorname{cost}_R(\pi),\qquad
\Wb_R(N)=\max_{F\in\mathrm{UNSAT},\ |\langle F\rangle|\le N}W_R(F).
\]
Here $\operatorname{cost}_R$ sums the weights of the distinct nodes of the derivation DAG. Optimal-proof selection is free, and the cost of discovering a proof is \emph{not} charged. This matches the user's minimal-work, worst-case formulation over the multisign DAG hypergraph. Write $\mathrm{LB}(R)$ for ``$\Wb_R$ is not bounded by any polynomial'' and
\[
\mathrm{B3}(R):\quad \mathrm{SAT}\in\mathrm P\ \Longrightarrow\ \exists C,k\ \forall N:\ \Wb_R(N)\le C(N+1)^k .
\]
\textbf{Key structural fact.} $\Wb_R$ is an \emph{oracle-free} arithmetic function. It depends only on $R$, which is fixed before any machine is considered. Hence $\mathrm{LB}(R)$ is an oracle-free sentence.

\section{Results}

\begin{definition}[Relativized bridge]
For an oracle $O$, let $\mathrm{B3}^O(R)$ be $\mathrm{NP}^O\subseteq\mathrm P^O\Rightarrow \Wb_R(N)\le\mathrm{poly}(N)$. For B2, let $\mathrm{B2}^O(R)$ say that for every deterministic oracle machine $A^O$ deciding $\mathrm{SAT}$ (equivalently, deciding $\mathrm{SAT}$ correctly with oracle access) there is a polynomial $p_A$ with $\Wb_R(N)\le p_A(N+T^{\rm wc}_{A^O}(N))$. A proof of B2/B3 \emph{relativizes} if it proves $\mathrm{B2}^O$/$\mathrm{B3}^O$ for every oracle $O$.
\end{definition}

\begin{theorem}[Relativization barrier for the bridge]\label{thm:rel}
For every fixed system $R$: if $\mathrm{B3}^O(R)$ holds for all $O$, or if $\mathrm{B2}^O(R)$ holds for all $O$, then $\Wb_R$ is polynomially bounded, i.e.\ $\neg\mathrm{LB}(R)$. Consequently, if $\mathrm{LB}(R)$ is true, then no relativizing argument proves $\mathrm{B3}(R)$ or $\mathrm{B2}(R)$.
\end{theorem}
\begin{proof}
Let $O=\mathrm{TQBF}$. By Baker--Gill--Solovay, $\mathrm P^O=\mathrm{NP}^O=\mathrm{PSPACE}$. In particular there is a polynomial-time oracle machine $A^O$ that decides $\mathrm{SAT}$ on all inputs, including malformed ones, with $T^{\rm wc}_{A^O}(N)\le q(N)$ for a polynomial $q$: it asks one oracle query after a polynomial-time parse. $\mathrm{B3}^O(R)$ has a true hypothesis, and its conclusion makes no reference to $O$. So $\Wb_R$ is polynomially bounded. $\mathrm{B2}^O(R)$ applied to $A^O$ gives $\Wb_R(N)\le p_A(N+q(N))$, again a polynomial. The final sentence is the contrapositive.
\end{proof}

\noindent\textbf{Interpretation.} The bridge cannot be proved by any argument that treats the hypothetical SAT algorithm as a black box with step-by-step simulation. Any successful proof must use the internal code of the algorithm, for example its formalized correctness (the reflection route below), or some other non-relativizing property. A decider with a TQBF oracle step has no propositional rendering of that step in a fixed $R$. This explains why the reflection/Cook-translation route is the only kind of bridge the corpus has found. It also shows that every ``trace-to-proof'' compiler that treats oracle-like macro steps uniformly must fail; the corpus TRACE firewalls (LSPV-5/409) are instances.

\begin{corollary}[No weakening from choosing a system with a known lower bound]\label{cor:known}
Let $R$ be any system for which $\mathrm{LB}(R)$ is already a theorem, e.g.\ resolution (Haken 1985 for PHP; Urquhart 1987 for Tseitin), cutting planes (Pudl\'ak 1997), or bounded-depth Frege (Ajtai; Pitassi--Beame--Impagliazzo; Kraj\'{\i}\v{c}ek--Pudl\'ak--Woods). Then $\mathrm{B3}(R)\Longleftrightarrow \mathrm P\neq\mathrm{NP}$, and B3$(R)$ fails relative to TQBF.
\end{corollary}
\begin{proof}
($\Rightarrow$) $\mathrm{B3}(R)$ together with $\mathrm{LB}(R)$ gives $\mathrm{SAT}\notin\mathrm P$. ($\Leftarrow$) If $\mathrm{SAT}\notin\mathrm P$, the implication holds vacuously. The relativized failure is Theorem~\ref{thm:rel}.
\end{proof}

\begin{proposition}[Difficulty exchange]\label{prop:exchange}
Suppose $R'$ p-simulates $R$ with cost blow-up $\Wb_{R'}(N)\le s(N,\Wb_R(N))$ for a polynomial $s$. Then $\mathrm{B3}(R)\Rightarrow\mathrm{B3}(R')$ and $\mathrm{LB}(R')\Rightarrow\mathrm{LB}(R)$.
\end{proposition}
\begin{proof}
Polynomial composition, applied in both directions.
\end{proof}
\noindent Strengthening $R$ makes the bridge easier and the lower bound harder; weakening $R$ does the opposite. For every $R$, $\mathrm{B3}(R)\wedge\mathrm{LB}(R)\Longleftrightarrow \mathrm P\neq\mathrm{NP}\wedge\mathrm{LB}(R)$. So the pair is never weaker than P$\neq$NP, and it is strictly stronger unless $\mathrm{LB}(R)$ follows from P$\neq$NP. No choice of $R$ moves the difficulty away; the only choice is where it sits.

\begin{corollary}[Location of the current target]\label{cor:ef}
The corpus (LSPV-1/357) records that RES$+$VARDEF inside $R_{\rm mix341}$ is the extended-resolution system ER$_2$ at polynomial cost. Then $\mathrm{LB}(R_{\rm mix341})\Rightarrow\mathrm{LB}(\mathrm{ER})$, and hence superpolynomial Extended Frege lower bounds. The user's shared target therefore contains the open Extended Frege lower-bound problem as a sub-obligation. Conversely, the conditional reflection criterion (KNOWN THEOREM: Cook 1975, Kraj\'{\i}\v{c}ek--Pudl\'ak 1989, as specialized in LSPV-1/357) gives $\mathrm{B3}(R_{\rm mix341})$ under the extra hypothesis that some polynomial-time SAT decider has PV/$S^1_2$-provable rejection soundness. That hypothesis is not known to follow from SAT$\in$P.
\end{corollary}
\begin{proof}
Proposition~\ref{prop:exchange} with $R=\mathrm{ER}_2$ embedded in $R'=R_{\rm mix341}$ (unit paid nodes), plus the standard polynomial equivalence of ER and EF (Cook--Reckhow 1979).
\end{proof}

\begin{proposition}[What a successful bridge must look like]
If $\mathrm{B3}(R)$ is proved for $R\supseteq\mathrm{ER}$, the proof must (i) be non-relativizing (Theorem~\ref{thm:rel}, provided LB$(R)$ is the eventual target) and (ii) not require the correctness of \emph{every} polynomial-time decider to be internally provable (the G\"odelian firewall, LSPV-1/357). The weakest currently formulated sufficient condition is
\[
\mathrm{SAT}\in\mathrm P\ \Longrightarrow\ \exists\ \text{p-time }A\ \text{with } \mathrm{PV}\vdash \forall F\,[A(F)=\mathrm{UNSAT}\to \neg\mathrm{Sat}(F)].
\]
Pich--Santhanam (2019, arXiv:1905.12935) note that the \emph{negation} of the full PV-provable statement ``P=NP via $f$'' for every $f$ already follows from P$\neq$NP, and also from EF not being p-bounded. So this condition sits in exactly the same open region. \textbf{OPEN.}
\end{proposition}

\begin{remark}[Algebrization, CANDIDATE]\label{rem:alg}
In the Aaronson--Wigderson convention, take $A=\mathrm{PSPACE}$-complete and $\tilde A$ its multilinear extension. Then $\mathrm{NP}^{\tilde A}\subseteq\mathrm P^{A}$-type collapses hold. Since the conclusion of B3 is oracle-free, the same argument suggests that B3 cannot be proved by algebrizing techniques when LB$(R)$ holds. Aaronson--Wigderson and Impagliazzo--Kabanets--Kolokolova (2009) both warn that algebrization is not closed under modus ponens and is delicate for implications, so this remains a CANDIDATE.
\end{remark}
