\section{Objective and conventions}
$R$ is a fixed propositional refutation system in the Cook--Reckhow sense: a
deterministic polynomial-time verifier $V_R(F,\pi)$ with
$F\in\UNSAT\iff\exists\pi\,V_R(F,\pi)=1$, and $\mathrm{cost}_R(\pi)\ge|\pi|$
(DAG size of the derivation, every reduction step charged). The shared objective is
\[
\Wbar_R(N)=\max_{F\in\UNSAT,\ |\langle F\rangle|\le N}\ \min_{\pi:F\vdash_R\bot}\mathrm{cost}_R(\pi).
\]
The minimum over $\pi$ grants free knowledge of the best reduction sequence. This is a
legitimate convention for a \emph{lower bound on executing} a derivation: search can only
add work. Its consequence is that $\Wbar_R$ is a \emph{nondeterministic} (coNP-type)
measure, so lower bounds on it are statements about proof length, not running time,
until a transfer theorem is supplied.
\begin{definition}
$\mathrm{LB}(R)$: $\Wbar_R(N)\ne N^{O(1)}$.\quad
$\mathrm{Bridge}(R)$: $\SAT\in P\Rightarrow\Wbar_R(N)\le N^{O(1)}$.
\end{definition}
Then $\mathrm{Bridge}(R)\wedge\mathrm{LB}(R)\Rightarrow\SAT\notin P$ is immediate.

\section{Results}
\begin{theorem}[Vacuity and splitting; PROVED/EXACT]\label{thm:vac}
For every $R$: (a) $P\ne NP\Rightarrow\mathrm{Bridge}(R)$;
(b) $\mathrm{Bridge}(R)\wedge\mathrm{LB}(R)\iff (P\ne NP)\wedge\mathrm{LB}(R)$;
(c) $\mathrm{Bridge}(R)\iff [\,P=NP\Rightarrow R\text{ is polynomially bounded}\,]$.
\end{theorem}
\begin{proof}
(a) the antecedent $\SAT\in P$ is false. (b) ``$\Rightarrow$'': the bridge and LB give
$\SAT\notin P$. ``$\Leftarrow$'': by (a). (c) is the definition, since $R$ polynomially
bounded means exactly $\Wbar_R(N)\le N^{O(1)}$.
\end{proof}
\begin{corollary}[Bridge for lower-boundable systems is a reformulation; PROVED/EXACT]
If $\mathrm{LB}(R)$ is a theorem then $\mathrm{Bridge}(R)\iff P\ne NP$. This applies to
resolution and tree-like resolution (Haken 1985; Urquhart 1987; Ben-Sasson--Wigderson 2001),
cutting planes (Pudl\'ak 1997; random CNF: Fleming et al.\ 2017, Hrube\v{s}--Pudl\'ak 2017),
bounded-depth Frege (Ajtai; Beame--Impagliazzo--Pitassi; Kraj\'\i\v{c}ek--Pudl\'ak--Woods),
Polynomial Calculus and Sum-of-Squares (Razborov; Grigoriev; Schoenebeck), and regular and
depth-restricted $\mathrm{Res}(\oplus)$ (Efremenko--Garlik--Itsykson STOC 2024;
Alekseev--Itsykson STOC 2025; Efremenko--Itsykson 2025; Bhattacharya--Chattopadhyay 2025;
Byramji--Impagliazzo 2025).
\end{corollary}
\begin{proof}
By Theorem~\ref{thm:vac}(b) with $\mathrm{LB}(R)$ true: $\mathrm{Bridge}(R)\Rightarrow P\ne NP$,
and $P\ne NP\Rightarrow\mathrm{Bridge}(R)$ by (a).
\end{proof}
\noindent In world-ledger terms (\path{research/agents/claude/world-ledger/}) these
$\mathrm{Bridge}(R)$ rows fall in the category ``equivalent to $P\ne NP$''. So the
weakest-sufficient-bridge strategy gives a genuine split of the problem only for systems $R$ whose
lower bound is \emph{not} yet known. Those are Frege and stronger, together with unrestricted
$\mathrm{Res}(\oplus)$ and $\mathrm{AC}^0[p]$-Frege.

\begin{theorem}[Provable-soundness bridge; KNOWN THEOREM]\label{thm:pv}
Let $A$ be a polynomial-time algorithm and suppose
$\mathrm{PV}\vdash\forall F\,(A(F)=0\to\neg\exists a\,\mathrm{Sat}(F,a))$.
Then there are polynomial-size EF refutations of every $F$ with $A(F)=0$ (Cook 1975; see
Kraj\'\i\v{c}ek, \emph{Bounded Arithmetic, Propositional Logic and Complexity Theory},
Ch.~9; Cook--Nguyen 2010, Ch.~X). More generally, for a sound r.e.\ theory
$T\supseteq\mathrm{PV}$ the system $R_T=\mathrm{EF}+\{\|\mathrm{Refl}_Q\|:T\vdash\mathrm{Refl}_Q\}$
p-simulates $Q_A$ whenever $T$ proves $A$ sound (Kraj\'\i\v{c}ek--Pudl\'ak 1989).
\end{theorem}
\begin{corollary}[PROVED from Thm.~\ref{thm:pv}]
If $\SAT\in P$ via a $T$-provably sound decider, then $\Wbar_{R_T}(N)\le N^{O(1)}$.
Contrapositively, $\mathrm{LB}(\mathrm{EF})$ rules out every $\mathrm{PV}$-provably sound
polynomial-time SAT decider. It does \emph{not} rule out a correct decider whose soundness
is unprovable in $\mathrm{PV}$. That remaining gap is exactly the unrestricted bridge.
\end{corollary}

\begin{lemma}[Matched-theory ladder collapses for weak $R$; PROVED from known theorems]
Weak systems correspond to weak theories: bounded-depth Frege to $V^0$, $\mathrm{AC}^0[2]$-Frege
to $V^0(2)$ (Cook--Nguyen). Their provable-soundness bridges cover only deciders definable in
the matched class ($\mathrm{AC}^0$, $\mathrm{AC}^0[2]$). $\SAT$ lies in neither class,
unconditionally: Majority is in $P$, hence $\mathrm{AC}^0$-reducible to $\SAT$, and
Majority $\notin\mathrm{AC}^0[2]$ (Razborov--Smolensky). So for these $R$ the matched
bridge is trivially true, and $\mathrm{LB}(R)$ yields only circuit lower bounds that are
already known.
\end{lemma}

\begin{theorem}[Pointwise bridge fails for resolution; KNOWN THEOREM + PROVED composition]
There is a family $F_n$ (Tseitin formulas on constant-degree expanders) decided in polynomial
time by GF(2) elimination, with $\min_\pi\mathrm{cost}_{\mathrm{Res}}(\pi)=2^{\Omega(n)}$
(Urquhart 1987; Ben-Sasson--Wigderson 2001). So no statement ``$R$ simulates every
polynomial-time decider instance by instance'' holds for $R=\mathrm{Res}$. Any proposed hard
family has to be checked against $R$ at least as strong as every candidate algorithm.
Tseitin dies to $\mathrm{Res}(\oplus)$ and PC over $\mathbb F_2$; PHP dies to cutting planes
and EF. This is priority 4 of the directive.
\end{theorem}

\section{Barrier-transfer and literature preflight}
Obstacles: (O1) $\mathrm{LB}(R)$ for any $R$ where the bridge is non-vacuous (Frege/EF);
(O2) proving the bridge for deciders whose soundness is unprovable.
\begin{itemize}
\item Feasible interpolation, the main lower-bound engine for $R$ up to CP, \emph{fails} for
EF/Frege under RSA/factoring hardness (Kraj\'\i\v{c}ek--Pudl\'ak 1998; Bonet--Pitassi--Raz 2000).
This is a directly transferable barrier for O1.
\item Optimisation duality (LP/SOS hierarchies as proof systems) is a directly transferable theorem:
SOS-degree lower bounds are $\mathrm{LB}$ results, and by the Corollary they lie in the
reformulation regime.
\item Current frontier for O1 below Frege: $\mathrm{Res}(\oplus)$ lower bounds stop at depth
around $N^{2-\epsilon}$ (2024--2025 cited above). Unrestricted $\mathrm{Res}(\oplus)$, the
first system where the bridge is not known to be vacuous and Tseitin is easy, is the live
frontier.
\item PQ ledger: collision-extraction/special-soundness (item 6) and worst/average reductions
(item 5) are FAILED TRANSFER here. A cryptographic binding or average-case assumption would make
the bridge conditional, and Theorem~\ref{thm:vac} shows the bridge is already vacuous under
$P\ne NP$.
\item Trap found in search tooling: ``$P=NP\Rightarrow$ EF polynomially bounded'' is \emph{not}
a known theorem. Under $P=NP$ only the algorithm-dependent trivial system $Q_A$ is known to be
p-bounded. EF inherits this only via Theorem~\ref{thm:pv}'s provability hypothesis.
\end{itemize}
