\section{Collapse of the bridge for lower-bounded calculi}
\begin{theorem}[PROVED/EXACT]\label{thm:collapse}
Let $R$ be any calculus for which $\overline W_R(N)\ge N^{\omega(1)}$ is a theorem
(e.g.\ resolution, polynomial calculus over any field, cutting planes, $\mathrm{AC}^0$-Frege,
with cost $\ge$ proof length). Then $\textsc{Bridge}(R)\iff \SAT\notin\Pclass$.
\end{theorem}
\begin{proof}
($\Rightarrow$) If $\SAT\in\Pclass$, the bridge gives $\overline W_R\le\mathrm{poly}$, contradicting the
lower bound. ($\Leftarrow$) If $\SAT\notin\Pclass$ the antecedent is false and the implication holds.
\end{proof}
Consequence: for these $R$ the ``weakest sufficient bridge'' is not weaker than the target;
it \emph{is} the target. The multisign DAG hypergraph calculus, if it is
simulated by resolution or by any of the above systems, inherits this collapse. A run that
reports partial progress on $\textsc{Bridge}(R)$ for such $R$ is reporting partial progress on
$\Pclass\neq\NP$ itself and must be audited at that standard.

\begin{corollary}[PROVED/EXACT, standard meta-level sense]\label{cor:rel}
For $R$ as in Theorem~\ref{thm:collapse}, $\textsc{Bridge}(R)$ has no relativizing proof and no
algebrizing proof.
\end{corollary}
\begin{proof}
$\overline W_R$ is oracle-independent. For $O$ PSPACE-complete, $\NP^O=\Pclass^O$ (BGS), so the
relativized bridge $\NP^O\subseteq\Pclass^O\Rightarrow\overline W_R\le\mathrm{poly}$ is false; a relativizing
proof would establish it. Algebrization: Aaronson--Wigderson give $A,\tilde A$ with
$\NP^{\tilde A}\subseteq\Pclass^{A}$; the algebrized bridge fails identically.
\end{proof}
Hence every proposed transfer theorem that treats the SAT algorithm as a black box
(oracle access, input/output behaviour only) is FALSIFIED/RETIRED for these $R$ before any
combinatorics is examined. This is the precise reason the directive's free-optimal-sequence
device cannot by itself reach arbitrary algorithms.

\section{The free-optimal-sequence principle: exact domain}
\begin{proposition}[PROVED/EXACT]\label{prop:free}
Let $\mathcal A$ be any procedure such that every rejecting run on UNSAT $F$ of time $t$
\emph{contains} (can be mapped in $\mathrm{poly}(t)$ to) an $R$-derivation $\pi_F$ with
$\operatorname{cost}_R(\pi_F)\le \mathrm{poly}(t)$. Then $T_{\mathcal A}(N)\ge \overline W_R(N)^{\Omega(1)}$.
Charging no strategy-discovery cost is sound.
\end{proposition}
\begin{proof}
$\operatorname{cost}_R(\pi_F)\ge\min_\pi\operatorname{cost}_R(\pi)$; take the worst $F$.
\end{proof}
KNOWN instances: DPLL $\to$ tree resolution; CDCL with restarts $\to$ resolution
(Beame--Kautz--Sabharwal 2004; Pipatsrisawat--Darwiche 2011 for the converse simulation);
Gr\"obner/linear-algebra solvers $\to$ PC/Nullstellensatz. The hypothesis ``every polynomial-time
SAT algorithm admits such an extraction'' is \emph{exactly} $\textsc{Bridge}(R)$ in the
pointwise form; by Theorem~\ref{thm:collapse} it is OPEN and equivalent-strength to the target.

\section{The strong-calculus case: reduction to feasible UNSAT soundness}
\begin{theorem}[KNOWN THEOREM, assembled]\label{thm:ef}
Suppose $\SAT\in\Pclass$ via a deterministic algorithm $A$ such that
$\mathrm{PV}\vdash\forall F\,\forall x\;[A(F)=0\rightarrow\neg\mathrm{Sat}(F,x)]$.
Then $\overline W_{\mathrm{EF}}(N)\le\mathrm{poly}(N)$.
\end{theorem}
\begin{proof}[Sketch]
Cook's translation yields polynomial-size, uniformly generated EF proofs of
$\|\mathrm{Sound}_A\|_n$. For UNSAT $F$ the computation $A(F)=0$ is a fixed polynomial-length
tableau; EF verifies its constant instance in polynomial size, substitutes $F$ into
$\|\mathrm{Sound}_A\|_n$ and derives $\neg\mathrm{Sat}(F,x)$, i.e.\ refutes $F$.
\end{proof}

\begin{proposition}[PROVED/EXACT; SAT/UNSAT asymmetry]\label{prop:asym}
From any polynomial-time SAT decider $A$ one obtains a polynomial-time decider $A'$ whose
\emph{SAT} answers are PV-provably sound: $A'$ runs the downward self-reduction with $A$
as oracle and outputs $1$ only after checking the found assignment.
\end{proposition}
\begin{proof}
$A'(F)=1$ only if a checked $x$ with $\mathrm{Sat}(F,x)$ is produced; this is a PV-provable
property of the verifier, independent of $A$'s correctness. Correctness of $A$ makes $A'$
total and correct.
\end{proof}
No analogous transformation is known for UNSAT answers, since a 0-answer supplies no
witness. Therefore
\[
\textsc{Bridge}(\mathrm{EF})\ \Longleftarrow\
\bigl[\SAT\in\Pclass\Rightarrow\exists A\in\Pclass\text{ deciding }\SAT\text{ with PV-provable UNSAT soundness}\bigr],
\]
and the bracket (``feasible $\Pclass=\NP$ for the coNP side'') is \textbf{OPEN}. Even if
$\textsc{Bridge}(\mathrm{EF})$ were proved, a superpolynomial EF lower bound would still be
required, which is itself OPEN and subject to the Pich--Santhanam and
Khaniki--Pich--Sokolov difficulty results. World-ledger classification: the EF route
splits the target into two OPEN pieces, neither known to be easier than $\Pclass\neq\NP$.
