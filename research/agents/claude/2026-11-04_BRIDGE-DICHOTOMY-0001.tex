\noindent\textbf{Status.} Elementary logical and standard-literature analysis; no new lower bound. Labels: KNOWN THEOREM, PROVED/EXACT (logic), OPEN. \textbf{P versus NP remains OPEN.}

\section*{Setting}
$R$ is a Cook--Reckhow proof system (polynomial-time verifier). Define
$\Wb_R(N)=\max_{F\in\mathrm{UNSAT},|F|\le N}\min_{\pi:F\vdash_R\bot}\mathrm{cost}_R(\pi)$ with $\mathrm{cost}_R$ proof size (the free-optimal-strategy convention; the minimum charges nothing for finding $\pi$, which only strengthens lower bounds). Bridge:
\[ B3_R:\quad \SAT\in P\ \Longrightarrow\ \Wb_R(N)\le \mathrm{poly}(N). \]

\begin{proposition}[Circularity, PROVED/EXACT]\label{p1}
If $\Wb_R$ is superpolynomial \emph{as a known theorem} (e.g.\ $R$=Resolution, bounded-depth Frege, Cutting Planes, depth-limited $\mathrm{Res}(\oplus)$), then $B3_R\iff \SAT\notin P$. Hence any proof of $B3_R$ for such $R$ is already a proof of $P\neq NP$; it is not a weaker sufficient step.
\end{proposition}
\begin{proof} $B3_R$ is $(\SAT\in P)\Rightarrow(\Wb_R\ \mathrm{poly})$ and the consequent is false, so $B3_R$ is equivalent to $\neg(\SAT\in P)$. \qed\end{proof}

\begin{remark} Consequence for the project: the exact lower bounds already proved in weak calculi (Resolution-based $R_{\mathrm{mix341}}$ etc.) cannot be combined with a ``bridge'' without circularity. A noncircular B3 must use a feature of $R$ that Resolution lacks, namely the ability to certify the correctness of an arbitrary polynomial-time decider.\end{remark}

\begin{proposition}[Cook--Krajicek translation, KNOWN THEOREM]\label{p2}
Suppose $A$ is a polynomial-time SAT decider and the theory $PV$ proves its soundness $\forall F,\tau\,(A(F)=0\to F(\tau)=0)$. Then Extended Frege has polynomial-size refutations of every unsatisfiable $F$; i.e.\ $B3_{EF}$ holds along such $A$.
\end{proposition}
\emph{Sketch.} Translate the $PV$ proof into poly-size EF proofs of the propositional soundness formulas $\|A(F)=0\to\neg F(\tau)\|_n$ (Cook 1975); evaluate $A$ on the concrete $F$ in polynomial size to obtain $A(F)=0$; cut. The missing ingredient for a fixed $R$ and arbitrary unknown $A$ is exactly the provability hypothesis: $A$ is a parameter of the existence claim, not an input of $R$.

\begin{proposition}[Dichotomy table, synthesis]
\begin{center}\begin{tabular}{p{3.3cm}p{4.2cm}p{5.5cm}}
Regime of $R$ & Lower bound $\Wb_R$ & Status of $B3_R$\\\hline
Weak ($\mathrm{Res}$, $AC^0$-Frege, CP, depth-limited $\mathrm{Res}(\oplus)$) & KNOWN superpoly & $\iff P\neq NP$ (Prop.~\ref{p1}); no help\\
Intermediate (dag-like $\mathrm{Res}(\oplus)$, $AC^0[p]$-Frege) & OPEN & no known route; plausible but unproved\\
Strong ($\supseteq$ EF) & OPEN; superpoly would give $NP\ne coNP$ (Cook--Reckhow) & holds given $PV$-provable soundness (Prop.~\ref{p2}); unconditional form tied to optimal proof systems (Krajicek--Pudlak)
\end{tabular}\end{center}
\end{proposition}
\noindent So the program must either (i) find intermediate/strong $R$ where a lower bound is a real open target and $B3_R$ is plausibly provable, or (ii) accept that a Resolution-type lower bound plus transfer is the P$\neq$NP problem itself. Direct hard families (signed hypergraph, Tseitin, PHP) test (i) only through $R$ strictly stronger than the calculi where the bounds are already known.
