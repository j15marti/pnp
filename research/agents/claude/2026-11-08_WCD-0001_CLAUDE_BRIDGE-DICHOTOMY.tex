\noindent\textbf{Status.} P versus NP remains OPEN. Nothing below is a lower bound.

\section*{Setup}
$R$ is a Cook--Reckhow system (derivations checkable in time $\mathrm{poly}$ in $|\pi|+|F|$). Let $W_R(N)=\max_{|F|\le N,F\in\mathrm{UNSAT}}\min_\pi |\pi|$; the minimum is free (strategy search uncharged). Bridge: $B_R:\ \mathrm{SAT}\in\mathrm P\Rightarrow W_R\le\mathrm{poly}$.

\section*{Results}
\begin{thm}[PROVED/EXACT: bridge $=$ goal when a lower bound is known]
If $W_R$ is unconditionally superpolynomial, then $B_R\iff \mathrm{SAT}\notin\mathrm P$.
\end{thm}
\begin{proof} The consequent is false, so the implication holds iff the antecedent is false. \end{proof}
\emph{Consequence.} For resolution, polynomial calculus, cutting planes, bounded-depth Frege, a proof of $B_R$ is a proof of $\mathrm P\ne\mathrm{NP}$. Concretely, pigeonhole formulas are decided in polynomial time by counting yet need exponential resolution size (Haken), so $W_R$ is not a lower bound on algorithmic work for weak $R$.

\begin{thm}[KNOWN THEOREM, Cook--Reckhow]
$W_R\le\mathrm{poly}$ for some $R$ iff $\mathrm{NP}=\mathrm{coNP}$. Hence $B_R$ with $W_R$ superpolynomial for \emph{all} $R$ is not what is needed; we need one fixed $R$.
\end{thm}
\begin{thm}[KNOWN THEOREM, Cook translation; CONDITIONAL use]
If a polynomial algorithm $M$ for SAT has $\mathrm{PV}_1\vdash$ ``$M$ is sound on UNSAT'', then extended Frege has polynomial-size proofs of all UNSAT $F$ (via the translated soundness statements). So $B_{\mathrm{EF}}$ holds under hypothesis $H_{\rm prov}$: some polynomial SAT algorithm has provable soundness.
\end{thm}
\begin{lem}[PROVED/EXACT: asymmetry]
Levin-style universal search is provably correct on the satisfiable side (outputs are checked). The unsatisfiable side relies on a time bound whose correctness is exactly the unprovable part; the gap is the same side where derivations are needed.
\end{lem}
\begin{thm}[Dichotomy for the multisign DAG hypergraph system $H$; CONDITIONAL on $H$ being a Cook--Reckhow system]
Either (i) $H$ is p-simulated by a system with a known superpolynomial bound, whence $B_H\iff \mathrm{SAT}\notin\mathrm P$ by Theorem 1; or (ii) $H$ p-simulates EF, whence a superpolynomial $W_H$ implies an EF lower bound, a major open problem; or (iii) $H$ lies strictly between (Frege-like, restricted extension), and both bridge and lower bound are open.
\end{thm}
\noindent Case (iii) is where progress is possible: it needs a system strong enough that $B_H$ is derivable from a hypothesis weaker than $H_{\rm prov}$, and weak enough that a lower-bound method exists. Known methods stall: feasible interpolation fails for Frege-type systems under standard assumptions, and EF lower bounds are not provable in $\mathrm{PV}_1$ by model-theoretic means (literature review; not independently verified).

\section*{Red-team notes}
\begin{enumerate}
\item ``Minimal reduced logical state'' is undefined until a normal form is fixed; different reductions can give incomparable states. No claim here depends on it.
\item Free strategy discovery makes $W_R$ a nondeterministic measure; it cannot exceed what a verifier-based argument already gives, so no new power arises from granting it.
\item Time--space: peak memory of a DAG derivation is a separate measure (pebbling); not audited this run.
\end{enumerate}
