\noindent\textbf{Status.} \textbf{P versus NP remains OPEN.} This report proves two
classification statements about the shared objective
\[
\Wbar_R(N)=\max_{\substack{F\in\mathrm{UNSAT}\\ |\langle F\rangle|\le N}}\ \min_{\pi:F\vdash_R\bot}\operatorname{cost}_R(\pi)
\]
and about the weakest sufficient bridge
$\SAT\in\mathrm P\Rightarrow \Wbar_R(N)\le\mathrm{poly}(N)$.
Theorem~\ref{thm:mixEF} is \status{CONDITIONAL} only on the native $\Rmix$ grammar
consisting of the documented rule types (open item [P2-D1]); Theorems~\ref{thm:ladder} and
\ref{thm:bridgeEF} are \status{PROVED/EXACT} modulo cited \status{KNOWN THEOREM}s. No lower bound
is proved. No schedule, role, allocation, ownership region, or root \texttt{main.tex} text is changed.

\section*{Bottom line}
\begin{enumerate}
\item Granting the optimal derivation for free is legitimate: $\min_\pi\operatorname{cost}_R(\pi)$ is
  minimum proof size, and $\Wbar_R$ is the worst-case proof-size function of $R$
  (Cook--Reckhow). Nothing in this report disputes that step.
\item For every fixed $R$ with a \emph{known} superpolynomial lower bound (Resolution,
  Cutting Planes, bounded-depth Frege, Polynomial Calculus), the bridge is \emph{literally equivalent}
  to $\SAT\notin\mathrm P$. For such $R$ the ``weakest bridge'' is not weaker than the whole problem.
\item The project's fixed calculus $\Rmix$ (RES + AND3 + GF(2) XOR + LOCAL-CONVERT + paid two-input DEF)
  is polynomially equivalent to Extended Resolution, hence to Extended Frege (Theorem~\ref{thm:mixEF}).
  So the target ``$\Wbar_{\Rmix}$ is superpolynomial'' [P2-B1] is the
  \emph{Extended Frege lower-bound problem}, one of the central open problems of proof complexity.
\item For $\EF$ (hence $\Rmix$) the bridge is equivalent to
  $\mathrm{P}=\mathrm{NP}\Rightarrow\EF$ is p-bounded, and to a
  \emph{certification upgrade}: if SAT is in P, then some polynomial-time SAT algorithm has
  polynomial-size $\EF$ proofs of its own soundness (Theorem~\ref{thm:bridgeEF}).
  This open question is the precise missing link [P2-B2/B3].
\end{enumerate}
Thus the full plan for the project calculus is equivalent to the conjunction
\[
\boxed{\ \EF\ \text{is not p-bounded}\quad\wedge\quad\bigl(\mathrm P=\mathrm{NP}\Rightarrow \EF\ \text{is p-bounded}\bigr).\ }
\]
Both conjuncts are \status{OPEN}.

\section{Model and quantifiers}
A Cook--Reckhow system $R$ for UNSAT CNFs is a polynomial-time relation $V_R(F,\pi)$
that is sound (accepted $F$ are unsatisfiable) and complete. $\operatorname{cost}_R$ is a cost
polynomially related to encoded length $|\pi|$. For $\Rmix$ this holds: a derived node costs one and
has size at most $O(N+\#\text{nodes})$ bits (literals over source plus defined variables, or an explicit
GF(2) row), so $\#\text{nodes}\le|\pi|\le\mathrm{poly}(N+\#\text{nodes})$.
Source clauses are free leaves; distinct shared DAG nodes count once. $R$ is \emph{p-bounded} if
$\Wbar_R(N)\le N^{O(1)}$. $R'\ \ge_p R$ means $R'$ p-simulates $R$ on UNSAT CNFs.
Write $\mathrm{Br}(R)$ for the statement $\SAT\in\mathrm P\Rightarrow R$ is p-bounded,
and $\mathrm{LB}(R)$ for ``$\Wbar_R$ is superpolynomial''.

\section{The bridge ladder}
\begin{theorem}[Ladder; \status{PROVED/EXACT}]\label{thm:ladder}
\begin{enumerate}
\item[(a)] If $R'\ge_p R$, then $\mathrm{Br}(R)\Rightarrow\mathrm{Br}(R')$ and $\mathrm{LB}(R')\Rightarrow\mathrm{LB}(R)$.
\item[(b)] For every $R$: $\mathrm{Br}(R)\wedge\mathrm{LB}(R)\Rightarrow\SAT\notin\mathrm P$, and
  $\SAT\notin\mathrm P\Rightarrow\mathrm{Br}(R)$.
\item[(c)] If $\mathrm{LB}(R)$ is a theorem, then $\mathrm{Br}(R)\Leftrightarrow\SAT\notin\mathrm P$.
  This applies to Resolution [Haken], Cutting Planes [Pudl\'ak 1997], bounded-depth Frege
  [Ajtai; Kraj\'i\v{c}ek--Pudl\'ak--Woods; Pitassi--Beame--Impagliazzo], and Polynomial Calculus
  in characteristic $\ne 2$ on Tseitin formulas [Buss--Grigoriev--Impagliazzo--Pitassi].
\end{enumerate}
\end{theorem}
\begin{proof}
(a) Simulation transfers polynomial upper bounds up and superpolynomial lower bounds down.
(b) The first is modus tollens; the second holds because the implication has a false premise.
(c) Combine (b) with $\mathrm{LB}(R)$.
\end{proof}
\begin{remark}
Strengthening $R$ makes the bridge easier and the lower bound harder. At the bottom
(Resolution) the lower bound is known and the bridge \emph{is} P$\ne$NP. At the top, for the
algorithm-dependent clocked system $Q_A$ of LSPV-1/336 the bridge is trivial, but $Q_A$ depends on the
unknown algorithm, so this gives no fixed system. The shared objective therefore has value only for
an $R$ whose lower bound is open and whose bridge is believed provable. Theorem~\ref{thm:mixEF} shows
that the project's $R$ sits exactly at $\EF$.
\end{remark}

\section{The project calculus is Extended Frege}
\begin{theorem}[$\Rmix\equiv_p\ER\equiv_p\EF$; \status{CONDITIONAL} on {[P2-D1]}]\label{thm:mixEF}
Assume the native $\Rmix$ grammar consists of the documented rule types:
binary RES; typed AND3 on at most three variables; binary GF(2) row XOR; LOCAL-CONVERT between a
bounded-arity local constraint and its clauses or rows; and paid conservative signed two-input DEF
$y\leftrightarrow(\ell_1\circ\ell_2)$ with fresh $y$. Assume also that each rule instance is checkable in
polynomial time and is locally sound by a fixed finite truth-table argument. Then
$\Rmix\equiv_p\ER\equiv_p\EF$ on UNSAT CNFs, with node cost. Consequently
$\mathrm{LB}(\Rmix)\Leftrightarrow\mathrm{LB}(\EF)$ and $\mathrm{Br}(\Rmix)\Leftrightarrow\mathrm{Br}(\EF)$.
\end{theorem}
\begin{proof}
($\Rmix\ge_p\ER$.) RES+DEF is the fragment $E=\mathrm{ER2}$ already recorded in LSPV-1/376.
Extended Resolution with arbitrary extension definitions is p-simulated by two-input definitions:
introduce each extension formula gate by gate, with signed literals giving $\wedge,\vee$ and negation.
The defining clauses of each two-input gate are the DEF facets. This adds $O(1)$ nodes per gate.
$\ER\equiv_p\EF$ is a \status{KNOWN THEOREM} (Cook--Reckhow 1979; Kraj\'i\v{c}ek's 1995 book, Ch.~4).

($\EF\ge_p\Rmix$.) Cook (1975) shows $\EF$ p-simulates every proof system whose soundness is
provable in $\PV$; see also Kraj\'i\v{c}ek--Pudl\'ak 1990, \status{KNOWN THEOREM}.
Soundness of $\Rmix$ is the statement ``every node is true under any assignment satisfying the
source and the definitions.'' It is proved in $\PV$ by induction on the DAG order, which is
polynomial induction on node index. Each step is a fixed local soundness check:
RES, AND3, LOCAL-CONVERT and DEF have bounded arity; an XOR step is $\mathbb F_2$ addition of two
explicit rows, whose soundness is the identity $\langle a,z\rangle+\langle b,z\rangle=\langle a+b,z\rangle$,
a $\PV$-provable bit-parity identity. Fresh defined variables are handled by extending any
satisfying assignment by the definitions. Hence $\EF$ p-simulates $\Rmix$.

Costs: node count and encoded length are polynomially related, as noted in Section~1.
The two consequences follow from Theorem~\ref{thm:ladder}(a) in both directions.
\end{proof}
\begin{remark}[Consistency with the corpus hostile controls]
Every hard family retired in the corpus for $\Rmix$ is easy for $\EF$: pigeonhole by Cook's ER proof,
Tseitin by GF(2) rows, and affine, 2-SAT and Horn cases. This is consistent with
Theorem~\ref{thm:mixEF}: no family is currently known to require superpolynomial $\EF$ proofs.
Feasible interpolation, the main lower-bound engine for CP and bounded-depth systems, provably fails
for $\EF$ under standard cryptographic assumptions (Kraj\'i\v{c}ek--Pudl\'ak 1998, RSA). This is a
\emph{barrier}, not an assumption of any proof here. Monotone-interpolation or width methods
cannot reach $\Wbar_{\Rmix}$ unless that barrier is avoided.
\end{remark}

\section{The bridge for EF is a certification upgrade}
\begin{definition}
For a deterministic algorithm $A$ with polynomial clock $p$, let $\Ref_A^n(x,y,z)$ be the
polynomial-size propositional formula stating: if $y$ is a valid $p(n)$-step computation tableau of $A$ on
the length-$n$ CNF code $x$ ending in output UNSAT, then the assignment $z$ falsifies some clause of $x$.
Say $A$ is \emph{$\EF$-certified} if $\Ref_A^n$ has $\EF$ proofs of size $n^{O(1)}$.
\end{definition}
\begin{theorem}[\status{PROVED/EXACT} from cited \status{KNOWN THEOREM}s]\label{thm:bridgeEF}
The following are equivalent.
\begin{enumerate}
\item[(i)] $\mathrm{Br}(\EF)$: if $\SAT\in\mathrm P$ then $\Wbar_\EF(N)\le\mathrm{poly}(N)$.
\item[(ii)] Certification upgrade: if $\SAT\in\mathrm P$, then some correct polynomial-time SAT decider is $\EF$-certified.
\end{enumerate}
By Theorem~\ref{thm:mixEF}, under [P2-D1] the same holds for $\mathrm{Br}(\Rmix)$.
\end{theorem}
\begin{proof}
(ii)$\Rightarrow$(i). Let $A$ be correct and $\EF$-certified, and let $F$ be UNSAT of length $n$.
Then $A(F)=\mathrm{UNSAT}$ with an explicit tableau $\tau$ of size $\mathrm{poly}(n)$.
Substituting the constants $F,\tau$ for $x,y$ in the certified proof of $\Ref_A^n$ keeps it an $\EF$ proof,
since extension axioms remain instances after substitution, and the size does not grow.
$\EF$ verifies the closed hypothesis ``$\tau$ is a valid tableau of $A$ on $F$ with output UNSAT'' in
$\mathrm{poly}(n)$ lines by evaluating a variable-free formula. Modus ponens yields the
statement that $z$ falsifies a clause of $F$, i.e.\ $\neg F(z)$. This gives a polynomial-size refutation.

(i)$\Rightarrow$(ii). Assume $\SAT\in\mathrm P$. By (i), every UNSAT $F$ has an $\EF$ refutation of size
$\le r(|F|)$ for a polynomial $r$. Because $\mathrm P=\mathrm{NP}$ and NP search problems
self-reduce to their decision versions, a polynomial-time procedure finds either a satisfying assignment
or an $\EF$ refutation of size $\le r(|F|)$. Let $A'$ run it, check the object found with the fixed
assignment checker or the fixed $\EF$ checker, and output UNSAT only if the $\EF$ checker accepts.
$A'$ is correct, total and polynomial-time. Its soundness formula splits into two parts.
First, a valid tableau ending in UNSAT contains an accepted check of some $\pi$ against $x$. This is
a fixed property of the program text, provable in $\PV$, so it has polynomial-size $\EF$ proofs by
Cook's translation. Second, an $\EF$ refutation of $x$ accepted by the checker excludes a satisfying
$z$. This is the reflection principle for $\EF$, which has polynomial-size $\EF$ proofs:
$S^1_2$ proves $\mathrm{RFN}(\EF)$, and Cook/Buss translation applies
(Kraj\'i\v{c}ek--Pudl\'ak 1990; Kraj\'i\v{c}ek's 1995 book, Ch.~9), \status{KNOWN THEOREM}. Cutting on
the intermediate statement gives an $\EF$ proof of $\Ref_{A'}^n$ of size $n^{O(1)}$.
\end{proof}
\begin{corollary}[\status{PROVED/EXACT}]
$\mathrm{Br}(\EF)\Leftrightarrow\bigl(\mathrm P=\mathrm{NP}\Rightarrow\EF\text{ is p-bounded}\bigr)$, and
$\mathrm{LB}(\EF)$ implies that no polynomial-time SAT decider is $\EF$-certified.
The latter consequence is known in the $\PV$-provability form (Cook--Kraj\'i\v{c}ek 2007).
\end{corollary}

LSPV-1/360 gave the TQBF-oracle non-relativization requirement. The codex 2026-11-11 reports

\end{remark}

\section{Red-team audit of the shared objective}
\begin{description}
\item[Free optimal sequence.] Correct and harmless: $w_R(F)$ is minimum proof size. Search cost is irrelevant to a lower bound on $\Wbar_R$.
\item[Pointwise transfer.] Already \status{FALSIFIED/RETIRED} in the corpus for Resolution: Tseitin
  with Gaussian elimination (2026-11-01 redirect). That retirement does not apply to $\EF$, where
  Tseitin is easy.
\item[Relativizing bridge.] \status{FALSIFIED/RETIRED} (LSPV-1/360): relative to TQBF,
  $\SAT\in\mathrm P^{\mathrm{TQBF}}$ while $\Wbar_{\mathrm{Res}}$ is exponential. For $\EF$, a
  relativizing proof of $\mathrm{Br}(\EF)$ would prove $\EF$ p-bounded unconditionally,
  which gives $\mathrm{NP}=\mathrm{coNP}$ and destroys the plan. Any valid $\mathrm{Br}$ proof must use
  the \emph{code} of the algorithm. Certification upgrade is exactly such a code-dependent statement:
  it is ill-typed for oracle machines, because $\Ref$ of an oracle machine is not a tautology.
\item[Adaptive strategy switching / shared DAGs.] Already absorbed. $\EF$ allows arbitrary reuse
  and switching between algebraic and clausal reasoning through extension variables. A lower bound on
  $\Wbar_{\Rmix}$ is therefore automatically robust to them; the price is that it is an $\EF$ lower bound.
\item[Stale/overstated claims found.] None newly found. Corpus documents treating $\Rmix$
  hardness as a combinatorial question \emph{separate} from Extended Frege should be read with
  Theorem~\ref{thm:mixEF} in mind. Every claimed superpolynomial $\Rmix$ minimum would be a
  superpolynomial $\EF$ lower bound and should be audited at that standard.
\end{description}

\section{Literature check (this run)}
\begin{itemize}
\item Fleming, Grosser, Pitassi, Robere, \emph{Provable Reductions in TFNP}, ECCC TR26-109 (2026),
  \url{https://eccc.weizmann.ac.il/report/2026/109/}. This is new to the corpus. It defines the proof
  systems $\langle\EF,R\rangle$: a polynomial-time reduction to a TFNP problem $R$ plus an $\EF$ proof
  of correctness. If $\EF$ proves $R\in\mathrm{FP}$, then $\langle\EF,R\rangle\equiv\EF$; this is the
  proof-system form of (ii)$\Rightarrow$(i). For strong systems $P$ there is a polynomial-time $R_P$
  with $\langle\EF,R_P\rangle\equiv P$. Clue: systems beyond $\EF$ differ from $\EF$ \emph{only} by
  unprovable correctness of a polynomial-time search. That is the same gap as the certification
  upgrade. \emph{Classification:} DIRECTLY TRANSFERABLE THEOREM for the reformulation;
  it gives no lower bound.
\item Egidy--Gla\ss er, \emph{Optimal Proof Systems for Complex Sets are Hard to Find}, STOC 2025,
  arXiv:2408.07408. Relativized evidence that the existence of (p-)optimal proof systems resists
  relativizing proofs. This is consistent with the need for a code-dependent bridge.
  \emph{Classification:} USEFUL STRUCTURAL ANALOGY, no direct transfer.
\item Kraj\'i\v{c}ek--Pudl\'ak, JSL 54 (1989): $\mathrm E=\mathrm{NE}$ implies a p-optimal system exists.
  Since P$=$NP implies E$=$NE by padding, P$=$NP gives \emph{some} p-bounded p-optimal system, but not
  a known fixed one. This is why it does not prove $\mathrm{Br}(\EF)$. \emph{Classification:} FAILED
  TRANSFER for the fixed-$R$ bridge.
\item Pich--Santhanam, arXiv:2312.08163 / JACM 2026, already in the corpus: $\EF$ lower bounds plus
  witnessing hypotheses give P$\ne$NP. This is the most developed published attack on exactly the
  second conjunct.
\item Cook 1975 (STOC); Cook--Kraj\'i\v{c}ek, JSL 72 (2007); Kraj\'i\v{c}ek--Pudl\'ak, Z.\ Math.\ Logik 36 (1990);
  Kraj\'i\v{c}ek--Pudl\'ak, Inf.\ Comput.\ 140 (1998); Haken TCS 39 (1985); Pudl\'ak JSL 62 (1997).
\end{itemize}
