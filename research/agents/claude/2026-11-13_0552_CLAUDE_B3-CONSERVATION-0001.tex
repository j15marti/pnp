\section{Shared target (restated, quantifiers exact)}
Fix a sound, complete, polynomial-time-checkable refutation system $R$ with
nonnegative node weights, fixed independently of inputs. For UNSAT CNF $F$,
$W_R(F)=\min_{\pi:F\vdash_R\bot}\mathrm{cost}_R(\pi)$ (the optimal derivation is
granted for free; search cost is not charged), and
\[
\overline W_R(N)=\max_{F\in\mathrm{UNSAT},\ |\langle F\rangle|\le N} W_R(F)
\quad(\max\emptyset:=0).
\]
Write $\mathrm{PB}(R)$ for $\exists C,k\ \forall N:\ \overline W_R(N)\le C(N+1)^k$,
$\mathrm{LB}(R)$ for $\forall C,k,N_0\ \exists N\ge N_0:\ \overline W_R(N)>C(N+1)^k$,
and $\mathrm{B3}(R)$ for $\mathrm{SAT}\in\mathrm P\Rightarrow \mathrm{PB}(R)$.

\section{Results}
\begin{lemma}[Quantifier negation; PROVED/EXACT]\label{l0}
$\mathrm{LB}(R)\iff\neg\mathrm{PB}(R)$.
\end{lemma}
\begin{proof}
($\Rightarrow$) immediate with $N_0=0$. ($\Leftarrow$) Given $C,k,N_0$, put
$C'=\max\{C,\max_{N<N_0}\overline W_R(N)/(N+1)^k\}$ (finite since each
$\overline W_R(N)$ is finite). By $\neg\mathrm{PB}$ some $N$ has
$\overline W_R(N)>C'(N+1)^k$; necessarily $N\ge N_0$ and the inequality holds for $C\le C'$.
\end{proof}

\begin{lemma}[Disjunctive normal form of B3; PROVED/EXACT]\label{l1}
For every fixed $R$:\quad $\mathrm{B3}(R)\iff \mathrm{SAT}\notin\mathrm P\ \vee\ \mathrm{PB}(R)$,
equivalently (Lemma~\ref{l0}) $\mathrm{B3}(R)\iff(\mathrm{LB}(R)\Rightarrow\mathrm{SAT}\notin\mathrm P)$.
\end{lemma}
\noindent Propositional; no property of $R$ is used.

\begin{corollary}[Difficulty conservation; PROVED/EXACT given cited KNOWN THEOREMS]\label{c1}
(i) If $\mathrm{LB}(R)$ is proved, then $\mathrm{B3}(R)$ is provably \emph{equivalent} to
$\mathrm{SAT}\notin\mathrm P$; proving B3 for such $R$ \emph{is} proving P$\ne$NP.
(ii) Any proof of $\mathrm{SAT}\notin\mathrm P$ proves $\mathrm{B3}(R)$ for \emph{every} $R$.
(iii) Route B closure $\mathrm{LB}(R)\wedge\mathrm{B3}(R)$ is equivalent to
$\mathrm{LB}(R)\wedge\mathrm{SAT}\notin\mathrm P$.\\
KNOWN THEOREM instances of (i), with unit node weights and the systems' usual
encodings (polynomially related to $|\langle F\rangle|$): ordinary DAG Resolution
(Haken 1985, PHP; Urquhart 1987, Tseitin), bounded-depth Frege
(Ajtai 1988; Kraj\'\i\v{c}ek--Pudl\'ak--Woods 1995; Pitassi--Beame--Impagliazzo 1993),
Cutting Planes (Pudl\'ak 1997), Polynomial Calculus over any field
(Razborov 1998; Impagliazzo--Pudl\'ak--Sgall 1999). For each, ``B3'' is a
\emph{reformulation} of P$\ne$NP in the sense of the world ledger
(\texttt{research/agents/claude/world-ledger/}), not progress.
\end{corollary}

\begin{lemma}[Anti-monotone strength trade-off; PROVED/EXACT]\label{l3}
Say $R\le_W R'$ if there is a polynomial $q$ with $W_{R'}(F)\le q(|\langle F\rangle|+W_R(F))$
for all UNSAT $F$. Then $\mathrm{PB}(R)\Rightarrow\mathrm{PB}(R')$, hence
$\mathrm{B3}(R)\Rightarrow\mathrm{B3}(R')$ and $\mathrm{LB}(R')\Rightarrow\mathrm{LB}(R)$.
\end{lemma}
\begin{proof}
$\overline W_{R'}(N)\le q(N+C(N+1)^k)$, a polynomial (take $q$ monotone). The rest is Lemma~\ref{l1}/\ref{l0}.
\end{proof}
\noindent Consequence: strengthening $R$ makes the bridge easier and the lower bound
harder; weakening does the reverse. By Corollary~\ref{c1}(i) the only systems for
which B3 could carry \emph{less} than the full difficulty of P$\ne$NP are systems
for which no superpolynomial lower bound is currently known.

\begin{lemma}[Strong-system bridge via provable correctness; CONDITIONAL, KNOWN THEOREM application]\label{l4}
Let $G$ be the statement: \emph{if $\mathrm{SAT}\in\mathrm P$, then some polynomial-time $A$
deciding UNSAT has soundness $\forall F,\alpha\,(A(F){=}\mathrm{UNSAT}\to\alpha\not\models F)$
provable in Cook's theory PV.} Then $G\Rightarrow\mathrm{B3}(R)$ for every $R$ with
$\mathrm{ER}\le_W R$ (ER = extended resolution, p-equivalent to Extended Frege).
\end{lemma}
\begin{proof}[Proof sketch from known theorems]
Cook (1975): every PV-provable $\forall$-statement has polynomial-size, polynomial-time
constructible ER/EF proofs of its propositional translations. Instantiate the
translated soundness tautology at the bits of a fixed $F$ with $A(F)=\mathrm{UNSAT}$,
verify the computation of $A$ on $F$ inside EF in size polynomial in $|F|+T_A(F)$, and
conclude a refutation of $F$. This gives B1 for EF with respect to $A$, hence
$\mathrm{PB}(\mathrm{ER})$, and Lemma~\ref{l3} lifts it to $R$.
\end{proof}
\noindent $G$ is \textbf{OPEN}. Literature (search log below) records that
``PV proves P=NP'' is equivalent to PV defining a polynomial algorithm that outputs a
satisfying assignment or an EF refutation, and that the conjecture ``no
polynomial-time $f$ is PV-provably a SAT algorithm'' follows from P$\ne$NP and
also from EF being not polynomially bounded. So $G$ is not a theorem we can cite;
it isolates the remaining content of B3 for strong $R$ as a \emph{provability-of-correctness}
gap rather than a proof-size gap. LB(ER) is itself a decades-old open problem.

\begin{corollary}[Trilemma for the historical $R_{\mathrm{mix341}}$; CANDIDATE classification]
Relative to current knowledge, one of the following describes $R_{\mathrm{mix341}}$ once its byte grammar
[P2-D1] is fixed: (a) $\mathrm{ER}\le_W R_{\mathrm{mix341}}$ (e.g.\ if its charged
auxiliary gates are unrestricted extension definitions): then LB is at least as
hard as superpolynomial EF lower bounds, while B3 reduces to $G$ (Lemma~\ref{l4});
(b) some known superpolynomial lower bound applies to $R_{\mathrm{mix341}}$: then
B3 is exactly P$\ne$NP (Corollary~\ref{c1}); (c) neither: then both arrows are open
and B3 must be proved by a $R_{\mathrm{mix341}}$-specific translation that is
non-relativizing ([P2-L5-296]) and that does not pass through $G$.
\end{corollary}
\noindent Note on (b)/(c): $R_{\mathrm{mix341}}$ contains resolution plus binary
GF(2) row addition, resembling Res($\oplus$)/Res-Lin. Superpolynomial lower bounds for
\emph{unrestricted} Res($\oplus$) remained open as of late 2025 (only tree-like,
regular and depth-restricted versions are known), so (b) is not currently
established even if a Res($\oplus$) simulation were proved.

\section{Falsifiers proposed}
\begin{enumerate}
\item A ZFC proof of $\mathrm{B3}(\mathrm{Res})$ that does not prove $\mathrm{SAT}\notin\mathrm P$: impossible by Corollary~\ref{c1}; any claimed one hides a P$\ne$NP proof or an error.
\item A claimed bridge that uses the decider only as a black box (input/output and time): fails for every $R$ with LB proved, by the BGS oracle argument [P2-L5-296].
\item A claimed proof of $G$ from SAT$\in$P alone: would yield PB(EF) under SAT$\in$P; check against the literature equivalence for ``PV proves P=NP''.
\end{enumerate}
