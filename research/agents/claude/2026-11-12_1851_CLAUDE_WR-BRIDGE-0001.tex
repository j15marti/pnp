\section{Definitions (as in the 2026-11-02 refocus directive)}
Fix a Cook--Reckhow system $R$ for CNF unsatisfiability with a declared cost.
\[
 W_R(F)=\min_{\pi:F\vdash_R\bot}\mathrm{cost}_R(\pi),\qquad
 \Wb_R(N)=\max_{F\in\UNSAT,\ |\langle F\rangle|\le N}W_R(F).
\]
Write $\mathrm{PB}(R)$ for $\exists C,k\ \forall N:\ \Wb_R(N)\le C(N+1)^k$,
$\LB(R)$ for $\neg\mathrm{PB}(R)$, and
$\Bthree(R)$ for $\SAT\in\mathrm P\Rightarrow\mathrm{PB}(R)$.
The inner minimum is a nondeterministic quantity: the free choice of the
best derivation is exactly the existential proof quantifier. That choice is
legitimate for a lower bound \emph{inside} $R$, as the directive says. It
also means $\LB(R)$ for strong $R$ is a statement of
$\mathrm{NP}\neq\mathrm{coNP}$ type, which is stronger than
$\mathrm P\neq\mathrm{NP}$. All theorems below concern \emph{provability}
(in ZFC, say) of these statements; their truth values are not at issue.

\section{Conservation of difficulty}
\begin{theorem}[PROVED/EXACT]
For every $R$: (a) $\Bthree(R)\wedge\LB(R)\Rightarrow\mathrm P\neq\mathrm{NP}$;
(b) $\mathrm P\neq\mathrm{NP}\Rightarrow\Bthree(R)$;
(c) consequently, given a proof of $\LB(R)$, $\Bthree(R)$ is provably
equivalent to $\mathrm P\neq\mathrm{NP}$.
\end{theorem}
\begin{proof}
(a) Contrapositive of $\Bthree$. (b) A false antecedent makes $\Bthree$
true. (c) With $\LB(R)$ established, $\Bthree(R)$ reads
$\SAT\in\mathrm P\Rightarrow\bot$.
\end{proof}

\begin{corollary}[PROVED/EXACT, using KNOWN THEOREMS]
For every system $R$ already known to have a superpolynomial $\Wb_R$ lower
bound on CNFs, $\Bthree(R)$ is exactly as hard as P versus NP. This includes
Resolution (Haken 1985, PHP; Urquhart 1987 and Ben-Sasson--Wigderson 2001,
Tseitin and random 3-CNF), Cutting Planes (Pudlák 1997, clique--colouring),
bounded-depth Frege (Ajtai; Kraj\'{\i}\v{c}ek--Pudl\'ak--Woods;
Pitassi--Beame--Impagliazzo, PHP), Polynomial Calculus (Razborov 1998), and
bounded-depth $\mathrm{Res}(\oplus)$ (Alekseev--Itsykson STOC 2025;
Efremenko--Itsykson CCC 2025; Bhattacharya--Chattopadhyay arXiv:2507.23008;
Byramji--Impagliazzo ECCC TR25-118).
\end{corollary}

So the ``weakest sufficient bridge'' is never weaker than the target once
the lower bound for that system is known. Splitting P$\neq$NP into
$\LB(R)$ and $\Bthree(R)$ makes progress only for a system $R$ whose lower
bound is \emph{not} yet known.

\section{Monotonicity: the two halves pull in opposite directions}
\begin{theorem}[PROVED/EXACT]
Let $S$ polynomially simulate $R$ in cost, so there is a polynomial $p$
with $W_S(F)\le p(|\langle F\rangle|+W_R(F))$ for all $F\in\UNSAT$. Then
$\Bthree(R)\Rightarrow\Bthree(S)$ and $\LB(S)\Rightarrow\LB(R)$.
\end{theorem}
\begin{proof}
$\Wb_S(N)\le p(N+\Wb_R(N))$, and polynomials compose.
\end{proof}
Strengthening $R$ makes the bridge easier and the lower bound harder, and
weakening $R$ does the opposite. Any $R$ the project fixes faces this
trade-off.

\paragraph{Application to the project system.}
LSPV-1/592 proves: if the native calculus satisfies its hypotheses A1--A4
(arbitrary CNF source, paid fresh AND2 definitions with facets,
cross-parent signed Resolution, faithful weights), then Extended Resolution
compiles into it at linear cost. By Theorem~2, $\LB(R_{\rm mix341})$ would
then imply $\LB(\mathrm{ER})$, and ER and EF are p-equivalent on CNFs. A
superpolynomial ER/EF lower bound has been open since Cook--Reckhow
(1979). Under A1--A4, then, [P2-B1] contains the EF lower-bound problem.
\textbf{CANDIDATE (new target, below):} the converse simulation. Every
native rule (SOURCE, AND3, XOR, RES, DEF) has bounded or affine
local soundness that PV can prove, so EF should p-simulate the calculus.
If it does, $R_{\rm mix341}\equiv_p\mathrm{EF}$, [P2-B1] \emph{is} the EF
lower-bound problem, and [P2-B2] \emph{is} the $\Bthree(\mathrm{EF})$
problem.

\section{The bridge for strong systems: what is known}
\begin{theorem}[KNOWN THEOREM]
(i) (Cook 1975; see Kraj\'{\i}\v{c}ek, \emph{Proof Complexity}, 2019.)
Suppose some polynomial-time $A$ has $\mathrm{PV}\vdash$ ``$A$ decides SAT''.
Then EF is polynomially bounded.
(ii) (Folklore, via polynomial-size $\Sigma_1$-completeness proofs; Pudlák
1986.) For a consistent recursively axiomatised $T\supseteq I\Delta_0+\exp$,
let $R_T$ be the system whose proofs of $F$ are $T$-proofs of
$\ulcorner F\in\UNSAT\urcorner$. Suppose some polynomial-time $A$ has
$T\vdash$ ``$A$ is correct on UNSAT outputs''. Then $\mathrm{PB}(R_T)$.
\end{theorem}
So $\Bthree$ is proved for these systems \emph{except} in one case:
$\mathrm P=\mathrm{NP}$ holds only through algorithms whose correctness the
base theory cannot prove. That residual case is the whole content of
[P2-B2]. It is the finite-incompleteness territory of Pudl\'ak (2017,
\emph{Incompleteness in the finite domain}).

\paragraph{A published replacement for B3.}
Pich and Santhanam (arXiv:2312.08163, ``Towards P$\neq$NP from Extended Frege
lower bounds'') prove several conditional implications. One of them: if
$S^1_2$ proves that a polynomial-time function generates anticheckers for
SAT, then EF not p-bounded implies P$\neq$NP. Arteche--Khaniki--Pich--Santhanam
(ICALP 2024, arXiv:2405.02232) give an analogous link for implicit EF, with
conclusion $\#\mathrm P\not\subseteq\mathrm{FP/poly}$.

\begin{corollary}[CONDITIONAL]
Assume LSPV-1/592 A1--A4 for the native system, and
$S^1_2\vdash$ ``a p-time function generates SAT anticheckers''. Then
$\LB(R_{\rm mix341})\Rightarrow\mathrm P\neq\mathrm{NP}$.
\end{corollary}
\begin{proof}
Theorem~2 with ER $\le_p R_{\rm mix341}$ gives $\LB(\mathrm{ER})$, hence EF
is not p-bounded. Apply Pich--Santhanam.
\end{proof}
This replaces the project's self-made B3 obligation with a hypothesis
stated in the literature, which an outside expert can check. It does
\emph{not} remove any OPEN arrow. $\LB(R_{\rm mix341})$ remains OPEN and at
least as hard as $\LB(\mathrm{EF})$. The $S^1_2$-provability hypothesis
remains OPEN, and Pich--Santhanam note it is plausible but unproved.
