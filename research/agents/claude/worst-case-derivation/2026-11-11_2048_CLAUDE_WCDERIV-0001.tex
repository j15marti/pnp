\section{Objects and quantifiers}
\begin{definition}
Fix a sound, complete, polynomial-time-checkable refutation system $R$
(Cook--Reckhow) with nonnegative node weights. For an unsatisfiable CNF $F$,
$W_R(F)=\min_{\pi:F\vdash_R\bot}\mathrm{cost}_R(\pi)$, and
\[
\Wb_R(N)=\max_{F\in\UNSAT,\ |\langle F\rangle|\le N}\ W_R(F).
\]
$\mathrm{LB}(R)$: $\forall k\,\exists^\infty N\ \Wb_R(N)>N^k$.
$\mathrm{Bridge}(R)$: $\mathrm{SAT}\in\mathrm P\Rightarrow\exists C,d\ \forall N\
\Wb_R(N)\le C(N+1)^d$.
\end{definition}
The minimum over $\pi$ is a \emph{nondeterministic} measure: strategy discovery
is not charged. This is legitimate for a lower bound, exactly as the directive
says: for every procedure $A$ that outputs $R$-derivations,
$\mathrm{time}_A(F)\ge \mathrm{cost}_R(\pi_A(F))\ge W_R(F)$ whenever cost is
dominated by output length. The whole difficulty is that an arbitrary SAT
algorithm need not output $R$-derivations; Bridge$(R)$ is exactly the statement
that this does not matter.

\section{Theorem A: PV-reflection bridge (KNOWN THEOREM + CONDITIONAL transfer)}
For a polynomial-time algorithm $A$ let $\mathrm{Sound}(A)$ be the open
(universally quantified, $\Pi^b_1$) PV statement
$\forall F\,\forall a\ \bigl(\mathrm{Sat}(F,a)\to f_A(F)=1\bigr)$:
\emph{whenever $A$ rejects, $F$ really is unsatisfiable}.

\begin{theorem}[CONDITIONAL; relies on Cook 1975 and the corpus EF$\to R_{\mathrm{mix341}}$ simulation]
Suppose (a) $A$ is a polynomial-time algorithm that rejects every unsatisfiable
CNF (true completeness on UNSAT; no provability needed), and (b)
$\mathrm{PV}\vdash\mathrm{Sound}(A)$. Then $\Wb_{\mathrm{EF}}(N)\le\mathrm{poly}(N)$,
and, conditional on the EF$\to$ER$\to$RES+DEF simulation of LSPV-1/406 being a
native accepted $R_{\mathrm{mix341}}$ simulation ([P2-D1] OPEN),
$\Wb_{R_{\mathrm{mix341}}}(N)\le\mathrm{poly}(N)$.
\end{theorem}
\begin{proof}
Cook (1975): if PV proves an open formula $\varphi(x)$, its propositional
translations $\|\varphi\|_n$ have EF proofs of size $\mathrm{poly}(n)$,
constructible in polynomial time. Apply this to $\mathrm{Sound}(A)$ at length
$n=|\langle F\rangle|$. Substitute the constant bits of a fixed unsatisfiable $F$
(EF is closed under substitution with polynomial overhead). The auxiliary
variables of $\|\cdot\|$ encoding the computation of $f_A$ on the now-constant
input are forced by a polynomial-size EF evaluation of a variable-free circuit,
yielding $f_A(F)=0$ by (a). Modus ponens gives an EF proof of
$\neg\mathrm{Sat}(F,a)$, i.e.\ of $\neg F(a)$ in the variables $a$, which
converts to an EF (hence ER) refutation of the CNF $F$ with polynomial
overhead. Finally ER is a fragment of RES+fresh binary DEF with unit positive
weights, so the cost is polynomial.
\end{proof}

\begin{remark}[exact asymmetry]
Only \emph{soundness} of rejection must be PV-provable; \emph{completeness}
(that $A$ rejects all of UNSAT) is used only as a true fact. So the bridge holds
for every polynomial SAT algorithm except those whose rejections are correct for
reasons PV cannot verify. This is strictly weaker than requiring a PV proof that
$A$ decides SAT, and incomparable with LSPV-1/406's $\mathsf{AC}_k$ hypothesis.
\end{remark}

\begin{corollary}[PROVED/EXACT given Theorem A]
$\mathrm{LB}(R_{\mathrm{mix341}})$ implies: no polynomial-time algorithm both
rejects every unsatisfiable CNF and has PV-provable soundness. I.e.\ [P2-B1]
alone yields ``P$=$NP has no PV-certified witness'' --- a meta-mathematical
independence statement, \textbf{not} P$\ne$NP. The entire residual gap is the
reflection step
\[
\textsf{TRUE-TO-PV}:\quad A \text{ correct}\ \Longrightarrow\ \mathrm{PV}\vdash
\mathrm{Sound}(A')\ \text{for some correct polynomial } A',
\]
which is \textbf{OPEN} and for which no technique is known.
\end{corollary}

\section{Theorem B: bridge-collapse dilemma (PROVED/EXACT, elementary)}
\begin{theorem}
For every proof system $R$: (i) $\mathrm{P}\ne\mathrm{NP}\Rightarrow
\mathrm{Bridge}(R)$ (vacuously). (ii) If $\mathrm{LB}(R)$ is a theorem, then
$\mathrm{Bridge}(R)\iff \mathrm{P}\ne\mathrm{NP}$. (iii) If $R$ p-simulates EF,
then $\mathrm{LB}(R)\Rightarrow\mathrm{LB}(\mathrm{EF})$.
\end{theorem}
\begin{proof}
(i) false antecedent. (ii) given LB$(R)$, Bridge$(R)$ yields SAT$\notin$P by
contraposition; conversely use (i). (iii) if $R\ge_p\mathrm{EF}$ then
$W_R(F)\le\mathrm{poly}(W_{\mathrm{EF}}(F))$, so a polynomial EF upper bound
would give one for $R$.
\end{proof}
\noindent\textbf{Consequence.} The ``weakest sufficient bridge'' is never weaker
than the goal for any system whose lower bound is already known: Resolution
(Haken), regular Res($\oplus$) (Efremenko--Garl\'ik--Itsykson, STOC 2024),
bounded-depth Res($\oplus$) (Byramji--Impagliazzo 2025), bounded-depth Frege,
Cutting Planes, PC. For those, Bridge$(R)$ \emph{is} P$\ne$NP. The split
``LB + Bridge'' is a genuine division of labor only for systems at or above the
current lower-bound frontier. Since the corpus records
$R_{\mathrm{mix341}}\ge_p$ ER $\equiv_p$ EF (CONDITIONAL on [P2-D1]),
part (iii) puts [P2-B1] at least as hard as superpolynomial Extended Frege
lower bounds, open since Cook--Reckhow (1979); not even Frege lower bounds are
known. This is a \textbf{calibration/barrier}, not a refutation.
