\section{Target, model and quantifiers}
\begin{definition}[Multisign DAG hypergraph system $H$]
A CNF $F$ over $x_1,\dots,x_n$ is the hypergraph whose hyperedges are clauses, each vertex
occurrence carrying a sign. A reduction state is a DAG whose sources are the hyperedges
of $F$. One step adds a node $C\cup D$ with two in-arcs from existing nodes $C\cup\{x\}$ and
$D\cup\{\bar x\}$ (signed merge on a vertex of opposite signs); optionally weakening.
The reduced state is reached when the empty hyperedge $\bot$ is present.
$\operatorname{cost}_H(\pi)$ is the number of nodes (or total bit length; the two differ by at
most a factor $O(n\log n)$).
\end{definition}
\[
\overline W_R(N)=\max_{F\in\mathrm{UNSAT},\,|\langle F\rangle|\le N}\ \min_{\pi:F\vdash_R\bot}\operatorname{cost}_R(\pi),
\qquad \mathrm{Bridge}(R):\ \mathrm{SAT}\in\mathrm P\Rightarrow \overline W_R(N)\le\mathrm{poly}(N).
\]
Strategy discovery is not charged; the minimum is over all legal derivations. This is
exactly the proof-size measure of Cook--Reckhow, so $\overline W_R$ is the worst-case
proof complexity of $R$.

\section{Results of this run}
\begin{lemma}[PROVED/EXACT, elementary]\label{l:ident}
For every UNSAT $F$, the minimal reduced logical state of $H$ is the singleton $\{\bot\}$,
independent of $F$; and $\min_\pi \operatorname{cost}_H(\pi)=S_{\mathrm{Res}}(F)$, the minimum DAG
resolution refutation size, up to the additive $|F|$ source count. Hence
$\overline W_H=\overline W_{\mathrm{Res}}$.
\end{lemma}
\begin{proof}
A signed merge on a vertex of opposite signs is a resolution step, and the reduction
DAG is a resolution DAG; conversely any resolution refutation is a legal reduction
sequence. Soundness of resolution forces every terminal state of an UNSAT $F$ to contain
$\bot$, and $\bot$ subsumes every clause, so the minimal state carries no information
about $F$; all information sits in the work.
\end{proof}

\begin{theorem}[KNOWN THEOREM, lower-bound half for $H$]
$\overline W_H(N)\ge 2^{\Omega(N/\log N)}$, witnessed by random 3-CNF at constant clause
density (Chv\'atal--Szemer\'edi 1988; Ben-Sasson--Wigderson 2001). The bound survives
Res$(k)$ for $k=O(\sqrt{\log n})$ (ECCC TR25-096, not independently reviewed here),
polynomial calculus (Ben-Sasson--Impagliazzo; Alekhnovich--Razborov), and regular
Res$(\oplus)$ (Bhattacharya--Chattopadhyay--Dvo\v{r}\'ak, ECCC 2024). For cutting planes the
known random-CNF bound is for $\Theta(\log n)$-CNF (Fleming--Pankratov--Pitassi--Robere 2017),
not 3-CNF; OPEN for random 3-CNF to our search.
\end{theorem}

\begin{theorem}[PROVED/EXACT: bridge collapse]\label{t:collapse}
Let $R$ be any fixed system for which $\overline W_R(N)\ne N^{O(1)}$ is known
unconditionally. Then $\mathrm{Bridge}(R)\iff \mathrm P\ne\mathrm{NP}$.
\end{theorem}
\begin{proof}
If $\mathrm P\neq\mathrm{NP}$ the antecedent is false. If $\mathrm P=\mathrm{NP}$ the
consequent is false by hypothesis, so the implication is false.
\end{proof}
Consequence. For $H=$ Resolution, Res$(k)$, PC, bounded-depth Frege, regular Res$(\oplus)$,
and (if Braun, arXiv:2609.23015, survives review) unrestricted Res$(\oplus)$, proving the
bridge \emph{is} proving P$\ne$NP. The decomposition gives no leverage at the weak end.
In particular, the user's free-optimal-sequence clarification is correct for the
lower-bound half, but the full difficulty is moved into the bridge: a polynomial-time
SAT algorithm need not output any $R$-derivation.

\begin{theorem}[KNOWN THEOREM / CONDITIONAL: the strong end]\label{t:opt}
If $R$ is an optimal proof system (it exists if NE$=$coNE, Kraj\'i\v{c}ek--Pudl\'ak 1989) then
$\mathrm{NP}=\mathrm{coNP}\Rightarrow R$ is polynomially bounded, so $\mathrm{Bridge}(R)$ holds
unconditionally (as P$=$NP implies NP$=$coNP). But then $\overline W_R$ superpolynomial
$\iff$ NP$\ne$coNP (Cook--Reckhow 1979), strictly stronger as a target than P$\ne$NP.
\end{theorem}

\begin{theorem}[KNOWN THEOREM, reflection; CONDITIONAL bridge for EF]\label{t:ef}
If $A$ is a polynomial-time decider and the theory PV (equivalently $S^1_2$ for this
purpose) proves the $\Pi^b_1$ UNSAT-soundness statement
$\forall F\,\forall x\,(A(F)=\mathrm{UNSAT}\to F(x)=0)$, then EF $p$-simulates the
Cook--Reckhow system $P_A$ whose proof of $F$ is the run of $A$ (Cook 1975;
Kraj\'i\v{c}ek--Pudl\'ak 1990). If additionally $A$ is complete (true, not necessarily
provable), EF is polynomially bounded. Hence
\[\mathrm{Bridge}(\mathrm{EF})\ \Leftarrow\ \mathbf U:\ \mathrm{SAT}\in\mathrm P\Rightarrow\exists A\ \text{complete with PV}\vdash\mathrm{Sound}_{\mathrm{UNSAT}}(A).\]
\end{theorem}
Only UNSAT-soundness, not completeness or running time, must be provable. Search-to-decision
self-reduction makes SAT answers self-certifying but does not help the UNSAT direction.
A 2026 clue in the same direction: Fleming--Grosser--Pitassi--Robere (ECCC TR26-109) show
that if EF proves a search problem is in FP then $\langle\mathrm{EF},R\rangle\equiv\mathrm{EF}$.

\paragraph{Synthesis (CANDIDATE ranking, not reviewed).}
Weak $R$: lower bound KNOWN, bridge $\equiv$ P$\ne$NP. Optimal $R$: bridge free, lower bound
$\equiv$ NP$\ne$coNP. Intermediate $R=$ EF or Frege: both halves OPEN; the bridge reduces to
premise $\mathbf U$, so the residual obligation for the program at EF is
LB(EF) together with ruling out P$=$NP via algorithms with PV-unprovable UNSAT-soundness.
LB(EF) alone implies only that no complete poly-time $A$ has PV-provable soundness,
a consistency-style conclusion, not P$\ne$NP.
