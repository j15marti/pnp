\section{Target, model and quantifiers}
\begin{definition}[Multisign DAG hypergraph system $H$]
A CNF $F$ over $x_1,\dots,x_n$ is the hypergraph whose hyperedges are clauses, each vertex
occurrence carrying a sign. A reduction state is a DAG whose sources are the hyperedges
of $F$. One step adds a node $C\cup D$ with two in-arcs from existing nodes $C\cup\{x\}$ and
$D\cup\{\bar x\}$ (signed merge on a vertex of opposite signs); optionally weakening.
The reduced state is reached when the empty hyperedge $\bot$ is present.
$\operatorname{cost}_H(\pi)$ is the number of nodes (or total bit length; the two differ by at
most a factor $O(n\log n)$).
\end{definition}
\[
\overline W_R(N)=\max_{F\in\mathrm{UNSAT},\,|\langle F\rangle|\le N}\ \min_{\pi:F\vdash_R\bot}\operatorname{cost}_R(\pi),
\qquad \mathrm{Bridge}(R):\ \mathrm{SAT}\in\mathrm P\Rightarrow \overline W_R(N)\le\mathrm{poly}(N).
\]
Strategy discovery is not charged; the minimum is over all legal derivations. This is
exactly the proof-size measure of Cook--Reckhow, so $\overline W_R$ is the worst-case
proof complexity of $R$.

\section{Results of this run}
\begin{lemma}[PROVED/EXACT, elementary]\label{l:ident}
For every UNSAT $F$, the minimal reduced logical state of $H$ is the singleton $\{\bot\}$,
independent of $F$; and $\min_\pi \operatorname{cost}_H(\pi)=S_{\mathrm{Res}}(F)$, the minimum DAG
resolution refutation size, up to the additive $|F|$ source count. Hence
$\overline W_H=\overline W_{\mathrm{Res}}$.
\end{lemma}
\begin{proof}
A signed merge on a vertex of opposite signs is a resolution step, and the reduction
DAG is a resolution DAG; conversely any resolution refutation is a legal reduction
sequence. Soundness of resolution forces every terminal state of an UNSAT $F$ to contain
$\bot$, and $\bot$ subsumes every clause, so the minimal state carries no information
about $F$; all information sits in the work.
\end{proof}

\begin{theorem}[KNOWN THEOREM, lower-bound half for $H$]
$\overline W_H(N)\ge 2^{\Omega(N/\log N)}$, witnessed by random 3-CNF at constant clause
density (Chv\'atal--Szemer\'edi 1988; Ben-Sasson--Wigderson 2001). The bound survives
Res$(k)$ for $k=O(\sqrt{\log n})$ (ECCC TR25-096, not independently reviewed here),
polynomial calculus (Ben-Sasson--Impagliazzo; Alekhnovich--Razborov), and regular
Res$(\oplus)$ (Bhattacharya--Chattopadhyay--Dvo\v{r}\'ak, ECCC 2024). For cutting planes the
known random-CNF bound is for $\Theta(\log n)$-CNF (Fleming--Pankratov--Pitassi--Robere 2017),
not 3-CNF; OPEN for random 3-CNF to our search.
\end{theorem}

\begin{theorem}[PROVED/EXACT: bridge collapse]\label{t:collapse}
Let $R$ be any fixed system for which $\overline W_R(N)\ne N^{O(1)}$ is known
unconditionally. Then $\mathrm{Bridge}(R)\iff \mathrm P\ne\mathrm{NP}$.
\end{theorem}
\begin{proof}
If $\mathrm P\neq\mathrm{NP}$ the antecedent is false. If $\mathrm P=\mathrm{NP}$ the
consequent is false by hypothesis, so the implication is false.
\end{proof}
Consequence. For $H=$ Resolution, Res$(k)$, PC, bounded-depth Frege, regular Res$(\oplus)$,
and (if Braun, arXiv:2609.23015, survives review) unrestricted Res$(\oplus)$, proving the
bridge \emph{is} proving P$\ne$NP. The decomposition gives no leverage at the weak end.
In particular, the user's free-optimal-sequence clarification is correct for the
lower-bound half, but the full difficulty is moved into the bridge: a polynomial-time
SAT algorithm need not output any $R$-derivation.

\begin{theorem}[KNOWN THEOREM / CONDITIONAL: the strong end]\label{t:opt}
If $R$ is an optimal proof system (it exists if NE$=$coNE, Kraj\'i\v{c}ek--Pudl\'ak 1989) then
$\mathrm{NP}=\mathrm{coNP}\Rightarrow R$ is polynomially bounded, so $\mathrm{Bridge}(R)$ holds
unconditionally (as P$=$NP implies NP$=$coNP). But then $\overline W_R$ superpolynomial
$\iff$ NP$\ne$coNP (Cook--Reckhow 1979), strictly stronger as a target than P$\ne$NP.
\end{theorem}

\begin{theorem}[KNOWN THEOREM, reflection; CONDITIONAL bridge for EF]\label{t:ef}
If $A$ is a polynomial-time decider and the theory PV (equivalently $S^1_2$ for this
purpose) proves the $\Pi^b_1$ UNSAT-soundness statement
$\forall F\,\forall x\,(A(F)=\mathrm{UNSAT}\to F(x)=0)$, then EF $p$-simulates the
Cook--Reckhow system $P_A$ whose proof of $F$ is the run of $A$ (Cook 1975;
Kraj\'i\v{c}ek--Pudl\'ak 1990). If additionally $A$ is complete (true, not necessarily
provable), EF is polynomially bounded. Hence
\[\mathrm{Bridge}(\mathrm{EF})\ \Leftarrow\ \mathbf U:\ \mathrm{SAT}\in\mathrm P\Rightarrow\exists A\ \text{complete with PV}\vdash\mathrm{Sound}_{\mathrm{UNSAT}}(A).\]
\end{theorem}
Only UNSAT-soundness, not completeness or running time, must be provable. Search-to-decision
self-reduction makes SAT answers self-certifying but does not help the UNSAT direction.
A 2026 clue in the same direction: Fleming--Grosser--Pitassi--Robere (ECCC TR26-109) show
that if EF proves a search problem is in FP then $\langle\mathrm{EF},R\rangle\equiv\mathrm{EF}$.

\paragraph{Synthesis (CANDIDATE ranking, not reviewed).}
Weak $R$: lower bound KNOWN, bridge $\equiv$ P$\ne$NP. Optimal $R$: bridge free, lower bound
$\equiv$ NP$\ne$coNP. Intermediate $R=$ EF or Frege: both halves OPEN; the bridge reduces to
premise $\mathbf U$, so the residual obligation for the program at EF is

\paragraph{Source.} Kraj\'i\v{c}ek, \emph{A note on SAT algorithms and proof complexity}
(text read from \texttt{karlin.mff.cuni.cz/\~{}krajicek/sat.pdf}). Definitions used verbatim
in substance: a proof system $P$ is \emph{decent} if restriction (D1), proofs of true
sentences (D2), modus ponens (D3) and extraction of $\varphi$ from a proof of
$\neg\mathrm{sat}_n(u,\neg\varphi,v)$ (D4) are polynomial-time. $\mathrm{Alg}(P)$ is the class
of SAT algorithms $M$ whose reflection formulas
$\mathrm{ref}^M_{k,n}=\mathrm{reject}^M_{k,n}\to\neg\mathrm{sat}_n$ have polynomial-size
$P$-proofs. Theorem 2.1 there: if $P$ is decent, $M\in\mathrm{Alg}(P)$ and $\tau_n$ needs
$P$-proofs of size $\ell(n)$, then $M$ needs time $\ell(n)^{\Omega(1)}$ to reject $\neg\tau_n$.

\begin{theorem}[KNOWN THEOREM, Kraj\'i\v{c}ek; restated in the present notation]\label{t:alg}
For every decent $R$ and every SAT algorithm $M\in\mathrm{Alg}(R)$, and every UNSAT $F$,
$\min_{\pi:F\vdash_R\bot}\operatorname{cost}_R(\pi)\le t_M(F)^{O(1)}$, where $t_M(F)$ is the
time $M$ needs to reject $F$. Consequently $\overline W_R$ is a lower bound on the worst-case
running time of every algorithm in $\mathrm{Alg}(R)$, and on no other algorithm a priori.
\end{theorem}

\begin{theorem}[PROVED/EXACT, immediate from Theorem~\ref{t:alg}; not externally reviewed]\label{t:refl}
For every decent $R$ the following are equivalent:
(i) Bridge$(R)$;
(ii) P$=$NP implies $\mathrm{Alg}(R)$ contains a complete polynomial-time SAT algorithm;
(iii) P$=$NP implies $\mathrm{Alg}(R)$ contains every complete polynomial-time SAT algorithm
(for every admissible choice of $\mathrm{reject}^M$).
\end{theorem}
\begin{proof}
(i)$\Rightarrow$(iii): if $R$ is polynomially bounded, every tautology, in particular every
$\mathrm{ref}^M_{k,n}$ of a sound $M$, has a polynomial-size proof; their size is $k^{O(1)}$
with $k$ polynomial in $n$. (iii)$\Rightarrow$(ii): P$=$NP supplies one complete
polynomial-time algorithm. (ii)$\Rightarrow$(i): Theorem~\ref{t:alg} with
$t_M(F)\le|F|^{O(1)}$ bounds $\overline W_R(N)$ polynomially.
\end{proof}
Interpretation for the program. Granting the optimal derivation for free (the user's
clarification) is correct and costs nothing, but the resulting lower bound reaches exactly
the algorithms whose soundness $R$ can verify cheaply. The bridge is therefore not a
simulation statement about all algorithms; it is the statement that every hypothetical
P$=$NP algorithm is $R$-provably sound. For $R$ with a known lower bound (Resolution,
Res$(k)$, PC, CP, AC$^0$-Frege) Theorem~\ref{t:refl} plus Theorem~\ref{t:collapse} gives the
KNOWN unconditional statement (Kraj\'i\v{c}ek, Cor.~2.2) that $\mathrm{Alg}(R)$ contains no
complete polynomial-time algorithm; this is consistency-type evidence, not P$\ne$NP.
Resolution does not prove its own soundness (Atserias--Bonet 2004), so even the
algorithms whose runs \emph{are} resolution refutations need not lie in Alg(Res).

\begin{theorem}[PROVED/EXACT, not externally reviewed; closes run-1 next lemma]\label{t:U}
Premise $\mathbf U$ of Theorem~\ref{t:ef} is equivalent to Bridge(EF).
\end{theorem}
\begin{proof}
($\Rightarrow$) is Theorem~\ref{t:ef}. ($\Leftarrow$) Assume Bridge(EF) and P$=$NP, and fix a
polynomial $p$ with every UNSAT $F$ having an EF refutation of size $\le p(|F|)$. The
language $L=\{(F,w):\exists\pi\supseteq w,\ |\pi|\le p(|F|),\ \mathrm{Chk}_{EF}(F,\pi)=1\}$
is in NP, hence in P via some (unverified) decider $B$. Let $A(F)$ build $\pi$ bit by bit by
prefix search with $B$, then output UNSAT iff $\mathrm{Chk}_{EF}(F,\pi)=1$, and SAT
otherwise. $A$ runs in polynomial time. It is complete: for UNSAT $F$ a refutation of size
$\le p(|F|)$ exists and correct prefix search finds one; for SAT $F$ no refutation exists,
so the check fails. Its UNSAT-soundness is PV-provable \emph{independently of $B$}:
by the definition of $A$ as a PV function, PV proves $A(F)=\mathrm{UNSAT}\to
\mathrm{Chk}_{EF}(F,\pi_A(F))=1$, and PV proves the reflection principle of EF,
$\mathrm{Chk}_{EF}(F,\pi)=1\to F(x)=0$ (Cook 1975; Kraj\'i\v{c}ek 1995, Ch.~9).
\end{proof}
Consequence: premise $\mathbf U$ is \emph{not} a weaker target than the bridge; uniform,
nonuniform (Theorem~\ref{t:refl}(ii)) and size forms coincide. The same argument works for
any $R$ with polynomial-time checker whose reflection principle is provable in a theory
$T$ that $R$ simulates (for Frege with $T=\mathrm{VNC}^1$: CONDITIONAL on checking the
Cook--Nguyen statement; not done this run). For weak $R$ the second hypothesis fails
(Atserias--Bonet), consistent with Theorem~\ref{t:collapse}.

\begin{theorem}[PROVED/EXACT, elementary; unreviewed]\label{t:gap}
Let $Q$: ``P$=$NP and EF p-bounded $\Rightarrow$ Frege has polynomial-size proofs of
$\|\mathrm{Con}_{EF}\|$''. Then $\mathrm{Bridge}(\mathrm{Frege})\iff\mathrm{Bridge}(\mathrm{EF})\wedge Q$.
\end{theorem}
\begin{proof}
Forward: Frege $\le_p$ EF gives Bridge(EF); under P$=$NP, Frege is p-bounded, so the
polynomial-size tautologies $\mathrm{Con}_{EF}$ have short Frege proofs. Backward: under P$=$NP,
EF is p-bounded and $Q$ gives short Frege proofs of $\mathrm{Con}_{EF}$, hence Frege
p-simulates EF (Buss 1991), hence Frege is p-bounded.
\end{proof}
So the routes $(\mathrm{LB}(\mathrm{EF}),\mathrm{Bridge}(\mathrm{EF}))$ and
$(\mathrm{LB}(\mathrm{Frege}),\mathrm{Bridge}(\mathrm{EF})\wedge Q)$ trade a weaker lower bound
for exactly $Q$; neither dominates.

\begin{theorem}[PROVED, sketch; unreviewed]\label{t:frU}
Under Bridge(Frege) and P$=$NP, the prefix-search algorithm $A_{Fr}$ with a log-depth Frege
checker is complete, polynomial-time and in $\mathrm{Alg}(\mathrm{Frege})$, independently of the
inner decider. (Trace variables supply the checker wires; log depth makes $\mathrm{Valid}_{Fr}$
a polynomial-size formula; Frege proves its own reflection, Buss 1991.) Open obligation:
reflection from partial consistency for Frege, to be checked against Kraj\'i\v{c}ek 1995, Ch.~9.
\end{theorem}

\paragraph{Observation (KNOWN in substance).} Under P$=$NP every disjoint NP pair is
P-separable, so every proof system has trivial feasible interpolation and every p-bounded
system is automatable. All known conditional evidence against Frege $\ge$ EF is
cryptographic (Kraj\'i\v{c}ek--Pudl\'ak 1998; Bonet--Pitassi--Raz 2000) and is void under the
bridge hypothesis. Hence $Q$ is OPEN with no known falsifying tool. Structurally,
interpolation-based LB and the bridge live in incompatible worlds.

\paragraph{Side closure (PROVED, elementary).} For a certifying (proof-logging DPLL/CDCL)
decider $A$, Sound$_{A,n}$ is EF-provable via resolution reflection, so lemma SR$(A)$ holds;
but no such $A$ is complete polynomial-time (Haken; Chv\'atal--Szemer\'edi). The DPLL-class
target of \texttt{LIVING\_LOG.md} is therefore RETIRED as uninformative.
