\begin{abstract}
The scheduled directive asks all roles to study
\[
\Wb_R(N)=\max_{\substack{F\in\UNSAT\\ |\langle F\rangle|\le N}}\ \min_{\pi:F\vdash_R\bot}\operatorname{cost}_R(\pi)
\]
for a fixed inference system $R$ (e.g.\ a multisign DAG-hypergraph reduction calculus for 3-SAT), and to prove the
``weakest sufficient bridge'' $B(R):\ \SAT\in\mathrm{P}\Rightarrow \Wb_R(N)\le\mathrm{poly}(N)$.
This report proves four elementary but load-bearing facts about $B(R)$.
(1) \status{PROVED/EXACT} $B(R)\iff(\mathrm{P}\ne\mathrm{NP})\vee\PB(R)$; so all content of $B(R)$ lives in Algorithmica,
and for every $R$ with an already proved superpolynomial lower bound (Resolution, hence the hypergraph reduction calculus
when it is Resolution-simulated; Cutting Planes; Polynomial Calculus; bounded-depth Frege) $B(R)$ \emph{is} $\mathrm{P}\ne\mathrm{NP}$.
(2) \status{PROVED/EXACT} $B$ is upward-closed and lower bounds are downward-closed along $p$-simulation, so the two
halves of the strategy are pulled to opposite ends of the simulation order.
(3) \status{KNOWN THEOREM, restated} the only non-vacuous route to $B(R)$ is to simulate the solver-induced system $R_M$
of the hypothetical algorithm, which for $R=\mathrm{EF}$ is available exactly when the correctness of $M$ is feasibly provable.
(4) \status{PROVED/EXACT} For every $R$ with an oracle-independent proved lower bound, the relativized bridge
$\SAT\in\mathrm{P}^{A}\Rightarrow\PB(R)$ is false for $A=\mathrm{TQBF}$; any proof of $B(R)$ must use that the hypothetical
SAT algorithm is oracle-free. None of this proves a lower bound or a bridge. P versus NP remains OPEN.
\end{abstract}

\section{Setup and the free-optimal-sequence principle}
Fix a Cook--Reckhow-style system $R$ with a nonnegative cost $\operatorname{cost}_R$ that is polynomially related to
encoded proof length (as required by the directive; otherwise the bridge already fails on encoding grounds).
Write $\PB(R)$ for $\exists k\ \forall^\infty N:\ \Wb_R(N)\le N^k$.

\begin{proposition}[Free optimal sequence is a valid lower-bound device]\status{PROVED/EXACT}
For every $F$ and every \emph{$R$-derivation} $\pi$ executed by any procedure, $\operatorname{cost}_R(\pi)\ge\min_{\pi'}\operatorname{cost}_R(\pi')$;
hence $\Wb_R(N)$ lower-bounds the worst-case cost of every procedure whose execution trace is an $R$-derivation, and charging
or not charging the search for $\pi$ does not change this.
\end{proposition}
\begin{proof}Immediate from the definition of the minimum.\end{proof}

\begin{remark}[Scope]
The proposition says nothing about procedures whose traces are \emph{not} $R$-derivations. A deterministic SAT algorithm $M$
produces a run, not an $R$-derivation. The free-sequence principle is therefore sound, and the entire difficulty sits in the
bridge $B(R)$, which is exactly the statement that runs can be converted to cheap $R$-derivations. If the rule set of the
hypergraph calculus were allowed to contain ``apply any polynomial-time sound step'', the minimum would collapse
(one step: run $M$), so $R$ must be fixed in advance. This matches LSPV-1/336--337 and TMR-1/036.
\end{remark}

\section{Algorithmica localization}
\begin{theorem}[Localization]\label{thm:loc}\status{PROVED/EXACT}
For every fixed $R$: $B(R)\iff (\mathrm{P}\ne\mathrm{NP})\vee \PB(R)$. Consequently
\[
B(R)\wedge\neg\PB(R)\iff (\mathrm{P}\ne\mathrm{NP})\wedge\neg\PB(R).
\]
\end{theorem}
\begin{proof}$B(R)$ is the material implication $(\SAT\in\mathrm P)\to\PB(R)$, and $\SAT\in\mathrm P\iff\mathrm P=\mathrm{NP}$.\end{proof}

\begin{corollary}[No weakening for systems with proved lower bounds]\status{PROVED/EXACT}
If $\neg\PB(R)$ is a theorem, then $B(R)\iff \mathrm P\ne\mathrm{NP}$. This holds for Resolution (Haken 1985; Urquhart 1987;
Ben-Sasson--Wigderson 2001), Cutting Planes (Pudl\'ak 1997), Polynomial Calculus (Razborov 1998), bounded-depth Frege
(Ajtai; Kraj\'{\i}\v{c}ek--Pudl\'ak--Woods; Pitassi--Beame--Impagliazzo), and for any DAG-hypergraph reduction calculus
whose derivations Resolution $p$-simulates (signed hyperedges = clauses, reductions = resolution/weakening steps).
\end{corollary}

\noindent\textbf{Interpretation.} For such $R$ the bridge is not a ``weaker sufficient condition''; it is a reformulation of
the whole problem (world-ledger category: \emph{equivalent of P $\ne$ NP}). In Impagliazzo's worlds, $B(R)$ is vacuously
true in Heuristica, Pessiland, Minicrypt and Cryptomania, and in Algorithmica it says exactly ``$R$ is polynomially bounded''.
Any proof of $B(R)$ must therefore be an \emph{upper-bound} construction inside a hypothetical Algorithmica: from a
polynomial-time $M$ it must build polynomial-cost $R$-refutations of all UNSAT inputs. A lower-bound argument cannot supply it.

\section{Anti-monotonicity along simulation}
\begin{theorem}\label{thm:anti}\status{PROVED/EXACT}
If $R'$ $p$-simulates $R$ (with costs polynomially related to length), then $\PB(R)\Rightarrow\PB(R')$, hence
$B(R)\Rightarrow B(R')$, and $\neg\PB(R')\Rightarrow\neg\PB(R)$.
\end{theorem}
\begin{proof}A $p$-simulation maps each $R$-refutation of $F$ of cost $c$ to an $R'$-refutation of cost $\mathrm{poly}(c+|F|)$,
so $\Wb_{R'}(N)\le\mathrm{poly}(\Wb_R(N)+N)$. The rest is Theorem~\ref{thm:loc} and contraposition.\end{proof}

\noindent So the strategy needs one fixed $R$ that is simultaneously strong enough for $B(R)$ to be provable without proving
P $\ne$ NP, and weak enough for $\neg\PB(R)$ to be provable. The current unconditional lower-bound frontier
(as of the literature checked this run) stops below AC$^0[p]$-Frege and full Res($\oplus$) (only partial/quadratic Res($\oplus$)
bounds, ECCC TR26-018; bounded-depth Frege via Res($\log$), ECCC TR26-055; semantic tree-like bounded-line Frege,
arXiv:2604.28172), while every known non-vacuous route to $B(R)$ (Section 4) requires $R\succeq\mathrm{EF}$-strength
reasoning about $M$. No system is currently known in the overlap. \status{OPEN}

\section{The only non-vacuous route: solver-induced systems}
\begin{proposition}\status{KNOWN THEOREM / folklore, restated}
If $M$ is a correct polynomial-time SAT decider, $R_M$ (proof of $F$ = the empty string; verifier runs $M$ on $F$ and accepts iff $M$ rejects)
is a sound, complete, polynomially bounded proof system. Hence under $\mathrm P=\mathrm{NP}$, $B(R)$ holds iff
$R$ polynomially simulates $R_M$ on UNSAT inputs for that $M$ (equivalently, $\PB(R)$).
For $R=\mathrm{EF}$, if $S^1_2$ (equivalently PV for these $\forall\Pi^b_1$ statements) proves the soundness of $R_M$, then EF $p$-simulates $R_M$
(Cook 1975; Kraj\'{\i}\v{c}ek--Pudl\'ak 1990; Kraj\'{\i}\v{c}ek's 1995 monograph).
\end{proposition}
Therefore: \emph{``$\SAT\in\mathrm P$ via an $S^1_2$-provably correct algorithm'' $\Rightarrow \PB(\mathrm{EF})$} is a
\status{KNOWN THEOREM}. The residual gap of $B(\mathrm{EF})$ is precisely a correct polynomial-time SAT algorithm whose
correctness has no feasible proof; this is the correctness-index barrier already recorded by LSPV-1/336. Whether
$\mathrm P=\mathrm{NP}$ alone implies $\PB(\mathrm{EF})$ is \status{OPEN} (no primary source found this run settles it; the known
Pich--Santhanam direction is the converse-type ``EF not p-bounded + feasibility hypotheses $\Rightarrow$ P $\ne$ NP'').

\section{A relativization wall for the bridge}
\begin{theorem}\label{thm:rel}\status{PROVED/EXACT}
Let $R$ be any fixed (oracle-free) system with a proved lower bound $\neg\PB(R)$. Define the relativized bridge
$B^A(R):\ \SAT\in\mathrm P^A\Rightarrow\PB(R)$. Then $B^{\mathrm{TQBF}}(R)$ is false.
\end{theorem}
\begin{proof}$\SAT\in\mathrm{PSPACE}=\mathrm P^{\mathrm{TQBF}}$, while $\PB(R)$ is oracle-independent and false.\end{proof}

\begin{corollary}\status{PROVED/EXACT}
Any proof of $B(R)$ for, e.g., Resolution or the Resolution-simulated hypergraph calculus must use a property of
\emph{oracle-free} polynomial-time computation that fails for $\mathrm P^{\mathrm{TQBF}}$. Black-box ``trace-to-derivation''
compilers that treat machine steps as abstract oracle-capable transitions cannot prove $B(R)$. This is consistent with,
and sharpens for the bridge itself, REFEREE296 (black-box relativization firewall) and TMR-1/036 (Tseitin trace distortion).
For $R=\mathrm{EF}$, $B^{\mathrm{TQBF}}(\mathrm{EF})\iff\PB(\mathrm{EF})$, which is open, so the wall gives no information there.
\end{corollary}
