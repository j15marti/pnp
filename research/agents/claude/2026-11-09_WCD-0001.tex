\title{WCD-0001: What the worst-case derivation bridge can and cannot do}\date{2026-11-09}\maketitle
\noindent\textbf{Status: P vs NP remains OPEN.} Nothing below is a lower bound on any natural system.

\section*{Setup}
$R$ is a refutation system (Cook--Reckhow: a poly-time checker for $\pi\vdash_R\bot$ on CNF $F$); $\mathrm{cost}_R(\pi)=|\pi|$ (checker time is polynomial in $|\pi|$, so size is the right proxy).
$W_R(N)$ as in the directive. $\mathrm{BRIDGE}(R)$: $\mathrm{SAT}\in\mathsf P\Rightarrow W_R\le\mathrm{poly}$.

\begin{prop}[free strategy = nondeterminism; PROVED/EXACT, folklore]
For Cook--Reckhow $R$, $W_R\le\mathrm{poly}$ iff $R$ is polynomially bounded; this implies $\mathsf{NP}=\mathsf{coNP}$ (Cook--Reckhow). Uncharged strategy search erases the deterministic-time structure: the quantity is proof size, not algorithm time.
\end{prop}

\begin{thm}[Dichotomy; PROVED/EXACT]
If $W_R$ is known superpolynomial (e.g.\ resolution, Haken; bounded-depth Frege; polynomial calculus), then $\mathrm{BRIDGE}(R)\iff \mathrm{SAT}\notin\mathsf P$.
\end{thm}
\begin{proof}($\Leftarrow$) vacuous. ($\Rightarrow$) $W_R$ superpoly and BRIDGE give $\mathrm{SAT}\notin\mathsf P$.\end{proof}
\noindent So a bridge for any system with a proved lower bound is $\mathsf P\neq\mathsf{NP}$ in disguise; the lower bound for such $R$ adds no information. A non-circular route needs $R$ whose lower bound is itself unknown.

\begin{thm}[Optimal systems; PROVED/EXACT]
If $R$ is an optimal proof system (polynomially simulates every proof system), then $\mathrm{BRIDGE}(R)$ holds, and $W_R$ superpoly $\iff \mathsf{NP}\neq\mathsf{coNP}$.
\end{thm}
\begin{proof}$\mathsf P=\mathsf{NP}\Rightarrow\mathsf{NP}=\mathsf{coNP}\Rightarrow$ some $Q$ is p-bounded; $R$ simulates $Q$. Converse by Cook--Reckhow.\end{proof}
\noindent Consequence: via optimal $R$ the target is exactly $\mathsf{NP}\ne\mathsf{coNP}$, strictly more than needed. Optimal systems are not known to exist (Krajíček--Pudlák: implied by $\mathsf{NE}=\mathsf{coNE}$).

\begin{thm}[Weakest bridge is uniform reflection; PROVED/EXACT, standard]
Let $M$ decide SAT in time $n^k$ and let $\mathrm{Ref}^n_M$ be the unsatisfiable CNF (Tseitin form) of ``$C_M(x)=0\wedge F_x(y)$'' over formula bits $x$ ($|x|=n$). For a system $R$ closed under substitution and able to evaluate a poly-size circuit on constants in poly size (Frege, EF, and stronger):
\[ W_R\le\mathrm{poly}\iff \mathrm{Ref}^n_M \text{ has } \mathrm{poly}(n)\ R\text{-refutations.}\]
\end{thm}
\begin{proof}($\Rightarrow$) $\mathrm{Ref}^n_M$ is UNSAT of poly size (that $M$ is correct makes it unsatisfiable). ($\Leftarrow$) For UNSAT $F$, substitute $x:=F$ in the refutation; evaluate $C_M(F)=0$ gate by gate on constants; the remainder is $F(y)$ itself, giving a refutation of $F$ of size $\mathrm{poly}$.\end{proof}
\noindent Hence the minimal sufficient bridge for Frege/EF is: \emph{if $\mathsf P=\mathsf{NP}$, some poly SAT algorithm has feasibly (R-)provable correctness}. This is the propositional form of provability of $\mathsf P=\mathsf{NP}$ in bounded arithmetic; I did not find an unconditional proof or refutation, so it is OPEN/CANDIDATE.

\section*{Implications for the five roles}
\begin{enumerate}
\item Lower-Bound Lead: do not target a system with a known lower bound (circular, Dichotomy); target $R\supseteq$ Frege/EF and prove BRIDGE via reflection; separately a lower bound there is $\mathsf{NP}\ne\mathsf{coNP}$-strength (Cook--Reckhow).
\item Reduction Builder: ``fixed combined inference system'' must declare Cook--Reckhow; a non-checkable cost measure re-admits free semantic answers (retired).
\item Hard families: survival against resolution/PC/bounded-depth is necessary but carries no bridge weight; a family must be tested against EF-style reflection-provable structure.
\item Resource Auditor: charge checker time/peak space of verifying $\pi$, not discovery.
\item Red Team: attack Dichotomy hypotheses and the reflection reduction's closure assumptions (substitution, constant evaluation).
\end{enumerate}
