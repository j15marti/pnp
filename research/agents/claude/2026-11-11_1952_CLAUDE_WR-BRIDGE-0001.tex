\section{Objects and exact quantifiers}
Fix one Cook--Reckhow refutation system $R$: a verifier $V_R(F,\pi)$
running in time $\poly(|F|+|\pi|)$, sound (accepts only for
$F\in\UNSAT$) and complete. Fix a cost with
$|\pi|^{1/c}-c\le\operatorname{cost}_R(\pi)\le c|\pi|^{c}+c$ for a constant
$c$ (node count of a DAG with polynomially bounded labels qualifies).
As in the shared directive,
\[
 W_R(F)=\min_{\pi:\,V_R(F,\pi)=1}\operatorname{cost}_R(\pi),\qquad
 \Wb_R(N)=\max_{F\in\UNSAT,\ |\langle F\rangle|\le N} W_R(F).
\]
The minimum is taken over \emph{all} valid derivations; the optimal one
is granted for free. The three bridges in corpus notation
(LSPV-1/334, /336):
\begin{align*}
 TR(R)&:\ \forall\text{ correct total }A\ \exists p_A\ \forall F\in\UNSAT:\
   W_R(F)\le p_A(|F|+T_A(F)),\\
 B3(R)&:\ \SAT\in\mathsf P\implies \Wb_R(N)\le\poly(N),\\
 LB(R)&:\ \Wb_R\ \text{is superpolynomial}.
\end{align*}
$TR(R)\Rightarrow B3(R)$, and $B3(R)\wedge LB(R)\Rightarrow \SAT\notin\mathsf P$.

\section{Theorem 1: the free-strategy convention is neutral for the bridge}
Let $W^{\mathrm{srch}}_R(F)$ be the least running time of a fixed
deterministic algorithm that, given $F$, outputs a valid $R$-refutation
(search charged), with $\Wb^{\mathrm{srch}}_R$ its worst case.
Unconditionally $W_R(F)\le\poly(W^{\mathrm{srch}}_R(F))$ since output
length is bounded by time.

\begin{theorem}[Search neutrality under the bridge hypothesis]
If $\SAT\in\mathsf P$ then there is one fixed algorithm $S_R$ that on every
$F\in\UNSAT$ outputs a valid $R$-refutation $\pi$ with
$|\pi|\le L_R(F)$ (minimum proof length) in time $\poly(|F|+L_R(F))$.
Consequently, under $\SAT\in\mathsf P$,
$\Wb^{\mathrm{srch}}_R(N)\le\poly(N+\Wb_R(N))$, and $B3(R)$ has the same
truth value whether or not strategy discovery is charged.
\end{theorem}
\begin{proof}
The language
$E_R=\{(F,1^s,u):\exists v,\ |uv|\le s,\ V_R(F,uv)=1\}$ is in $\mathsf{NP}$;
by Cook--Levin it reduces to $\SAT$ with a fixed polynomial blow-up, hence
$E_R\in\mathsf P$. For $s=1,2,\dots$ test $(F,1^s,\varepsilon)\in E_R$; the
first success is $s=L_R(F)$. Then extend $u$ one bit at a time (and test
termination with $V_R$), keeping membership in $E_R$; at most
$s$ steps of $O(1)$ queries each. Total time
$\poly(|F|+L_R(F))$. The cost/length polynomial relation transfers the
bound to $\operatorname{cost}_R$.
\end{proof}
\noindent\emph{Reading.} The user's clarification (``finding the optimal
sequence can only add work'') is correct, and Theorem~1 shows it is also
\emph{lossless}: on the hypothesis side of $B3$, search is polynomial
anyway. So the free-choice relaxation neither helps nor hurts the bridge;
it only fixes the lower-bound target as the stronger (proof-size) form.
This matches the literature folklore that $\mathsf P=\mathsf{NP}$ makes
every proof system automatable \cite{atserias}.

\section{Theorem 2: frontier collapse}
\begin{theorem}
If $LB(R)$ is a theorem (of ZFC), then $B3(R)$ is equivalent (in ZFC) to
$\SAT\notin\mathsf P$.
\end{theorem}
\begin{proof}
$B3(R)$ is the implication $\SAT\in\mathsf P\Rightarrow\neg LB(R)$,
i.e.\ $\SAT\notin\mathsf P\vee\neg LB(R)$; with $LB(R)$ provable this is
$\SAT\notin\mathsf P$. Conversely $\SAT\notin\mathsf P$ makes $B3(R)$
vacuously true.
\end{proof}
\noindent\emph{Consequence.} For every system on the current proved
lower-bound frontier---resolution (Haken; Ben-Sasson--Wigderson),
cutting planes (Pudl\'ak), polynomial calculus/Nullstellensatz,
bounded-depth Frege (Ajtai; Kraj\'{\i}\v{c}ek--Pudl\'ak--Woods;
Pitassi--Beame--Impagliazzo), and tree-like/regular/bounded-depth
Res$(\oplus)$---the ``weakest sufficient bridge'' is \emph{exactly}
$\mathsf P\ne\mathsf{NP}$, not a weaker statement. The bridge can be
strictly weaker than the goal only for $R$ beyond that frontier
(AC$^0[p]$-Frege, $\mathrm{TC}^0$-Frege, Frege, EF, or the project's mixed
calculi when they p-simulate such systems), and there $LB(R)$ is itself a
long-standing open problem. LSPV-1/334 already showed $TR(\mathrm{Res})$
is FALSE; Theorem~2 adds that $B3(\mathrm{Res})$ is not false but is the
whole problem. \textbf{Allocation implication:} hard-instance work on
resolution/PC/CP lower bounds cannot by itself shorten the path; the
target $R$ must be at least as strong as the weakest system whose lower
bound is open.

\section{Theorem 3 (known): the EF instantiation}
\begin{theorem}[Cook 1975; Kraj\'{\i}\v{c}ek--Pudl\'ak]
If some polynomial-time $A$ decides $\SAT$ and the soundness statement
$\forall F\,\forall a\,(A(F)=\mathrm{UNSAT}\rightarrow \neg\mathrm{Sat}(F,a))$
is provable in $\mathrm{PV}$, then $\Wb_{\mathrm{EF}}(N)\le\poly(N)$.
Moreover $\mathrm{PV}\vdash\mathsf P=\mathsf{NP}$ iff $\mathrm{PV}$ defines a
polynomial-time function giving either a satisfying assignment or an EF
refutation (the same for $S^1_2$).
\end{theorem}
Hence $LB(\mathrm{EF})$ alone yields only the \emph{consistency} of
$\mathsf P\ne\mathsf{NP}$ with PV \cite{krajicek-survey}; the residual gap
in $B3(\mathrm{EF})$ is precisely ``$\SAT\in\mathsf P\Rightarrow$ some
polynomial SAT algorithm has PV-provable soundness'' --- the
provability gap already isolated as F1 in LSPV-1/361 and the bounded-arithmetic
criterion of LSPV-1/357. \textbf{ADOPTED, not new.}

\section{Theorem 4: explicit hard families are incompatible with $TR(R)$}
Call $\{F_n\}$ \emph{explicit} if $1^n\mapsto F_n$ is polynomial-time and
$|F_n|\ge n$.
\begin{theorem}
If $TR(R)$ holds, then every explicit family of unsatisfiable formulas
has $W_R(F_n)\le\poly(|F_n|)$. Hence $TR(R)\wedge LB(R)$ can only be
witnessed by formulas outside the range of every polynomial-time generator.
\end{theorem}
\begin{proof}
$H=\{F_n\}$ is polynomial-time recognizable: on input $x$, compute $F_n$
for $n\le|x|$ (only those can equal $x$ since $|F_n|\ge n$) and compare.
The decider $A_H$ of LSPV-1/334 (validate; if $x\in H$ answer UNSAT; else
exhaustive search) is total and correct and has $T_{A_H}(F_n)\le\poly(n)$.
Apply $TR(R)$ to this single $A_H$.
\end{proof}
\noindent This sharpens the LSPV-1/334 necessary condition into a hard
rule for the Hypergraph \& Proof Certificates role: PHP, Tseitin on
explicit expanders, ordering principles, signed-hypergraph families
generated by a fixed program, and any ``constructed'' worst case are
useless for the universal $TR$ route, whatever $R$ is. They remain
admissible for the conditional $B3$ route (Theorem~4 says nothing about
$B3$), which is a further reason the directive is right to prioritize $B3$
over $TR$. For the non-explicit candidates: dense random 3-CNF with
$m\ge Cn^{7/5}$ clauses has polynomial-size non-satisfiability witnesses
almost surely (Feige--Kim--Ofek \cite{fko}), so it fails for any $R$ that
p-simulates the FKO certificate verifier; the constant-density regime and
Kraj\'{\i}\v{c}ek-style proof-complexity-generator $\tau$-formulas (indexed by
strings outside the generator range) remain OPEN candidates.
