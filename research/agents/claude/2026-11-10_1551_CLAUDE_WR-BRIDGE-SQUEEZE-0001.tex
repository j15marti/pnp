\begin{abstract}
The shared five-role directive targets
$\Wbar_R(N)=\max_{F\in\mathrm{UNSAT},\,|\langle F\rangle|\le N}\min_{\pi:F\vdash_R\bot}\mathrm{cost}_R(\pi)$
and the bridge $B_R:\ \mathrm{SAT}\in\mathrm P\Rightarrow \Wbar_R(N)\le\poly(N)$.
We record three facts about the choice of the fixed calculus $R$.
(1) \textbf{Squeeze (PROVED/EXACT, elementary).} For every calculus $R$ whose lower bound
$\mathrm{LB}_R$ (superpolynomial $\Wbar_R$) is already proved, $B_R$ is \emph{equivalent} to
$\mathrm{SAT}\notin\mathrm P$; for such $R$ the ``weakest sufficient bridge'' is a reformulation of
$\mathrm P\neq\mathrm{NP}$, not a weakening of it.
(2) \textbf{Reflection route (KNOWN THEOREM, composed).} For the fixed system $R_T$ of a theory
$T\supseteq\mathrm{PV}$ (with $R_{\mathrm{PV}}\equiv_p\mathrm{EF}$), $B_{R_T}$ holds whenever
$\mathrm{SAT}\in\mathrm P$ is witnessed by a polynomial-time decider whose soundness is $T$-provable.
(3) \textbf{Extensionality barrier (PROVED/EXACT).} For every sound polynomial-time decider $A$
and every consistent recursively axiomatized $T\supseteq\mathrm{PV}$ there is a polynomial-time
$A'$ with $A'(F)=A(F)$ for all $F$ and $T\nvdash\Sound_{A'}$. Hence no argument that uses the
hypothesis $\mathrm{SAT}\in\mathrm P$ only through the extension of a given decider can obtain the
provable soundness needed in (2); it must \emph{construct} a new, provably sound decider.
No lower bound on $\Wbar_R$ for any $R\ge_p\mathrm{EF}$ is claimed. $B_R$ for the historical
$R_{\mathrm{mix341}}$ ([P2-B2]) and P versus NP remain \textbf{OPEN}.
\end{abstract}

\section{Setting and quantifiers}
A Cook--Reckhow UNSAT system $R$ is a polynomial-time verifier $V_R(F,\pi)$, sound
($V_R(F,\pi)=1\Rightarrow F\in\mathrm{UNSAT}$) and complete. Following the directive, the derivation
$\pi$ is chosen \emph{freely} (optimal sequence given for free); $\mathrm{cost}_R(\pi)\ge|\pi|^{\Omega(1)}$
is assumed polynomially related to encoded proof length, so $\Wbar_R$ is polynomially equivalent to
the worst-case minimum proof size $S_R(N)$. Write
\[
\mathrm{LB}_R:\ \Wbar_R(N)\ne N^{O(1)},\qquad
B_R:\ \mathrm{SAT}\in\mathrm P\ \Rightarrow\ \Wbar_R(N)\le N^{O(1)} .
\]
$\mathrm{LB}_R\wedge B_R\Rightarrow\mathrm{SAT}\notin\mathrm P$ is the intended two-arrow spine.
The quantifier order is essential and is the one recorded in REFEREE296 and LSPV-5/409:
$R$ is fixed \emph{before} the hypothesized decider $A$; the system $Q_A$ that depends on $A$ is
trivially polynomially bounded and is excluded.

\section{Squeeze theorem}
\begin{theorem}[Squeeze; PROVED/EXACT]\label{thm:squeeze}
For every fixed $R$: if $\mathrm{LB}_R$ is a theorem, then $B_R\iff\mathrm{SAT}\notin\mathrm P$.
If $R\le_p R'$ ($R'$ polynomially simulates $R$), then $B_R\Rightarrow B_{R'}$ and
$\mathrm{LB}_{R'}\Rightarrow\mathrm{LB}_R$.
\end{theorem}
\begin{proof}
($\Rightarrow$) $B_R$ and $\mathrm{LB}_R$ give $\mathrm{SAT}\notin\mathrm P$ by contraposition.
($\Leftarrow$) If $\mathrm{SAT}\notin\mathrm P$, the antecedent of $B_R$ is false. Simulation:
$S_{R'}(N)\le\poly(S_R(\poly N))$.
\end{proof}
\begin{corollary}
$B_{\mathrm{Res}}$, $B_{\mathrm{AC^0\text{-}Frege}}$, $B_{\mathrm{CP}}$,
$B_{\mathrm{PC}}$ (all systems with known superpolynomial lower bounds:
Haken 1985; Ajtai 1988, Kraj\'{\i}\v{c}ek--Pudl\'ak--Woods, Pitassi--Beame--Impagliazzo 1993;
Pudl\'ak 1997; Razborov 1998, Alekhnovich--Razborov) are each \emph{equivalent} to $\mathrm P\ne\mathrm{NP}$.
In the world-ledger taxonomy they are equivalents of $\mathrm P\ne\mathrm{NP}$, not progress.
\end{corollary}
\begin{remark}[The trade-off]
Theorem~\ref{thm:squeeze} squeezes every choice of $R$ between two frontier problems.
If $R$ is weak enough that $\mathrm{LB}_R$ is known, $B_R$ is the whole of $\mathrm P\ne\mathrm{NP}$.
Moving $R$ up the simulation order makes $B_R$ easier (Theorem~\ref{thm:refl} gives a conditional proof
for $R\ge_p\mathrm{EF}$), but $\mathrm{LB}_R$ for $R\ge_p\mathrm{EF}$, or even for Frege, is a
long-standing open problem with no superpolynomial bound known. The directive's step
``establish superpolynomial $\Wbar_R$ in a fixed, sufficiently powerful system'' is therefore the
EF/Frege lower-bound problem as soon as $R$ is strong enough for any known $B_R$ argument.
For the historical $R_{\mathrm{mix341}}$: it contains Resolution and GF(2) steps (so Tseitin is easy,
LSPV-2/341), and both $\mathrm{LB}$ ([P2-B1]) and $B$ ([P2-B2]) are OPEN. If its rules are locally
sound and polynomial-time checkable, then $R_{\mathrm{mix341}}\le_p\mathrm{EF}$ (standard; PV proves
local soundness), so by Theorem~\ref{thm:squeeze} $B_{\mathrm{mix341}}\Rightarrow B_{\mathrm{EF}}$:
any proof of [P2-B2] contains a proof of $B_{\mathrm{EF}}$, which is subject to
Section~\ref{sec:barrier}. (The local-soundness hypothesis for the recovered /341 grammar is itself
open, [P2-D1]; this sentence is CONDITIONAL on it.)
\end{remark}

\section{The reflection route (known theorems, composed)}
\begin{definition}
For a consistent, recursively axiomatized theory $T\supseteq\mathrm{PV}$ that is sound for
$\Pi^0_1$ sentences, $R_T$ accepts $(F,\pi)$ iff $\pi$ is a $T$-proof of
$\mathrm{Unsat}(\underline F)\equiv\forall x\,\neg\mathrm{Sat}(\underline F,x)$.
For a polynomial-time machine $A$,
$\Sound_A\equiv\forall F\,\forall x\,(A(F)=0\rightarrow\neg\mathrm{Sat}(F,x))$, a $\Pi^0_1$ (indeed
$\forall\Pi^b_1$) sentence.
\end{definition}
\begin{theorem}[KNOWN THEOREM: Cook 1975; Kraj\'{\i}\v{c}ek--Pudl\'ak 1990; composition]\label{thm:refl}
If $A$ runs in time $n^k$, $A(F)=0$ for every UNSAT $F$, and $T\vdash\Sound_A$, then
$S_{R_T}(N)\le N^{O(k)}$. For $T=\mathrm{PV}$ the propositional translations give polynomial-size EF
refutations, so $S_{\mathrm{EF}}(N)\le\poly(N)$. Equivalently (contrapositive):
$\mathrm{LB}_{\mathrm{EF}}\Rightarrow$ no polynomial-time SAT decider has a PV-provable soundness statement.
\end{theorem}
\begin{proof}[Proof sketch]
An $R_T$-proof of a given UNSAT $F$ is: the fixed proof of $\Sound_A$ (size $c_A$, independent of $F$),
a $T$-proof of the true closed equation $A(\underline F)=0$ (PV proves true polynomial-time
computations on numerals by proofs polynomial in the running time), and one instantiation/modus
ponens. For $T=\mathrm{PV}$ Cook's translation turns PV proofs of $\forall\Pi^b_1$ statements into
polynomial-size EF proofs of their translations; EF plus the $\|\Sound_A\|$ tautologies
polynomially simulates the system $Q_A$ (Kraj\'{\i}\v{c}ek--Pudl\'ak 1990).
\end{proof}
\begin{remark}
$R_{\mathrm{ZFC}}$ is a single fixed system chosen before $A$. Theorem~\ref{thm:refl} thus proves
$B_{R_{\mathrm{ZFC}}}$ in every scenario where $\mathrm P=\mathrm{NP}$ has a ZFC-certified witness.
The only scenario left open for $B_{R_T}$ is: $\mathrm{SAT}\in\mathrm P$ is true but \emph{no}
polynomial-time decider has $T$-provable soundness. This localizes the residual of [P2-B2] (and the
``formal reflection of arbitrary code'' prohibition [P2-C4]) to a provability gap.
\end{remark}

\section{Extensionality barrier}\label{sec:barrier}
\begin{theorem}[PROVED/EXACT]\label{thm:ext}
Let $T\supseteq\mathrm{PV}$ be consistent and recursively axiomatized with a polynomial-time proof checker,
and let $A$ be any polynomial-time machine with $\Sound_A$ true. Define $A'(F)$: enumerate all
strings $p$ of length $\le\log_2|F|$ over $T$'s proof alphabet; if one is a $T$-proof of $0=1$,
output $0$ (``UNSAT''); otherwise output $A(F)$. Then $A'$ runs in polynomial time,
$A'(F)=A(F)$ for every $F$, and $T\nvdash\Sound_{A'}$.
\end{theorem}
\begin{proof}
There are $|\Sigma|^{\log_2|F|}=|F|^{\log_2|\Sigma|}$ candidates, each checked in polynomial time, so
$A'$ is polynomial-time. Since $T$ is consistent no candidate is a proof of $0=1$, so $A'=A$ pointwise.
Suppose $T\vdash\Sound_{A'}$. Reason in $T$: if some $p$ is a $T$-proof of $0=1$, take the explicitly
satisfiable formula $F_p=(x_1)$ padded with copies to length $\ge2^{|p|}$; $T$ proves $\mathrm{Sat}(F_p,1)$,
and $T$ proves $A'(F_p)=0$ (the enumeration finds $p$), contradicting $\Sound_{A'}$. Hence
$T\vdash\mathrm{Con}_T$, contradicting G\"odel's second incompleteness theorem.
\end{proof}
\begin{corollary}[Extensionality barrier for $B_{R_T}$; PROVED/EXACT]\label{cor:ext}
Any proof of $B_{R_T}$ that derives the needed short $R_T$-proofs from the hypothesis
``$A$ is a polynomial-time machine with $L(A)=\mathrm{SAT}$'' by reflecting on $A$ itself fails for the
extensionally identical $A'$. A valid proof of $B_{R_T}$ (and, via Theorem~\ref{thm:squeeze} and the
CONDITIONAL simulation above, of [P2-B2]) must either (a) build from any extensional witness a
\emph{different} decider whose soundness $T$ proves, or (b) produce $R_T$-proofs by a route that is not
reflection on the given code. Route (a) is a uniform ``provable-witness'' transformation for which no
candidate is known; Theorem~\ref{thm:ext} shows it cannot be the identity map on code.
\end{corollary}
\begin{remark}[Relation to the directive's free-sequence principle]
Granting the optimal derivation for free is sound for lower bounds: $\min_\pi$ is exactly
nondeterministic proof size, and $\mathrm{LB}_R$ for \emph{every} $R$ is equivalent to
$\mathrm{NP}\ne\mathrm{coNP}$ (Cook--Reckhow 1979), stronger than $\mathrm P\ne\mathrm{NP}$.
The deterministic content lives entirely in $B_R$: a polynomial-time decider is not obliged to
execute any $R$-derivation. Algorithm-to-proof extraction is a theorem only for specific algorithm
classes (DPLL $\to$ tree Resolution; CDCL $\to$ Resolution, Pipatsrisawat--Darwiche 2011,
Atserias--Fichte--Thurley 2011); Theorem~\ref{thm:ext} shows why no such extraction can rely on
provable properties of arbitrary code.
\end{remark}
