\noindent\textbf{Status.} P versus NP remains \textbf{OPEN}. This report proves (Theorem~2) that the
conditional bridge
\[
\text{B3}_R:\qquad \SAT\in P\ \Longrightarrow\ \exists C,k\ \forall N:\ \Wb_R(N)\le C(N+1)^k
\]
holds \emph{unconditionally as an implication} for one explicit, fixed, sound, complete, effectively
specified calculus $\RKP$ (a size-gated Krajíček--Pudlák reflection system), and then shows exactly where
the difficulty went: $\RKP$ is a Cook--Reckhow (polynomial-time verifiable) system only conditionally, and
a superpolynomial lower bound for $\RKP$ would already imply $\NP\ne\coNP$ (Theorem~3). The result does
\emph{not} close B3 for the historical $R_{\mathrm{mix341}}$, does not supply any lower bound, and is not
progress toward P$\ne$NP by itself. It is a precise map of the B3/LB trade-off. Additive report; no schedules,
roles, allocations, trackers, or root \texttt{main.tex} regions were changed (this routine owns none).

\section*{Fixed objects and quantifiers}
Inputs are binary strings; $\langle F\rangle$ is a fixed CNF encoding, $N=|\langle F\rangle|$. For a calculus
$R$ with a cost on proofs, $W_R(F)=\min_{\pi:F\vdash_R\bot}\mathrm{cost}_R(\pi)$ (free optimal choice of
$\pi$; proof search is not charged) and $\Wb_R(N)=\max\{W_R(F): F\in\UNSAT, |\langle F\rangle|\le N\}$,
with $\max\emptyset=0$. In this report $\mathrm{cost}_R(\pi)=|\pi|$ (bit length). Verification work is
recorded separately, as the directive requires. A \emph{Cook--Reckhow (CR) system} is a polynomial-time
verifier $V(F,\pi)$ with $V(F,\pi)=1\Rightarrow F\in\UNSAT$; every CR system is given by a clocked pair
$(M,k)$: run TM $M$ on $(F,\pi)$ for $(|F|+|\pi|+2)^k$ steps.

\section{The calculus}
\begin{definition}[$\RKP$]
An $\RKP$-proof of a CNF $F$ is either (rule 0) an ordinary Resolution refutation of $F$, or (rule 1) a
tuple $w=(\langle M,k\rangle,\pi,1^{s})$ with $n=|\langle F\rangle|$, $L=n+|\pi|$, such that
\begin{enumerate}
\item[(a)] size gate: $|\langle M,k\rangle|\le\log_2(L+2)$;
\item[(b)] padding: $s=(L+2)^k$;
\item[(c)] acceptance: $M$ accepts $(F,\pi)$ within $(L+2)^k$ steps;
\item[(d)] bounded reflection: $\mathrm{SOUND}(M,k,L)$, i.e.\ for every $(G,\sigma)$ with
$|\langle G\rangle|+|\sigma|\le L$ and every assignment $y$: if $M$ accepts $(G,\sigma)$ within
$(|\langle G\rangle|+|\sigma|+2)^k$ steps then $y$ does not satisfy $G$.
\end{enumerate}
\end{definition}
This is a size-gated, padded, finite-reflection variant of the Krajíček--Pudlák construction (JSL 54(3),
1989), stated directly at the verifier level rather than via EF proofs of propositional
$\|\mathrm{Ref}_Q\|$ formulas. Its definition does not depend on any hypothesis.

\begin{lemma}\label{lem:basic}
\status{PROVED/EXACT}
(1) $\RKP$ is sound and refutationally complete. (2) Checks (a)--(c) take time polynomial in $|w|$.
(3) The set $D=\{(\langle M,k\rangle,1^{(L+2)^k}): |\langle M,k\rangle|\le\log_2(L+2),\ \mathrm{SOUND}(M,k,L)\}$
is in $\coNP$ and is sparse (at most $L+3$ members at each relevant length). Hence the $\RKP$ verifier is a
deterministic $P^{D}$ procedure making one query to a sparse $\coNP$ set.
\end{lemma}
\begin{proof}
(1) Rule 0 is Resolution. For rule 1, (d) applied to $(G,\sigma)=(F,\pi)$ and (c) give that no $y$
satisfies $F$. Completeness: rule 0. (2) Each check is a parse or a simulation of at most $(L+2)^k\le s$
steps. (3) A witness of non-membership is $(G,\sigma,y)$ plus an accepting run of length
$\le(L+2)^k$, polynomial in the input length; at most $2^{\lfloor\log_2(L+2)\rfloor+1}\le 2(L+2)$
descriptions pass the gate, each determining one string per $L$ (sparse up to a constant factor).
\end{proof}

\section{B3 holds for $\RKP$}
\begin{theorem}[B3 for $\RKP$]\label{thm:b3}
\status{PROVED/EXACT}
If $\SAT\in P$ then there are $C,k$ with $\Wb_{\RKP}(N)\le C(N+1)^k$ for every $N$, and every such
witness proof is verifiable in time polynomial in $N$.
\end{theorem}
\begin{proof}
Let $A$ decide SAT in time $\le(n+2)^c$ (on all binary inputs; malformed inputs are rejected). Let $M_A$ ignore
$\pi$, run $A(F)$ and accept iff $A$ answers UNSAT; take $k=c+1$. Because $A$ is correct, $\mathrm{SOUND}(M_A,k,L)$
holds for every $L$. Let $n_0$ be least with $|\langle M_A,k\rangle|\le\log_2(n_0+2)$; this is a constant
depending only on $A$. For $F\in\UNSAT$ with $n\ge n_0$, $w=(\langle M_A,k\rangle,\varepsilon,1^{(n+2)^k})$ is an
$\RKP$ proof of length $O((n+2)^k)$. For $n<n_0$ use rule 0; the finitely many such $F$ contribute a constant.
Verification work: (a)--(c) are polynomial; (d) is membership in $D\in\coNP$, and $\SAT\in P$ gives
$\coNP=P$, so (d) is decided in time polynomial in $(n+2)^k$.
\end{proof}

\begin{remark}
The hypothesis is used only to obtain $A$ and to make (d) cheap; the calculus is fixed beforehand (F1 does
not fire). The argument relativizes \emph{only together with} the calculus: relative to $O$ one must use
$\RKP^{O}$ with oracle machines, so REFEREE296's barrier for oracle-free $R$ is respected (F4 does not fire).
For oracle-free $\RKP$ the argument is a non-relativizing use of $\SAT\in P$ in the unrelativized world, which
is consistent.
\end{remark}

\section{Where the difficulty went}
\begin{theorem}[sandwich for the lower bound]\label{thm:sandwich}
\status{PROVED/EXACT, modulo the cited KNOWN THEOREMS}
Let $\mathrm{LB}_{\RKP}$ be: $\Wb_{\RKP}(N)$ is not bounded by any polynomial. Then
\[
\text{PH does not collapse to }\Sigma^p_3\ \Longrightarrow\ \mathrm{LB}_{\RKP}\ \Longrightarrow\ \NP\ne\coNP .
\]
\end{theorem}
\begin{proof}
Right arrow. $\RKP$ $p$-simulates every CR system unconditionally: if $Q=(M,k)$ is CR and $\pi$ is a
$Q$-proof of $F$ with $L=n+|\pi|\ge L_0(M,k)$, then $(\langle M,k\rangle,\pi,1^{(L+2)^k})$ is an $\RKP$ proof
of size polynomial in $L$ ((d) holds because $Q$ is sound). If $\NP=\coNP$ there is a polynomially bounded CR
system (Cook--Reckhow 1979, KNOWN THEOREM), hence $\Wb_{\RKP}$ is polynomial. Left arrow (contrapositive).
If $\Wb_{\RKP}$ is polynomial, guess a short proof and verify it with one query to the sparse set $D$
(Lemma~\ref{lem:basic}); so $\UNSAT\in\NP^{D}\subseteq\NP/\mathrm{poly}$ (sparse oracle answers become
polynomial advice), i.e.\ $\coNP\subseteq\NP/\mathrm{poly}$, and Yap (1983, KNOWN THEOREM) gives
$\mathrm{PH}=\Sigma^p_3$.
\end{proof}

\begin{corollary}[the B3/LB trade-off made exact]
\status{PROVED/EXACT}
For $\RKP$ the two-premise route $\mathrm{LB}_R\wedge\mathrm{B3}_R\Rightarrow P\ne NP$ reduces to the single
premise $\mathrm{LB}_{\RKP}$, which is at least as strong as $\NP\ne\coNP$ and at most as strong as
``PH is infinite at level 3''. Conversely, for any oracle-free CR system with an already proved
superpolynomial lower bound (Resolution via Haken; REFEREE332), $\mathrm{B3}_R\iff\SAT\notin P$.
\end{corollary}
\noindent So the weakest bridge is never the bottleneck on its own: either the calculus is weak (lower bound
available, B3 is P$\ne$NP itself) or strong enough that B3 is provable (lower bound is $\NP\ne\coNP$-strength
and its verifier needs a sparse $\coNP$ check). This is classified as a \emph{reformulation / world-separator}
in the sense of the project world ledger, not progress.

\begin{proposition}[the size gate is necessary; F2 control]\label{prop:full}
\status{PROVED/EXACT}
Let $R_{\mathrm{full}}$ be $\RKP$ without gate (a). Then $\Wb_{R_{\mathrm{full}}}(N)=O(N)$ unconditionally.
\end{proposition}
\begin{proof}
For $F\in\UNSAT$ let $M_F$ accept exactly input $(F,\varepsilon)$, with $|\langle M_F\rangle|=O(n)$ and $k=1$.
$\mathrm{SOUND}(M_F,1,L)$ holds because $F$ is unsatisfiable. The proof $(\langle M_F,1\rangle,\varepsilon,1^{n+2})$
has length $O(n)$. Its check (d) is as hard as deciding UNSAT for $F$: this is the directive's banned hidden
validity oracle, so $R_{\mathrm{full}}$ is inadmissible and its linear $\Wb$ is a FAMILY FAILURE of every
proposed hard family, not a lower-bound statement.
\end{proposition}
