\section{Shared objective and notation}
For a fixed sound, complete Cook--Reckhow refutation system $R$ for CNF (3-CNF where stated), with a cost function $\mathrm{cost}_R$ that is at least the number of derived lines and at most polynomial in the proof bit length,
\[
W_R(F)=\min_{\pi:F\vdash_R\bot}\mathrm{cost}_R(\pi),\qquad
\Wbar_R(N)=\max_{\substack{F\in\mathrm{UNSAT}\\ |\langle F\rangle|\le N}} W_R(F).
\]
In the language of the directive: $F$ is a signed (multisign) clause hypergraph, an $R$-derivation is a DAG of reductions, the minimal reduced logical state of an UNSAT input is $\bot$, and $W_R(F)$ is the minimal work to reach it, with the best sequence given for free. For SAT inputs the reduced state is a witness of polynomial size, so the worst case lives on UNSAT inputs, as the directive states. Write
\[
\mathrm{B3}(R):\quad \mathrm{SAT}\in\mathrm{P}\ \Longrightarrow\ \Wbar_R(N)\le N^{O(1)},\qquad
\mathrm{LB}(R):\quad \Wbar_R(N)\ne N^{O(1)}.
\]
B3 is the directive's ``weakest sufficient bridge''. $\mathrm{B3}(R)\wedge\mathrm{LB}(R)\Rightarrow \mathrm{P}\ne\mathrm{NP}$ is immediate.

\section{New result 1: for systems with known lower bounds the bridge \emph{is} the goal}

\begin{theorem}[Collapse; PROVED/EXACT]\label{thm:collapse}
Let $R$ be a fixed system for which $\mathrm{LB}(R)$ is an unconditional theorem. Then
\[
\mathrm{B3}(R)\iff \mathrm{P}\ne\mathrm{NP}.
\]
\end{theorem}
\begin{proof}
($\Leftarrow$) If $\mathrm{P}\ne\mathrm{NP}$ then $\mathrm{SAT}\notin\mathrm{P}$ and the implication $\mathrm{B3}(R)$ holds vacuously.
($\Rightarrow$) If $\mathrm{B3}(R)$ held and $\mathrm{P}=\mathrm{NP}$, then $\mathrm{SAT}\in\mathrm{P}$ and $\Wbar_R$ is polynomially bounded, contradicting the theorem $\mathrm{LB}(R)$.
\end{proof}

\noindent\textbf{KNOWN THEOREM inputs (3-CNF, bounded width, unconditional):} $\mathrm{LB}$ holds for Resolution (Urquhart 1987, Tseitin on degree-3 expanders; Chv\'atal--Szemer\'edi 1988, random 3-CNF), for bounded-depth Frege (Ben-Sasson 2002 and Pitassi--Rossman--Servedio--Tan 2016, Tseitin on bounded-degree expanders), and for Polynomial Calculus over fields of characteristic $\ne2$ (Buss--Grigoriev--Impagliazzo--Pitassi 2001, Tseitin; Ben-Sasson--Impagliazzo 1999 and Alekhnovich--Razborov 2001, random 3-CNF). Cutting Planes lower bounds are known for random $\Theta(\log n)$-CNF (Fleming--Pankratov--Pitassi--Robere 2017; Hrube\v{s}--Pudl\'ak 2017); this report does \emph{not} assert the 3-CNF case for CP.

\begin{corollary}[Consequence for the directive]
For every system in the preceding list, proving $\mathrm{B3}(R)$ is \emph{exactly as hard as proving} $\mathrm{P}\ne\mathrm{NP}$. Hard-instance families that defeat these systems (priority~4 of the directive) supply no evidence toward a bridge; they only confirm $\mathrm{LB}(R)$, which was already known. A bridge can be genuinely weaker than the goal only for systems $R$ for which $\mathrm{LB}(R)$ is \emph{not} known, i.e.\ at Frege strength or above, where $\mathrm{LB}(R)$ is itself the central open problem of proof complexity.
\end{corollary}

\begin{remark}
The same argument applies to every restricted fragment of $R_{\mathrm{mix341}}$ on which the project has proved an unconditional superpolynomial minimum (for example a Resolution-only restriction). Any proposed ``fragment bridge'' $\mathrm{SAT}\in\mathrm{P}\Rightarrow\Wbar_{\text{fragment}}$ polynomial is equivalent to $\mathrm{P}\ne\mathrm{NP}$ and should be filed as a reformulation, not progress (compare the world-ledger rule in \texttt{AGENTS.md}).
\end{remark}

\section{New result 2: oracle and algebrization firewall for the bridge}
Say a proof of $\mathrm{B3}(R)$ \emph{relativizes} if the same argument proves, for every oracle $O$,
\[
\mathrm{B3}^O(R):\quad \mathrm{SAT}\in\mathrm{P}^O\ \Longrightarrow\ \Wbar_R(N)\le N^{O(1)}.
\]
(The right-hand side does not mention machines, so it is oracle-independent.)

\begin{theorem}[Relativization barrier for B3; PROVED/EXACT]\label{thm:rel}
For every $R$ with $\mathrm{LB}(R)$ an unconditional theorem, $\mathrm{B3}^{\mathrm{TQBF}}(R)$ is false. Hence no relativizing argument proves $\mathrm{B3}(R)$ for Resolution, bounded-depth Frege, or PC in characteristic $\ne 2$.
\end{theorem}
\begin{proof}
$\mathrm{NP}\subseteq\mathrm{PSPACE}=\mathrm{P}^{\mathrm{TQBF}}$, so $\mathrm{SAT}\in\mathrm{P}^{\mathrm{TQBF}}$, while $\Wbar_R$ is superpolynomial by $\mathrm{LB}(R)$.
\end{proof}

\begin{corollary}[Algebrization; PROVED/EXACT]\label{cor:alg}
Let $\widetilde{A}$ be the multilinear extension of $A=\mathrm{TQBF}$ over a finite field, as in Aaronson--Wigderson. Then $\mathrm{P}^{\widetilde A}\supseteq \mathrm{P}^{A}\supseteq\mathrm{NP}$, so $\mathrm{SAT}\in\mathrm{P}^{\widetilde A}$ and $\mathrm{B3}^{\widetilde A}(R)$ is false for the same systems. An argument for B3 that survives replacement of the decider by an oracle machine with oracle access to a low-degree extension therefore also fails for them.
\end{corollary}
\noindent Theorem~\ref{thm:rel} is consistent with Theorem~\ref{thm:collapse}: for these $R$, B3 is equivalent to $\mathrm{P}\ne\mathrm{NP}$, which does not relativize (Baker--Gill--Solovay 1975). For Extended Frege, $\mathrm{LB}(\mathrm{EF})$ relative to TQBF is unknown, so the theorem does not apply; that agrees with the corpus finding (LSPV-1/357) that the only known positive EF bridge, Cook's PV translation, uses the \emph{code} of the decider and is nonrelativizing.

\section{Where an honest bridge could still live (CANDIDATE / OPEN)}
\begin{enumerate}
\item \textbf{KNOWN THEOREM (Cook 1975; see LSPV-1/357):} if some polynomial-time decider $A$ has a PV (equivalently $S^1_2$-$\forall\Pi^b_1$) proof of rejection soundness, then $\Wbar_{\mathrm{EF}}$ is polynomial. Bridge gap: $\mathrm{SAT}\in\mathrm{P}\Rightarrow$ ``some correct decider is PV-provably sound'' is OPEN, and LSPV-1/357 shows it fails for \emph{all} deciders by a G\"odel wrapper.
\item \textbf{CONDITIONAL (corpus, LSPV-1/336):} if $R$ is size-optimal, then B3 holds; existence of an optimal proof system is OPEN (Kraj\'i\v{c}ek--Pudl\'ak 1989).
\item \textbf{OPEN target, stated sharply:} find a fixed $R$ with \emph{no known} $\mathrm{LB}(R)$ and an argument for $\mathrm{B3}(R)$ that is nonrelativizing in the sense of Theorem~\ref{thm:rel}, then prove $\mathrm{LB}(R)$. Theorems~\ref{thm:collapse}--\ref{thm:rel} say both halves must be done at Frege strength or above.
\end{enumerate}

\section{Literature check this run (obstacles: the bridge, and $\mathrm{LB}$ at EF strength)}
\begin{itemize}
\item Pich--Santhanam, \emph{Towards $P\ne NP$ from Extended Frege lower bounds}, arXiv:2312.08163 / ECCC TR23-199. EF lower bounds for specific tautologies imply $\mathrm{P}\ne\mathrm{NP}$ \emph{under extra hypotheses} (an $S^1_2$-provable hardness of an E function, or feasible anticheckers). \textbf{Useful structural analogy}: it is a conditional B3-type bridge for EF whose hypotheses are bounded-arithmetic provability, matching item~1 above. Not an unconditional bridge.
\item Fleming--Grosser--Pitassi--Robere, \emph{Provable Reductions in TFNP}, ECCC TR26-109 (2026). Systems $\langle\mathrm{EF},R\rangle$: a refutation is a polynomial-time reduction from the false-clause search problem to $R$ plus an EF proof that the reduction is correct; every sufficiently strong proof system is equivalent to some $\langle\mathrm{EF},R_P\rangle$. \textbf{Analogy, not a transfer}: it identifies the ``fixed combined inference system'' (Role 2) as EF-provable correctness of a reduction. The bottleneck is the same as Cook's: B3 needs \emph{provable} correctness, not truth. The false-clause search problem is total and easy, so $\mathrm{SAT}\in\mathrm{P}$ adds nothing there.
\item Lu--Santhanam--Tzameret, ITCS 2026: unconditional AC$^0[p]$-Frege hardness for a DNF family whose tautologyhood is open. Relevant only to $\mathrm{LB}$ for systems where Theorem~\ref{thm:collapse} already applies.
\item Cross-domain transfer screen: control-theory barrier certificates and coding-theory decoding certificates both need the certificate to be \emph{checkable in the target calculus}. Relying on the plant/decoder simply being correct is the same gap as above. \textbf{Misleading analogy} if used without a provability step.
\item PQ ledger: no PQ technique applies to Theorems~\ref{thm:collapse}--\ref{thm:rel}. Item~5 (worst-case/average-case architecture) is a FAILED TRANSFER for B3: it needs black-box extraction of an $R$-proof from a decider, and that is relativizing, so Theorem~\ref{thm:rel} blocks it for every $R$ with a known lower bound.
\end{itemize}
