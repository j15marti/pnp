\section{Objective, as assigned}
For a fixed Cook--Reckhow refutation system $R$ for 3-CNF (polynomial-time
verifier $V_R(F,\pi)$, sound and complete), set
\[
 s_R(F)=\min_{\pi:\,F\vdash_R\bot}\operatorname{cost}_R(\pi),\qquad
 \Wbar_R(N)=\max_{\substack{F\in\UNSAT\\ |\langle F\rangle|\le N}} s_R(F).
\]
We assume cost is faithful: $\operatorname{cost}_R(\pi)\le|\pi|\le
(|F|+\operatorname{cost}_R(\pi))^{O(1)}$ (true for unit-node DAG cost in any
system whose lines have polynomially bounded size, such as $R_{\rm mix341}$ and
Resolution). The assigned weakest bridge is
\[
 \Bridge(R):\qquad \SAT\in\Pc\ \Longrightarrow\ \Wbar_R(N)\le\mathrm{poly}(N).
\]
The user's point is correct and is kept: $s_R$ is a \emph{minimum over all
derivations}, and the search for the optimal derivation is free. This makes
$\Wbar_R$ a \emph{nondeterministic} measure. By Cook--Reckhow,
``$\Wbar_R$ superpolynomial for every $R$'' is equivalent to
$\NP\ne\mathsf{coNP}$ (KNOWN THEOREM). The bridge is the step that brings this
back to a single $R$ and to $\Pc$ versus $\NP$.

\section{Main results}
\begin{definition}[P-subset boundedness]
$\PSB(R)$ holds if for every $S\in\Pc$ with $S\subseteq\UNSAT$ there is a
polynomial $p_S$ such that $s_R(F)\le p_S(|F|)$ for all $F\in S$.
\end{definition}

\begin{theorem}[Bridge dichotomy; PROVED/EXACT]\label{thm:dich}
For every Cook--Reckhow system $R$,
\[
 \Bridge(R)\iff\bigl(\Pc\ne\NP\bigr)\ \vee\ \PSB(R).
\]
Moreover, if $\Pc=\NP$ then $\PSB(R)$ is equivalent to $R$ being polynomially
bounded.
\end{theorem}
\begin{proof}
($\Leftarrow$) If $\Pc\ne\NP$ then $\SAT\notin\Pc$ and $\Bridge(R)$ holds
vacuously. If $\PSB(R)$ and $\SAT\in\Pc$, then $S=\UNSAT$ (restricted to valid
3-CNF encodings) is a P-subset of $\UNSAT$, so $s_R(F)\le p_S(|F|)$ for all
unsatisfiable $F$, that is, $\Wbar_R(N)\le p_S(N)$.
($\Rightarrow$) Assume $\Bridge(R)$ and suppose $\Pc=\NP$. Then $R$ is
polynomially bounded, hence trivially $\PSB(R)$.
The final sentence follows: under $\Pc=\NP$, $\UNSAT\in\Pc$ gives one direction,
and polynomial boundedness implies $\PSB$ for the other.
\end{proof}

\begin{lemma}[Explicit families are P-subsets]\label{lem:explicit}
Let $(F_n)$ be unsatisfiable, computable from $1^n$ in polynomial time, with
$|F_n|\ge n$. Then $S=\{F_n:n\ge1\}\in\Pc$.
\end{lemma}
\begin{proof}
On input $G$, for $n=1,\dots,|G|$ build $F_n$ (time $\mathrm{poly}(n)$) and
compare it with $G$. Any $n$ with $F_n=G$ satisfies $n\le|F_n|=|G|$.
\end{proof}

\begin{corollary}[Explicit hardness collapses the bridge; PROVED/EXACT]\label{cor:explicit}
If $R$ has superpolynomial $s_R(F_n)$ on some polynomial-time-constructible
UNSAT family, then $\neg\PSB(R)$, so $\Bridge(R)\iff\Pc\ne\NP$.
A proof of $\Bridge(R)$ for such an $R$ is a proof of $\Pc\ne\NP$; there is
nothing weaker to prove.
\end{corollary}

Known instances (KNOWN THEOREMS, explicit 3-CNF Tseitin formulas on explicit
bounded-degree expanders): Resolution (Urquhart 1987; Ben-Sasson--Wigderson
2001), Polynomial Calculus over characteristic $\ne2$
(Buss--Grigoriev--Impagliazzo--Pitassi 2001), and bounded-depth Frege
(Pitassi--Rossman--Servedio--Tan 2016; Håstad 2021; Håstad--Risse, SICOMP
2025, which improves the exponent to roughly $n^{1/d}$). For these $R$,
$\Bridge(R)$ is \emph{exactly} $\Pc\ne\NP$. This sharpens the corpus remark
(codex 2026-11-11 SAT\_POLYNOMIAL\_ASSUMPTION\_AUDIT, \S12) that the bridge is
``logically equivalent to the separation once a lower bound is proved''.
Here the lower bound only has to hold on some P-subset, and the same conclusion
reaches every system for which this is known.

\begin{theorem}[PSB systems exist iff optimal systems exist; PROVED/EXACT]\label{thm:opt}
Some Cook--Reckhow system $R$ satisfies $\PSB(R)$ if and only if an optimal
proof system for UNSAT 3-CNF exists, that is, a system $Q^\ast$ with
$s_{Q^\ast}(F)\le p_Q(|F|+s_Q(F))$ for every system $Q$.
\end{theorem}
\begin{proof}
($\Leftarrow$) Let $Q^\ast$ be optimal and $S\in\Pc$, $S\subseteq\UNSAT$.
Define $Q_S$: the string $0F$ proves $F$ whenever $F\in S$; the string $1\tau$
proves $F$ when $\tau$ is a complete truth table refuting $F$. Then $Q_S$ is a
Cook--Reckhow system with $s_{Q_S}(F)=|F|+1$ on $S$. Optimality gives
$s_{Q^\ast}(F)\le p(|F|)$ on $S$, so $\PSB(Q^\ast)$ holds.

($\Rightarrow$) Assume $\PSB(R)$. Fix an enumeration $(M_i,k_i)$ of clocked
deterministic machines. Machine $M_i$ accepts $(x,y)$ when it does so within
$(|x|+|y|)^{k_i}+k_i$ steps. For $n,m$, let $\mathrm{Ref}_{i,n,m}$ be the
3-CNF, obtained by the Tseitin encoding, of the circuit statement
``$M_i$ accepts $(x,y)$ with $|x|=n,|y|=m$, and assignment $z$ satisfies the
3-CNF encoded by $x$''. It is constructible from $(i,1^n,1^m)$ in time
$\mathrm{poly}_i(n+m)$ and has size at least $n+m$. For fixed $i$ the family
$\{\mathrm{Ref}_{i,n,m}\}$ is in $\Pc$, by the argument of
Lemma~\ref{lem:explicit} over the pairs $(n,m)$. If $M_i$ is sound, meaning it
never accepts a pair $(x,y)$ with $x$ satisfiable, then every member is
unsatisfiable, and $\PSB(R)$ gives
$s_R(\mathrm{Ref}_{i,n,m})\le p_i(n+m)$.

Define $Q^\ast$. A proof of $F$ is either a truth-table refutation, or a tuple
$\langle i,\pi,1^{|\mathrm{Ref}_{i,|F|,|\pi|}|},\rho\rangle$ such that
$M_i$ accepts $(F,\pi)$ within its clock and $V_R$ accepts $\rho$ as a
refutation of $\mathrm{Ref}_{i,|F|,|\pi|}$. The padding makes verification time
polynomial in the proof length.
\emph{Soundness:} $\rho$ certifies that $\mathrm{Ref}_{i,|F|,|\pi|}$ is
unsatisfiable. Fixing $x=F$ and $y=\pi$ satisfies the acceptance part, so no
$z$ satisfies $F$. This holds even if $M_i$ is unsound on other inputs.
\emph{Optimality:} every Cook--Reckhow system $Q$ has a sound clocked verifier
$M_i$. Given a shortest $Q$-proof $\pi$ of $F$,
$s_{Q^\ast}(F)\le |i|+|\pi|+\mathrm{poly}_i(|F|+|\pi|)+p_i(|F|+|\pi|)$,
which is a polynomial in $|F|+s_Q(F)$.
\end{proof}

\begin{corollary}[CONDITIONAL on the KP conjecture]\label{cor:kp}
If no optimal proof system exists (the Krajíček--Pudlák conjecture, OPEN), then
$\Bridge(R)\iff\Pc\ne\NP$ for \emph{every} Cook--Reckhow system $R$.
\end{corollary}
\begin{proof}
By Theorem~\ref{thm:opt}, $\neg\PSB(R)$ for every $R$. Then apply
Theorem~\ref{thm:dich}.
\end{proof}

\begin{corollary}[Constructivity barrier for the joint programme; PROVED/EXACT]\label{cor:constr}
Suppose the two user priorities are achieved by one $R$ through a non-vacuous
bridge: $\PSB(R)$ and $\Wbar_R$ superpolynomial. Then
(i) $\Pc\ne\NP$;
(ii) an optimal proof system exists;
(iii) no polynomial-time-constructible UNSAT family is hard for $R$, so the
lower bound on $\Wbar_R$ can only be witnessed non-constructively.
\end{corollary}
\begin{proof}
(i) follows from Theorem~\ref{thm:dich}, (ii) from Theorem~\ref{thm:opt}, and
(iii) from Lemma~\ref{lem:explicit} together with $\PSB(R)$.
\end{proof}

\section{What this means for the five assignments}
\begin{itemize}
\item \textbf{General SAT Lower-Bound Lead (weakest transfer theorem).}
The weakest sufficient bridge is now identified exactly. Apart from proving
$\Pc\ne\NP$ outright, the only route to $\Bridge(R)$ is to prove $\PSB(R)$:
$R$ must have short proofs of every polynomial-time-recognizable set of
unsatisfiable formulas. By Theorem~\ref{thm:opt}, that alone would settle the
open existence question for optimal proof systems. Krajíček--Pudlák (1989)
show this existence follows from $\mathsf{NE}=\mathsf{coNE}$; Sadowski (2002)
shows it is equivalent to a recursive presentation of the easy subsets of
TAUT. The bridge is not a weakening of the problem in any useful sense.
\item \textbf{SAT-Circuit Reduction Builder (fixed combined system).}
To have any chance at $\PSB$, the fixed system must at least be closed under the
reflection tautologies of every polynomial-time solver. That means it must sit
at the level of Extended Frege or above. Resolution-type rule sets on the
multisign DAG hypergraph cannot satisfy $\PSB$ (Corollary~\ref{cor:explicit}).
$R_{\rm mix341}$ with paid VARDEF has extension power. No lower bound is known
for it, and none would be expected short of an EF-type breakthrough.
\item \textbf{Hypergraph \& Proof Certificates (hard UNSAT families).}
Every \emph{explicit} family you construct and prove hard for $R$ certifies
$\neg\PSB(R)$, and with it $\Bridge(R)\equiv\Pc\ne\NP$
(Corollary~\ref{cor:explicit}). Explicit hard families are still valuable as
hostile controls and as falsifiers of pointwise transfer, but they cannot feed a
non-circular bridge.
\item \textbf{Time-Space Resource Auditor.}
The equivalence is insensitive to polynomial changes in the cost model, given
cost faithfulness. Audit that $\operatorname{cost}_R$ is polynomially related
to $|\pi|$ for the chosen rules, in particular VARDEF cost against line size.
\item \textbf{Independent Proof Red Team.}
Use Corollaries~\ref{cor:explicit} and~\ref{cor:kp} as the standing test. Any
proposed proof of $\Bridge(R)$ that applies to an $R$ with a known explicit
lower bound is a claimed proof of $\Pc\ne\NP$, and must be audited as one
(relativization: Theorem~\ref{thm:dich} relativizes, so it gives no barrier
evasion by itself). Pointwise simulation, $s_R(F)\le\mathrm{poly}(T_A(F))$, is
already FALSIFIED by hard-coding: a decider that recognises $F_n$ syntactically
and otherwise runs brute force. This repeats the corpus result TMR-4/041, which
we do not count as new.
\end{itemize}
