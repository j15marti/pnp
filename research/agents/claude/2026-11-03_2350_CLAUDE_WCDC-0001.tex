\section{Setting (exact definitions)}
A \emph{derivation system} $R$ is a Cook--Reckhow proof system for $\UNSAT$: a polynomial-time
verifier $V_R(F,\pi)$ that is sound ($V_R(F,\pi)=1\Rightarrow F\in\UNSAT$) and complete.
A ``multisign DAG hypergraph reduction sequence'' in the directive is a derivation
$\pi$ from $F$ to $\bot$ in a fixed finite rule set; any such rule set with
polynomial-time checkable steps is an instance of this definition. The cost is
$\operatorname{cost}_R(\pi)=|\pi|$ (DAG size; width/space variants in \S5).
\[
\Wbar_R(N)=\max_{F\in\UNSAT,\ |\langle F\rangle|\le N}\ \min_{\pi:F\vdash_R\bot}\operatorname{cost}_R(\pi).
\]

\begin{proposition}[free strategy is exactly nondeterminism; PROVED/EXACT]\label{p:free}
$\Wbar_R(N)$ is the worst-case proof size of $R$. Granting the optimal sequence for free
costs nothing in meaning: it \emph{is} the nondeterministic (NP-style) quantity.
Hence $\Wbar_R\le\mathrm{poly}$ for some $R$ iff $\mathsf{NP}=\mathsf{coNP}$ (Cook--Reckhow 1979).
\end{proposition}
\begin{proof} Immediate from definitions; the equivalence is the Cook--Reckhow theorem.\end{proof}

Consequence: a lower bound on $\Wbar_R$ for \emph{one fixed} $R$ is a proof-complexity lower
bound. It says nothing about $\mathsf P$ vs $\mathsf{NP}$ unless a bridge is added, and by
Prop.~\ref{p:free} it cannot say anything about $\mathsf{NP}$ vs $\mathsf{coNP}$ either.

\section{The bridge $B_R:\ \mathrm{SAT}\in\mathsf P\Rightarrow\Wbar_R(N)\le\mathrm{poly}(N)$}

\begin{theorem}[bridge dichotomy; PROVED/EXACT given cited theorems]\label{t:dich}
\begin{enumerate}
\item If $\Wbar_R$ is already known to be superpolynomial, then $B_R$ is \emph{equivalent} to
$\mathrm{SAT}\notin\mathsf P$, i.e.\ to $\mathsf P\ne\mathsf{NP}$. This covers Resolution (Haken 1985;
Urquhart 1987; Chv\'atal--Szemer\'edi 1988), Cutting Planes (Pudl\'ak 1997), Polynomial Calculus
and bounded-depth Frege (Ajtai 1994; Beame et al.\ 1992/1996). So for these $R$ the
``weakest sufficient bridge'' \emph{is} the target and gives no progress
(ledger category: equivalent to $\mathsf P\ne\mathsf{NP}$).
\item For $R$ with no known superpolynomial lower bound (Frege, Extended Frege (EF), stronger),
$B_R$ is not known to hold, and a superpolynomial bound on $\Wbar_R$ is itself a famous
open problem.
\item Consequently any route through $B_R$ needs an $R$ that is simultaneously (a) strong
enough for $B_R$ to be plausible and (b) weak enough for a lower bound to be provable.
No such $R$ is known; at present (a) forces $R\supseteq$ Frege-strength (Thm.~\ref{t:bridge}) and (b)
is open for Frege.
\end{enumerate}
\end{theorem}
\begin{proof} (1) $B_R$ is an implication whose consequent is known false, so it is
equivalent to the negation of its antecedent. (2),(3) restate the state of the literature
and Thm.~\ref{t:bridge}.\end{proof}

\section{A conditional bridge: it costs exactly the certified soundness of the decider}

Let $M$ decide SAT in time $N^c$. Let $\mathrm{Tr}_M(x,w)$ be the CNF encoding of the deterministic
run of $M$ on input bits $x$ with trace variables $w$, and let $\mathrm{out}(w)$ be the
output bit ``UNSAT''. Define the tautology
\[
\mathrm{Sound}_{M,N}(x,y,w)\;:=\;\neg\big[\mathrm{Tr}_M(x,w)\wedge \mathrm{out}(w)\wedge \mathrm{Eval}(x,y)\big],
\]
where $\mathrm{Eval}(x,y)=1$ iff the CNF encoded by $x$ is satisfied by $y$. It has size $\mathrm{poly}(N)$ and is a
tautology iff $M$ never answers ``UNSAT'' on a satisfiable input.

\begin{theorem}[feasible-correctness bridge; PROVED/EXACT, standard translation]\label{t:bridge}
Let $R$ be a Frege system or any system that p-simulates it and is closed under substitution of
constants for variables. If $M$ decides SAT in polynomial time and $\mathrm{Sound}_{M,N}$ has $R$-proofs of size
$\mathrm{poly}(N)$ (hypothesis $H_{R,M}$), then $\Wbar_R(N)\le\mathrm{poly}(N)$.
\end{theorem}
\begin{proof} Fix $F\in\UNSAT$, $|\langle F\rangle|=N$. Substitute the constants $x:=\langle F\rangle$ into the
given proof of $\mathrm{Sound}_{M,N}$ (size unchanged up to constants). With $x$ constant,
$\mathrm{Tr}_M(\langle F\rangle,w)$ is a CNF whose clauses are locally propagating; unit propagation
along the time layers derives each $w_t$ and hence $\mathrm{out}(w)$ in $\mathrm{poly}(N)$ steps (this is Resolution-sized,
hence Frege-sized). Also $\mathrm{Eval}(\langle F\rangle,y)$ reduces by constant evaluation to the conjunction of the
clauses of $F$, available as hypotheses. Modus ponens and conjunction rules, available in Frege, give $\bot$. Total size is polynomial.
\end{proof}

\begin{remark}[what the theorem says about the directive]
The bridge is not ``free'' from $\mathrm{SAT}\in\mathsf P$: the extra input is a \emph{short $R$-proof
that the decider is sound}. The difficulty cannot be removed by the free choice of strategy,
because a universal system ``accept $(M,\text{run})$'' is unsound without that proof
(soundness of $M$ is a coNP statement). This is the Cook--Reckhow/Krajíček--Pudlák
correspondence between provability in $\mathsf{PV}$/bounded arithmetic and EF proof size. Whether
$\mathsf P=\mathsf{NP}\Rightarrow$ ``EF is p-bounded'' holds \emph{unconditionally} is OPEN (CANDIDATE
statement for the Red Team, not claimed).
\end{remark}

\begin{remark}[direction]
The converse fails to be informative: $\Wbar_R\le\mathrm{poly}$ implies only $\mathsf{NP}=\mathsf{coNP}$,
which is consistent with $\mathsf P\ne\mathsf{NP}$. So even a proved equivalence of the form
``$\Wbar_R$ poly $\iff$ $\mathrm{SAT}\in\mathsf P$'' would be a statement strictly stronger than anything the
proof-system quantity alone can deliver; one-sided $B_R$ plus a lower bound is the only logical
shape that works: $B_R\wedge[\Wbar_R\ \text{superpoly}]\Rightarrow\mathsf P\ne\mathsf{NP}$.
\end{remark}

\section{Fixed combined inference systems and strategy switching (item 4 of the directive)}
\begin{proposition}[PROVED/EXACT]\label{p:union}
For finitely many systems $R_1,\dots,R_k$ let $R_\cup$ allow any step of any $R_i$ (adaptive switching,
mid-proof). Then $\Wbar_{R_\cup}\le\min_i\Wbar_{R_i}$, and $R_\cup$ p-simulates each $R_i$.
A lower bound for $R_\cup$ must therefore beat the strongest component.
\end{proposition}
So a ``sufficiently powerful combined system'' that contains Frege-level steps inherits the Frege barrier;
one without them inherits the unconditional lower bounds and hence (Thm.~\ref{t:dich}.1) a bridge equivalent
to $\mathsf P\ne\mathsf{NP}$. There is no middle ground at present.

\section{Time/space audit (Time--Space Auditor role)}
\begin{itemize}
\item Size, width, and clause-space are distinct costs. Resolution has strong known space and
size--space trade-off bounds (Atserias--Dalmau 2008 width$\Rightarrow$space; Ben-Sasson--Nordstr\"om 2011
trade-offs for pebbling formulas): KNOWN THEOREMS, but all are for systems already below the bridge threshold
(Thm.~\ref{t:dich}.1), so they cannot be turned into SAT-algorithm lower bounds without the bridge.
\item In Thm.~\ref{t:bridge} the proof size is $\mathrm{poly}$ but the peak space of the derived
proof is $\Theta(\text{trace length})$ unless the trace is re-derived layerwise; any space-bounded
version of the bridge needs an analogous space-bounded soundness proof (CANDIDATE; not worked out).
\item Charging the derivation cost as $\mathrm{poly}(N)$ time versus the SAT algorithm's time: the bridge loses
a polynomial factor in the exponent (degree depends on the length of the soundness proof), so it
can be used only for the P-versus-superpolynomial boundary, not for fine-grained ARC/FLMV exponents.
\end{itemize}

\item Pich--Santhanam, \emph{Strong co-nondeterministic lower bounds for NP cannot be proved feasibly}, STOC 2021:
$\mathsf{PV}$ cannot prove that an NP language has no subexponential co-nondeterministic circuits (unconditional
unprovability). Follow-up: Li--Oliveira, arXiv:2305.15235 and arXiv:2404.11841 (unprovability of fixed-polynomial
lower bounds in weak theories). \emph{Relevance:} it explains why Thm.~\ref{t:bridge}'s hypothesis
$H_{R,M}$ is the right obstruction: lower bounds that are themselves feasibly provable are obstructed.
\emph{Classification: ANALOGY needing a new transfer theorem} (the models differ: unprovability in a theory vs.\ size

near-optimal Resolution refutations is NP-hard. \emph{Relevance:} confirms the directive's observation that strategy
search is genuinely harder than execution, so free strategy gives a valid lower-bound reading (Prop.~\ref{p:free}); it also shows that for Resolution
the min over sequences is far from computable cheaply. KNOWN THEOREM, directly consistent; no new bridge.

and Krajíček, \emph{Proof Complexity Generators} (Cambridge, 2025): the generator-tautology programme is the
only known route toward lower bounds beyond Resolution-like systems, and it is explicitly tied to
