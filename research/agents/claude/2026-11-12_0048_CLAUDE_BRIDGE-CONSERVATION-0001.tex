\section{Objects}
Fix a Cook--Reckhow proof system $R$ for UNSAT CNF: a polynomial-time checker $V_R(F,\pi)$ that is
sound and complete. Fix a cost $\operatorname{cost}_R$ polynomially related to proof length:
$|\pi|^{1/c}-c\le\operatorname{cost}_R(\pi)\le c(|\pi|+1)^c$ (this holds for unit-paid distinct-node
cost when node encodings are polynomial, cf.\ [P2-D3]; it must be verified separately for each
concrete system). Following the directive and [P2-D2],
\[
W_R(F)=\min_{\pi:F\vdash_R\bot}\operatorname{cost}_R(\pi),\qquad
\Wb_R(N)=\max_{F\in\UNSAT,\ |\langle F\rangle|\le N}W_R(F).
\]
The minimum ranges over all valid derivations, chosen freely; the search for the best derivation is
\emph{not} charged. Write
\begin{itemize}
\item $\mathbf{B1}(R)$: $\Wb_R(N)\ne N^{O(1)}$ (proof2 [P2-B1]);
\item $\mathbf{B2}(R)$: $\SAT\in P\Rightarrow\Wb_R(N)\le\poly(N)$ (proof2 [P2-B2], the
weakest sufficient bridge of the directive).
\end{itemize}

\section{The conservation law}
\begin{theorem}[\status{PROVED/EXACT}]\label{thm:cons}
For every fixed $R$:
\begin{enumerate}
\item[(a)] $\mathbf{B1}(R)\wedge\mathbf{B2}(R)\Rightarrow P\neq NP$.
\item[(b)] $P\ne NP\Rightarrow\mathbf{B2}(R)$. Hence $\mathbf{B2}(R)$ has no counterexample unless
$P=NP$; red-teaming $\mathbf{B2}(R)$ by searching for counterexamples cannot succeed short of
exhibiting $P=NP$.
\item[(c)] If $\mathbf{B1}(R)$ is a theorem, then $\mathbf{B2}(R)\iff P\ne NP$.
\item[(d)] If $R'$ polynomially cost-simulates $R$ (each $R$-refutation of $F$ maps to an
$R'$-refutation of $F$ of cost polynomial in the original cost and $|F|$), then
$\mathbf{B2}(R)\Rightarrow\mathbf{B2}(R')$ and $\mathbf{B1}(R')\Rightarrow\mathbf{B1}(R)$.
\end{enumerate}
\end{theorem}
\begin{proof}
(a) Contrapositive of $\mathbf{B2}$, using Cook--Levin for $\SAT\notin P\Rightarrow P\ne NP$.
(b) If $\SAT\notin P$ the implication $\mathbf{B2}(R)$ is true because its hypothesis is false.
(c) ``$\Leftarrow$'' is (b); ``$\Rightarrow$'' is (a) with $\mathbf{B1}(R)$ discharged.
(d) Simulation gives $W_{R'}(F)\le p(W_R(F)+|F|)$ for a fixed polynomial $p$, hence
$\Wb_{R'}(N)\le p(\Wb_R(N)+N)$ (empty maxima are $0$). So a polynomial bound on $\Wb_R$, whether
or not it is derived under $\SAT\in P$, gives one on $\Wb_{R'}$; this proves
$\mathbf{B2}(R)\Rightarrow\mathbf{B2}(R')$, and its contrapositive gives
$\mathbf{B1}(R')\Rightarrow\mathbf{B1}(R)$.
\end{proof}

\begin{corollary}[\status{PROVED/EXACT given cited KNOWN THEOREMS}]\label{cor:known}
For each of the following systems $S$, $\mathbf{B1}(S)$ is a known theorem, hence
$\mathbf{B2}(S)\iff P\ne NP$: ordinary DAG Resolution (Haken 1985; Urquhart 1987;
Chv\'atal--Szemer\'edi 1988; Ben-Sasson--Wigderson 2001); Cutting Planes (Pudl\'ak 1997);
bounded-depth Frege (Ajtai 1988; Kraj\'{\i}\v{c}ek--Pudl\'ak--Woods 1995; Pitassi--Beame--Impagliazzo 1993);
Polynomial Calculus (Razborov 1998; Alekhnovich--Razborov 2001); the corpus-recorded fixed
syntactic $\mathrm{Res}(PC_1/\mathbb F_3)$ via Pang arXiv:2610.00837v2 Thm.\ 7.2 (as recorded in
[P2-B3 evidence, LSPV-1/482, 492]; not re-audited here).
\end{corollary}

\begin{remark}
The directive's clarification that the optimal derivation may be chosen for free is sound for
$\mathbf{B1}$: a lower bound on $W_R$ lower-bounds every procedure that must produce an $R$-proof.
It does not help $\mathbf{B2}$: Corollary~\ref{cor:known} already grants free optimal choice to
Resolution, and $\mathbf{B2}(\mathrm{Res})$ is still exactly as hard as $P\ne NP$. This sharpens,
for natural unpadded systems, the generic-property firewall of LSPV-1/462 ($R_{\mathrm{clock}}$) and
the $Q_A$-simulation equivalence of LSPV-5/409.
\end{remark}

\section{A Resolution test for proposed bridge arguments}\label{sec:restest}
\begin{theorem}[\status{PROVED/EXACT}; red-team filter]\label{thm:restest}
Let $\Phi$ be a property of proof systems and suppose an argument establishes
$\forall R\,[\Phi(R)\Rightarrow\mathbf{B2}(R)]$. If $\Phi(S)$ holds for any $S$ in
Corollary~\ref{cor:known}, the same argument proves $P\neq NP$ unconditionally.
\end{theorem}
\begin{proof}
$\Phi(S)\Rightarrow\mathbf{B2}(S)$, and $\mathbf{B1}(S)$ is known; apply Theorem~\ref{thm:cons}(a).
\end{proof}
\noindent\textbf{Consequence for all five roles.} Each of the following is shared by Resolution and
therefore cannot, by itself or in combination, carry the bridge: soundness; refutational completeness;
polynomial-time checking; containing the Resolution rule; DAG sharing with paid-once nodes; free
choice of the optimal derivation; closure under restriction; implicational completeness; locality of
rules; bounded fan-in. A bridge argument for $R_{\mathrm{mix341}}$ must use a property that
\emph{fails for Resolution, Cutting Planes, PC, bounded-depth Frege and Res$(PC_1/\mathbb F_3)$}.
Any proposed bridge write-up should state that property explicitly and show that it fails for each
of those systems; otherwise the argument is either a full proof of $P\neq NP$ or wrong.
(Status of the filter: \status{PROVED/EXACT}; whether any such property exists: \status{OPEN}.)

\section{The only known bridge mechanism: reflection}
\begin{theorem}[\status{KNOWN THEOREM}; Cook 1975, Kraj\'{\i}\v{c}ek--Pudl\'ak 1989]\label{thm:refl}
Let $A$ be a polynomial-time algorithm and suppose Cook's theory $PV$ proves
$\forall F\,[A(F)=0\rightarrow F\text{ is unsatisfiable}]$. Then the propositional translations of
this statement have polynomial-size Extended Frege proofs, so $EF$ polynomially simulates the
system $Q_A$ of LSPV-5/409. If moreover $A$ decides $\SAT$, then $\Wb_{EF}(N)\le\poly(N)$.
\end{theorem}
\begin{corollary}[\status{CONDITIONAL}]
$\mathbf{B2}(EF)$ follows from the hypothesis
$\mathrm{H}_{PV}$: ``if $\SAT\in P$ then some polynomial-time $\SAT$ algorithm is $PV$-provably
correct on its rejections.'' $\mathrm{H}_{PV}$ is OPEN and is the precise gap: a true but
$PV$-unprovably correct $\SAT$ algorithm is not excluded by anything known. Contrapositively
(\status{KNOWN THEOREM}), an $EF$ lower bound yields only that no polynomial-time algorithm is
$PV$-provably a correct $\SAT$ refuter, a consistency statement, not $P\ne NP$.
\end{corollary}
\noindent\textbf{Literature clue (KNOWN, scope as stated by the authors).} Arteche, Khaniki, Pich and
Santhanam (ICALP 2024, arXiv:2405.02232) obtain a \emph{conditional} bridge of exactly this
reflection type for Implicit EF: if $iEF$ efficiently proves a standard hardness hypothesis, then any
superpolynomial $iEF$ lower bound implies $\#P\not\subseteq FP/\mathrm{poly}$. Their mechanism
is a formalization of the soundness of the LFKN sum-check protocol inside $iEF$. They also record
an observation of Pich--Santhanam (2023): an unconditional link ``$EF$ lower bounds $\Rightarrow$
$NP\not\subseteq P/\mathrm{poly}$'' would already imply $NEXP\not\subseteq P/\mathrm{poly}$.
\status{ANALOGY/CANDIDATE}: a uniform analogue of that meta-barrier for $\mathbf{B2}(EF)$ is not
derived here; transfer requires a new argument.

\section{Where $R_{\mathrm{mix341}}$ must sit: a dichotomy}
\begin{proposition}[\status{CONDITIONAL} on the open grammar node P2-D1]\label{prop:dich}
Exactly one of the following describes the route through $R_{\mathrm{mix341}}$.
\begin{enumerate}
\item[(i)] \emph{Weak position.} $R_{\mathrm{mix341}}$ is polynomially cost-simulated by some $S$
with a known lower bound for the same hard families (for example, if paid DEF is restricted
to bounded-support definitions over original variables, as in the restricted calculi of
[P2-L1-499] and LSPV-1/503). Then once $\mathbf{B1}(R_{\mathrm{mix341}})$ is proved,
$\mathbf{B2}(R_{\mathrm{mix341}})\iff P\ne NP$ (Theorem~\ref{thm:cons}(c)), and
Theorem~\ref{thm:restest} applies with no reflection mechanism available.
\item[(ii)] \emph{Strong position.} $R_{\mathrm{mix341}}$ polynomially simulates a system with no known
lower bound. Two concrete thresholds: (ii-a) if its GF(2) rows combine with clauses so that it
simulates DAG-like Res$(\oplus)$, then $\mathbf{B1}(R_{\mathrm{mix341}})$ implies a superpolynomial
DAG-like Res$(\oplus)$ lower bound, still OPEN in the literature (known only for tree-like, regular,
and bounded-depth fragments: Efremenko--Garl\'{\i}k--Itsykson; Alekseev--Itsykson;
Bhattacharya--Chattopadhyay arXiv:2507.23008; Byramji--Impagliazzo); (ii-b) if paid DEF admits
unrestricted extension definitions, $R_{\mathrm{mix341}}\ge_p$ Extended Resolution $\equiv_p EF$
(Cook--Reckhow 1979), so $\mathbf{B1}$ is the EF lower-bound problem, and the only known
$\mathbf{B2}$ mechanism is Theorem~\ref{thm:refl}.
\end{enumerate}
\end{proposition}
\noindent(The two cases are exhaustive in the sense that any known lower bound either applies
to a simulating system or does not; ``exactly one'' refers to the classification relative to
the current literature, which can change.)
