\section{Target, model and quantifiers}

\begin{definition}
A \emph{refutation system} $R$ is a Cook--Reckhow system for UNSAT: a polynomial-time
verifier $V_R(F,\pi)$ such that $F\in\UNSAT$ iff $\exists\pi\,V_R(F,\pi)=1$. A cost
$\mathrm{cost}_R(\pi)\ge|\pi|^{\Omega(1)}$ is fixed with $R$ (size, or a charged work measure on the
multisign DAG hypergraph derivation; any polynomially related choice gives the same
polynomial/superpolynomial verdict). Set
\[
\Wbar_R(N)=\max_{F\in\UNSAT,\ |\langle F\rangle|\le N}\ \min_{\pi:\,F\vdash_R\bot}\mathrm{cost}_R(\pi).
\]
The minimum is over all valid derivations, granting free knowledge of the best one,
exactly as the directive permits.
\end{definition}

\noindent\textbf{Bridge}$(R)$: \quad $\SAT\in\mathrm{P}\implies\Wbar_R(N)\le\mathrm{poly}(N)$.

\noindent\textbf{Candidate of this run.} Determine the exact logical strength of
Bridge$(R)$ for fixed $R$, and test the pointwise premise ``every SAT algorithm $A$ has
$\mathrm{time}_A(F)\ge\min_\pi\mathrm{cost}_R(\pi)$.''
Falsifiers tried: (i) a fixed $R$ with a known unconditional lower bound;
(ii) an oracle world with $\mathrm{P}^O=\mathrm{NP}^O$; (iii) a polynomial-time algorithm
outside $R$ on $R$-hard instances.

\section{Results}

\begin{proposition}[\status{PROVED/EXACT}]\label{prop:eq}
For every fixed $R$:\quad Bridge$(R)\iff\bigl(\mathrm{P}\neq\mathrm{NP}\ \vee\ \Wbar_R\le\mathrm{poly}\bigr)$.
\end{proposition}
\begin{proof}
Bridge$(R)$ is the implication $\neg(\mathrm{P}\ne\mathrm{NP})\to(\Wbar_R\le\mathrm{poly})$,
which is logically the disjunction.
\end{proof}

\begin{corollary}[\status{PROVED/EXACT} from \status{KNOWN THEOREM}s]\label{cor:weak}
If $\Wbar_R$ is unconditionally superpolynomial, then Bridge$(R)\iff\mathrm{P}\neq\mathrm{NP}$.
This holds (known theorems) for $R\in\{$tree-like resolution, resolution (Haken 1985 pigeonhole;
Urquhart 1987 Tseitin), cutting planes (Pudl\'ak 1997), polynomial calculus (Razborov 1998;
Buss--Grigoriev--Impagliazzo--Pitassi Tseitin mod $q$), bounded-depth Frege (Ajtai; Beame--Impagliazzo--Pitassi; Kraj\'i\v{c}ek--Pudl\'ak--Woods)$\}$.
\end{corollary}

\noindent\emph{Consequence for the program.} For every system where the lower-bound half is
already a theorem, the ``weakest sufficient bridge'' is not weaker than the goal: it
\emph{is} P $\neq$ NP. Building hard UNSAT families for such $R$ (the Hypergraph \&
Proof Certificates role) re-proves known results unless $R$ is beyond the current frontier
(Frege, Extended Frege). For Frege/EF the lower-bound half is itself OPEN.
A statement equivalent to P $\neq$ NP is a reformulation, not progress
(cf.\ world ledger, \texttt{research/agents/claude/world-ledger/}).

\begin{proposition}[\status{PROVED/EXACT}; relativization gate]\label{prop:rel}
Let $R$ be any fixed system with $\Wbar_R$ superpolynomial (Corollary~\ref{cor:weak}).
Then Bridge$(R)$ has no relativizing proof.
\end{proposition}
\begin{proof}
Take the Baker--Gill--Solovay oracle $O$ (e.g.\ a PSPACE-complete set) with
$\mathrm{P}^O=\mathrm{NP}^O$, so $\SAT\in\mathrm{P}^O$. The quantity $\Wbar_R$ is
defined without reference to any oracle, so it stays superpolynomial. Hence the relativized
implication $\SAT\in\mathrm{P}^O\Rightarrow\Wbar_R\le\mathrm{poly}$ is false relative to $O$.
\end{proof}
\noindent So any proof of the bridge must contain a step that uses the code of the
hypothetical SAT algorithm in a non-black-box way. The natural non-relativizing
mechanism is \emph{provability}:

\begin{theorem}[\status{KNOWN THEOREM}: Cook 1975; Kraj\'i\v{c}ek--Pudl\'ak 1990]\label{thm:pv}
If a polynomial-time algorithm $A$ decides SAT and the theory PV (or $S^1_2$) proves
``$A(F)=0\Rightarrow F$ is unsatisfiable,'' then Extended Frege has polynomial-size
refutations of every UNSAT formula: $\Wbar_{\mathrm{EF}}\le\mathrm{poly}$.
\end{theorem}
\begin{corollary}[\status{CONDITIONAL}]
If $\Wbar_{\mathrm{EF}}$ is superpolynomial (OPEN), then no polynomial-time SAT algorithm
has a PV-provable soundness statement. This is a ``feasible'' form of P $\neq$ NP, not
P $\neq$ NP itself.
\end{corollary}
\noindent\textbf{Exact missing hypothesis (OPEN).} Bridge(EF) would follow from
``every polynomial-time SAT algorithm can be replaced by one whose soundness PV proves.''
No such theorem is known; a true but PV-unprovable soundness statement is not excluded.
Pich--Santhanam (arXiv:2312.08163, 2023) give conditional routes from EF-unprovability of
circuit lower bounds to P $\neq$ NP; their hypotheses (I)--(II) are themselves OPEN.

\begin{proposition}[\status{COMPUTATIONAL} + \status{KNOWN THEOREM}; pointwise transfer FALSIFIED for $R=$ resolution]\label{prop:tseitin}
The premise $\mathrm{time}_A(F)\ge\min_{\pi:F\vdash_R\bot}\mathrm{cost}_R(\pi)$ is false for
$R$ = resolution and $A$ = Gaussian elimination over $\mathbb{F}_2$: on Tseitin formulas of
constant-degree expanders, resolution needs $2^{\Omega(n)}$ (Urquhart 1987;
Ben-Sasson--Wigderson 2001), whereas $A$ runs in $O(n^3)$ bit operations.
\end{proposition}
Exact finite check (\texttt{wbar/tseitin\_treeres.py}, deterministic, odd charge at one vertex).
Minimum tree-like resolution size is computed exactly via the decision-tree correspondence
(size $=2L-1$, $L$ = minimum leaves of a falsified-clause search tree):

\noindent\textbf{Reading.} ``Minimizing over all valid reduction sequences'' is a valid lower
bound \emph{only for algorithms whose executions are $R$-derivations of comparable cost}
(e.g.\ DPLL/CDCL traces are resolution refutations). Granting the best sequence for free
is harmless; the problem is that an arbitrary algorithm need not run any $R$-sequence. This is
the gap Bridge$(R)$ must close, and Proposition~\ref{prop:rel} shows it cannot be closed by
black-box simulation.

\section{Multisign DAG hypergraph reduction system}
If the reduction rules on the signed hypergraph are local sound rules on clauses (merge
complementary signs, subsume, delete), the system is p-simulated by resolution, the lower
half is known (Corollary~\ref{cor:weak}), and Bridge is equivalent to P $\neq$ NP. If the DAG
may introduce defined auxiliary nodes (shared subformulas, extension variables), the system
reaches Extended Frege strength; then the lower half is OPEN and the only known bridge is the
provability-conditional Theorem~\ref{thm:pv}. A ``minimal reduced logical state'' for an UNSAT
input is $\bot$; its minimal work is by definition $\min_\pi\mathrm{cost}_R(\pi)$, so the
multisign framing adds no quantity beyond $\Wbar_R$ unless the rule set is specified.
