\begin{document}\maketitle
\noindent\textbf{Status.} P versus NP remains OPEN. Additive report; no schedule, allocation, ownership or root
\texttt{main.tex} changed. Not an LSPV run. Setting: $R$ is a Cook--Reckhow proof system (polynomial-time
verifier), $\mathrm{cost}_R(\pi)=|\pi|$ (a time/space-charged variant only increases cost, so lower bounds
below transfer), and $\Wb_R(N)=\max_{F\in\UNSAT,|F|\le N}\min_{\pi}\mathrm{cost}_R(\pi)$. Free choice of the
best $\pi$ is granted; this only makes $\Wb_R$ smaller, so it is legitimate for lower bounds.
Write $\mathrm{Br}_R$ for the bridge statement $\SAT\in\mathrm P\Rightarrow\Wb_R\le\mathrm{poly}$.

\section{Results}
\begin{proposition}[Known-bound systems: the bridge is P$\neq$NP itself]\status{PROVED/EXACT}
If $\Wb_R\ge N^{\omega(1)}$ is already a theorem (resolution, cutting planes, polynomial calculus, bounded-depth
Frege, AC$^0[p]$-Frege on suitable families), then $\mathrm{Br}_R$ is logically equivalent to $\SAT\notin\mathrm P$.
\end{proposition}
\emph{Proof.} The consequent is false, so $A\Rightarrow\bot$ is $\neg A$. $\square$
So a bridge for such $R$ cannot be proved without proving P$\neq$NP, and any ``transfer'' argument for them
must already contain a P$\neq$NP proof.

\begin{proposition}[Cook--Reckhow]\status{KNOWN THEOREM}
$\Wb_R\le\mathrm{poly}$ iff $R$ is polynomially bounded; this implies NP$=$coNP. Hence
$\Wb_R$ superpolynomial for \emph{every} $R$ is equivalent to NP$\ne$coNP, a statement strictly at least as
strong as P$\ne$NP (NP$\ne$coNP$\Rightarrow$P$\ne$NP; the converse is open).
\end{proposition}

\begin{proposition}[Optimal $R$: the bridge holds, but the target becomes NP$\ne$coNP]\status{PROVED/EXACT}
If $R$ is optimal (every proof system $Q$ has a polynomial $p_Q$ with $\Wb_R\le p_Q(\Wb_Q)$ pointwise,
i.e.\ $R$ simulates $Q$), then $\mathrm{Br}_R$ holds, and $\Wb_R$ superpolynomial $\iff$ NP$\ne$coNP.
\end{proposition}
\emph{Proof.} $\SAT\in\mathrm P\Rightarrow$ NP$=$coNP $\Rightarrow$ some $Q$ is polynomially bounded $\Rightarrow$
$R$ is. Second claim: same argument plus Proposition 2. $\square$

\noindent Existence of an optimal system for TAUT is a known open problem (it follows from NE$=$coNE,
Kraj\'{\i}\v{c}ek--Pud\l\'ak 1989 \status{KNOWN THEOREM}; no unconditional explicit one is known). So route (b) needs
(i) an optimal $R$ and (ii) a superpolynomial lower bound for it, and (ii) \emph{is} NP$\ne$coNP.

\begin{proposition}[Provable correctness gives the bridge]\status{PROVED/EXACT, easy direction}
Suppose $\SAT\in\mathrm P$ via a machine $M$ and there are polynomial-size $R$-proofs of the uniform-in-$n$
family of statements ``$M$ outputs $0$ on $F$ $\Rightarrow F$ unsatisfiable'' (soundness of $M$ in $R$). Then
$\Wb_R\le\mathrm{poly}$.
\end{proposition}
\emph{Proof.} For $F\in\UNSAT$ concatenate the polynomial-size computation trace of $M$ on $F$ with the soundness
proof instantiated at $F$. $\square$

\begin{corollary}[Trichotomy]\status{PROVED/EXACT}
For a fixed system $R$, exactly one holds: (a) a superpolynomial bound is known, so $\mathrm{Br}_R\equiv\SAT\notin\mathrm P$;
(b) $R$ is optimal, so the program reduces to NP$\ne$coNP; (c) $R$ is intermediate (Frege, Extended Frege and
relatives; no bound known and no optimality known), and $\mathrm{Br}_R$ is a \emph{new} statement.
\end{corollary}

\section{What route (c) requires}
By Proposition 4, $\mathrm{Br}_{EF}$ would follow from: if $\SAT\in\mathrm P$ then some deciding $M$ has soundness
provable by polynomial-size EF proofs (roughly, in PV/$S^1_2$). Levin's universal algorithm decides SAT whenever any
algorithm does, but its soundness proof essentially needs the very unsatisfiability proofs being sought, so the
needed provability is not automatic. \status{OPEN} whether P$=$NP forces a provably sound algorithm; I know no
unconditional proof and no oracle separation checked in this run (\status{CANDIDATE} reading: a relativized
world with SAT$^A\in\mathrm P^A$ but $\mathrm{EF}^A$ superpolynomial would show $\mathrm{Br}_{EF}$ cannot be proved by
relativizing means; not verified here).

\section{Consequences for the five-agent plan}
\begin{itemize}
\item \textbf{Hard-instance families} (Tseitin, pigeonhole, hypergraph constructions) giving lower bounds for
resolution/algebraic/circuit-compressed systems are \emph{valuable only for route (c)} systems; against systems of
type (a) they are already known, and they do not touch the bridge (Prop.\ 1).
\item The \emph{weakest sufficient bridge} is therefore not weak: it is equivalent in strength to a provability
form of ``P$=$NP is feasibly witnessed.'' Target T-BRIDGE: characterize $\mathrm{Br}_{EF}$ (or $\mathrm{Br}_{Frege}$)
in terms of provably-correct SAT algorithms, and decide whether the reflection principle for $M$ can be
bypassed by free choice of $\pi$ (this relaxation is already built in: it equals the nondeterministic cost).
\item Adaptive strategy switching does not help the lower bound: the min over $\pi$ already includes any switching.
\item A hypergraph-DAG ``minimal reduced state'' lower bound must be a bound on $\Wb_R$ for a specific $R$ in class (c);
state which, and verify it is not simulated by a known-bounded system in the whole class.
\end{itemize}
