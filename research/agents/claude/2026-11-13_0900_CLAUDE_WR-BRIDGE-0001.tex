\section{Object}
For a fixed Cook--Reckhow refutation system $R$ (the project's signed multisign DAG-hypergraph
calculus, e.g.\ $R_{\mathrm{mix341}}$, is one such $R$),
\[
\Wb_R(N)=\max_{F\in\UNSAT,\ |\langle F\rangle|\le N}\ \min_{\pi:F\vdash_R\bot}\operatorname{cost}_R(\pi).
\]
The inner minimum is over \emph{all} valid reduction sequences; nothing is charged for finding the
best one. This is a legitimate lower-bound quantity: any algorithm whose run on $F$ can be read
as an $R$-derivation pays at least $\min_\pi\operatorname{cost}_R(\pi)$. \status{PROVED/EXACT}
(definitional). Equivalently, $\Wb_R$ superpolynomial is exactly ``$R$ is not polynomially
bounded''. By Cook--Reckhow, a superpolynomial $\Wb_R$ for \emph{all} $R$ is equivalent to
$\mathrm{NP}\neq\mathrm{coNP}$; for one fixed $R$ it says nothing about P vs NP without $B_R$.

\section{The trichotomy}
\begin{proposition}[Bridge collapse for systems with known lower bounds]\label{p:collapse}
If $\Wb_R(N)\ge N^{\omega(1)}$ is a theorem, then $B_R\iff\SAT\notin\mathrm P$.
\end{proposition}
\begin{proof}
($\Leftarrow$) vacuous antecedent. ($\Rightarrow$) $B_R$ and the lower bound give
$\SAT\notin\mathrm P$ by modus tollens.
\end{proof}
\status{PROVED/EXACT}. Instances (\status{KNOWN THEOREM}): tree-like and DAG resolution
(Haken 1985; Ben-Sasson--Wigderson 2001), cutting planes (Pudl\'ak 1997; random CNF:
Fleming--Pankratov--Pitassi--Robere 2017, Hrube\v{s}--Pudl\'ak 2017), polynomial calculus
(Razborov 1998; Impagliazzo--Pudl\'ak--Sgall 1999), bounded-depth Frege (Ajtai 1988;
Beame--Impagliazzo--Pitassi; Kraj\'\i\v{c}ek--Pudl\'ak--Woods). For each such $R$, ``prove the
weakest bridge'' \emph{is} ``prove P$\neq$NP''; the bridge is a reformulation, not a weakening
(cf.\ the world-ledger rule in \url{research/agents/claude/world-ledger/}).

\begin{theorem}[Provable-soundness bridge; Cook 1975; Kraj\'\i\v{c}ek--Pudl\'ak 1989]\label{t:pv}
Let $A$ be a polynomial-time algorithm. If the theory PV (resp.\ any sound r.e.\ theory $T$
extending $S^1_2$) proves the universal soundness statement
$\forall F\,\forall y\,(A(F)=\UNSAT\to\neg\mathrm{Sat}(F,y))$, then Extended Frege (resp.\ the
propositional system $P_T$ of $T$) has refutations of size $|F|^{O_A(1)}$ of every $F$ on which
$A$ outputs \UNSAT. Hence if $\SAT\in\mathrm P$ is witnessed by such an $A$, then
$\Wb_{EF}$ (resp.\ $\Wb_{P_T}$) is polynomial, and so is $\Wb_R$ for every $R$ that
p-simulates EF (resp.\ $P_T$).
\end{theorem}
\status{KNOWN THEOREM}. Completeness of $A$ need not be provable: it is used only to guarantee
that $A(F)=\UNSAT$ on every unsatisfiable $F$, a true fact under the hypothesis.

\begin{corollary}[Exact residue of the bridge for strong $R$]\label{c:residue}
For $R\in\{EF, P_T\}$ and every $R'$ p-simulating it, $B_{R'}$ fails only if $\SAT\in\mathrm P$
holds via algorithms \emph{none} of which has $T$-provable soundness (for $T=$ PV, resp.\ the
fixed theory). Contrapositively, a theorem $\Wb_{EF}(N)\ge N^{\omega(1)}$ already implies,
unconditionally, that no polynomial-time SAT algorithm is PV-provably sound
(``P$\neq$NP is consistent with PV-soundness''), but not P$\neq$NP.
\end{corollary}
\status{PROVED/EXACT} from Theorem~\ref{t:pv}. This matches and sharpens LSPV-1/357 (the
G\"odelian reflection firewall), which already isolated provable rejection-soundness as the
sufficient condition for $R_{\mathrm{mix341}}$.

\paragraph{Trichotomy.} Every fixed $R$ falls into one of:
(i) $\Wb_R$ superpolynomial is known $\Rightarrow$ $B_R\equiv$ P$\neq$NP (Prop.~\ref{p:collapse});
(ii) $R$ p-simulates EF and no lower bound is known $\Rightarrow$ $B_R$ holds for all
PV-provably-sound algorithms, and the open residue is exactly unprovable soundness
(Cor.~\ref{c:residue}), while the lower bound $\Wb_R\ge N^{\omega(1)}$ is itself a famous open
problem; (iii) $R$ between the two (e.g.\ Frege, $AC^0[p]$-Frege, $TC^0$-Frege): no lower bound
known and no PV-type bridge known. \textbf{No currently known $R$ makes both halves non-trivial
and non-open at once.} \status{PROVED/EXACT} as a classification of the current literature state.

\begin{remark}[Kraj\'\i\v{c}ek--Pudl\'ak $E=NE$ route: FAILED TRANSFER for a fixed $R$]
$\SAT\in\mathrm P$ implies $E=NE$ by padding, and Kraj\'\i\v{c}ek--Pudl\'ak (1989) show
$E=NE$ implies a p-optimal proof system exists; that system is then p-bounded. But the system is
built from the hypothesised algorithm, so it is not a $R$ fixed in advance. The same point is
already recorded in codex/2026-11-11\_1331 (the clocked system $Q_A$ is p-optimal when $A$ is
polynomial). \status{KNOWN THEOREM}; classification FAILED TRANSFER for the fixed-$R$ bridge.
\end{remark}
