\section{Setting}
$R$ is a fixed derivation system: derivations are polynomial-time checkable, sound, and complete for UNSAT (every UNSAT $F$ has one). $\mathrm{cost}_R(\pi)$ is the number of paid inferences (or encoded size), polynomially related to $|\pi|$.
$W_R(F)=\min_\pi \mathrm{cost}_R(\pi)$, $\overline W_R(N)=\max_{|F|\le N}W_R(F)$. Strategy discovery is not charged.
\begin{itemize}
\item $\mathrm{LB}_R$: $\overline W_R(N)\ne N^{O(1)}$.\quad $\mathrm{Br}_R$: $\SAT\in P\Rightarrow \overline W_R(N)=N^{O(1)}$.
\end{itemize}
(In the corpus, $\mathrm{Br}_R$ is B3 for $R=\mathrm{Rmix341}$.)

\section{Results}
\begin{proposition}[Combination]\status{PROVED/EXACT}
$\mathrm{LB}_R\wedge\mathrm{Br}_R\Rightarrow P\ne NP$. (Contrapositive of $\mathrm{Br}_R$.) Both are needed; neither alone gives anything.
\end{proposition}

\begin{proposition}[Free strategy = nondeterminism]\status{PROVED/EXACT}
$\overline W_R(N)\le N^{O(1)}$ iff $R$ is polynomially bounded in the sense of Cook--Reckhow. If this holds for \emph{some} system then $NP=coNP$; hence ``$\mathrm{LB}_R$ for every $R$'' is equivalent to $NP\ne coNP$, strictly stronger than $P\ne NP$. Granting a free optimal sequence therefore moves the question from $P$ vs $NP$ to proof complexity of the one chosen $R$.
\end{proposition}

\begin{theorem}[Weak-$R$ trap]\status{PROVED/EXACT}
If $\mathrm{LB}_R$ is a known unconditional theorem, then $\mathrm{Br}_R\iff \SAT\notin P$, i.e.\ $\mathrm{Br}_R\iff P\ne NP$ (given $P\subseteq NP$ and $\SAT\in P\iff P=NP$).
\end{theorem}
\begin{proof}
$\mathrm{LB}_R$ makes the consequent of $\mathrm{Br}_R$ false, so $\mathrm{Br}_R$ is equivalent to the negation of its antecedent. \qed
\end{proof}
\noindent\status{KNOWN THEOREM} (stated from memory of the literature, not re-derived here) systems with unconditional super-polynomial lower bounds include (DAG-like) Resolution (Haken 1985: PHP; Urquhart 1987: expander Tseitin; Chv\'atal--Szemer\'edi 1988 and Ben-Sasson--Wigderson 2001: random $k$-CNF), and Cutting Planes with a general-class lower bound (Pudl\'ak 1997, clique--coclique). For all of these $\mathrm{Br}_R$ is a pure reformulation of $P\neq NP$; proving it needs the whole problem. Adding AND3, GF(2) XOR, and local CONVERT (as in Rmix341) does not escape this unless the enlarged system provably beats the known hard families; XOR kills expander Tseitin and the known short EF/Frege proofs kill PHP, so the corpus's hard families have to change for any mixed system that is strong enough, and then $\mathrm{LB}_R$ is no longer a known theorem.

\begin{theorem}[Strong-$R$ side]\status{KNOWN THEOREM + CONDITIONAL}
Let $A$ be a polynomial-time SAT decider whose correctness (``$A(F)=0\Rightarrow F$ unsatisfiable'') is provable in PV (or $S^1_2$). Then every UNSAT $F$ has an Extended Frege proof of size $\mathrm{poly}(|F|)$ (Cook's PV$\to$EF translation: the run of $A$ plus the instantiated soundness lemma). So $\mathrm{Br}_{EF}$ holds \emph{for P=NP witnessed by a feasibly provably-correct algorithm}. Without the provability proviso, $\mathrm{Br}_{EF}$ is OPEN to me: I found no source asserting $P=NP\Rightarrow EF$ polybounded.
\end{theorem}
\noindent So the only region where $\mathrm{Br}_R$ is not trivially a reformulation is $R\supseteq EF$ (or a system simulating the relevant theory), and there $\mathrm{LB}_R$ is the open Frege/EF lower-bound problem.

\begin{proposition}[Optimal-system bridge exists but is non-explicit]\status{KNOWN THEOREM (Kraj\'i\v{c}ek--Pudl\'ak 1989, from memory) + derived}
$P=NP\Rightarrow E=NE=coNE\Rightarrow$ an optimal proof system exists (and, since $NP=coNP$ gives a polybounded system, any optimal one is polybounded). So $\mathrm{Br}_{R^*}$ holds for the optimal $R^*$, but $R^*$ is defined from the unknown machine witnessing $NE=coNE$; no lower bound can be proved for an unspecified $R^*$ except by proving it for all $R$, i.e.\ $NP\ne coNP$.
\end{proposition}

\begin{corollary}[Design constraint]\status{PROVED/EXACT as a summary}
Any route in this program must pick $R$ with (i) $\mathrm{Br}_R$ provable from $\SAT\in P$ --- forces $R$ to be at least EF-strong or to be tied to the algorithm; and (ii) $\mathrm{LB}_R$ provable --- forces $R$ to be weak or the hard family to dodge all known short proofs. The two requirements pull in opposite directions; the program's live window is a system strictly between Resolution$+$XOR$+$AND3 and EF (e.g.\ bounded-depth Frege with parity gates, $AC^0[2]$-Frege, PC/IPS variants), where $\mathrm{LB}_R$ is partly known but $\mathrm{Br}_R$ is not.
\end{corollary}
