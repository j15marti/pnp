\section{Objective}
For a derivation system $R$ with cost $\mathrm{cost}_R$, $\overline W_R(N)=\max_{F\in\UNSAT,|F|\le N}\min_{\pi:F\vdash_R\bot}\mathrm{cost}_R(\pi)$.
Target bridge $\mathrm{B}(R)$: $\SAT\in\mathrm P\Rightarrow \overline W_R(N)\le\mathrm{poly}(N)$.
Free choice of the optimal $\pi$ is granted (it only lowers $\overline W_R$, so lower bounds on it are stronger).

\section{Results}
\begin{proposition}[Hardwiring collapse] \status{PROVED/EXACT}
If derivations need not be verified (cost counts only the steps of $\pi$, with an unchecked rule ``$F\vdash\bot$ is semantically true''), then $\overline W_R(N)=O(N)$. Hence $R$ must be a Cook--Reckhow system: $\pi$ is checkable in time $\mathrm{poly}(|\pi|+|F|)$, soundness being local to rules, and the checking cost is charged.
\end{proposition}
\begin{proof} Take $\pi_F=$ the single axiom $F\vdash\bot$, cost $|F|$. \end{proof}

\begin{proposition}[Quantifier order] \status{PROVED/EXACT}
Let derivations be runs of algorithms sound on all inputs. Then $\max_F\min_A t_A(F)=O(N)$ (hardwire $A$ per $F$); letting $A$ depend on $N$ only gives $\min_A\max_F t_A(F)=$ the $\mathrm P/\mathrm{poly}$ upper bound for SAT; the uniform $\min_A\max_F$ is $\SAT\in\mathrm P$. By Karp--Lipton ($\SAT\in\mathrm P/\mathrm{poly}\Rightarrow\mathrm{PH}=\Sigma_2^p$) the free-sequence $\max\min$ with per-length advice cannot be equated with P.
\end{proposition}

\begin{theorem}[Vacuity of the bridge for systems with known lower bounds] \status{PROVED/EXACT}
If $R$ has an unconditional superpolynomial lower bound ($R\in\{$resolution, constant-depth Frege, cutting planes, polynomial calculus, \dots$\}$), then $\mathrm B(R)\iff \SAT\notin\mathrm P$.
\end{theorem}
\begin{proof} The consequent of $\mathrm B(R)$ is false, so $\mathrm B(R)$ is equivalent to the negation of its antecedent. \end{proof}
Consequence: for such $R$ the bridge is \emph{not} a weaker sufficient step than P$\neq$NP; it is P$\neq$NP. A lower bound for $R$ plus $\mathrm B(R)$ gives nothing new.

\begin{theorem}[Optimal systems] \status{PROVED/EXACT}
(a) If $R$ is p-optimal (or merely optimal) then $\mathrm B(R)$ holds. (b) For such $R$, $\overline W_R$ superpolynomial $\iff \mathrm{NP}\neq\mathrm{coNP}$.
\end{theorem}
\begin{proof} $\SAT\in\mathrm P\Rightarrow\mathrm{NP}=\mathrm{coNP}\Rightarrow$ (Cook--Reckhow) some system is p-bounded; $R$ simulates it with polynomial blow-up, so $R$ is p-bounded. (b) is Cook--Reckhow applied to $R$. \end{proof}
Existence of an optimal system is open (Kraj\'{\i}\v{c}ek--Pud\l{}\'ak: follows from $\mathrm{NE}=\mathrm{coNE}$). So any bridge-valid $R$ is "as strong as optimal" for P=NP instances, and the lower-bound half is then at least as hard as $\mathrm{NP}\neq\mathrm{coNP}$, strictly stronger than P$\neq$NP and open.

\begin{theorem}[Conditional bridge for Extended Frege] \status{KNOWN THEOREM} (Cook 1975; Kraj\'{\i}\v{c}ek, \emph{Bounded Arithmetic, Propositional Logic and Complexity Theory})
If a poly-time $M$ decides SAT and $\mathrm{PV}$ (equivalently, up to translation, $S^1_2$) proves $\forall F,w:\,M(F)=\text{reject}\to\neg\mathrm{Sat}(F,w)$, then EF has polynomial-size refutations of every unsatisfiable $F$. Contrapositive: an EF superpolynomial lower bound shows no poly-time SAT algorithm has feasibly provable soundness. This is the sharpest form of the bridge available: $\mathrm B'(\mathrm{EF})$: ``$\SAT\in\mathrm P$ \emph{with PV-provable correctness} $\Rightarrow\overline W_{\mathrm{EF}}\le\mathrm{poly}$''. Related recent work: Pich--Santhanam, arXiv:2312.08163 (EF lower bounds imply P$\neq$NP under further hypotheses on anticheckers).
\end{theorem}
