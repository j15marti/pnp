\section{Shared objective and model}
Fix a Cook--Reckhow refutation system $R$ for CNF (polynomial-time verifier $V_R$, sound, complete).
For UNSAT $F$ let $L_R(F)$ be the minimum bit length of an $R$-refutation and
$W_R(F)$ the minimum charged node count, with the optimal derivation chosen freely (the user's
``free knowledge of the best sequence''). Worst case:
\[
\overline W_R(N)=\max_{\substack{F\in\mathrm{UNSAT}\\|\langle F\rangle|\le N}} W_R(F),\qquad
\overline L_R(N)=\max_{\substack{F\in\mathrm{UNSAT}\\|\langle F\rangle|\le N}} L_R(F).
\]
\textbf{Locality hypothesis (LOC).} There is a polynomial $q$ such that every refutation $\pi$ of
$F$ with $w$ charged nodes has an encoding of at most $q(|F|+w)$ bits. ER/RES-style
systems over width-$\le 3$ clauses with $O(\log)$-bit variable names, including the project's
$R_{\mathrm{mix341}}$ node convention, satisfy LOC; Frege with unbounded line size does
not unless lines are charged by size. Under LOC, $W_R\le L_R\le q(|F|+W_R)$, so
$\overline W_R$ and $\overline L_R$ are polynomially related.

\begin{definition}
$\mathrm{B3}(R)$: $\ \mathrm{SAT}\in\mathsf P\implies \overline W_R(N)\le\poly(N)$.\\
$\mathrm{Opt}(R)$ (size-optimality): for every Cook--Reckhow system $Q$ there is a polynomial
$p_Q$ with $L_R(F)\le p_Q(|F|+L_Q(F))$ for all UNSAT $F$.\\
$\mathrm{LB}(R)$: $\overline W_R(N)$ is not bounded by any polynomial.
\end{definition}

\section{Main theorem}
\begin{theorem}[Exact form of the weakest bridge]\label{thm:exact}
Under LOC, for every fixed $R$:
\[
\boxed{\ \mathrm{B3}(R)\iff \bigl(\mathsf P\neq\mathsf{NP}\bigr)\ \lor\ \mathrm{Opt}(R).\ }
\]
Moreover, if $\mathsf P=\mathsf{NP}$ then the following are equivalent:
(i) $\overline W_R$ polynomial; (ii) $\mathrm{Opt}(R)$; (iii) $R$ is p-optimal
(polynomial-time translations from every $Q$).
\end{theorem}
\begin{proof}
($\Leftarrow$) If $\mathsf P\ne\mathsf{NP}$ the antecedent of B3 is false. If $\mathrm{Opt}(R)$
and $A$ is a polynomial-time SAT decider, let $Q_A$ accept the empty proof for $F$ iff $A(F)=\mathrm{UNSAT}$;
$Q_A$ is Cook--Reckhow with $L_{Q_A}\equiv 0$. Then $L_R(F)\le p_{Q_A}(|F|)$ and
$W_R\le L_R$ gives B3 (this direction is the corpus's CLOCKED336 bridge).

($\Rightarrow$) Suppose B3$(R)$ and $\mathsf P=\mathsf{NP}$. Then $\overline W_R(N)\le p(N)$; by LOC
$L_R(F)\le q(|F|+p(|F|))\le q(|F|+p(|F|)+L_Q(F))$ for every $Q$, which is Opt$(R)$.

For the equivalences under $\mathsf P=\mathsf{NP}$: (i)$\Rightarrow$(ii) as just shown;
(ii)$\Rightarrow$(i) by applying (ii) to $Q_A$; (iii)$\Rightarrow$(ii) trivially.
(i)$\Rightarrow$(iii): the prefix language
$\{(F,u):\exists v,\ |uv|\le q(|F|+p(|F|)),\ V_R(F,uv)=1\}$ is in $\mathsf{NP}=\mathsf P$, so
bit-by-bit self-reduction outputs an $R$-refutation of any UNSAT $F$ in time $\poly(|F|)$,
ignoring the given $Q$-proof; that is a polynomial-time translation.
\end{proof}

\begin{corollary}[There is no strictly weaker fixed-$R$ bridge]\label{cor:noweak}
Any proof of $\mathrm{B3}(R)$ that does not already prove $\mathsf P\ne\mathsf{NP}$ must prove
$\mathsf P=\mathsf{NP}\Rightarrow\mathrm{Opt}(R)$. So the ``weakest sufficient bridge'' of the
directive is not weaker than conditional size-optimality of $R$; it is exactly that.
Every attempted B3 proof must, in a hypothetical $\mathsf P=\mathsf{NP}$ world, turn an
\emph{unknown} polynomial SAT decider into short $R$-proofs for \emph{all} UNSAT formulas.
\end{corollary}

\begin{corollary}[The two-arrow program, classified]\label{cor:lb}
(a) $\mathrm{B3}(R)\wedge\mathrm{LB}(R)\Rightarrow\mathsf P\ne\mathsf{NP}$.
(b) $\mathrm{Opt}(R)\wedge\mathrm{LB}(R)\Rightarrow\mathsf{NP}\ne\mathsf{coNP}$ (an optimal system
that is not polynomially bounded leaves no system that is; Cook--Reckhow).
(c) $\mathsf P\ne\mathsf{NP}$ alone implies B3$(R)$ for every $R$, including weak $R$ such as
Resolution where LB is known (Haken 1985). So B3 is \emph{true} in our world if P$\ne$NP, but can be
\emph{proved} only through (P$\ne$NP) or (P$=$NP$\Rightarrow$Opt).
In World-Ledger terms (research/agents/claude/world-ledger), B3$(R)$ is an \emph{implication},
not progress by itself; B3$(R)\wedge$LB$(R)$ is the real target.
\end{corollary}

\begin{corollary}[Free optimal derivation is a nondeterministic measure]
Granting the best derivation for free is sound for lower bounds against every algorithm whose
execution trace \emph{is} an $R$-derivation. But for an arbitrary decider $A$, $Q_A$ is itself a
Cook--Reckhow system whose unique ``derivation'' is $A$'s run, so ``$\mathrm{LB}$ for every $R$''
is exactly $\mathsf{NP}\ne\mathsf{coNP}$ (Cook--Reckhow). For one fixed $R$, the gap between
free-minimum hardness and arbitrary-algorithm hardness is exactly $\neg\mathrm{Opt}(R)$.
\end{corollary}

\section{Provability-stratified version (KNOWN THEOREM, organized for the project)}
For a theory $T\supseteq\mathrm{PV}$ with associated propositional system $R_T$
(EF for PV/$S^1_2$, Cook 1975; Kraj\'{\i}\v{c}ek--Pudl\'ak 1990), if some polynomial-time $A$
has $T\vdash\mathrm{RS}_A$ (rejection soundness), then $\overline L_{R_T}$ is polynomial.
Hence the only world in which B3$(R_T)$ fails is
\[
\mathrm{Gap}_T:\quad \mathrm{SAT}\in\mathsf P\ \wedge\ \forall A\in\mathsf P\
\bigl(A\text{ correct}\Rightarrow T\nvdash\mathrm{RS}_A\bigr).
\]
LSPV-1/357's G\"odel wrapper makes \emph{one} correct decider unprovable; it does not produce
$\mathrm{Gap}_T$, which needs \emph{all} correct deciders unprovable. Consequence (folklore,
Kraj\'{\i}\v{c}ek's book): an EF lower bound implies PV proves rejection soundness of no
polynomial-time SAT algorithm. That is \textbf{not} $\mathsf P\ne\mathsf{NP}$.
