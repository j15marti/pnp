\section*{Status}
\textbf{P versus NP remains OPEN.} Propositions~1--4 below are \textbf{PROVED/EXACT}
(elementary, within the stated definitions). Cited proof-complexity results are
\textbf{KNOWN THEOREM}. No load-bearing arrow is closed by this report.
The central finding is a \emph{classification}: for every fixed calculus $R$ for
which a superpolynomial lower bound is already known, the requested bridge is
\emph{logically equivalent} to $\SAT\notin\Pc$. Per the world-ledger rule
(a statement equivalent to $\Pc\neq\NP$ is a reformulation, not progress), the
shared objective is non-trivial only for systems $R$ without a known lower bound.

\section*{Objects and quantifiers}
Fix a Cook--Reckhow refutation system $R$ (polynomial-time verifier, sound, complete).
For unsatisfiable CNF $F$ let $w_R(F)=\min_{\pi:F\vdash_R\bot}\operatorname{cost}_R(\pi)$
(free choice of the optimal derivation, as the directive grants), and
\[
 \overline W_R(N)=\max_{F\in\UNSAT,\ |\langle F\rangle|\le N} w_R(F).
\]
$\mathrm{LB}(R)$: $\overline W_R(N)\neq N^{O(1)}$.
$\mathrm{Bridge}(R)$: $\SAT\in\Pc\Rightarrow \overline W_R(N)\le\poly(N)$.
$\mathrm{PS}(R)$ (pointwise simulation, secondary target): for every correct deterministic
UNSAT-decider $A$ there is a polynomial $p_A$ with $w_R(F)\le p_A(\mathrm{time}_A(F))$ for all $F\in\UNSAT$.
Costs are encoded proof length or any cost polynomially related to it.

\section{Logical collapse of the bridge}
\begin{proposition}[PROVED/EXACT]
For every fixed $R$: $\mathrm{Bridge}(R)\iff(\SAT\notin\Pc\ \vee\ \neg\mathrm{LB}(R))$.
Consequently $\mathrm{Bridge}(R)\wedge\mathrm{LB}(R)\iff \SAT\notin\Pc\wedge \mathrm{LB}(R)$.
\end{proposition}
\begin{proof}
Material implication: $(a\Rightarrow b)\equiv(\neg a\vee b)$ with $a=\SAT\in\Pc$, $b=\neg\mathrm{LB}(R)$.
\end{proof}

\begin{corollary}[PROVED/EXACT, given KNOWN THEOREMS]
Let $R$ be any of: tree or DAG Resolution, Res$(k)$, Cutting Planes, Polynomial Calculus
(any field), bounded-depth Frege, Nullstellensatz, or any $R$ with a proved superpolynomial
$\overline W_R$. Then $\mathrm{Bridge}(R)\iff \Pc\neq\NP$.
\end{corollary}
\begin{proof}
$\mathrm{LB}(R)$ is a known theorem for each listed system (Haken 1985 for Resolution on
$\mathrm{PHP}$; Pudlák 1997 for CP; Razborov 1998 / Impagliazzo--Pudlák--Sgall 1999 for PC;
Ajtai 1988, Kraj\'{\i}\v{c}ek--Pudlák--Woods 1995, Pitassi--Beame--Impagliazzo 1993 for bounded-depth Frege).
Apply Proposition~1.
\end{proof}

\begin{proposition}[PROVED/EXACT]
$\exists R\,[\mathrm{Bridge}(R)\wedge\mathrm{LB}(R)]\iff \Pc\neq\NP.$
\end{proposition}
\begin{proof}
($\Rightarrow$) $\mathrm{LB}(R)$ makes the bridge's consequent false, so its antecedent is false.
($\Leftarrow$) If $\Pc\ne\NP$, take $R=$ Resolution: Bridge holds vacuously and LB is Haken's theorem.
\end{proof}

\paragraph{Interpretation.}
The directive's ``weakest sufficient bridge'' does not decompose the problem when $R$
is a system with a provable lower bound: \emph{all} the difficulty lives in the bridge,
and proving the bridge is proving $\Pc\neq\NP$. The decomposition is non-trivial only
when both $\mathrm{Bridge}(R)$ and $\mathrm{LB}(R)$ are separately open, i.e.\ for
Frege, Extended Frege (EF), and stronger systems. This is consistent with the free-sequence
framing: granting the optimal derivation makes $w_R$ a \emph{nondeterministic}
measure, so $\mathrm{LB}(R)$ for every $R$ is $\NP\neq\mathrm{coNP}$ (Cook--Reckhow 1979),
strictly stronger than the deterministic target.

\section{Pointwise simulation is falsified for every system with an explicit lower bound}
\begin{proposition}[PROVED/EXACT]
Let $\{F_n\}$ be an unsatisfiable family recognizable in polynomial time
(given $G$, decide whether $G=F_n$ for some $n$) with $w_R(F_n)\ge n^{\omega(1)}$.
Then $\mathrm{PS}(R)$ is false.
\end{proposition}
\begin{proof}
Let $A$ output UNSAT if the recognizer accepts $G$, and otherwise run truth-table search.
$A$ is a correct total decider with $\mathrm{time}_A(F_n)\le\poly(n)$, while
$w_R(F_n)$ is superpolynomial, so no polynomial $p_A$ exists.
\end{proof}
Every lower-bound family cited in Corollary~1 (PHP$^{n+1}_n$, Tseitin on explicit expanders,
clique--colouring, random-restriction families with explicit seeds) is polynomial-time recognizable,
so $\mathrm{PS}(R)$ is \textbf{FALSIFIED} for all those systems. This agrees with and generalizes the
corpus result REFEREE295 (GF(2) Tseitin decider vs.\ Resolution) from one family to the whole class.
It also matches the KNOWN characterization of Beyersdorff--Sadowski: a language has a p-optimal
proof system iff its polynomial-time subsets admit a suitable recursive presentation; a system
satisfying $\mathrm{PS}$ must give short proofs to every polynomial-time easy subset of $\UNSAT$.

\begin{proposition}[PROVED/EXACT]
$\mathrm{PS}(R)$ holds iff $R$ p-simulates the system $R_A$ for every correct deterministic decider $A$,
where an $R_A$-proof of $F$ is $1^{\mathrm{time}_A(F)}$ and the verifier runs $A$ on $F$ for that many steps.
If $R$ is p-optimal then $\mathrm{PS}(R)$ holds, and $\mathrm{PS}(R)\Rightarrow\mathrm{Bridge}(R)$.
\end{proposition}
\begin{proof}
$R_A$ is a Cook--Reckhow system exactly when $A$ is correct, and its proof length is $\mathrm{time}_A(F)+O(1)$;
a p-simulation of $R_A$ by $R$ is precisely a polynomial $p_A$ as in PS (the translation is computable by
running $A$ and then the translation function). Under $\SAT\in\Pc$, a polynomial $A$ gives
$\overline W_R(N)\le p_A(N^{c})$.
\end{proof}
KNOWN THEOREM context: Kraj\'{\i}\v{c}ek--Pudlák (1989) show a p-optimal system for TAUT exists iff an
optimal acceptor for TAUT exists, and that $\mathsf E=\mathsf{NE}$ implies p-optimal systems exist;
$\Pc=\NP$ implies $\mathsf E=\mathsf{NE}$ by padding. But the p-optimal system so obtained is built from the
hypothesis, so it is \emph{not} a fixed $R$ chosen in advance, which is what the bridge requires.
Existence of a p-optimal system for TAUT remains \textbf{OPEN}.
Combined with Proposition~3, the pointwise route survives only for $R$ with no explicit-family lower bound,
and its truth for such $R$ would need $R$ to dominate all deterministic algorithms (an optimal-acceptor-type
statement), which is OPEN.

\section{The only non-vacuous targets}
\begin{enumerate}
\item \textbf{Bridge(EF) / Bridge(Frege).} KNOWN THEOREM (Cook 1975 correspondence; Kraj\'{\i}\v{c}ek's monograph):
if a polynomial-time SAT algorithm $A$ has its correctness provable in Cook's theory PV, then EF has
polynomial-size refutations of all unsatisfiable CNFs (propositional translations of the PV proof plus a
run of $A$). Thus Bridge(EF) would follow from ``every $\Pc=\NP$ witness is PV-provable''. That statement is
not known and is generally believed hard; it is a \textbf{CANDIDATE} reduction of Bridge(EF), not a proof.
\item \textbf{LB(EF)}: superpolynomial EF lower bounds are \textbf{OPEN} (no technique beyond
bounded-depth Frege is known; see Kraj\'{\i}\v{c}ek's monograph). Neither half of the decomposition is
closer for EF than the original problem.
\item \textbf{Restricted transfers that are exact.} For CDCL/DPLL solvers, the trace yields a Resolution
refutation (Beame--Kautz--Sabharwal 2004; Pipatsrisawat--Darwiche 2011), so
$\mathrm{time}_A(F)\ge w_{\mathrm{Res}}(F)^{\Omega(1)}$ for $A$ in that class. This is a
\textbf{directly transferable theorem} for restricted algorithms only and transfers to no general $A$
(Proposition~3 kills the pointwise form for general $A$).
\end{enumerate}

\section*{Barrier-transfer preflight}
Obstacle 1: the bridge is the problem (Propositions 1--2). Transfer domain: logic/proof complexity
(Cook--Reckhow, Kraj\'{\i}\v{c}ek--Pudlák): \emph{directly transferable theorem} confirming the obstruction.
Obstacle 2: algorithms exploit easy subfamilies (Proposition~3). Transfer domain: coding/cryptanalysis
(PQ ledger item ``hidden structure may be a weakness''): the recognizer decider is exactly the algorithmic
exploitation of hidden structure; \emph{exact object transfer} of the lesson, no new hardness.
Optimization duality (LP integrality gaps vs.\ algorithms) is a \emph{misleading analogy}: gaps lower-bound
a relaxation, not an arbitrary algorithm, the same scope error as $\mathrm{PS}$.
