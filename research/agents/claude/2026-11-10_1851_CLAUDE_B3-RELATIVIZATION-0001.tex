\begin{abstract}
For every fixed sound, complete, oracle-free proof system $R$ for UNSAT CNF, the
directive's program is to prove (LB$_R$) $\Wbar_R(N)$ is superpolynomial and
(B3$_R$) $\mathrm{SAT}\in\Pc\Rightarrow\Wbar_R(N)\le\mathrm{poly}(N)$.
We prove three short facts.
(1) \textbf{PROVED/EXACT.} If LB$_R$ holds, the relativized bridge B3$_R^{O}$ is
\emph{false} for $O=\TQBF$, and the algebrized bridge is false for the multilinear
extension of $\TQBF$. Hence any successful choice of $R$ forces a
non-relativizing, non-algebrizing proof of B3$_R$. In particular, a bridge that
translates the run of an arbitrary correct polynomial-time SAT decider into an
$R$-derivation using only its input--output behaviour cannot work.
(2) \textbf{PROVED/EXACT.} For every $R$ with an \emph{already proved}
superpolynomial lower bound (Resolution, Cutting Planes, bounded-depth Frege,
Polynomial Calculus), B3$_R$ is logically equivalent to $\Pc\neq\NP$; the
statement $\exists R\,(\mathrm{LB}_R\wedge \mathrm{B3}_R)$ is equivalent to
$\Pc\neq\NP$. It reformulates the problem; it does not make it easier.
(3) \textbf{KNOWN THEOREM / CONDITIONAL.} For $R\ge_p$ Extended Frege, B3$_R$ follows
from the existence of a polynomial-time SAT algorithm whose correctness is provable
in Cook's theory PV (Cook 1975). Then LB$_R$ implies superpolynomial EF lower
bounds, a problem open since 1979. For the historical $R_{\mathrm{mix341}}$ this transfer is
CONDITIONAL on the corpus's claimed EF$\to$ER$\to$RES+DEF simulation (LSPV-1/406).
The difficulty is conserved: a weak $R$ puts all of it into B3, and a strong $R$ puts all
of it into LB. P versus NP remains \textbf{OPEN}.
\end{abstract}

\section{Setting and exact quantifiers}
Fix an encoding $\langle F\rangle$ of CNF formulas and a proof system $R$ given by a
deterministic polynomial-time verifier $V_R(\langle F\rangle,\pi)$ that is sound
(accepts only UNSAT $F$) and complete. $R$ is \emph{oracle-free}: $V_R$ queries no oracle.
Fix a nonnegative cost $\mathrm{cost}_R(\pi)$ that depends only on $(\langle F\rangle,\pi)$, such as
the number of distinct paid DAG nodes or $|\pi|$. Following the
2026-11-02 directive
(\texttt{research/coordination/2026-11-02\_1204\_WORST\_CASE\_MINIMUM\_DERIVATION\_FIVE\_ROLE\_REFOCUS.md}):
\[
W_R(F)=\min_{\pi:\,V_R(F,\pi)=1}\mathrm{cost}_R(\pi),\qquad
\Wbar_R(N)=\max_{F\in\mathrm{UNSAT},\,|\langle F\rangle|\le N}W_R(F).
\]
\begin{itemize}
\item[LB$_R$:] $\forall k,C,N_0\ \exists N\ge N_0:\ \Wbar_R(N)>C(N+1)^k$.
\item[B3$_R$:] $\mathrm{SAT}\in\Pc\ \Longrightarrow\ \exists C,k\ \forall N:\ \Wbar_R(N)\le C(N+1)^k$.
\end{itemize}
The inner minimum lets the prover choose the best derivation for free. As the directive
says, this is a legitimate lower-bound strategy: every procedure that outputs an
$R$-derivation pays at least $W_R(F)$. The quantity $\Wbar_R$ measures proof
\emph{existence}. By Cook--Reckhow, ``$\Wbar_R$ superpolynomial for \emph{every} $R$''
is equivalent to $\NP\neq\mathrm{coNP}$. Only B3 connects proof existence to
arbitrary \emph{algorithms}. This report studies B3 alone.

\begin{definition}[Relativized and algebrized bridge]
For an oracle $O$ let
\[
\mathrm{B3}_R^{O}:\quad \NP^{O}\subseteq\Pc^{O}\ \Longrightarrow\ \exists C,k\ \forall N:\ \Wbar_R(N)\le C(N+1)^k .
\]
Since $R$ is oracle-free, the consequent \emph{does not depend on $O$}. Following
Aaronson--Wigderson, for an oracle $A$ and a low-degree extension $\tilde A$, let
$\mathrm{B3}_R^{A,\tilde A}$ be the same statement with hypothesis
$\NP^{\tilde A}\subseteq\Pc^{A}$.
A proof of B3$_R$ \emph{relativizes} if it proves B3$_R^{O}$ for every $O$.
It \emph{algebrizes} if it proves B3$_R^{A,\tilde A}$ for every pair $(A,\tilde A)$.
\end{definition}

\section{Results}

\begin{theorem}[PROVED/EXACT: oracle refutation of the bridge]\label{thm:rel}
Let $R$ be any oracle-free system satisfying LB$_R$. Then $\mathrm{B3}_R^{\TQBF}$ is false,
and $\mathrm{B3}_R^{\TQBF,\widetilde{\TQBF}}$ is false, where $\widetilde{\TQBF}$ is the
multilinear extension over finite fields. Consequently B3$_R$ has no relativizing and no
algebrizing proof.
\end{theorem}
\begin{proof}
Baker--Gill--Solovay give $\Pc^{\TQBF}=\NP^{\TQBF}=\PSPACE$, so the hypothesis of
$\mathrm{B3}_R^{\TQBF}$ holds. Its conclusion is the oracle-independent statement
``$\Wbar_R$ is polynomially bounded'', which LB$_R$ refutes. So the implication is false.
For algebrization, $\widetilde{\TQBF}$ is computable in $\PSPACE$ (Aaronson--Wigderson,
using the arithmetization of Babai--Fortnow--Lund/Shamir). Hence
$\NP^{\widetilde{\TQBF}}\subseteq\PSPACE=\Pc^{\TQBF}$ and the hypothesis holds again.
The conclusion is unchanged and false.
\end{proof}

\begin{corollary}[PROVED/EXACT dichotomy for every fixed $R$]
For every oracle-free $R$, either the program fails at LB$_R$ (because $\Wbar_R$ is
polynomially bounded), or every proof of B3$_R$ is non-relativizing and
non-algebrizing. There is no third case.
\end{corollary}

\begin{remark}[Operational red-team test]
Suppose a proposed proof of B3$_R$ only uses that the decider $A$ is correct, total and
polynomial-time, and turns its run into $R$-steps one transition at a time. Then the same
proof works for $A$ with oracle gates, because a query to $O$ is just one more transition
whose answer is taken as correct. By Theorem~\ref{thm:rel} such a proof is false whenever
LB$_R$ is true. This explains the corpus evidence that uncertified rejecting traces are
insufficient (\texttt{[P2-L5-409]}, \texttt{[P2-B7]}). It also explains why the identity system
$Q_A$ of LSPV-5/409 does not help: relative to $\TQBF$, $Q_A^{O}$ is a valid
polynomially bounded oracle proof system that no LB-satisfying $R$ can simulate.
A viable bridge must use the \emph{code} of $A$, for example a proof of its correctness in
a fixed theory that $R$ can reflect. This matches every known sufficient condition
(Theorem~\ref{thm:cook}, and Pich--Santhanam in LSPV-1/406).
\end{remark}

\begin{proposition}[PROVED/EXACT: B3 is all of P$\neq$NP for weak systems]\label{prop:weak}
If LB$_R$ is \emph{proved}, then B3$_R\iff\mathrm{SAT}\notin\Pc\iff\Pc\neq\NP$.
Moreover $\Pc\neq\NP\iff\exists R\,(\mathrm{LB}_R\wedge\mathrm{B3}_R)$.
\end{proposition}
\begin{proof}
The consequent of B3$_R$ is the negation of LB$_R$. So with LB$_R$ the implication is
equivalent to the negation of its hypothesis. Cook--Levin gives $\mathrm{SAT}\in\Pc\iff\Pc=\NP$.
For the second claim: ($\Leftarrow$) LB$_R\wedge$B3$_R$ excludes $\mathrm{SAT}\in\Pc$.
($\Rightarrow$) Take $R=$ Resolution. Haken (1985) gives $2^{\Omega(h)}$ lower bounds on
$\mathrm{PHP}^{h+1}_h$, whose encoding has $N=h^{O(1)}$ bits, so LB$_{\mathrm{Res}}$ holds.
B3$_{\mathrm{Res}}$ then holds vacuously when $\Pc\neq\NP$.
\end{proof}
The same holds for every system with a known unconditional superpolynomial bound:
Cutting Planes (Pudlák 1997), bounded-depth Frege (Ajtai; Krajíček--Pudlák--Woods;
Pitassi--Beame--Impagliazzo), and PC/Res(PC) over any field (Razborov; Impagliazzo--Pudlák--Sgall;
Pang 2026 for Res(PC$_r$), as already used in \texttt{[P2-B3 evidence, LSPV-1/482]}).
For each of these, the ``weakest sufficient bridge'' is exactly as hard as the full problem.
In world-ledger terms it is an \emph{equivalent of $\Pc\neq\NP$}, a reformulation and not
progress.

\begin{theorem}[KNOWN THEOREM, Cook 1975; see Krajíček 1995, Ch.~9]\label{thm:cook}
Suppose some polynomial-time algorithm $A$ decides SAT and PV proves the universal
($\Pi^b_1$) statement ``if $A$ rejects $F$ then no assignment satisfies $F$''. Then every
UNSAT $F$ has Extended Frege refutations of size polynomial in $|\langle F\rangle|$.
Hence B3$_R$ holds, with this strengthened hypothesis, for every $R$ that polynomially
simulates EF.
\end{theorem}
\textit{Sketch.} The propositional translations of a PV-provable $\Pi^b_1$ formula have
polynomial-size EF proofs. Instantiate the translation at $F$, add the polynomial-size
EF evaluation of $A$'s rejecting computation on the fixed input $F$, and conclude
$\neg F$. This argument uses a proof \emph{about the code} of $A$, so it does not relativize,
as Theorem~\ref{thm:rel} requires.

\begin{corollary}[CONDITIONAL on the corpus simulation claim]
LSPV-1/406 records an exact-original-source EF$\to$ER$\to$RES+DEF simulation inside
$R_{\mathrm{mix341}}$. If that simulation is polynomial, then
(a) \texttt{[P2-B1]} (LB for $R_{\mathrm{mix341}}$) implies superpolynomial ER/EF lower bounds,
which have been open since Cook--Reckhow 1979. Strong cryptographic evidence suggests feasible
interpolation does not apply to EF (Krajíček--Pudlák 1998, Bonet--Pitassi--Raz 2000).
(b) \texttt{[P2-B2]} (B3 for $R_{\mathrm{mix341}}$) holds under ``$\Pc=\NP$ via a
PV-provably correct algorithm''. The remaining B3 gap is therefore exactly the gap between
$\Pc=\NP$ and \emph{feasibly provable} $\Pc=\NP$.
\end{corollary}
