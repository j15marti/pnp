\noindent\textbf{Status summary.} P versus NP remains \textbf{OPEN}. This report proves
no lower bound and no transfer theorem. It locates where the work sits: for every fixed
calculus $R$, at least one half of Route~B ([P2-B1] lower bound, [P2-B2]/B3 bridge) is
either exactly the original problem or a long-standing open problem in proof complexity.
Granting a free optimal derivation sequence does not touch this obstruction.

\section{Objects}
For a fixed sound and complete propositional calculus $R$ with polynomial-time checkable
proofs (Cook--Reckhow), let
\[
\overline W_R(N)=\max_{\substack{F\in\mathrm{UNSAT}\\|\langle F\rangle|\le N}}\
\min_{\pi:F\vdash_R\bot}\operatorname{cost}_R(\pi),
\qquad
\mathrm{B3}(R):\quad \mathrm{SAT}\in\mathrm P\ \Longrightarrow\ \overline W_R(N)\le\mathrm{poly}(N).
\]
The minimum over $\pi$ already grants free knowledge of the optimal sequence, as in
[P2-D2]. Route~B is the conjunction
$\mathrm{LB}(R)\wedge\mathrm{B3}(R)\Rightarrow \mathrm{P}\ne\mathrm{NP}$, where
$\mathrm{LB}(R)$ says that $\overline W_R$ is superpolynomial ([P2-B1]). This step is trivial:
$\mathrm{LB}(R)$ contradicts the conclusion of $\mathrm{B3}(R)$, so $\mathrm{SAT}\notin\mathrm P$.

\section{Results}

\begin{theorem}[PROVED/EXACT, elementary: bridge collapse for lower-bounded $R$]\label{thm:collapse}
If $\mathrm{LB}(R)$ is an unconditional theorem, then $\mathrm{B3}(R)$ is logically
equivalent to $\mathrm P\ne\mathrm{NP}$.
\end{theorem}
\begin{proof}
($\Rightarrow$) Combine $\mathrm{B3}(R)$ with $\mathrm{LB}(R)$ as above.
($\Leftarrow$) If $\mathrm P\ne\mathrm{NP}$, the antecedent $\mathrm{SAT}\in\mathrm P$ is false,
so the implication holds vacuously.
\end{proof}

\noindent The following calculi satisfy $\mathrm{LB}(R)$ on strict 3-CNF (KNOWN THEOREMS):
ordinary DAG Resolution (Chv\'atal--Szemer\'edi 1988 random 3-CNF; Urquhart 1987 and
Ben-Sasson--Wigderson 2001 for Tseitin on expanders, $2^{\Omega(N/\log N)}$ in encoded bits);
Polynomial Calculus and PCR over any field on random 3-CNF (Alekhnovich--Razborov;
Ben-Sasson--Impagliazzo); bounded-depth Frege for fixed depth (Ajtai; Pitassi--Beame--Impagliazzo;
Kraj\'{\i}\v{c}ek--Pudl\'ak--Woods; formulas of non-constant width need a splitter);
the distinct fixed $\mathrm{Res}(\mathrm{PC}_r/\mathbb F_p)$ of [P2-B3] and [P2-L5-424]
(Pang arXiv:2610.00837, conditional on that external theorem). For each of these $R$, a
``weaker'' bridge does not exist: proving $\mathrm{B3}(R)$ \emph{is} proving
$\mathrm P\ne\mathrm{NP}$, and no lower-bound evidence for $R$ makes $\mathrm{B3}(R)$ easier.

\begin{theorem}[KNOWN THEOREM, Cook 1975; Kraj\'{\i}\v{c}ek 1995 Ch.~9: a provable bridge for strong $R$]\label{thm:cook}
Let $R\ge_p\mathrm{EF}$, i.e.\ $R$ p-simulates Extended Frege. Suppose $A$ is a polynomial-time
SAT decider whose soundness on UNSAT answers,
\[
\forall F\,\forall a\ \bigl(A(F)=\textsc{unsat}\rightarrow \neg\,\mathrm{Sat}(F,a)\bigr),
\]
is provable in Cook's theory $\mathrm{PV}$ (equivalently, in $S^1_2$ for this
$\forall\Pi^b_1$ form). Then $\overline W_R(N)\le\mathrm{poly}(N)$.
\end{theorem}
\begin{proof}[Proof sketch (standard)]
The propositional translation of the PV proof gives polynomial-size EF proofs of
$\|\mathrm{Sound}_A\|_n$ for each length $n$. For an UNSAT input $F$ with $A(F)=\textsc{unsat}$,
substitute the bits of $F$. Then derive the value of the circuit for $A$ on $F$, which takes
polynomial-size EF proofs by evaluating gate by gate with extension variables, and modus ponens gives $\neg F$.
Completeness of $A$ ensures $A(F)=\textsc{unsat}$ for every UNSAT $F$. Then p-simulate EF in $R$.
\end{proof}

\begin{proposition}[PROVED/EXACT finite basis check + KNOWN THEOREM: ER2 is EF-strong]\label{prop:er2}
In the distinct signed Boolean calculus ER2 (charged binary RES plus globally fresh charged
two-input signed AND definitions $q\leftrightarrow(\ell_a\wedge\ell_b)$ on \emph{distinct} input
IDs with three defining facets, per [P2-L5-448]), every one of the 16 binary Boolean functions
is the signed literal of a circuit with at most 3 such gates. XOR and XNOR need 3, the
constants need 2 because $\ell\wedge\neg\ell$ is illegal, and the rest need at most 1. Each
gate's facets $(\neg q\vee\ell_a),(\neg q\vee\ell_b),(q\vee\neg\ell_a\vee\neg\ell_b)$ are exactly
its CNF. Hence every Tseitin-style binary extension of Extended Resolution is simulated with
constant overhead, so $\mathrm{ER2}\equiv_p\mathrm{ER}\equiv_p\mathrm{EF}$
(Cook--Reckhow 1979; Kraj\'{\i}\v{c}ek). The script
\path{research/agents/claude/bridge-dichotomy/verify_er2_gate_basis.py} checks the
gate counts and facets exhaustively. It runs in under 1\,s, so no chunking was needed, and its
output is \path{verify_er2_gate_basis_results.json}.
\end{proposition}

\begin{corollary}[Bridge dichotomy: calibration, not progress]\label{cor:dichotomy}
For every fixed $R$:
\begin{enumerate}
\item If $\mathrm{LB}(R)$ is known (Theorem~\ref{thm:collapse}), then $\mathrm{B3}(R)\equiv\mathrm P\ne\mathrm{NP}$.
\item If $R\ge_p\mathrm{ER2}$ (for example ER2 itself, or any completion of $R_{\mathrm{mix341}}$
that admits ER2-style fresh AND definitions with RES), then $\mathrm{LB}(R)$ implies
superpolynomial Extended Frege lower bounds for 3-CNF. That is a central open problem
since Cook--Reckhow 1979, and it would separate EF from p-boundedness, i.e.\ give a
superpolynomial lower bound for the strongest standard proof system. In this regime
$\mathrm{B3}(R)$ is \emph{already true} for every PV-verifiable polynomial-time SAT
algorithm (Theorem~\ref{thm:cook}). The bridge's only open case is a polynomial-time SAT
algorithm whose correctness $\mathrm{PV}$ cannot prove.
\item In the intermediate regime (for example general Res($\oplus$), for which no superpolynomial
size lower bound is known; see Bhattacharya--Chattopadhyay arXiv:2507.23008 for depth
$O(N^{2-\epsilon})$ and Byramji--Impagliazzo arXiv:2511.20023), both halves are open.
Theorem~\ref{thm:cook} does not apply unless $R\ge_p\mathrm{EF}$.
\end{enumerate}
Contrapositive of Theorem~\ref{thm:cook} (KNOWN): $\mathrm{LB}(\mathrm{EF})$ implies that
\emph{no} polynomial-time SAT algorithm has PV-provable soundness and completeness.
\end{corollary}

\begin{remark}[What the free optimal sequence does and does not buy]
Minimizing over all $\pi$ gives a valid lower bound on executing any $R$-derivation, and it
correctly ignores search cost. But an arbitrary SAT algorithm need not execute an
$R$-derivation at all. The whole gap between Route~B and P vs NP is the
algorithm$\to$certificate transfer B3, which the free-sequence relaxation leaves untouched.
LSPV-5/409's identity system $Q_A$ ([P2-B2]) is the exact form of this gap.
Theorem~\ref{thm:cook} identifies the one known mechanism that closes it: provability of
$A$'s soundness in a theory whose proofs translate into $R$.
\end{remark}

\section{Implications for the five assignments}
\begin{itemize}
\item \textbf{Lower-Bound Lead (LSPV-1):} Choose $R$ knowing the dichotomy. A B3 effort for
Resolution, PC or $\mathrm{Res}(\mathrm{PC}_r/\mathbb F_p)$ is a direct P$\ne$NP effort.
For EF-strong $R$, the target is ``SAT algorithms with PV-unprovable soundness''. This links to
reflection-style bridges already in the corpus (LSPV-1/336, /337, /353), so it is not new.
\item \textbf{Reduction Builder (LSPV-2):} Settle whether the historical $R_{\mathrm{mix341}}$ DEF rule
reaches ER2. If it does, [P2-B1] is at least as hard as EF lower bounds (case 2). If it does not,
state which case-3 system it is.
\item \textbf{Hard families (LSPV-3):} Families that are hard for Resolution, PC or $\mathrm{Res}(\mathrm{PC})$
(Tseitin, random 3-CNF, PHP) Tseitin and PHP have polynomial EF/ER proofs: Tseitin via Gaussian elimination
(already polynomial in Frege), PHP by Cook 1976. So they cannot witness $\mathrm{LB}(R)$ for
EF-strong $R$. Random 3-CNF is a candidate only, since short EF refutations are neither known nor excluded. The literature has no explicit candidate family with a proved
superpolynomial EF lower bound.
\item \textbf{Resource Auditor (LSPV-4):} Theorem~\ref{thm:cook} has polynomial overhead that depends on
$A$'s running time and the length of its PV proof. Charging is consistent with [P2-D3].
\item \textbf{Red Team (LSPV-5):} A shortcut check for any claimed $\mathrm{LB}(R)$ family:
test whether it has short ER2 proofs (Gaussian elimination, counting, Cook PHP). Any such
proof falsifies hardness for EF-strong $R$.
\end{itemize}
