\begin{align}

\begin{theorem}[Coupled fixed-point FLMV certificate, conditional on (1)--(2)]
Put $\lambda=1/2+\xi$, $\zeta=\max\{\eta,\xi\}$ and
\[
K_*=\max\left\{11/10+\pi,\;3/5+\theta+\tau,\;
\frac{3/5+\theta+\nu}{1/2-\xi}\right\},\qquad
L_*=1/2+\zeta.
\]
For every finite guest exponent pair $K_0,L_0$, repeated application of the \emph{hypothesized} hereditary compiler satisfies
\[
K_j\le K_*+\lambda^j\max(0,K_0-K_*),\quad
L_j\le L_*+\lambda^j\max(0,L_0-L_*).
\]
The fixed point enters some strict FLMV Theorem~5.7 forbidden box with $c\ge6/5$ exactly when
\[
\boxed{
\zeta<1/90,\quad
11/10+\pi<c_\zeta,\quad 3/5+\theta+\tau<c_\zeta,\quad
\theta+\nu+c_\zeta\xi<(c_\zeta-6/5)/2, }\tag{3}
\]
where $c_\zeta\in[6/5,1.20557)$ is the unique nearby root of
\[
c_\zeta^3+(1+2\zeta)c_\zeta^2-c_\zeta-2=0.
\]
Boundary equality does not give entry into FLMV. Equation (3) characterizes the \emph{formal recurrence-certificate route}, not the actual unconstructed $\Phi$.
\end{theorem}
\begin{proof}
For $\lambda<1$, substitution in (1)--(2) gives the asserted geometric upper bounds by induction. For $6/5\le c\le c_0$, the complete FLMV envelope is
\[
d_*(c)=d_2(c)=\frac{2+c-c^3}{2c^2},
\]
where $c_0$ is the root of $c^3+c^2-c-2=0$ near $1.2055694$. For $c>1$ in this interval, the first two competitors are below $d_2$ since
\[
d_2-d_1=\frac{(c-1)^2(c+2)}{2c^2}>0,\qquad
d_2-d_3=\frac{(c-1)(4-c^3-c^2)}{6c^3}>0.
\]
For every $k\ge4$, $d_k\le c^{-k}\le c^{-4}<d_2$, since
\[
d_2-c^{-4}=\frac{(c-1)(c+1)(2-c^3)}{2c^4}>0
\]
on this interval. Thus this also certifies the \emph{entire infinite tail}. Moreover $d_*(6/5)=23/45=1/2+1/90$, and $d_2(c)$ decreases strictly to $1/2$ at $c_0$. For each $\zeta<1/90$, $L_*<d_*(c)$ precisely for $6/5\le c<c_\zeta$; the strict fixed-point time bound $K_*<c_\zeta$ is equivalent to the last three inequalities in (3), after multiplying the third candidate by $1/2-\xi$. One can then choose $c$ strictly between $\max(K_*,6/5)$ and $c_\zeta$. If any inequality fails, no such $c$ exists. \end{proof}
