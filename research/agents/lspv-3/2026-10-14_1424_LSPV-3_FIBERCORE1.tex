\section{CONSTRUCTION OR PROOF ATTEMPT}
Let $P$ be an explicitly represented binary CSP with each domain $D_v\subseteq\{0,1,2\}$ and permitted-pair relation $R_{uv}$. This is the actual BINARYPROJ1/MERGEGUARD1 postprojection kernel, not an inequality-only graph. Put $U=\{v:|D_v|=3\}$, $t=|U|$. A fiber $P[e]$, $e\in\{0,1,2\}^{U}$, excludes color $e_v$ at each $v\in U$; every restricted domain has at most two colors, so it becomes a 2-CNF $\Phi_e$ in polynomial time, with each forbidden source pair recorded.

\begin{theorem}[Source-grounded failed-fiber cylinder]
For every $e$ such that $P[e]$ is unsatisfiable, a polynomial-time algorithm produces $S\subseteq U$, the restriction $\rho=e|_S$, and a polynomial-length source-checkable contradiction such that
\[
\forall f\in\{0,1,2\}^{U}:\quad f|_S=\rho\Longrightarrow P[f]\in\mathsf{UNSAT}.
\]
\end{theorem}
\begin{proof}
Replace each two-color source variable by a Boolean choice. A forbidden source pair becomes a clause $(u\ne a)\vee(v\ne b)$, with fixed singleton values simplified, of width at most two. An empty clause gives an immediate contradiction. Otherwise an unsatisfiable 2-CNF has a variable $z$ with implication paths $z\leadsto\neg z$ and $\neg z\leadsto z$. Take simple paths, each of at most $2b-1$ edges for $b$ Boolean variables, and retain their clause origins. Define $S$ to contain every original three-color variable mentioned by a retained source clause. If $f|_S=e|_S$, then every variable appearing in the retained clauses has the same two-choice or singleton domain as before; all retained clauses and implication paths remain valid. Hence $P[f]$ is unsatisfiable. Translation, origins, paths and excluded labels are checked in polynomial time. 
\end{proof}

\begin{theorem}[Exact union with no polynomial-size guarantee]
Let $\rho_e$ be one certified pattern for each failed fiber $e$, and let $B_P$ be the union of their cylinders. Then
\[
B_P=\{e:P[e]\in\mathsf{UNSAT}\},\quad
P\in\mathsf{UNSAT}\iff B_P=\{0,1,2\}^U.
\]
\end{theorem}
\begin{proof}
Each cylinder is sound by the first theorem, while every failed fiber belongs to its own learned cylinder. A source satisfying assignment $c$ yields $2^t$ good exclusions with $e_v\ne c_v$ for all $v\in U$. This is exact, but enumerating all bad fibers may cost $3^t$.
\end{proof}

\begin{theorem}[Abstract width-three global-cover barrier]
Given an explicit list of cylinders, each fixing at most three coordinates of $\{0,1,2\}^t$, deciding if their union covers the whole cube is coNP-complete.
\end{theorem}
\begin{proof}
An uncovered vector is an NP witness for noncoverage. For a Boolean 3-CNF $F$ introduce singleton forbidden patterns $e_i=2$ for each variable and one forbidden pattern per clause fixing all its variables to their falsifying Boolean values. An uncovered ternary vector exists if and only if it is a Boolean satisfying assignment of $F$. The reduction is polynomial.
\end{proof}
\noindent This is a theorem about \emph{arbitrary} cylinder lists. It does not imply that all such lists are realizable as certificates from this source CSP class, nor that global coverage checking is compulsory for source decision.

\begin{theorem}[Source-aware replay gap]
For every $t\ge2$ there is an $O(t)$-size CSP with an immediate polynomial witness and a bad fiber whose extracted implication-path certificate records all $t$ excluded-color variables, although one exclusion is a complete source-level explanation.
\end{theorem}
\begin{proof}
Pin $x_0=0$, give $x_1,\ldots,x_t$ the full ternary domains and constrain each $x_{i-1}=x_i$. The unique source solution is all zero. The fiber is satisfiable exactly when no excluded $e_i$ equals zero. For $e=(1,\ldots,1,0)$ the semantic bad pattern is simply $e_t=0$. The restricted 2-SAT contradiction propagates along all $t$ equality edges; if any edge is dropped, zero on the pinned component and two on the other component satisfy the restricted source. Thus the declared path extractor records all $t$ edges and their ternary endpoints. This is an extractor limitation, not an arbitrary-algorithm lower bound.
\end{proof}

\begin{theorem}[Exact witness-overlap moment]
Let $W$ be the set of distinct full-three-variable projections of source witnesses and $m(e)=|\{w\in W:\forall i,\ e_i\ne w_i\}|$. Then
\[
\sum_e m(e)=|W|2^t,\qquad
\sum_e m(e)^2=\sum_{w,w'\in W}2^{t-d_H(w,w')}.
\]
For $W\ne\varnothing$ the uniform success probability is at least
\[
\frac{|W|^2(2/3)^t}{\sum_{w,w'\in W}2^{-d_H(w,w')}}.
\]
\end{theorem}
\begin{proof}
One projected witness is compatible with $2^t$ exclusions. Two witnesses share $2^{t-d_H}$ exclusions. Count first and second moments and apply Cauchy--Schwarz to nonzero $m(e)$.
\end{proof}
