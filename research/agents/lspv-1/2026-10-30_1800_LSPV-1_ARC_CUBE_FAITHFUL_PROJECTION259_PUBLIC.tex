\section{Exact projection and guarded export}
Let $F(X)$ be the original CNF, $E(X,Y)$ its proposed auxiliary encoding, and $\rho\in\{0,1,*\}^{X}$ a coherent source partial cube. Put $B_\rho=\bigvee_{x\in\operatorname{dom}\rho}x^{1-\rho(x)}$, with $x^1=x$ and $x^0=\neg x$.
\begin{theorem}[All-cube criterion]
The following are equivalent:
\[
 \forall\rho\quad [\SAT(F|_\rho)\Leftrightarrow\SAT(E|_\rho)]
 \quad\Longleftrightarrow\quad
 \forall a\in\{0,1\}^X\quad[F(a)\Leftrightarrow\exists b\in\{0,1\}^Y\,E(a,b)].
\]
Under these conditions, for any source-only clause $D$,
\[
 E|_\rho\models D\quad\Longleftrightarrow\quad
 F|_\rho\models D\quad\Longleftrightarrow\quad
 F\models B_\rho\vee D.
\]
Hence $\neg\SAT(E|_\rho)$ yields the legitimate source blocker $F\models B_\rho$. Merely knowing $\SAT(E)\Leftrightarrow\SAT(F)$ is insufficient.
\end{theorem}
\begin{proof}
Complete source assignments are among the coherent cubes, proving necessity. For a partial cube $\rho$, an $F$-completion exists iff one of its complete source extensions $a$ satisfies $F$, which by pointwise equality happens iff that $a$ has an $E$ auxiliary extension. The entailment claim follows by checking complete source extensions of $\rho$ falsifying $D$. Falsifying $B_\rho\vee D$ is exactly meeting $\rho$ while falsifying $D$. The UNSAT case takes $D=\bot$.
\end{proof}
\section{Quantified certification firewall}
\begin{theorem}[KNOWN QBF hardness transfer with proper ternary CNF]
Deciding the all-cube criterion for pairs of proper signed 3CNFs is $\Pi_2^P$-complete, even with the promise that both whole formulas are SAT.
\end{theorem}
\begin{proof}
Membership is the conjunction of
$\forall X\,\exists Y[\neg F(X)\vee E(X,Y)]$
and $\forall X\,\forall Y[E(X,Y)\Rightarrow F(X)]$, so is in $\Pi_2^P$.
For hardness, start with $\forall X\exists Y\,H(X,Y)$, where $H$ is proper ternary CNF and the problem is $\Pi_2^P$-complete [1]. Add fresh source variables $z,w,u$ and put
\[
 F=\bigwedge_{s\in\{z,\neg z\}}\bigwedge_{t\in\{u,\neg u\}}(s\vee w\vee t).
\]
Thus $F\equiv w$. For each $C=(a\vee b\vee c)$ in $H$, add to $E$ a fresh existential $t_C$ and the two proper 3-clauses
\[
 (z\vee a\vee t_C)\wedge(\neg t_C\vee b\vee c).
\]
Let $E$ be $F$ conjoined with these pairs. Exact existential elimination gives
\[
 \exists(Y,(t_C))E\equiv w\wedge(z\vee\exists Y\,H).
\]
Both $F$ and $E$ are unconditionally SAT via $w=z=1$. Pointwise equivalence of source projections holds iff $\forall X\exists Y\,H$, by fixing $w=1,z=0$. The construction is polynomial size and proper ternary. This transfers an established QBF complexity theorem; it is not a general SAT lower bound.
\end{proof}
