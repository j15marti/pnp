\section*{Exact excess identity for arbitrary explicit fan-in}
\begin{theorem}[Source-ancestry excess identity]\label{thm:weighted}
Let F be any UNSAT CNF and $\pi$ a \emph{pruned} valid sound finite refutation DAG, so all nodes are ancestors of the final contradiction. Suppose $\pi$ has r distinct used original clause sources, q distinct charged nodes, $k_v$ distinct named premise nodes for each charged v, and nonnegative charged weights $w_v$. Write $E=\sum_vk_v$ and $\beta=E-(r+q-1)$. Then $r\ge\mu(F)$ and $\beta\ge0$, and
\[
\boxed{\sum_vw_v-(\mu(F)-1)
=(r-\mu(F))+\beta+\sum_v[w_v-(k_v-1)].}
\tag{1}
\]
If each preselected rule has $w_v\ge k_v-1$, the three right-side components are nonnegative and every proof costs at least $\mu(F)-1$. This includes any explicitly charged fan-in and shared DAG, but not hidden source reads outside named premises.
\end{theorem}
\begin{proof}
The undirected dependency graph on $r+q$ nodes and E edges is connected, so its cyclomatic excess $\beta=E-(r+q-1)$ is nonnegative. Soundness and conservative definitions require the r used original source clauses to be UNSAT, so $r\ge\mu(F)$. Algebra gives
$\sum_v[w_v-(k_v-1)]=\sum_vw_v-E+q$, while
$E-q=r-1+\beta$. Substitution yields (1). This refers to one actual DAG and does not charge discovery or assert arbitrary-machine work.
\end{proof}

\begin{corollary}[Exact binary-unit proof-defect budget]\label{cor:binary}
In fixed Rmix341 with $k_v\in\{0,1,2\}$ and $w_v=1$, let u be the number of paid nodes with one named proof predecessor and $d_0$ the number with none. Then
\[
\boxed{q-(\mu(F)-1)=(r-\mu(F))+\beta+u+2d_0.}
\tag{2}
\]
Thus a refutation with total excess $e=q-\mu+1$ contains at most $\lfloor e/2\rfloor$ zero-premise VARDEF nodes, at most e unary conversions and cycle rank at most e, regardless of adaptivity, sharing and temporary enlargement.
\end{corollary}
\begin{proof}
With unit weights the local term in (1) is $2-k_v$, equal respectively to 2, 1 or 0 for premise count 0, 1 or 2.
\end{proof}

\begin{corollary}[The first two near-floor layers]
For a clause-minimally UNSAT F with m clauses, $\mu(F)=m$ and soundness requires all m originals in the contradiction ancestry, hence $r=m$. If $q=m-1$, the proof is a full binary source-once tree. If $q=m$, no paid zero-premise VARDEF can occur, and the proof topology must satisfy exactly one of $(u,\beta)=(1,0)$ or $(0,1)$. If $q=m+1$, there may be at most one paid VARDEF; if present it consumes both excess units and hence $u=\beta=0$. These are necessary \emph{not sufficient} syntactic conditions.
\end{corollary}
