\section{Main theorem: the sharp all-multigraph bound}
\begin{theorem}[PROVED/EXACT]
For every finite loopless cubic multigraph $H$ with $s$ vertices, parallel edges and disconnected components permitted,
\[
\boxed{\nu(H)\ge\left\lceil 2s/5\right\rceil.}
\]
The coefficient is sharp on a connected graph of ten vertices.
\end{theorem}
\begin{proof}
By the Tutte--Berge identity,
\[
s-2\nu(H)=\max_{S\subseteq V(H)}(o(H-S)-|S|).
\]
Fix $S$, let $z=|S|$, let $a$ count singleton odd components of $H-S$, and let $b$ count odd components of size at least three. A singleton odd component has exactly three edges crossing to $S$ because the graph is cubic and loopless. Every larger odd component has an odd degree-sum, hence at least one crossing edge. Consequently $3a+b\le3z$, while $s\ge z+a+3b$. Thus
\[
\begin{split}
s-5(o(H-S)-z)
&\ge(z+a+3b)-5(a+b-z)\\
&=6z-4a-2b\ge 2a\ge0.
\end{split}
\]
Therefore $o(H-S)-z\le s/5$ for every $S$, yielding $\nu(H)\ge 2s/5$, and integrality gives the ceiling.
For sharpness make a central vertex $o$ and three triples $(a_i,b_i,c_i)$, with two parallel edges $a_ib_i$ and single edges $a_ic_i,b_ic_i,oc_i$. There are ten vertices and fifteen edges. Deleting $o$ leaves three odd components and so maximum matching deficiency at least two. A matching of four edges exists, hence $\nu=4=2s/5$.
\end{proof}
\begin{corollary}[Exact algorithm; PROVED/EXACT]
Every valid signed exact-$(2,2)$ source admits a uniform deterministic exact SAT decider with
\[
\boxed{T(F)=O(m\log(m+2)+\delta^4+
2^{\lfloor3\delta/5\rfloor}(n+m+1))}
\]
word operations and $O(n+m)$ peak reusable words.
\end{corollary}
\begin{proof}
Construct a maximum matching of $\nu(H_F)$ edges using polynomial general-graph matching. Select original source variables labeling those matched edges; these hit $2\nu$ distinct unmatched clauses. For each of the other $\delta-2\nu$ clauses select an arbitrary incident source variable. De-duplicate selected variable names. Their number $h$ obeys
\[
h\le\nu+(\delta-2\nu)=\delta-\nu
\le\lfloor3\delta/5\rfloor.
\]
Enumerate assignments to the $h$ sources, simplify $B\wedge T$, and decide each resulting width-at-most-two formula by SCC. SAT in a branch iff the original contracted formula is SAT. A model can be lifted and replayed on the original formula. Every failed branch, input read, matching step, reconstruction, and check is charged. Sorting costs $O(m\log(m+2))$, conservative deterministic matching $O(\delta^4)$, each SCC branch $O(n+m+1)$. No decider is required to use this algorithm.
\end{proof}
