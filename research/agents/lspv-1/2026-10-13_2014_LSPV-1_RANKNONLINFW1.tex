\section{Exact projected-rank identity and enumeration escape}
Write $C(X)=\ker B_X$ for the binary cycle space and let $X-R$ retain the full
vertex set, hence also isolated vertices.

\begin{theorem}[Projected cycle-space rank]
For every finite undirected graph $X=(V,E)$ and $R\subseteq E$,
\[
 \dim\pi_R C(X)=|R|+\operatorname{cc}(X)-\operatorname{cc}(X-R).           \tag{4}
\]
\end{theorem}
\begin{proof}
The kernel of $\pi_R$ restricted to $C(X)$ is exactly the cycle space supported
on $E\setminus R$, namely $C(X-R)$.  Rank--nullity and
$\dim C(Y)=|E(Y)|-|V|+\operatorname{cc}(Y)$ give
\begin{align*}
\dim\pi_R C(X)
 &=\dim C(X)-\dim C(X-R)\\
 &=\bigl(|E|-|V|+\operatorname{cc}(X)\bigr)
   -\bigl(|E|-|R|-|V|+\operatorname{cc}(X-R)\bigr),
\end{align*}
which is (4).
\end{proof}

\begin{corollary}[Charged interface enumeration]
Assume $B_Xx=c$ is consistent and $Q$ reads only $x_R$ and is evaluable in
polynomial time.  Satisfiability of (1), and a witness when one exists, can be
found in
\[
 2^\rho\operatorname{poly}(N+|Q|)                          \tag{5}
\]
time and polynomial workspace, including elimination and reconstruction.
Consequently $\rho=O(\log N)$ is a uniform polynomial-time escape.
\end{corollary}
\begin{proof}
Gaussian elimination produces one solution $a$ and a basis for $C(X)$.  Its
projection gives an affine space $\pi_R(a)+\pi_R C(X)$ with exactly $2^\rho$
points.  Enumerate a basis-coordinate Gray code, evaluate $Q$ on each point,
and, on acceptance, solve
$B_{E\setminus R}x_{E\setminus R}=c+B_Rx_R$ to reconstruct the remaining
edges.  Each step is polynomial in the explicit input and circuit encoding.
If $Q$ has $h$ additional unconstrained Boolean variables, the honest bound is
$2^{\rho+h}\operatorname{poly}(N+|Q|)$, not (5).
\end{proof}

\section{High-rank nonlinear polynomial escape}
Let $X_t$ be the friendship graph of $t$ triangles sharing one central vertex.
It has $2t+1$ vertices, $3t$ edges, is connected, and has a width-two tree
decomposition with one bag per triangle.  Put $R=E$ and
\[
 Q_t(x)=\mathbf 1[|x|\le1].                               \tag{6}
\]

\begin{theorem}[\texttt{RANK-NONLINEARITY-FIREWALL1}]
The family $(X_t,Q_t)$ satisfies all of the following.
\begin{enumerate}
\item $\rho=t=\Theta(|E|)$.
\item $Q_t^{-1}(1)$ is not affine for $t\ge1$ with $|E|\ge2$.
\item For every charge vector $c$, one uniform algorithm decides (1) and
returns a witness in $O(|E|(|V|+|E|))$ bit operations under a direct incidence
scan, using polynomial workspace.
\end{enumerate}
Thus linear projected rank and a genuinely nonlinear residual do not imply
superpolynomial witness-search work.
\end{theorem}
\begin{proof}
Deleting all edges leaves $2t+1$ isolated vertices, so (4) gives
$\rho=3t+1-(2t+1)=t$.  The accepted set in (6) contains $0,e_i,e_j$ but not
$e_i+e_j$, hence is not affine.  The algorithm tests $x=0$ and then each unit
vector $e_i$, computing $B_Xx$ and comparing it with $c$.  These are all and
only the vectors of weight at most one, so the algorithm is correct for YES
and NO inputs.  Parsing, candidate generation, incidence evaluation, and
output are included in the displayed polynomial bound.
\end{proof}

The counterexample survives edge/vertex renaming, edge-order shuffling,
COPY/DEAD metadata, and interpreter wrappers after charging their polynomial
parsing.  Independent products remain easy by the same global candidate list.
It is deliberately a bounded-treewidth control: high cycle rank need not mean
large treewidth, and neither invariant alone is arbitrary-machine work.

\section{Exact conditional amplification lemma}
The redirected sufficient target can be stated without mistaking a percentage
loss for an exponent.

\begin{theorem}[Growing-factor amplification]
Let $s_j\to\infty$, $s_{j+1}\le A s_j$ for fixed $A\ge1$, and $H_j>0$.  If
for fixed $B>0$ and a function $r(s)\to\infty$,
\[
 H_{j+1}\ge H_j\,r(s_j)/B,                                \tag{7}
\]
then for every fixed $d$, $H_j/s_j^d\to\infty$ along the tail.  In particular
$H_j$ is superpolynomial in $s_j$ on this family.
\end{theorem}
\begin{proof}
Fix $d$.  Since $s_j\to\infty$ and $r(s)\to\infty$, choose $J$ such that
$r(s_j)/B\ge2A^d$ for all $j\ge J$.  Iterating (7) and the size inequality,
\[
 \frac{H_{J+k}}{s_{J+k}^d}
 \ge \frac{H_J(2A^d)^k}{(A^ks_J)^d}
 =2^k\frac{H_J}{s_J^d}\longrightarrow\infty.
\]
\end{proof}

This theorem is \EXACT{} as an implication, but application to UWI is
\COND{} because no verified construction proves (7) with $r(s)\to\infty$.
A fixed multiplier or fixed percentage does not dominate $A^d$ for arbitrary
$d$ and therefore cannot be reported as exponent amplification.
