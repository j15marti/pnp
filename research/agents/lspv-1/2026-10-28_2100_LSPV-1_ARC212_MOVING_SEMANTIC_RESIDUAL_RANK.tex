\section{Independent static-interface audit and its limited scope}
\begin{proposition}[Source-static coverage, inherited and checked]\label{prop:static}
Let a connected, proper strict signed exact-$(2,2)$ 3CNF have $n$ variables. Suppose an authenticated fixed global boundary $Y$ has $b>0$ names, there are $k$ nonempty disjoint private modules, and every original clause meets $Y$ or one of the private per-module syntactic clause-hitting sets $B_i$ of size at most $q$. Then
\[
 k\le 4b,\qquad n\le 3\Bigl(b+\sum_{i=1}^k |B_i|\Bigr)\le3b+12bq,
 \qquad b+q\ge\sqrt{n/3}-1/4.
\]
With $b=0$, connectedness gives $k=1$ and $q\ge n/3$.
\end{proposition}
\begin{proof}
Every nonempty private component has a clause-incidence connection to $Y$, consuming at least one of its $4b$ variable occurrences, proving $k\le4b$. The union $H=Y\cup\bigcup_i B_i$ hits every clause. The formula has $4n/3$ clauses because each clause uses three distinct names and each variable occurs exactly four times. A name in $H$ hits at most four clauses, so $4n/3\le4|H|$, proving the second inequality. Put $x=b+q$ and note
\[
 n\le 3b+12bq=12b(x+1/4-b)\le3(x+1/4)^2.
\]
The last inequality follows by completing the square. If $b=0$, the connected source has one private module and the hitting bound $n\le3q$ applies. All conclusions are about this specific static \emph{syntactic} decomposition, not runtime or moving semantic states.
\end{proof}

\section{Exact residual functions on a connected grid}
\noindent For an even integer $L\ge2$, let $G_L$ be the undirected $L\times L$ square grid (without wraparound), with $n=L^2$ vertices and $2L(L-1)$ edges. Define the \emph{monotone 2CNF Boolean function}
\[
 \Phi_L(x)=\bigwedge_{\{u,v\}\in E(G_L)}(x_u\vee x_v).
\]
The graph is connected with maximum degree $4$. The source is \emph{not} a strict signed exact-$(2,2)$ 3CNF, a necessary model firewall. Evaluating $\Phi_L$ on arbitrary assignments is not the same problem as deciding whether $\Phi_L$ has \emph{some} satisfying assignment.

\begin{lemma}[Every balanced grid cut has linearly many edges]\label{lem:cut}
For every set $S\subseteq V(G_L)$ with $|S|=n/2$, at least $L/2$ grid edges have one endpoint in $S$ and the other in $\overline S$.
\end{lemma}
\begin{proof}
For each row $i$, let $a_i\in\{0,\ldots,L\}$ be its number of vertices in $S$. Call a row mixed when $0<a_i<L$. If at least $L/2$ rows are mixed, each such row contributes a horizontal crossing edge. Otherwise there must exist a full row: without a full row, fewer than $L/2$ rows could contain any selected vertex, giving $|S|<(L/2)L=n/2$. By the symmetric complement argument there must also be an empty row. A full and an empty row imply
$\sum_{i=1}^{L-1}|a_{i+1}-a_i|\ge L$.
Each difference is a lower bound for the vertical crossing edges between its consecutive rows, proving the claim.
\end{proof}

\begin{lemma}[Greedy cross-cut induced matching]\label{lem:matching}
In a graph of maximum degree $\Delta$, any set of $E_\times$ cut edges contains a matching of at least $E_\times/(2\Delta^2)$ edges whose $2q$ endpoints induce exactly those $q$ edges in the original graph.
\end{lemma}
\begin{proof}
Choose a remaining cut edge $uv$ and delete all remaining cut edges incident to $N[u]\cup N[v]$, where $N[\cdot]$ is a closed neighborhood. Since $u\in N[v]$ and $v\in N[u]$, $|N[u]\cup N[v]|\le2\Delta$, so at most $2\Delta^2$ cut edges are deleted at each step. The chosen edges have no endpoint adjacency across distinct chosen pairs, hence constitute an induced matching. At least $E_\times/(2\Delta^2)$ choices occur.
\end{proof}

\begin{theorem}[Every-order complete-residual explosion, restricted]\label{thm:residual}
For every variable ordering $\pi$ of $\Phi_L$, there is a prefix of length $n/2$ after which the Boolean function has at least
\[
 \boxed{2^{L/64}=2^{\sqrt n/64}}
\]
distinct nonfalse residual functions on the remaining variables. Consequently every reduced ordered binary decision diagram, under every ordering, has at least that many width states at the balanced layer. Yet $\Phi_L$ is visibly satisfiable by the all-one assignment and its SAT decision is trivial.
\end{theorem}
\begin{proof}
Take $S$ to be the first $n/2$ variables of $\pi$. Lemma~\ref{lem:cut} gives at least $L/2$ crossing edges. Lemma~\ref{lem:matching} with $\Delta=4$ gives an induced matching $(u_i,v_i)$ of size $q\ge L/64$, with $u_i\in S$ and $v_i\notin S$.

For each subset $I\subseteq\{1,\ldots,q\}$, set $u_i=0$ iff $i\in I$, and set \emph{every other} variable in $S$ to $1$. Because chosen left endpoints are pairwise nonadjacent, every all-left clause is satisfied. The residual is a nonfalse monotone function on $\overline S$ comprising unchanged right-right clauses and unit requirements on right neighbors of the zeroed selected left vertices.

If $I\ne J$, choose $i\in I\triangle J$, say $i\in I\setminus J$. In the matching, $v_i$ is not adjacent to any selected $u_j$ with $j\ne i$. Complete the unassigned right variables by setting $v_i=0$ and all other right variables to $1$. That completion satisfies the residual for $J$, but violates the edge clause $(u_i\vee v_i)$ in the residual for $I$. Thus the $2^q$ residual Boolean functions are distinct and all satisfiable. An OBDD merging two such prefix histories could not produce both distinct suffix truth functions; hence the claimed width bound follows. This is an OBDD/function-evaluation restriction only. The all-one valuation witnesses satisfiability immediately.
\end{proof}

\section{A constructive polynomial-value rank theorem}
\begin{theorem}[Fully charged one-child polynomial-rank descent]\label{thm:rank}
Let $\mathcal C$ be a decidably encoded instance class containing a uniformly polynomial-time reduction of ordinary SAT, with an answer-preserving embedding into $\mathcal C$. Suppose there are fixed constants $a\ge1,c,d,e>0$, one fixed deterministic answer-oblivious algorithm $G$, a decidable base rule $B$, and a nonnegative integer rank $\rho$ such that for every initial instance $x\in\mathcal C$ of length $N$:
\begin{description}
\item[] (i) Every state $x_j=G^j(x)$ is in $\mathcal C$, all states along the actual trajectory have physical encoded length $|x_j|\le O(N^a)$, and $\rho(x)\le O(N^d)$ as a \emph{value}, not merely a binary description length.
\item[] (ii) When $x_j$ is not a base, $G$ creates exactly one fully materialized child $x_{j+1}$, preserves both YES and NO answers, and strictly decreases rank $\rho(x_{j+1})<\rho(x_j)$. Selection, parsing, source authentication, failed trials, memory access, representation changes, validation, and full physical output together cost $O((|x_j|+1)^c)$ time and $O((|x_j|+1)^e)$ space on one fixed declared machine model.
\item[] (iii) All rank-zero states are bases and the base procedure decides their true answer in $O((|x_j|+1)^c)$ time and $O((|x_j|+1)^e)$ space, without a SAT-answer oracle. There is no advice, free answer-class bit or uncharged second recursive call.
\end{description}
Then a deterministic decider for $\mathcal C$ uses
\[
 \boxed{T(N)=O(N^{d+ac}+N^{ac}),\qquad
 S(N)=O(N^{a\max(1,e)})}
\]
up to fixed encoding constants, and SAT belongs to P. Temporary intermediate \emph{growth} is permitted by hypothesis (i); neither $N^\lambda$ one-step shrink nor monotonically decreasing encoded length is required.
\end{theorem}
\begin{proof}
Rank is a nonnegative integer and decreases by at least one per nonbase step, so there are at most $O(N^d)$ such steps on the complete actual trajectory. Every reached encoding has length $O(N^a)$, and each charged step costs $O(N^{ac})$ time. A rank-zero base is reached, giving $O(N^{d+ac})$ total time. A tail-iterative implementation need retain only the current and next encoded states plus per-step workspace, using $O(N^a+N^{ae})$ space and no recursion stack. Each step preserves the answer on both branches of the \emph{one fixed algorithm}; the base decision therefore equals the initial decision. Composition with the polynomial SAT embedding is polynomial. No separate SAT/UNSAT solver or free answer-class selector is used.
\end{proof}

\begin{proposition}[Binary-rank loophole]\label{prop:counter}
Replacing the rank-value bound $\rho(x)\le N^d$ by a bound $|\mathrm{bin}(\rho(x))|\le N^d$ does not ensure polynomial total iterations.
\end{proposition}
\begin{proof}
For an $N$-bit input initialize an integer counter to $2^N-1$ using $O(N)$ bits. Each update decrements it and retains the same input, taking $\operatorname{poly}(N)$ bit operations and no asymptotically larger encoding. There are $2^N-1$ updates. Each strict decrease is genuine, yet the rank's \emph{value} is exponential in $N$. The counter is not a SAT-solving mechanism; it is a valid falsifier of the proposed runtime inference.
\end{proof}

\section{Model-native accounting and the five-role dependency DAG}
\noindent The rank theorem is deliberately model-parametric: it is valid \emph{separately} when its charged costs are proved on a single shared read/write tape (input counted in peak footprint), on a fixed-$k$ multitape machine with a separate read-only input (distinct visited work cells and access traffic), on native two-fully-writable tapes (charge both tapes' symbol changes), or on indexed RAM. Numerical exponents cannot be transferred between these models without a charged simulation. The conservative sweep simulation of a fixed-$k$ read-only-input multitape run with time $N^{t_m+o(1)}$ and visited workspace $N^{s_m+o(1)}$ yields a single-tape upper \emph{simulation} exponent $t_1\le t_m+\max(1,s_m)$ for the specified construction; this is not a lower bound, and does not retain the original time-space profile. Other models need their own simulation ledgers. A fixed universal interpreter's program parsing, common input layout, program-code length and interpreter overhead are also charged before applying any theorem.
