\section{Four falsifiable mechanisms declared before literature and corpus inspection}
\begin{description}
\item[] \textbf{H1 (proved as derived restricted lemma).} Split all runs at $T=N^2$: above it the desired time disjunction holds vacuously; below it ordinary physical work-head travel ensures $\log(S+2)=O(\log N)$. Thus the polynomial-time premise in the \emph{write--time} corollary can be removed, not in an unrelated RAM theorem.
\item[] \textbf{H2 (proved as a derived restricted lemma).} Source-cut injective coding still excludes exponentially many diagonal YES inputs when $D$ reaches a \emph{constant fraction} of $N/\log^2N$, a regime stronger than $D\le N^{1-\epsilon}$.
\item[] \textbf{H3 (falsified as model assertion).} Zero changed work symbols implies zero space. A one-head work-tape \emph{position} is an $m$-state unary counter even with blank contents. Every visited cell counts; a physical comparator has $D=0$, $S=m+1$ and $T=O(mN)$.
\item[] \textbf{H4 (falsified as transfer).} The same dichotomy holds on a two-input-head machine or indexed RAM. Two independent source heads pair the two bit shores in linear time with no work writes; random indexed probes have no compulsory head crossing.
\end{description}

\section{Exact model and quantified source}
Let $H(a,b)$ be the authenticated proper signed exact-$(2,2)$ 3CNF
component with 15 variables, 20 distinct clauses and model counts
$(772,272,272,0)$ for $ab=00,01,10,11$.
For $m$ copies place all Alice-dependent clauses first,
$16m$ invariant clauses in a middle interval, and all Bob-dependent
clauses last. Let $F_m(a,b)$ be the physical fixed-width input encoding,
$Y_a=F_m(a,\overline a)$, $N=N_m=\Theta(m\log(m+2))$ its \emph{formula}
bit length, and $G=G_m=\Theta(N)$ the number of invariant-gap
source cuts. Then $\SAT(F_m(a,b))\iff a\land b=0^m$;
every $Y_a$ is satisfiable, and for $a\ne a'$ at least one crossed
source $F_m(a,\overline{a'})$ or $F_m(a',\overline a)$ is unsatisfiable.

Fix \emph{one} deterministic finite-state interpreter $U$ with one
two-way sequential read-only input head and fixed $k\ge1$
initially blank read/write work tapes, each with one unit-move head,
a finite alphabet and detectable immutable left end.
Every eligible selectable program $p$ is placed within the same
fixed-width, read-only inline prefix \emph{left} of every gap cut,
and is globally SAT-sound:
\[
\forall p\in\mathcal P_m\;\forall F\in\UNSAT:\quad
 U(p\#F)\text{ does not accept}.
\tag{1}
\]
Programs can vary with $a$ and be selected with hindsight;
they need not accept other SAT inputs.
The program slot may be long, but $N$ always means \emph{formula}
bits, not $N+b$ bits of whole program-plus-source.
For an accepting run let $T$ be actual transitions,
$S$ total distinct visited work cells (including blank),
$D$ actual work-symbol changes (including blank restorations),
$H$ input head moves, $R$ reversals, $L$ emitted output,
all on this \emph{one} computation. In particular,
$S\le k(T+1)$; all parsing, strategy selection, failures,
source reads, children, output, caches and retained ancestor states
are charged. No second input/proof head, indexed jump, uncharged
output-head coordinate or initially nonblank advice is available.

\section{Existing chronicle injection, restated exactly}
At one source cut record all actual changes
$\mathrm W(j,x,\sigma)$ and all crossings
$\mathrm C(\mathrm{direction},\mathrm{full\ finite\ state},h_1,\ldots,h_k)$
in chronological order (work changes before a simultaneous crossing).
The initial blank configuration and the work-change log reconstruct
the complete work contents at every crossing.
Equal codes for accepted diagonal sources, \emph{even with different
left-inline programs}, splice into an accepted UNSAT crossed source
and violate (1). Therefore all qualifying codes are distinct.

By counting the $G$ cuts choose for each run one traversed at most
$\lfloor T/G\rfloor$ times. The established ARC205 inequality gives,
for any nonnegative common ceilings $T_0,D_0,S_0$,
\[
\boxed{
|\mathcal A_m(T_0,D_0,S_0)|\le
G\!\left(2^{\,B+1}-1\right),\quad
B=C_U(D_0+\lfloor T_0/G\rfloor+2)
       (\lceil\log_2(S_0+2)\rceil+1),
}
\tag{2}
\]
where $\mathcal A_m$ is the set of diagonal indices $a$
having \emph{some} globally sound accepting $p$ and one
run jointly satisfying the three ceilings. $C_U$ depends
only on fixed interpreter/tapes. There is no extra
$2^{|p|}$ factor. Equation (2) is the only non-elementary
input of the following theorem, and its independent
referee status is \emph{pending}.

\begin{theorem}[critical-log disjunction, no advance polynomial ceiling]
\label{th:critical}
For each fixed $U,k$ and source encoding above, there exist
$\delta,\tau>0$ and $m_0$ such that for every $m\ge m_0$,
with $\Lambda=\log_2N_m$,
\[
\left|\left\{a:\exists p\in\mathcal P_m\;\;
U(p\#Y_a)\downarrow\mathrm{accept},\;\;
D\le\delta\,\frac{N}{\Lambda^2},\;\;
T\le\tau\,\frac{N^2}{\Lambda^2}
\right\}\right|\le2^{m/2}.
\tag{3}
\]
Here both resource inequalities must hold on the \emph{same
accepting run}. Hence on at least $(1-2^{-m/2})2^m$
distinguished satisfiable sources, every globally sound accepting
inline program must either make
$D>\delta N/\Lambda^2$ actual changes or
take $T>\tau N^2/\Lambda^2$ transitions.
The theorem does \emph{not} assume $T\le N^K$ for any advance $K$.
\end{theorem}
\begin{proof}
Choose constants $\alpha,\beta>0$ and $m_1$ with
$m\ge\alpha N/\Lambda$ and $G\ge\beta N$
for $m\ge m_1$, possible by the physical encoding.
Let $T_0=\tau N^2/\Lambda^2$,
$D_0=\delta N/\Lambda^2$, and $S_0=k(T_0+1)$.
The estimate (2) applies with these \emph{joint ceilings}.
For all sufficiently large $m$, $T_0\le N^2$.
Thus $S_0\le k(N^2+1)$ and
\[
\lceil\log_2(S_0+2)\rceil+1
\le C_k\Lambda,\qquad
B\le C_UC_k\left(\delta+\frac{\tau}{\beta}\right)
                \frac{N}{\Lambda}+O(\Lambda).
\tag{4}
\]
Choose fixed positive $\delta,\tau$ so that
$C_UC_k(\delta+\tau/\beta)<\alpha/4$.
The $O(\Lambda)$ term, $\log_2G$ and fixed
encoding overhead are $o(N/\Lambda)$.
For sufficiently large $m$,
$B+1+\log_2G\le m/2$, and (2) yields (3).
Neither the numerical value of $K$ nor a running-time
bound on unqualified accepted computations was used.
\end{proof}

\begin{corollary}[no-$K$ version of the strict-exponent implication]
\label{co:nok}
For any $\epsilon>0$, if every accepted run in a proposed
globally sound covering inline portfolio has
$D\le N^{1-\epsilon}$, then on at least
$(1-2^{-m/2})2^m$ diagonal YES sources
its accepting runs have
$T>\tau N^2/\log_2^2 N$ for all large $m$.
More generally, one cannot obtain all-input coverage with
both $D=o(N/\log^2N)$ and $T=o(N^2/\log^2N)$
in this restricted model.
\end{corollary}
\begin{proof}
$N^{1-\epsilon}=o(N/\log^2N)$, so every indicated run
falls below the $D$ threshold in Theorem~\ref{th:critical}.
The conclusion applies whether the other runs are
polynomial, superpolynomial, or nonterminating. A run
with $T>N^2$ already satisfies the asserted time
alternative and requires no estimate of $\log S$.
\end{proof}
