\section{Pairwise-Helly theorem}
\begin{lemma}[Majority closure]
For every 2-CNF $B$, coordinatewise majority of any three models of $B$ is again a model of $B$.
\end{lemma}
\begin{proof}
If the majority falsifies a two-literal clause, at least two of three assignments falsify each of its literals. These sets intersect, contradicting that all three assignments satisfy the clause. Units are immediate.
\end{proof}
\begin{theorem}[Exact pairwise unit extension]
For every satisfiable 2-CNF $B$ and literal multiset $U=(u_1,\ldots,u_s)$,
\[
B\land\bigwedge_i u_i\in SAT
\ \Longleftrightarrow\
\forall i,j\in[s],\quad B\land u_i\land u_j\in SAT.
\]
Repetition $i=j$ tests singleton feasibility.
\end{theorem}
\begin{proof}
For $s\le2$, the claim is immediate. Suppose it holds below size $s\ge3$. By induction, for $a=1,2,3$ there is a model of $B$ satisfying all literals except possibly $u_a$. Majority of these three models satisfies $B$, and each $u_i$ holds in at least two models. Thus the majority satisfies every unit. Necessity is trivial.
\end{proof}

\section{Exact choice graph}
Create one choice vertex $(j,a)$ per literal $\ell_{ja}$ of $T_j$. Delete the choice if $B\land\ell_{ja}$ is UNSAT; join distinct-group choices when $B\land\ell_{ja}\land\ell_{hb}$ is UNSAT.
\begin{theorem}[Independent transversal]
For every such $F$, including shared signed source variables,
\[
F\in SAT\quad\Longleftrightarrow\quad
G_B\text{ has an independent transversal selecting one available choice per group}.
\]
There is a uniform source-checked compiler with at most
$1+3k+9\binom{k}{2}$ SCC-based 2-SAT calls,
$O((1+k^2)(n+m+1))$ word time, $O(n+m+k^2)$ peak words and $O(k^2)$ output records.
\end{theorem}
\begin{proof}
A model selects one true literal from each ternary, whose singleton and pair tests succeed. Conversely, pairwise-compatible selected literals have a common extension satisfying $B$ by the preceding theorem, and so satisfy all ternaries. Test each unary/pair combination with a charged SCC call, including source rereads and unsuccessful tests. One extra SCC call under the selected units reconstructs a full witness if requested.
\end{proof}

Graph three-colorability occurs exactly: for each vertex $v$, put $T_v=(x_{v,0}\lor x_{v,1}\lor x_{v,2})$, and for each edge $uv$ and color $c$ put $(\neg x_{u,c}\lor\neg x_{v,c})$ in $B$. Selecting compatible choices is precisely three-coloring. This NP-complete endpoint is KNOWN THEOREM and does not prove deterministic exponential time.

\section{Exact first interaction and charged amortization}
Take $k=2$, $n=6$ with $T_1=(x_1\lor x_2\lor x_3)$, $T_2=(y_1\lor y_2\lor y_3)$, and nine binary clauses $(\neg x_a\lor\neg y_b)$ for all $a,b\in[3]$.
\begin{theorem}[Nine-field SAT-bit obstruction]
The complete formula is clause-minimally UNSAT; each one-ternary marginal is SAT. For each binary clause encode the second signed-literal word as either $\neg y_b$ (active conflict) or $x_a$ (inactive tautology). On the all-active UNSAT formula, every exact deterministic source-word decider must read all nine independent fields, since changing an unread field produces an equal-length SAT instance. For the declared one-field-at-a-time query calculus,
$\rho_t=9-\#\{\mathrm{distinct\ conflict\ fields\ read}\}$ drops by at most one per probe, and nine probes are necessary and sufficient.
\end{theorem}
\begin{proof}
Any model needs one true $x_a$ and one true $y_b$, forbidden by their binary clause. Deleting that clause allows those two choices with all others false. Deleting a ternary allows its entire group false and the other group nonempty. Any single inactive field therefore changes UNSAT to SAT. Unread-word indistinguishability proves the lower bound.
\end{proof}
The source itself contains nine independent fields; this is a constant-size, restricted acquisition lower bound, not an exponential time theorem. For a complete table service over $k$ groups, $9\binom{k}{2}$ pair fields are independently selectable using optional negative binary clauses, giving a tight input-order table-probe bound. That stronger service is not required of a SAT-bit decider.

\begin{countertheorem}[No per-candidate-work lower bound]
Initially there are $3^k$ literal-choice selections with no conflicts. Adding one binary clause $(\neg x_{1,1}\lor\neg x_{2,1})$ excludes exactly $3^{k-2}$ selections at one constant-size source-record update. The resulting CNF remains SAT for all $k\ge2$.
\end{countertheorem}
\begin{proof}
Exactly the selections taking choice 1 in both first groups are removed, leaving arbitrary choices in the other $k-2$ groups. Selecting choice 2 in every group satisfies the new formula.
\end{proof}
Thus counting excluded witnesses and forcing one unit of work per excluded choice is FALSIFIED/RETIRED.
