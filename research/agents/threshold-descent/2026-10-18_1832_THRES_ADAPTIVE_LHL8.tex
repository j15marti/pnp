\section*{FORMAL PROOF}
Let $N=2^n$; let $X$ be a $k$-wise independent unbiased distribution on $N$ bits with support in the size-$Akn^2$ Boolean circuit class (uniform degree-less-than-$k$ polynomials on $\mathbb F_{2^n}$ followed by trace). Let $U$ be uniform on $N$ bits. $X$ has min-entropy at least $k$ because any fixed $k$ coordinates have a uniform distribution.
A randomized preprocessing phase observes at most $q\le k$ distinct source coordinates adaptively; transcript $T$ includes private preprocessing coins, indices and observed bits. After $T=t$, sample a \emph{fresh source-independent public seed} $R$ from a two-universal family $\mathcal H_t$ of functions $h_{t,R}:\{0,1\}^N\to\{0,1\}^B$. The family may depend on $t$ but its seed is independent of the remaining source bits. The output is $\Phi(V)=(T,R,h_{T,R}(V))$. The hash itself may process every source bit.
\begin{theorem}[ADAPTIVE-LHL-NOGO8]
\[
\boxed{\mathrm{TV}(\Phi(X),\Phi(U))
\le \frac12\,2^{(B+q-k)/2}
+\frac12\,2^{(B+q-N)/2}.}
\]
In particular, if $B+q\le k-\omega(1)$, no classifier, even unbounded, can distinguish these output distributions with constant advantage.
\end{theorem}
\begin{proof}
Condition first on the preprocessing coins. For each possible adaptive transcript $t$ with $\ell$ distinct observations, the transcript probability is exactly $2^{-\ell}$ under both $X$ and $U$, by $k$-wise independence. The conditional maximum point mass is at most $2^{\ell-k}$ for $X$ and $2^{\ell-N}$ for $U$ (Bayes' rule). For any conditional source $Z$ with min-entropy $h$, a two-universal hash satisfies
\[
\mathbb E_R\mathrm{Col}(h_{t,R}(Z))\le 2^{-B}+2^{-h};
\]
Cauchy--Schwarz gives the strong leftover-hash estimate
\[
\mathrm{TV}((R,h_{t,R}(Z)),(R,U_B))\le\tfrac12\,2^{(B-h)/2}.
\]
Apply this to both conditioned sources, use the triangle inequality, and average over the common transcript distribution; $\ell\le q$ yields the displayed bound. Data processing covers any subsequent target decision or randomized postprocessing.
\end{proof}
\begin{corollary}[Robust-MCSP promise-purity firewall]
Fix $0<\mu\le\nu<1$ and $0<\delta<1/2$; put $s=N^\mu$, $u=N^\nu$, $k=\lfloor s/(An^2)\rfloor$. Suppose a two-stage translator followed by any classifier has worst-case YES error $\alpha$ for all $CC(f)\le s$ and worst-case NO error $\gamma$ for all $d_H(f,\mathcal C_u)>\delta N$. Then, writing $H_2$ for binary entropy and $\eta_N=|\mathcal C_u|2^{-N(1-H_2(\delta))}=2^{-\Omega(N)}$,
\[
\boxed{
1-\alpha-\gamma-\eta_N
\le\tfrac12\bigl(2^{(B+q-k)/2}+2^{(B+q-N)/2}\bigr).
}
\]
Thus errors $\alpha=\gamma=1/3$ are impossible for large $N$ when $q+B\le k-\omega(1)$.
\end{corollary}
\begin{proof}
The chosen polynomial source $X$ lies entirely in source YES. Uniform $U$ is promised Hamming-far NO except with probability $\eta_N$ by a circuit-counting bound $\log|\mathcal C_u|=O(u\log(n+u))=o(N)$. The claimed translator/classifier would distinguish the sources with gap at least $1-\alpha-\gamma-\eta_N$, contradicting the theorem.
\end{proof}
