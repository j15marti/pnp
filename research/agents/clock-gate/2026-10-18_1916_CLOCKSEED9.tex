\section*{Definitions}
Let N=2^n, b>=1, t>=1, W_t={g in F_2^N:wt(g)=t}, L=binom(N,t), q=2^{-b}. Every g in W_t has a shared-negation point-DNF circuit of at most n+nt-1 gates. Let A be a random b-by-N binary matrix such that every nonzero F_2-linear character of its bN entries has absolute bias at most epsilon. Let alpha=(2^b-1)epsilon<1 and gamma=(2^{2b}-1)epsilon. The source f is fixed before and independent of A.
\begin{theorem}[SMALLBIAS-SECTION-COVER9, PROVED/EXACT]
For every fixed f in F_2^N,
\[
 \Pr_A[\nexists g\in W_t:Ag=Af]\le
 \min\left\{1,\frac{(1+\alpha)2^b}{L(1-\alpha)^2}
                   +\frac{\gamma+2\alpha}{(1-\alpha)^2}\right\}=:U.
\tag{9.1}
\]
For every fixed label y in F_2^b the probability that y is not attained by A(W_t) is also at most U. Consequently Pr_A[A(W_t) != F_2^b] <= 2^b U.
\end{theorem}
\begin{proof}
If f is in W_t, the first claim is trivial. Otherwise the nonzero vectors v_g=f+g are distinct, hence any two linearly independent over F_2. Put I_g=1[Av_g=0], X=sum_g I_g. Fourier expansion 1[Av=0]=2^{-b}sum_{u in F_2^b}(-1)^{u^TAv} shows q(1-alpha)<=E I_g<=q(1+alpha). For g!=h, nonzero (u,w) implies u v_g^T+w v_h^T is nonzero, so E I_g I_h <= q^2(1+gamma). Thus E X>=Lq(1-alpha) and Var X<=Lq(1+alpha)+L^2q^2(gamma+2alpha). Chebyshev at X=0 proves (9.1). Replacing I_g by 1[Ag=y] gives the same Fourier bounds because the weight-t words g are distinct nonzero vectors. Union-bound the 2^b possible labels.
\end{proof}
\begin{corollary}[EXPLICIT-SEED-SATURATION9, PROVED/EXACT with KNOWN THEOREM]
By the explicit Naor--Naor epsilon-biased generator, taking epsilon=2^{-4b} realizes the above matrix family from O(b+log(bN)) public seed bits, each matrix entry uniformly computable from its seed and index. Equations (9.1) and (9.2) imply
\[
 \Pr[\nexists g\in W_t:Ag=Af]\le(1+o(1))2^b/L+O(2^{-2b}),
\quad
 \Pr[A(W_t)\ne F_2^b]\le(1+o(1))2^{2b}/L+O(2^{-b})
\]
as b tends to infinity. For t=floor((s'-n+1)/n), s'=N^\tau with fixed 0<tau<1, log_2 L>=(1-tau-o(1))s'. Whenever b=Theta(s') and b<=(1-tau-eta)s', every fixed source YES or NO instance maps with overwhelming probability to an image label attained by a size-s' circuit. The stronger all-label statement holds for b<=(1-tau-eta)s'/2. These are failures of NO soundness for existential affine-hash-image semantics, not reductions to ordinary MCSP.
\end{corollary}
\section*{Single-vector marginal universality is insufficient}
\begin{theorem}[MARGINAL-NOGO9, PROVED/EXACT]
For b=1, odd t, and f=0^N, sample a uniform binary row with probability 1-p and the fixed all-ones parity row with probability p. Every nonzero v satisfies
\[
 \Pr[Av=0]\in[(1-p)/2,(1+p)/2],
\]
but Pr[nexists g in W_t:Ag=Af]>=p regardless of L=binom(N,t).
\end{theorem}
\begin{proof}
The parity-forced row maps every odd-weight g to one and f=0 to zero, giving no section collision. For even-weight nonzero v its collision probability on the forced branch is one; for odd-weight v it is zero. Mixing with a uniform row yields the displayed marginal interval. This does not claim avoidance of the entire small-circuit image because f=0 is itself cheap.
\end{proof}
\begin{theorem}[OUTPUT-SUPPORT-SEED-FLOOR9, PROVED/EXACT]
If a uniform r-bit public seed yields A_R v within total-variation distance kappa<1 of uniform on F_2^b for even one fixed nonzero v, then r>=b+log_2(1-kappa). A matrix epsilon-biased as above gives kappa<=alpha/2. Thus strong near-uniform b=N^\tau outputs cannot use an N^{o(1)}-bit seed. The O(b+log N) seed is short relative to N but not subpolynomial in N.
\end{theorem}
\begin{proof}
The output distribution has at most 2^r support points, whereas any distribution supported on m points has total-variation distance at least 1-m/2^b from uniform. For an epsilon-biased matrix each output atom has probability in 2^{-b}[1-alpha,1+alpha], giving total-variation at most alpha/2.
\end{proof}
\begin{lemma}[TV-ADAPTIVE-GUARD9, PROVED/EXACT]
If the seed law Q_f selected after observing source f is within total-variation zeta of the original epsilon-biased seed law Q, its section miss probability is bounded by U+zeta. An unrestricted source-adaptive selector can avoid this theorem by strongly changing its seed law; efficient YES/NO preservation is not demonstrated.
\end{lemma}
