\section{CONSTRUCTION OR PROOF ATTEMPT}
Let $M(q)=8\binom q3$ and let $Z$ be uniform among exact-$t$ clause sets.

\begin{theorem}[Local-fingerprint escape]
If every positive $Y$ contains some r-clause pattern $P$ from a family $\mathcal P_q$, then
\[
D(z)=\bigvee_{P\in\mathcal P_q}\bigwedge_{i\in P}z_i
\]
is a monotone DNF accepting all positives and
\[
\Pr[D(Z)=1]\le |\mathcal P_q|\frac{(t)_r}{(M)_r}.
\]
For $t=\Theta(q)$, $M=\Theta(q^3)$ and $|\mathcal P_q|\le q^c$, this is $O(q^{c-2r})$.
\end{theorem}
\begin{proof}
A fixed r-pattern is contained in a uniform exact-t set with probability $(t)_r/(M)_r$; apply a union bound.
\end{proof}

\begin{corollary}[Tseitin/parity positives fail]
Every standard cubic Tseitin 3-CNF contains a complete four-clause parity bundle on some 3-variable support. There are only $2\binom q3$ possible bundles. At $t=q$, a polynomial monotone DNF accepts every such positive but a uniform exact-q formula with probability at most
\[
2\binom q3\frac{(q)_4}{(M)_4}=O(q^{-5}).
\]
Thus CRITICALPRISM/Tseitin formulas are unsuitable as the positive distribution for this approximation route, regardless of graph, charge, or relabeling.
\end{corollary}

\begin{lemma}[Exact first-marginal matching]

\[
\Pr_{\mu_+}[C\in F]=\Pr_{\mu_-}[C\in F]=t/M(q).
\]
\end{lemma}
\begin{proof}
Signed variable permutations act transitively on clause coordinates and preserve SAT/UNSAT and exact weight. Summing equal coordinate marginals gives t.
\end{proof}
