\section{IDEAS PROPOSED BEFORE LITERATURE}
\noindent\textbf{CANDIDATE A.} A small controller $c$ and an affine payload $x$ allow nonlinear switching by the observed projection $P_0x$. Close the rowspace of $P_0$ under all affine branch matrices, then perform binary affine composition indexed by observations. Missing premise: uniformly polynomial-size closure and source authentication. Falsifier: same observation, divergent next observation.
\par\noindent\textbf{CANDIDATE B.} Replace the full observation table by a reachable-orbit table with charged discovery. Missing: a polynomial orbit-size promise. Falsifier: exponentially many reachable observations.
\par\noindent\textbf{CANDIDATE C.} Guess an unrestricted two-output PFCERT from the one-bit output alphabet. Missing: efficient coverage and exact semantic soundness. Falsifier: PSPACE-hard binary-horizon circuit iteration; a small output alphabet alone does not aid verification.
\par\noindent\textbf{CANDIDATE D.} Test transfer to the current five-symbol beat-relation CSP's exact cofactor DP. Missing: source compiler preserving integer prefix multiplicities and low closure rank. Falsifier: same masks with different exact counts.

\section{CONSTRUCTION OR PROOF ATTEMPT}
\noindent\textbf{Source semantics.} Fix binary payload $x\in\mathbb F_2^m$, controller $c\in\mathbb F_2^k$, a full-row-rank switch-projection $P_0\in\mathbb F_2^{r_0\times m}$, and an explicit or fully enumerably evaluated complete branch table $(A_{c,z},b_{c,z},f(c,z))$ for all $c$ and $z\in\mathbb F_2^{r_0}$. Given source, initial state, binary horizon $T$ and terminal circuit $g$, the physical step is
\[
 z=P_0x,\qquad c'=f(c,z),\qquad x'=A_{c,z}x+b_{c,z}.
\]
The switch may be nonlinear in $(c,z)$, including products of observed bits. Hidden payload bits cannot be queried by the switch unless explicitly included in $P_0$.

\begin{lemma}[Minimal common invariant observation]
Let $W_0=\operatorname{row}(P_0)$ and $W_{j+1}=W_j+\sum_{c,z}W_jA_{c,z}$. The chain stabilizes after at most $m-r_0$ strict enlargements to the smallest common invariant rowspace $W_*$ containing $W_0$. If $P$ is an independent-row basis of $W_*$, then for each $(c,z)$ there is a unique $B_{c,z}$ such that $PA_{c,z}=B_{c,z}P$. These data are constructed by polynomial-time exact binary elimination in the complete branch-table size and $m$.
\end{lemma}
\begin{proof}
A strict subspace inclusion increases dimension, bounded by $m$. Inductively every common invariant subspace containing $W_0$ contains each $W_j$, proving minimality. Right multiplication of basis rows by any branch matrix remains in the stable span, and independence gives unique coordinates for $B$.
\end{proof}
Let $r=\dim W_*$ and $H$ satisfy $P_0=HP$. Then the observation $h=(c,y)$, $y=Px$, is \emph{closed}:
\[
F(c,y)=\bigl(f(c,Hy),\ B_{c,Hy}y+Pb_{c,Hy}\bigr).
\]
There are $Q=2^{k+r}$ possible observations.

\begin{theorem}[Exact adaptively switched endpoint]
For all authenticated sources in the above grammar and all $T\ge0$, the endpoint $(c_T,x_T)$ is computed exactly by source-derived affine binary-lifting tables in $O(Q\log(T+2))$ affine compositions. Conservative simultaneous ordinary bit-machine bounds are
\[
T_{\mathrm{eval}}=O(M^4),\qquad S_{\mathrm{peak}}=O(M),\qquad
M=N+Q\lceil\log_2(T+2)\rceil(m+k+1)^3,
\]
where $N$ includes the entire source, input, and binary horizon. Thus $k+r=O(\log N)$ makes both resources polynomial in the actual input size even for exponentially long physical executions.
\end{theorem}
\begin{proof}
Store for every observation $h$ an exact affine payload transformer $x\mapsto M_j(h)x+v_j(h)$ and endpoint observation $F^{2^j}(h)$. At $j=0$ use the source one-step table. If $h'=F^{2^j}(h)$, compose the two consecutive segments using
\[
M_{j+1}(h)=M_j(h')M_j(h),\quad
v_{j+1}(h)=M_j(h')v_j(h)+v_j(h'),\quad
F^{2^{j+1}}(h)=F^{2^j}(h').
\]
Closure ensures that the next observation depends only on $h$, not on the hidden payload. Induction proves correctness of every composed segment, and scanning set bits of $T$ starting at the current observation yields the full endpoint. All $Q\lceil\log_2(T+2)\rceil$ maps, source-circuit lookup, parsing, multiplication, switching, cache retention and full readout are charged in a single conventional sequential-tape realization. Even allowing an $O(M)$ scan for each operand access and branch lookup yields the deliberately conservative joint $O(M^4)$ time and $O(M)$ space upper bound.
\end{proof}
\noindent\textbf{Certificate scope.} When $k+r=O(\log N)$, a verifier can compute the unique exact terminal label $g(c_T,x_T)$ and accept precisely that label. Global accepted-output multiplicity is one for this RESTRICTED grammar, not a universal PFCERT for arbitrary short programs.

\begin{lemma}[Rank obstruction]
Set $P_0=e_1$ and let a branch matrix have rows $(e_2,\dots,e_m,0)$. Then every $e_j=e_1A^{j-1}$ belongs to $W_*$, so $r=m$. For $m=2$, initial payloads $00$ and $01$ have the same projected bit but distinct next projected bits. The small-closure assumption cannot simply be omitted.
\end{lemma}
This obstructs only this \emph{explicit linear-observation-table construction}. The shift program itself remains easy via a different representation and no lower bound for unrestricted work follows.

\noindent\textbf{Matched-oracle two-output barrier.}
Fix one and the same oracle $O$ for $U^O$, verifier $V^O$, randomness $K_U^{t,O}$, and the classes $P^O,NP^O$. Suppose $V^O(n,p,w,\pi)$ is polynomial-time with polynomially bounded witnesses, covers every $n$-bit endpoint of every $|p|\le n-2$ program running for at most $t(n)$, and globally accepts at most two outputs $w$ for each $(n,p)$ over all proofs. Define $E^O_n=\{w:\exists |p|\le n-2,\pi\ V^O(n,p,w,\pi)=1\}$. It lies in $NP^O$ and satisfies $|E^O_n|\le2(2^{n-1}-1)=2^n-2$. Its infinite complement consists of strings of $K_U^{t,O}(w)\ge n-1$. Hirahara's proof relativizes through an $O(\log n)$ suffix enumeration using the same oracle and therefore forbids an infinite $P^O$ subset of these random strings. With a Baker--Gill--Solovay collapsing oracle $A$, $P^A=NP^A$ makes the complement of $E^A$ an infinite $P^A$ subset, contradiction. Thus the universal PFCERT schema is impossible in that oracle world. The current affine-source result relativizes and supplies no escape. Nonalgebrization remains OPEN.

\noindent\textbf{Five-symbol interface.}
For $R(a,b)=[b-a\bmod5\in\{1,3\}]$, TMR-2's exact DP must retain prefix \emph{integer multiplicities} in addition to future-domain masks. On an edgeless $n$-vertex graph all partial assignments have identical future masks, while the exact answer is $5^n$. Mask-only affine transfer is insufficient. No $O(\log N)$ rank invariant or arbitrary-T5 compiler is claimed.
