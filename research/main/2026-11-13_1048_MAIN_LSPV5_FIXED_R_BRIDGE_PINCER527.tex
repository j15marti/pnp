\[

complexity (minimum proof size), and the free choice is legitimate for a lower bound.

\section{Theorem 1 (PROVED/EXACT): the fixed-$R$ bridge pincer}
\begin{theorem}
Let $R$ be any fixed Cook--Reckhow refutation system for 3-CNF whose cost is polynomially
related to proof length, and let ``$R$ p-bounded'' mean $\overline W_R(N)\le\operatorname{poly}(N)$.
\begin{enumerate}
\item $\mathrm B_R \iff (\mathrm P=\mathrm{NP}\Rightarrow R\text{ p-bounded})$.
\item If ``$R$ is not p-bounded'' is a theorem, then $\mathrm B_R\iff \mathrm P\neq\mathrm{NP}$.
 This applies to tree-like and DAG Resolution (Haken 1985; Urquhart 1987;
 Ben-Sasson--Wigderson 2001), Cutting Planes (Pudlák 1997), bounded-depth Frege
 (Ajtai; Beame--Impagliazzo--Pitassi; Krajíček--Pudlák--Woods), Polynomial Calculus over
 any field (Razborov; Alekhnovich--Razborov), and any $R$ weaker than one of these.
\item If $\mathrm B_R$ is proved and $R$ is not p-bounded is proved, P$\neq$NP follows; hence
 for every $R$ in item 2, proving $\mathrm B_R$ is \emph{exactly} as hard as P$\neq$NP.
 The bridge is a non-trivial reduction of the problem only for systems $R$ whose
 superpolynomial lower bound is itself open (Frege, Extended Frege, and the
 project's mixed $R_{\mathrm{mix341}}$, which contains paid extension and GF(2) rules).
\end{enumerate}
\end{theorem}
\begin{proof}
SAT$\in$P $\iff$ P$=$NP. Item 1 is the definition. Item 2: the conclusion of $\mathrm B_R$ is
false, so $\mathrm B_R$ reduces to $\neg(\mathrm{SAT}\in\mathrm P)$. Item 3 is modus tollens.
\end{proof}
\textbf{Ledger classification} (world ledger categories): for $R$ in item 2, $\mathrm B_R$ is an
\emph{equivalent of P$\neq$NP}, i.e.\ a reformulation, not progress. For $R\supseteq$ EF it is
a genuine split into two open halves: (H1) $\mathrm B_R$ and (H2) $\overline W_R$
superpolynomial. H2 for EF is the central open problem of proof complexity; no
superpolynomial lower bound is known even for Frege.

\section{Theorem 2 (KNOWN THEOREM, transferred): the weakest known sufficient bridge}
\begin{theorem}[Cook 1975; Krajíček--Pudlák 1990]
(a) If SAT is decided by a polynomial-time algorithm $A$ and $\mathrm{PV}\vdash
\forall F\,\forall a\,[A(F)=0\to\neg\mathrm{Sat}(F,a)]$, then EF is p-bounded on 3-CNF, so
$\overline W_{\mathrm{EF}}$ is polynomial.
(b) EF p-simulates every proof system $Q$ whose reflection principle $\mathrm{RFN}(Q)$ is
provable in $S^1_2$.
\end{theorem}
Consequently $\mathrm B_{\mathrm{EF}}$ (and $\mathrm B_R$ for every $R$ that p-simulates EF with
polynomial node cost) follows from the weaker-looking hypothesis
\[
\mathrm{H}_{\mathrm{PV}}:\quad \mathrm P=\mathrm{NP}\ \Longrightarrow\ \exists\text{ poly-time SAT decider whose UNSAT-soundness is PV-provable.}
\]
H$_{\mathrm{PV}}$ is \textbf{OPEN}.  It is exactly where the corpus's REFEREE294 Gödel firewall
bites: a correct decider can be built whose correctness is unprovable in any fixed theory
(search for an inconsistency proof first), so ``every decider'' cannot work; H$_{\mathrm{PV}}$
asks only for \emph{some} decider.  Levin universal search does not supply one: it certifies
YES answers by witnesses but has no checkable certificate for NO answers, so its
UNSAT-soundness is not PV-provable without a coNP certificate, which is what is being sought.
\textbf{Red-team verdict:} the coNP certification asymmetry is the irreducible core of
$\mathrm B_R$; no argument restricted to running time can cross it.

\textbf{Krajíček--Pudlák route.} P$=$NP $\Rightarrow$ E$=$NE (padding) $\Rightarrow$ a
p-optimal system $Q^\ast$ exists, and $Q^\ast$ is then p-bounded.  This gives
$\mathrm{P}=\mathrm{NP}\Rightarrow\exists R\ (R\text{ p-bounded})$ with $R$ depending on the
hypothesis, i.e.\ the quantifier order $\forall\text{world}\,\exists R$, not the order
$\exists R\,\forall\text{world}$ that $\mathrm B_R$ requires.  \textbf{FAILED TRANSFER} as a proof of
$\mathrm B_R$ for the pre-fixed $R_{\mathrm{mix341}}$; it remains useful because any pre-fixed $R$
that is p-optimal unconditionally would satisfy $\mathrm B_R$, and unconditional existence of
p-optimal systems is itself OPEN.
