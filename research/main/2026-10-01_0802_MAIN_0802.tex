\paragraph{Fixed-code stacked rank.}

\[
\boxed{\rho(H)=\operatorname{rank}_{\mathbb F_2}[B_v]_v,\qquad

\[
W_c=\operatorname{span}(\{\mathbf1\}\cup c(\mathbb F_q)),\qquad
