\fbox{\parbox{.94\linewidth}{\textbf{Executive finding.}
The continuous and integer geometry of the static connected-interface proxy
can be stated exactly, but it is neither a machine lower bound nor a
simultaneous time--space frontier.  In the active region the continuous time
exponent is strictly convex, with unique minimum
$b^*=\sqrt{n/12}$ and limiting value $\sqrt{n/3}-1/4$; the first equality of
the two \emph{log proxies} is the permitted point $b=\lceil n/3\rceil,q=0$.
For $n=1200$, $b=9,10,11$ all minimize the integer time exponent at 20, but
only $b=9$ is Pareto-optimal: the alleged convergence point $(400,400)$ is
dominated in both proxies.  More importantly, the $2^b$ space coordinate is
the cost of storing a boundary table.  The same exhaustive decision route can
stream assignments in $O(N+b+q)$ peak bits, so equality of the plotted proxies
does not imply equality of measured time and space.  Adaptive method envelopes
also require either summed running time or summed live state.  No general SAT
frontier, $P=NP$, or $P\ne NP$ follows.}}
\end{center}

\section{BUILD I: exact continuous proxy geometry}
Fix $n>0$ and define the permitted relaxed lower envelope
\[
 q_0(b)=\max\{0,n/(12b)-1/4\},\qquad
 t(b)=b+q_0(b),\qquad s(b)=b.
\]
These are logarithms of the stated enumeration and stored-table proxies,
up to additive $O(\log N)$ polynomial factors.  They are not measured machine
exponents unless an implementation with those simultaneous costs is given.

\begin{theorem}[Continuous permitted frontier]
In the active region $0<b<n/3$,
\[
 t(b)=b+\frac{n}{12b}-\frac14,quad
 t'(b)=1-\frac{n}{12b^2},quad
 t''(b)=\frac{n}{6b^3}>0. \tag{1}
\]
Thus $t$ has the unique minimum
\[
 b^*=\sqrt{n/12},\qquad
 t(b^*)=\sqrt{n/3}-\frac14. \tag{2}
\]
Since $s'(b)=1$, the tangent slope in the $(s,t)$ plane is
$dt/ds=1-n/(12b^2)$.  The first permitted equality $t=s$ occurs when the
truncation sets $q_0=0$, namely $b=n/3$, and thereafter $t=s=b$.
\end{theorem}
\begin{proof}
Differentiate the active expression.  Strict positivity of the second
derivative gives the unique stationary minimum.  Equality $t=s$ is equivalent
to $q_0=0$.
\end{proof}

Two asymptotic scales expose what the plot does and does not say.  If
$b=\alpha\sqrt n$ with fixed $\alpha>0$, then
\[
 \frac{t}{\sqrt n}\longrightarrow
 \alpha+\frac{1}{12\alpha},\quad
 \frac{s}{\sqrt n}\longrightarrow\alpha,quad
 \frac{dt}{ds}\longrightarrow1-\frac{1}{12\alpha^2}. \tag{3}
\]
If $b=cn$, then $t/n,s/n\to c$ and the truncated curve has slope one.
Under canonical $N=\Theta(n\log n)$, the proxy minimum is
\[
 2^{\Theta(\sqrt n)}=
 2^{\Theta(\sqrt{N/\log N})}, \tag{4}
\]
which is subexponential but superpolynomial.  The equality point has exponent
$\Theta(n)=\Theta(N/\log N)$ and is much larger, not efficient convergence.

\section{BUILD II: exact integer Pareto frontier}
For integer $b\ge1$, let
\[
 q_n(b)=\max\left\{0,
 \left\lceil\frac{n}{12b}-\frac14\right\rceil\right\},
 \quad P_b=(b+q_n(b),b). \tag{5}
\]
The two coordinates are respectively the log time proxy and log stored-table
space proxy.

\begin{proposition}[Discrete nondominance]
Because the second coordinate is strictly increasing in $b$, $P_b$ is
Pareto-nondominated exactly when $b+q_n(b)$ is a strict record low among
$1,\ldots,b$.  Equivalently, retain the points where the discrete time
sequence drops below all earlier values.  The first permitted equality is
$b=\lceil n/3\rceil$, but for all sufficiently large $n$ it is dominated by a
point near $\sqrt{n/12}$.
\end{proposition}
\begin{proof}
Any potential dominator has smaller $b$.  Such a point dominates precisely
when its time coordinate is no larger.  The equality threshold follows from
$n/(12b)-1/4\le0$.
\end{proof}

For $n=1200$, the integer time minimum is 20 at $b=9,10,11$, with
$q=11,10,9$.  Yet $P_{10}$ and $P_{11}$ have the same time and larger space
than $P_9$, so only $P_9=(20,9)$ is Pareto-optimal.  The record frontier is
\[
(101,1),(52,2),(37,3),(29,4),(25,5),(23,6),(22,7),(21,8),(20,9).
\]
The equality point $(400,400)$ is dominated by $(20,9)$.  This is the exact
correction to interpreting the visual convergence as an optimum.

\section{BUILD III: permitted versus realizable endpoints}
The inequality alone gives a necessary permitted region, not realization for
every integer pair.  The meaning of $q=0$ also depends on the precise hitting
definition.

\begin{proposition}[$q=0$ equality realization under residual hitting]
For every $b\ge3$, there is a connected proper strict signed exact-$(2,2)$
3CNF with $n=3b$, a boundary $Y$ of size $b$, one connected private module,
and every source clause containing exactly one member of $Y$.  Consequently
$Y$ hits all clauses and the residual private-only hitting number is $q=0$;
the equality $n=3b$ is attained.
\end{proposition}
\begin{proof}
Take $2b$ private vertices and the connected 4-regular circulant graph with
edges of cyclic differences one and two.  It has $4b$ edges.  Partition these
edges into $b$ groups of four; for each edge $uv$ in group $j$, create the
clause on $y_j,u,v$.  Give the four occurrences of each boundary variable two
positive and two negative signs.  Independently, order the four incident
edges at each private vertex and give its first two occurrences positive and
last two negative.  Every variable is exact-$(2,2)$, all clause scopes are
distinct and proper, and connectedness follows from the private circulant.
There are $b+2b=3b$ variables and $4b=4n/3$ clauses, all hit by $Y$.
\end{proof}

If instead the phrase ``$B_i$ hits the module'' requires $B_i$ itself to hit
even boundary-touching clauses, empty $B_i$ is invalid for nonempty modules.
Under that stronger definition the $q=0$ endpoint is not realizable.  Reports
must therefore state which convention is used; the inequality and a plot do
not settle this semantic fork.

\section{BUILD IV: the stored-table space falsifier}
\begin{theorem}[Streaming exhaustive evaluation]
For a fixed authenticated hitting set $H$ of size $h=b+q$, the brute-force
decision route can enumerate all $2^h$ assignments one at a time, simplify or
scan the explicit source, and discard each assignment before the next.  Its
time is $2^h\operatorname{poly}(N)$ and its extra live assignment/counter
space is $O(h+\log N)$ bits, or $O(N+h)$ when the explicit source footprint is
counted.  It need not store a $2^b$ boundary table.
\end{theorem}
\begin{proof}
A binary counter generates each assignment.  Evaluate the residual condition
by a charged source scan and retain only the counter, current clause/literal
indices, and the current verification state.  No earlier row is required for
a one-shot decision.  If witness output is requested, retain the first
successful assignment.  This proves an upper construction, not a lower bound.
\end{proof}

Thus $S_{\rm proxy}=2^b\operatorname{poly}(N)$ describes a memoized
boundary-table implementation or a reusable-summary contract.  It is not
mandatory peak space even for the same static enumeration architecture.  At
$n=1200,b=400,q=0$, the plotted equality compares time $2^{400}$ with an
optional $2^{400}$-entry table; a streaming implementation has the same
exponential time proxy but only polynomial live space.  The crossing is not a
machine $T=S$ theorem.

\section{BUILD V: charged adaptive envelopes}
Let $A_1,\ldots,A_m$ be deterministic partial certified methods run by one
controller, each either returning a checked answer or UNKNOWN.  Let
$t_j(x),s_j(x)$ include discovery, source access, failed internal branches,
re-encoding, joins and reconstruction.  Assume at least one method answers
every input in the declared class.

\begin{theorem}[Portfolio accounting]
If the controller runs methods sequentially to their certified budgets, then
\[
 T(x)\le D(x)+\sum_{j\ \mathrm{attempted}}t_j(x),\qquad
 S(x)\le S_D(x)+\max_j s_j(x). \tag{6}
\]
If it dovetails all $m$ methods while retaining their states, and method $j^*$
is the first answering method in its own step count, then
\[
 T(x)\le D(x)+m\,t_{j^*}(x)+O(m),\qquad
 S(x)\le S_D(x)+\sum_{j=1}^m s_j(x). \tag{7}
\]
Therefore the coordinatewise ideal envelope
$(\min_j t_j,\min_j s_j)$ is not generally a jointly achievable pair.
\end{theorem}
\begin{proof}
Equation (6) is direct summation with state reuse.  Round-robin simulation
executes one step of each live method per round, yielding (7); every retained
configuration contributes to simultaneous peak space.  Discarding states can
reduce space only by paying recomputation time.  The fastest method and the
smallest-space method may differ, so independently minimized coordinates need
not occur on one execution.
\end{proof}

For polynomial $m$, dovetailing preserves a polynomial time exponent when one
method has one, but it does not prove that a solving method exists on every
input.  Exponentially many tool sequences or unbounded restarts invalidate the
polynomial overhead premise.  Randomized portfolio success does not prove a
deterministic all-input bound.
