## The adaptive affine theorem

Let $E$ be a common set of authenticated affine equations over $\mathbb F_2$ in variables $X$. A finite rooted tree $T$ is **complete** when every internal node $v$ queries an affine form $A_v(X)$ and has both children, adding $A_v=0$ and $A_v=1$ respectively. Empty cells may be discarded; every assignment follows exactly one nonempty leaf cell. Let $r$ be the rank of all linear parts queried anywhere in $T$.

At every nonempty leaf $\lambda$, suppose there is a binary-XOR refutation of $E$ plus the path assumptions. Put all refutations into one acyclic union proof DAG. Every vertex has an immutable ordered parent tuple and a normalized affine context. A vertex can be shared only when these contexts and parents agree. Common vertices have empty assumption ancestry. Let $q$ count the union of distinct paid XOR-inference vertices once.

**Theorem 407.1 (PROVED/RESTRICTED).** Under those hypotheses,

$$
W_{\mathrm{GF}(2)}(E)\le q+r.
$$

**Proof.** Tag each derived row at leaf $\lambda$ by the parity vector of path assumptions used in its derivation. Source rows in $E$ and genuinely common vertices have tag zero; a path assumption has its unit tag; binary XOR adds both the row and its tag. For a false terminal, add back the tagged assumptions. They cancel from the derivation and leave an unconditional $E$-derived affine row

$$
U_\lambda(X)=1+\sum_{e\in P_\lambda}\tau_{\lambda,e}(A_e(X)+b_e).
$$

If $x$ lies in leaf cell $\lambda$, every path equation vanishes at $x$, so $U_\lambda(x)=1$. Completeness of $T$ implies that no assignment zeros all rows $U_\lambda$; the affine system $\{U_\lambda=0\}$ is inconsistent. Consequently the augmented row $(0,1)$ lies in their GF(2) span.

Every linear part of $U_\lambda$ lies in the global query span, of dimension $r$. After adjoining the constant coordinate, all augmented $U_\lambda$ lie in a space of dimension at most $r+1$. Hence at most $r+1$ of them span $(0,1)$, and at most $r$ further binary XOR joins derive false. The tag transformation reuses each valid union inference vertex once; incompatible branch vertices were never identified. Thus the common-source proof has at most $q+r$ paid inferences. $\square$

This is an upper-bound/retraction control. It is not a lower bound, a proof-search algorithm, historical syntax authentication, or B3.

### Reconvergence firewall

The proof extends to an acyclic search DAG only when each reconvergent state carries the full normalized affine context and deterministic assignment routing remains complete. Two syntactic paths that query $x_1,x_2$ in opposite orders may merge because their RREF contexts agree. Opposite contexts $x=0$ and $x=1$ may not merge by dropping the assumption label. Row equality alone is not proof-node identity; if two equal rows have incompatible parents or tags, charging them once silently changes the calculus and can leak branch-local conclusions. General semantic state merging with omitted context therefore remains OPEN, rather than being smuggled into Theorem 407.1.

This is an independent abstract GF(2) checker.  It is not the historical
R_mix341 checker and does not authenticate any historical byte grammar.
"""

from __future__ import annotations

import itertools

def parity(x: int) -> int:
    return x.bit_count() & 1


def row_eval(row: tuple[int, int], x: int) -> int:
    return parity(row[0] & x) ^ row[1]


def xor_row(a: tuple[int, int], b: tuple[int, int]) -> tuple[int, int]:
    return a[0] ^ b[0], a[1] ^ b[1]


def rank(masks: list[int]) -> int:
    basis: dict[int, int] = {}
    for v in masks:
        while v:
            p = v.bit_length() - 1
            if p in basis:
                v ^= basis[p]
            else:
                basis[p] = v
                break
    return len(basis)


def rref(rows: list[tuple[int, int]], n: int) -> tuple[tuple[int, int], ...]:
    a = [[(m >> i) & 1 for i in range(n)] + [c] for m, c in rows]
    lead = 0
    for col in range(n):
        pivot = next((j for j in range(lead, len(a)) if a[j][col]), None)
        if pivot is None:
            continue
        a[lead], a[pivot] = a[pivot], a[lead]
        for j in range(len(a)):
            if j != lead and a[j][col]:
                a[j] = [u ^ v for u, v in zip(a[j], a[lead])]
        lead += 1
    out = []
    for row in a:
        if any(row):
            out.append((sum(row[i] << i for i in range(n)), row[-1]))
    return tuple(sorted(out))


def adaptive_tree(n: int, policy_code: int | None = None) -> dict:
    """Query x_d plus an adaptive combination of previously fixed bits."""
    nodes, leaves, query_masks = [], [], []
    digits = policy_code

    def twist_for(depth: int, prefix: int) -> int:
        nonlocal digits
        if depth == 0:
            return 0
        if digits is None:
            # Deliberately branch-dependent and deterministic.
            return ((prefix * 0x9E3779B1) ^ (prefix >> 1)) & ((1 << depth) - 1)
        base = 1 << depth
        t = digits % base
        digits //= base
        return t

    def rec(depth: int, prefix: int, path: list[dict]) -> None:
        if depth == n:
            leaves.append({"assignment": prefix, "path": list(path)})
            return
        twist = twist_for(depth, prefix)
        mask = (1 << depth) ^ twist
        node_id = len(nodes)
        node = {"id": node_id, "depth": depth, "prefix": prefix, "mask": mask}
        nodes.append(node)
        query_masks.append(mask)
        for value in (0, 1):
            bit = value ^ parity(twist & prefix)
            edge = {"node": node_id, "mask": mask, "value": value,
                    "row": (mask, value)}
            rec(depth + 1, prefix | (bit << depth), path + [edge])

    rec(0, 0, [])
    return {"n": n, "nodes": nodes, "leaves": leaves,
            "query_rank": rank(query_masks)}


def solve_subset(rows: list[tuple[int, int]], target: tuple[int, int]) -> int:
    for bits in range(1 << len(rows)):
        acc = (0, 0)
        for i, row in enumerate(rows):
            if bits >> i & 1:
                acc = xor_row(acc, row)
        if acc == target:
            return bits
    raise AssertionError("target outside row span")


def min_span(rows: list[tuple[int, int]], target: tuple[int, int]) -> tuple[int, int]:
    best = None
    best_bits = 0
    for bits in range(1 << len(rows)):
        if best is not None and bits.bit_count() >= best:
            continue
        acc = (0, 0)
        for i, row in enumerate(rows):
            if bits >> i & 1:
                acc = xor_row(acc, row)
        if acc == target:
            best, best_bits = bits.bit_count(), bits
    if best is None:
        raise AssertionError("target not spanned")
    return best, best_bits


def build_fixture(n: int) -> dict:
    tree = adaptive_tree(n)
    source = [(1 << i, 0) for i in range(n)] + [((1 << n) - 1, 1)]
    assert all(any(row_eval(e, x) for e in source) for x in range(1 << n))
    # One genuine common paid inference, included twice in every leaf proof.
    shared = xor_row(source[0], source[-1])
    paid = 1
    discharged = []
    max_path_paid = 0
    terminals = []
    for leaf in tree["leaves"]:
        x = leaf["assignment"]
        violated = next(e for e in source if row_eval(e, x))
        assumptions = [tuple(edge["row"]) for edge in leaf["path"]]
        # violated + selected assumptions = constant 1.
        bits = solve_subset(assumptions, xor_row(violated, (0, 1)))
        acc, tags = violated, 0
        steps = 0
        for i, a in enumerate(assumptions):
            if bits >> i & 1:
                acc = xor_row(acc, a)
                tags ^= 1 << i
                steps += 1
        assert acc == (0, 1)
        # Use the shared assumption-free node twice without changing the terminal.
        acc = xor_row(acc, shared)
        acc = xor_row(acc, shared)
        steps += 2
        assert acc == (0, 1)
        paid += steps
        max_path_paid = max(max_path_paid, steps + 1)
        u = (0, 1)
        for i, a in enumerate(assumptions):
            if tags >> i & 1:
                u = xor_row(u, a)
        assert u == violated and row_eval(u, x) == 1
        discharged.append(u)
        terminals.append({"assignment": x, "assumption_bits": tags,
                          "discharged": u, "paid": steps})
    assert all(any(row_eval(u, x) for u in discharged) for x in range(1 << n))
    unique_u = sorted(set(discharged))
    basis_size, basis_bits = min_span(unique_u, (0, 1))
    assert basis_size <= tree["query_rank"] + 1
    return {
        "n": n,
        "internal_nodes": len(tree["nodes"]),
        "leaves": len(tree["leaves"]),
        "query_rank": tree["query_rank"],
        "union_paid_q": paid,
        "common_upper_q_plus_r": paid + tree["query_rank"],
        "distinct_discharged_rows": len(unique_u),
        "contradictory_basis_rows": basis_size,
        "final_joins": basis_size - 1,
        "ideal_span_upper": max_path_paid + (basis_size - 1).bit_length(),
        "basis_digest": hashlib.sha256(json.dumps([unique_u, basis_bits]).encode()).hexdigest(),
        "terminals_digest": hashlib.sha256(json.dumps(terminals, sort_keys=True).encode()).hexdigest(),
    }


def exhaustive_policies_n3() -> dict:
    # Mixed-radix policy digits: 1 choice at depth 0, two base-2 digits at
    # depth 1, four base-4 digits at depth 2: 2^2 * 4^4 = 1024 policies.
    count = 0
    digest = hashlib.sha256()
    for code in range(1024):
        t = adaptive_tree(3, code)
        assignments = sorted(leaf["assignment"] for leaf in t["leaves"])
        assert assignments == list(range(8))
        assert t["query_rank"] == 3

        canon = ";".join(f'{nd["mask"]},{nd["prefix"]}' for nd in t["nodes"])
        digest.update(canon.encode())
        count += 1
    return {"policies": count, "all_cover_8_assignments": True,
            "all_query_rank": 3, "digest": digest.hexdigest()}

    leaf = t["leaves"][5]

    wrong_constant_rejected = any(row_eval(tuple(e["row"]), leaf["assignment"]) for e in wrong)
