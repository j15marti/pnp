\section*{MULTILEVEL-EXTENSIONCAP}
Let $\mu$ be any distribution on the exact $t$-slice of $\{0,1\}^M$ satisfying
\[
 \Pr_\mu[T\subseteq X]=p_k:=\frac{(t)_k}{(M)_k}
 \qquad\text{for every }T\in\binom{[M]}k.
\]
Fix $r$ with $k\le r\le t$, and let
\[
 \mathcal C\subseteq\binom{[M]}r.
\]
For $T\in\binom{[M]}k$, define
\[
 d_{\mathcal C}(T)=|\{S\in\mathcal C:T\subseteq S\}|.
\]
Then
\[
\boxed{
\sum_{S\in\mathcal C}\Pr_\mu[S\subseteq X]
\le
\frac{p_k}{\binom rk}
\sum_{T\in\binom{[M]}k}
\min\left\{
d_{\mathcal C}(T),\binom{t-k}{r-k}
\right\}.
}
\]
\textbf{Status: PROVED/EXACT.}

\paragraph{Proof.}
Double counting incidences between $k$-subsets and selected $r$-sets gives
\[
\binom rk\sum_{S\in\mathcal C}\Pr_\mu[S\subseteq X]
=
\sum_T\sum_{\substack{S\in\mathcal C\\S\supseteq T}}
\Pr_\mu[S\subseteq X].
\]
For fixed $T$, every summand is at most
\[
\Pr_\mu[T\subseteq X]=p_k,
\]
so the selected fiber is at most $d_{\mathcal C}(T)p_k$. Exact weight also gives the full-fiber identity
\[
\sum_{\substack{S\supseteq T\\|S|=r}}\Pr_\mu[S\subseteq X]
=
\binom{t-k}{r-k}p_k,
\]
because conditional on $T\subseteq X$, exactly $\binom{t-k}{r-k}$ $r$-supersets of $T$ occur. Taking the smaller bound for each $T$ and dividing by $\binom rk$ proves the claim. \qed
