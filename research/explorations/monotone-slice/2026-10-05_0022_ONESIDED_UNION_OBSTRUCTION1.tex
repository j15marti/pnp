\section*{Threshold hostile control}
For \(1\le w\le L\), let
\[
T_{L,w}(z)=\mathbf 1[\sum_{i=1}^{L}z_i\ge w].
\]
Let \(\nu\) be permutation-invariant on \(2^{[L]}\) and \(p_w=\Pr_{Y\sim\nu}[|Y|=w]\).

\begin{theorem}[ONESIDED-APPROX-UNION-OBSTRUCTION]
Let \(H=\bigvee_{A\in\mathcal A}\bigwedge_{i\in A}z_i\) be a monotone DNF with \(H\le T_{L,w}\) pointwise and
\[
\Pr_{Y\sim\nu}[T_{L,w}(Y)=1,H(Y)=0]\le\varepsilon.
\]
If \(p_w>0\), then
\[
|\mathcal A_w|\ge\left(1-\frac{\varepsilon}{p_w}\right)\binom Lw.
\]
Some \((w-1)\)-set is contained in at least
\[
\left(1-\frac{\varepsilon}{p_w}\right)(L-w+1)
\]
distinct width-\(w\) terms.
\end{theorem}

\begin{proof}
Pointwise \(H\le T_{L,w}\) forbids terms of size below \(w\). On the layer \(|Y|=w\), a term accepts \(Y\) iff it is a width-\(w\) term equal to \(Y\). Conditional layer-uniformity gives
\[
p_w(1-|\mathcal A_w|/\binom Lw)\le\varepsilon.
\]
Each selected \(w\)-set contains \(w\) different \((w-1)\)-cores. Double counting gives average core degree
\[
\frac{w|\mathcal A_w|}{\binom L{w-1}}
=
\frac{|\mathcal A_w|}{\binom Lw}(L-w+1),
\]
so some core has at least this degree.
\end{proof}

\begin{corollary}
If \(R\le(1-\varepsilon/p_w)(L-w+1)\), then \(H\) contains distinct width-\(w\) terms \(A_1,\ldots,A_R\) satisfying
\[
|\bigcup_{j=1}^{R}A_j|=w-1+R.
\]
\end{corollary}
