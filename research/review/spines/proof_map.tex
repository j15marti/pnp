\section*{Acquisition repair}
\begin{theorem}
If $q(N)=N^{v+o(1)}$, $|p|=O(N)$ and
$w(N)=\kappa\log_2N+o(\log N)$, then
\[
 (2^{w+1}-1)(q+N+|p|+w+1)
 =N^{\kappa+\max\{v,1\}+o(1)}.
\]
Thus the CC-0002 exponent $v+\kappa$ follows from that display only for
$v\ge1$ or under a separate domination hypothesis.
\end{theorem}
\begin{proof}
The factors have exponents $\kappa$ and $\max\{v,1\}$ respectively.  A
pointer-based implementation may remove repeated copying of read-only input,
but it is a distinct machine theorem and remains C2.
\end{proof}
