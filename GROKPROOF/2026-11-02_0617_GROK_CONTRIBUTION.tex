\section*{GROK Resource-Contradiction Update (2026-11-02)}

\begin{theorem}[Numerical Membership]
Let \(\beta(t) = \frac12 \bigl( \frac{t+2}{t^2 - t} \bigr)\). For every finite \(K \ge 2\), the pair
\[
(t', s') = \bigl(1.3 - \tfrac1{5K},\ 0.2\bigr)
\]
satisfies \( s' < \beta(t') \). Consequently the pair lies in the FLMV forbidden region (in the applicable random-access Turing-machine model).
\end{theorem}

\begin{proof}
Direct verification for the limiting case \(K=2\) yields \(t' = 1.2\), \(\beta(1.2) \approx 0.416 > 0.2\). The map \(K \mapsto 1.3 - 1/(5K)\) is increasing in \(K\) and approaches \(1.3\), while \(\beta\) is continuous and positive on \([1.2,1.3]\). The inequality holds uniformly.
\end{proof}

\begin{remark}
The membership is numerical only. It does not construct a transformation. The all-input charged reduction producing \(\le N^{1-\lambda+\zeta+o(1)}\) children of length \(\le N^{\lambda+o(1)}\) with total local work \(N^{1.1+o(1)}\) and live space \(N^{0.2+o(1)}\) (\(\lambda=1/(5K)\), \(\zeta=0.1\)) remains \OPEN.
\end{remark}

\begin{lemma}[Self-Reducibility Exponent Preservation]
Suppose \(A\) decides SAT in time \(O(N^K)\). The standard self-reduction that fixes a single variable and recurses on the two resulting instances of length \(N-O(1)\) produces a procedure whose leading time exponent remains \(K\), not a strictly smaller exponent.
\end{lemma}

\begin{proof}
Each recursive call costs \(O((N-1)^K) = O(N^K)\). Two calls yield a constant-factor overhead. The asymptotic exponent is unchanged. Multi-variable fixing that produces polynomially many subinstances of length \(N^\lambda\) with \(\lambda<1\) requires either an exponential number of branches or a discovery/verification procedure whose cost must be charged against the \(N^{1.1+o(1)}\) budget. No such low-cost uniform discovery is known for arbitrary exact-(2,2)-CNF-3 instances.
\end{proof}
