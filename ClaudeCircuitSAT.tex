\section*{Status (strict labels)}
\textbf{No superpolynomial lower bound for circuit-SAT is claimed. P versus NP remains OPEN.}
Labels: PROVED, KNOWN THEOREM, COMPUTATIONAL, CONDITIONAL, CANDIDATE, OPEN, FALSIFIED.

\section{Setup}
Circuit $C$ over $n$ inputs, $s$ gates of fan-in 2 (basis $B_2$: all 2-input gates, free output negation); wires $w=2s$ (counting fan-out copies as separate edges into gate inputs). $\mathrm{SAT}(C)$: does some $x\in\{0,1\}^n$ give $C(x)=1$?
Two separate quantities are tracked and \emph{must not be conflated}:
(a) $c_n(f)$ = minimum gates to \emph{build} a circuit for $f$ (construction cost);
(b) the \emph{time to decide} $\mathrm{SAT}(C)$ as a function of $(n,s,\text{parameters})$.
Only (b) is a SAT lower bound; (a) is the staircase we climb to see what structure forces cost.

\section{Construction ladder}
\begin{thm}[connectivity, PROVED/EXACT] If $f$ depends on all $n$ inputs, every $B_2$ circuit for $f$ has $\ge n-1$ gates, hence $\ge 2(n-1)$ wires, with equality for $\mathrm{AND}_n,\mathrm{XOR}_n$.\end{thm}
\begin{proof} The underlying undirected graph on inputs and gates must connect all $n$ input nodes to the output; a connected graph with $n$ source nodes and fan-in-2 gates needs $\ge n-1$ gates (each gate merges at most two components). Upper bound by a binary tree.\end{proof}

\section{From construction cost to SAT cost}
\begin{thm}[reading bound, PROVED] Any SAT algorithm for circuits that is correct on all circuits of size $s$ must read $\Omega(s)$ bits of $C$ in the worst case: changing one unread gate can flip satisfiability of the circuit $C\wedge g$ vs.\ $C\wedge\neg g$ style gadgets. \end{thm}
(This is only linear; included as the base rung.)

\begin{thm}[black-box query bound, PROVED, model-limited] If $C$ is accessible only by evaluation queries, deciding $\exists x\,C(x)=1$ needs $2^n$ queries deterministically (adversary answers 0 until the last point). Quantum: $\Omega(2^{n/2})$ (BBBV, KNOWN). \emph{This does not transfer to white-box circuits}; it is exactly the relativization barrier.\end{thm}

\section{Parametrized targets}
Natural parameters: $n$ (inputs), $s$ (size), treewidth $\mathrm{tw}(C)$, depth $d$, number of fan-out $>1$ wires $f$.
\begin{itemize}
\item Upper bounds (KNOWN): $2^n\mathrm{poly}(s)$; $2^{O(\mathrm{tw})}\mathrm{poly}(s)$ by dynamic programming over a tree decomposition.
\item CONDITIONAL lower bounds: under SETH, no $(2-\epsilon)^n\mathrm{poly}(s)$ algorithm; under SETH, treewidth-parametrized SAT lower bound $(2-\epsilon)^{\mathrm{tw}}$ for CNF primal pathwidth is in the literature \emph{(citation to be verified in a later run)}.
\item OPEN (target): an \emph{unconditional} bound of the form $T(n,s)\ge g(n)$ with $g$ superpolynomial in $n$ for a non-black-box model, e.g.\ for resolution-style / DPLL-style algorithms (known exponential for specific families; restricted models only).
\end{itemize}

\section{Wire-by-wire candidate program (CANDIDATE)}
Idea: in the ladder, define $\mu(C)$ = minimum number of \emph{fan-out wires} needed so that the circuit is a formula (tree). For $n=3$ compute the minimum $(s,f)$ Pareto front per function; then ask whether $\mathrm{SAT}$-hardness on a family of circuits with parameter $f=k$ scales like $2^{\Omega(k)}$ for a restricted algorithm class (backtracking over fan-out wires: $2^{f}$ branching suffices, KNOWN upper bound $2^{f}\mathrm{poly}$ since removing fan-out yields a formula, which is decidable in linear time for satisfiability-counting-free monotone cases but \emph{not} for general formulas---formula-SAT is still NP-hard, so this upper bound is FALSE as stated).
\textbf{Correction (FALSIFIED):} the claim ``$2^f\mathrm{poly}$ SAT in the number of fan-out wires'' is wrong: formulas (tree circuits, $f=0$) with repeated inputs are NP-hard. Fan-out of \emph{input} variables must be counted via $n$; only fan-out of internal gates is cheap to branch on, and that parameter $f_{\text{int}}$ gives $2^{f_{\text{int}}}\cdot 2^{n}$-type bounds, not an improvement. Retired.
