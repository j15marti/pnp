\section{Program}
Start with the simplest circuits, add inputs one at a time, and at each step record (i) the minimum number of wires needed to build the circuit, and (ii) the best provable lower bound on the work to decide satisfiability. Bounds may be parametrized by $n$ (inputs), $m$ (wires), and a structural width $w$ (treewidth or cutwidth).

\paragraph{Model.} Circuit: DAG over $\{x_1..x_n\}$, fan-in-2 gates from $B_2$, one output. $\#\text{wires}=2\cdot\#\text{gates}$ (gate input edges) plus the output designation. $\CSAT$: given the DAG, does some input make the output 1.

\section{Wire ledger (construction side)}
\begin{proposition}[connectivity] \status{PROVED/EXACT}
A fan-in-2 circuit whose output depends on all $n$ inputs has at least $n-1$ gates, hence at least $2(n-1)$ gate-input wires. The point function $\delta_a(x)=[x=a]$ meets this: an AND tree of $n$ literals, with negations absorbed into $B_2$ gates.
\end{proposition}
\begin{proof}
Contract the sub-DAG feeding the output: it is connected and contains $n$ input leaves. Each gate adds at most one to the number of components merged (two inputs, one output), so merging $n$ leaves needs $\ge n-1$ gates.
\end{proof}

\section{Lower bounds on solving (decision side)}
\begin{proposition}[probe bound, explicit description] \status{PROVED/EXACT}
Any deterministic algorithm that reads the circuit through gate-probes must probe $\Omega(m)$ gates in the worst case.
\end{proposition}
\begin{proof}
Let $I_0$: $k$ gates $g_i=\mathrm{OR}(x_i,\neg x_i)$ ANDed together (satisfiable). $I_j$: same but $g_j=\mathrm{AND}(x_j,\neg x_j)$ (unsatisfiable). They differ only in the type of gate $j$, so an algorithm that misses gate $j$ cannot separate $I_0$ from $I_j$. Hence $k=\Theta(m)$ probes. \qed
\end{proof}
This is a reading bound only; it says nothing about search over assignments.

\begin{proposition}[black-box bound] \status{KNOWN THEOREM}
If the circuit is accessible only by evaluation, $2^n$ evaluations are necessary (deterministic) and $\Omega(2^{n/2})$ quantumly (BBBV). Not a lower bound for white-box $\CSAT$.
\end{proposition}

\begin{proposition}[parametrized upper/lower pair] \status{KNOWN THEOREM} / \status{CONDITIONAL}
Upper bound $2^{O(w)}\cdot\mathrm{poly}(m)$ for treewidth $w$. Under ETH there is no $2^{o(w)}\mathrm{poly}(m)$ algorithm (via CNF-SAT with bounded primal treewidth; the exact base needs SETH and the citation is to be verified, CANDIDATE). The parameter that must grow is $w$: any unconditional $\CSAT$ lower bound of size $2^{\Omega(w)}$ would give $P\ne NP$ for circuits of $w=\omega(\log m)$.
\end{proposition}
