\section*{Status}
\textbf{P versus NP remains OPEN.} Nothing here is an unconditional Turing-machine lower bound.
All results are in a restricted \emph{edge-probe} model or are computational tables.
Labels: PROVED, KNOWN, COMPUTATIONAL, CONDITIONAL, CANDIDATE, OPEN.

\section{Plan}
Build circuits from the simplest case upward, tracking (i) gates $g$, (ii) wires $2g$ (fan-in 2), (iii) number $k$ of essential inputs,
and (iv) the number $C(g,k)$ of \emph{SAT-critical wires}: wires whose polarity flip changes satisfiability.
Parameters: $n$ inputs, $g$ gates, $k$ essential inputs; a bound may be parametrized, e.g.\ $\Omega(C)$ or $2^{\Omega(k)}$ conditionally.

\section{Model}
Circuit = AND-gates of fan-in 2 with negatable edges (AIG), all gates reachable from the output. Edge-probe model: the solver learns
one wire's endpoints/polarity per probe.

\begin{lemma}[PROVED: critical-wire adversary]
Any deterministic probe algorithm deciding SAT on a fixed circuit $\mathcal C$ must probe every critical wire of $\mathcal C$.
\end{lemma}
\emph{Proof.} If critical wire $e$ is unprobed, the algorithm's view is identical on $\mathcal C$ and $\mathcal C$ with $e$ flipped, whose answers differ. $\square$

So probe complexity $\ge C(g,k)$ for the best instance; trivially $C\le 2g$.

\begin{lemma}[PROVED: formulas of distinct variables are never critical]
If $k=g+1$ then $C(g,k)=0$.
\end{lemma}
\emph{Proof.} There are $2g$ edges, $g-1$ of them gate-to-gate (each non-output gate has $\ge1$ out-edge), so at most $g+1$ edges
come from inputs; $k=g+1$ essential inputs force equality: a read-once AND-tree on distinct variables, satisfiable under every polarity assignment. No flip changes the answer. $\square$

\section{Parametrized view}
\begin{itemize}
\item KNOWN: brute force gives $O(2^n\cdot m)$. CONDITIONAL: SETH implies no $(2-\varepsilon)^n\,\mathrm{poly}(m)$ algorithm for Circuit-SAT.
\item PROVED (model-limited): probe complexity $\ge C(g,k)=\Theta(g)$ if $\Theta(g)$-critical instances exist with $k$ small; this is only linear, and says
nothing about time.
\item Gap: linking critical-wire structure to time needs a bridge beyond probes (OPEN).
\end{itemize}
