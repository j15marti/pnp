\section{Setting and parameters}
A circuit $C$ has $n$ inputs, $g$ fan-in-2 gates (any of the 16 binary functions,
negation free) and $m$ wires (edges between inputs/gates; the output wire is not counted).
Circuit-SAT: given $C$, decide whether $\exists x\,C(x)=1$. Parameters of an instance:
$n$ (inputs), $m$ (wires), $k=|C^{-1}(1)|$ (solutions). A \emph{parametrized lower bound}
is a function $T(n,m,k)$ below the cost of every algorithm in a stated model. Two models:
\emph{black-box} (algorithm may only evaluate $C$) and \emph{white-box} (algorithm reads
the wiring, random access).

\section{Ladder, exact values for $n\le 3$ (COMPUTATIONAL)}
Minimum number of gates $c(f)$ over all $f:\{0,1\}^n\to\{0,1\}$, by number $k$ of
satisfying assignments. Exhaustive search over all computed-function sets
(\texttt{research/agents/claude/circuitsat/ladder.py}, raw data \texttt{ladder\_results.json}).

\section{Proved bounds}
\begin{lemma}[Connectivity, PROVED]\label{lem:conn}
If a fan-in-2 circuit computes $f$ that depends on all $n$ inputs, then $g\ge n-1$ and
$m\ge n+g-1\ge 2n-2$.\end{lemma}
\begin{proof} The cone of the output is a connected DAG on $n$ inputs plus $g'\le g$ gates,
so it has at least $n+g'-1$ edges. Each gate has at most 2 in-edges: $n+g'-1\le 2g'$, so $g'\ge n-1$.
\end{proof}

\begin{lemma}[Black-box bound, PROVED]\label{lem:bb}
Any deterministic algorithm that only evaluates $C$ and distinguishes UNSAT from
instances with exactly $k$ solutions needs $\ge 2^n-k+1$ evaluations.\end{lemma}
\begin{proof} Answer 0 to every query. If fewer than $2^n-k+1$ points were queried, at least $k$ unqueried
points remain and could all be the solution set. \end{proof}
So $T_{\rm bb}(n,k)=2^n-k+1$; for $k=1$ this is the full $2^n$, tight by exhaustive search.

\begin{lemma}[Reading bound, PROVED]\label{lem:read}
Any deterministic random-access algorithm deciding Circuit-SAT reads $\ge m$ wire cells
on worst-case instances of $m$ wires, so time $\Omega(m)$.\end{lemma}
\begin{proof} Let $B_\emptyset$ be the AND-chain of $m$ literal slots, all equal to $x_1$ (SAT), and
$B_j$ the same with slot $j$ equal to $\neg x_1$ (UNSAT). An algorithm that leaves slot $j$ unread
gives the same answer on $B_\emptyset$ and $B_j$. \end{proof}

\begin{proposition}[Parametrized summary, PROVED/KNOWN]
White-box time $\ge \Omega(m+n)$ unconditionally. Black-box time $\ge 2^n-k+1$.
The gap between $\Omega(m)$ and $2^n$ is exactly the open white-box question.
\end{proposition}

\section{Conditional bounds (CONDITIONAL)}
Since 3-SAT with $N$ variables and $O(N)$ clauses embeds as a Circuit-SAT instance with
$n=N$, $m=O(N)$, ETH gives $2^{\delta n}\mathrm{poly}(m)$ for some $\delta>0$ in the white-box model;
SETH gives $2^{(1-\varepsilon)n}$ impossible. These are hypotheses, not results here.
