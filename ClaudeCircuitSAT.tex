\section{Model and parameters}
A circuit $C$ has $n$ inputs, $s$ fan-in-2 gates (AND with free negation on wires, a De~Morgan basis), and $w=2s$ wires.
Circuit-SAT: is there $x\in\{0,1\}^n$ with $C(x)=1$? Parameter vector $\theta=(n,s,w)$; a bound is a function $T(n,s)$.
Decider model R (query/RAM): the description is read-only; a probe names a slot $(g,p)$ and returns the node feeding it.

\section{Ladder (wire-by-wire construction)}
\textbf{Rung 0} ($n=1$): functions $0,1,x,\bar x$: 0 gates, 0 wires. SAT is decided by reading nothing beyond the output node.\\
\textbf{Rung 1} ($n=2$): every function on 2 inputs; exhaustive search gives min gates 0/1/3 (XOR/XNOR need 3 gates $=6$ wires).\\
\textbf{Rung 2} ($n=3$): all 128 canonical functions (up to output negation); max 6 gates $=12$ wires (the 9 functions at 6 include $\oplus_3$).

\begin{lemma}[connectivity, PROVED/EXACT]\label{lem:conn}
If $f$ depends on all $n$ inputs, any fan-in-2 circuit for $f$ has $\ge n-1$ gates and $\ge 2(n-1)$ wires.
\end{lemma}
\emph{Proof.} Each input must reach the output; the underlying graph restricted to the cone is connected with $n$ sources, and each gate merges at most two components. $\square$

\section{A rigorous (linear) parametrized bound}
\begin{theorem}[critical-wire bound, PROVED/EXACT in model R]\label{thm:crit}
For every $n\ge2$ there is a family of circuits with $n$ inputs, $\Theta(n)$ gates, such that any deterministic decider for Circuit-SAT in model R makes at least $n$ probes on some input; hence time $\Omega(w)$ where $w$ is the wire count.
\end{theorem}
\emph{Proof.} Let $I_0=\bigvee_{i=1}^n\big(x_i\wedge\neg y_i\big)$ with each $y_i$ the wire from $x_i$; $I_0$ is UNSAT. Let $I_i$ rewire slot $y_i$ to $x_{j}$, $j\ne i$; then $x_i{=}1,x_j{=}0$ satisfies, so $I_i$ is SAT, and $I_i$ differs from $I_0$ in one slot only. If a decider on $I_0$ never probes slot $y_i$, its run is identical on $I_i$, contradiction. So it probes all $n$ slots. $\square$

\paragraph{Honest scope.} This is the trivial ``must read the critical wires'' bound, $T=\Omega(s)$. It is parametrized and rigorous but linear; it says nothing about the $2^{\Omega(n)}$ hoped for. Rung-wise computation shows wire cost of \emph{building} hard functions is small ($\le 12$ wires at $n=3$) compared with the work expected to \emph{decide} them, which is where the gap lives.
