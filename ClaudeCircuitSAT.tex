\section{Model and parameters}
A circuit $C$ has $n$ inputs, $g$ fan-in-2 gates (any of the 16 binary functions, free fan-out), $w=2g$ wires.
\SAT\ input: description of $C$ ($m=\Theta(g\log(n+g))$ bits); question: $\exists x\,C(x)=1$.
Parameters: $(n,g)$; lower bound is a function $T(n,g)$. Algorithms: $O(2^n g)$ brute force; trivial lower bound $\Omega(g)$.

\section{Ladder, step 1: wires needed per input count}
\begin{lemma}[Connectivity, PROVED]\label{lem:conn}
If $f$ depends on all $n$ variables, any fan-in-2 circuit for $f$ has $\ge n-1$ gates and $\ge 2(n-1)$ wires.
\end{lemma}
\begin{proof}
Every input must have a directed path to the output, so the underlying graph is connected on $n$ inputs plus $g$ gates;
a connected graph with $n+g$ vertices has $\ge n+g-1$ edges, and edges $=2g$ (each gate has two in-wires), giving $2g\ge n+g-1$, i.e.\ $g\ge n-1$.
\end{proof}
\begin{lemma}[Tight for unique-SAT, PROVED]
The $n$-literal AND (unique satisfying assignment) has size exactly $n-1$ gates, $2(n-1)$ wires; this is the minimum over all circuits with exactly one satisfying assignment.
\end{lemma}
\begin{proof} Upper bound: tree of ANDs. Lower bound: such $f$ depends on all $n$ variables; Lemma~\ref{lem:conn}.\end{proof}

\section{Ladder, step 2: reading bound for the decision problem}
\begin{theorem}[Reading lower bound, PROVED]\label{thm:read}
Any deterministic (indeed bounded-error randomized with $\Omega(1)$ gap, by the same pairing) algorithm for \SAT\ on the AND-of-literals family with $n$ literal slots must probe $\ge n-1$ description entries in the worst case.
\end{theorem}
\begin{proof}
Slot $j$ stores a literal $(a_j,\sigma_j)$ ($a_j\in[n]$, $\sigma_j\in\{\pm\}$); $C=\bigwedge_j \ell_j$, unsatisfiable iff two slots hold complementary literals of one variable.
The adversary answers $(a_j,\sigma_j)=(j,+)$ to every probe; if some slot $j$ is unprobed and a slot $i\ne j$ is probed, it can set slot $j$ to $(i,-)$: unsat, versus sat for the answered instance. Hence after $\le n-2$ probes (two slots unprobed, at least one slot probed when $n\ge 3$ and probes $\ge1$) the algorithm cannot decide; with 0 probes both outcomes are also possible. So $\ge n-1$ probes. Randomized: with fewer than $n/2$ probes, the pairing of the all-sat instance with $n/2$ unsat neighbors defeats success $>1/2+\epsilon$ (Yao). \qquad(Sketch for the randomized clause; deterministic part complete.)
\end{proof}
Meaning: wire-by-wire, every wire of a minimal-instance circuit must be inspected: $T(n,g)=\Omega(g)$, matching the wire count of Lemma~\ref{lem:conn}.

\section{Step 3: where the ladder stops (barriers)}
\begin{itemize}
\item \textbf{Superlinear size is the wall.} Best explicit circuit-size lower bounds over $B_2$ are $\approx 3.1n$ (Li--Yang 2022, KNOWN, from memory of the literature---re-verify cite). Gate elimination of this type gives linear bounds in $n$; a $T(n,g)=\omega(g)$ for \SAT\ would be a new separation technique.
\item \textbf{Parameterized, conditional.} ETH gives $T(n,g)\ge 2^{\delta n}$; SETH gives $2^{(1-\epsilon)n}\mathrm{poly}(g)$ is impossible for CNF-type circuits (CONDITIONAL, KNOWN reductions).
\item \textbf{Time--space.} Williams-style lower bounds $n^{2\cos(\pi/7)}\approx n^{1.8019}$ for SAT with $n^{o(1)}$ space (KNOWN); Buss--Williams show alternation trading cannot exceed it. Repository ledger row L13/L34--L37 (\texttt{research/agents/claude/world-ledger}, \texttt{tport}) records the portfolio analogues; one-tape crossing-sequence bounds (L38--L40) do not separate P from NP.
\end{itemize}
