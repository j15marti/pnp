\paragraph{Status.} P versus NP remains \textbf{OPEN}. No super-polynomial lower bound for Circuit-SAT is claimed. This document tracks a ladder of exact small cases and proved trivial/parameterized bounds. Labels: PROVED/EXACT, KNOWN THEOREM, COMPUTATIONAL, CONDITIONAL, CANDIDATE, OPEN.

\section{Model}
Circuit: DAG over inputs $x_1..x_n$, fan-in-2 gates from $B_2$ (all 16 binary gates, so negation is free), output gate. Size $=$ number of gates $g$; wires $w=2g$ (every gate has two in-wires). Circuit-SAT: given $C$, is there $x$ with $C(x)=1$? Parameters: $n$ (inputs), $g$ (gates), later treewidth/depth $k$.

\section{Ladder of exact results}
\begin{prop}[connectivity floor; PROVED/EXACT]\label{p1}
If $f$ depends on all $n$ inputs, any fan-in-2 circuit for $f$ has $g\ge n-1$ gates and $w\ge 2(n-1)$ wires.
\end{prop}
\emph{Proof.} Every input must have a directed path to the output, so the sub-DAG on the output's ancestors is connected with $n$ sources; a connected graph on $n+g'$ nodes has at least $n+g'-1$ edges while each gate contributes exactly 2 in-edges: $2g'\ge n+g'-1$, so $g'\ge n-1$. \qed

\begin{prop}[reading bound for the solver; PROVED/EXACT]\label{p2}
In the model where $C$ is given as a gate list with random access, any deterministic algorithm deciding Circuit-SAT reads $\ge g$ gate cells on the family $F_g$: ``AND of $g$ cells, each cell either an input wire or the constant $0$''. So time $\Omega(\text{size})$, linear in the parameter $g$.
\end{prop}
\emph{Proof.} Adversary answers ``input wire'' to every probe; if some cell is unread the answer is still undetermined (satisfiable iff that cell is not $0$). \qed

\section{Parameterized view}
\begin{itemize}
\item \textbf{Size parameter $g$:} unconditional $\Omega(g)$ reading bound (Prop.~\ref{p2}); matched by $O(g)$ for structured families, so no gap yet.
\item \textbf{Input parameter $n$:} brute force $2^n g$. Lower bounds $2^{(1-\epsilon)n}$ are CONDITIONAL on SETH (CNF-SAT$\le$Circuit-SAT); unconditional status OPEN.
\item \textbf{Structure parameter (treewidth $k$):} $2^{O(k)}g$ upper bound is KNOWN; a $2^{o(k)}$ lower bound is CONDITIONAL on ETH. Candidate: relate minimal wire count of a width-$k$ circuit to the cut-width needed to carry satisfiability information (communication across a cut).
\end{itemize}
