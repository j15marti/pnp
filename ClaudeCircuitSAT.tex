\section*{Status}
\textbf{P versus NP remains OPEN.} No unconditional superpolynomial lower bound for Circuit-SAT is claimed. Labels: PROVED/EXACT, KNOWN THEOREM, COMPUTATIONAL, CONDITIONAL, CANDIDATE, OPEN, FALSIFIED/RETIRED.

\section{Model and accounting}
A circuit is a DAG over $n$ inputs with fan-in-2 gates from the 10 non-degenerate binary operations (negation free). \emph{Wires} = fan-in edges $=2g$ for $g$ gates, plus one output tap. Define $C(f)$ = minimum gates, $W(f)=2C(f)$ minimum wires. $\mathrm{CSAT}_n$: given such a circuit on $n$ inputs, is it satisfiable? Three separate resources are tracked, never mixed:
(i) \emph{size} $W$ of circuits needed to \emph{express} instances; (ii) \emph{black-box queries} to evaluate; (iii) \emph{description reads} of the instance encoding.

\section{Ladder of exact results}
\begin{theorem}[connectivity, PROVED/EXACT]\label{thm:conn}
If $f$ depends on all $n$ inputs then $C(f)\ge n-1$ and $W(f)\ge 2(n-1)$; $\mathrm{AND}_n$ attains $n-1$.
\end{theorem}
\begin{proof}
Every input must have a directed path to the output, so the underlying undirected graph is connected on $n+g$ nodes, requiring $\ge n+g-1$ edges. Each gate contributes exactly 2 in-edges, so $2g\ge n+g-1$, i.e.\ $g\ge n-1$. A balanced tree of ANDs attains it.
\end{proof}
\begin{theorem}[exhaustive census, COMPUTATIONAL]
Over all Boolean functions (negation free), exact $C(f)$: $n=2$: 6 functions with 0 gates, 10 with 1. $n=3$: gate histogram $\{0{:}8,\,1{:}30,\,2{:}114,\,3{:}80,\,4{:}24\}$ (256 functions); parity$_3$ needs 2, majority$_3$ needs 4; $\mathrm{AND}_n$ needs $0,1,2,3$ for $n=1..4$ (matches Thm~\ref{thm:conn}). Source: \texttt{experiments/claude\_circuitsat/2026-10-25\_ccs\_min\_gates.py}, raw \texttt{results/raw/claude\_circuitsat/2026-10-25\_ccs\_min\_gates.json} (runs $\approx7$\,s; $n=4$ census deferred, to be chunked).
\end{theorem}

\section{Decision-side lower bounds, parameterized by $(n,s)$}
\begin{theorem}[needle / black-box bound, PROVED/EXACT in the oracle model]\label{thm:needle}
Any deterministic algorithm that only evaluates the circuit $C:\{0,1\}^n\to\{0,1\}$ as a black box needs $2^n$ evaluations in the worst case to decide $\mathrm{CSAT}$ (even promised at most one satisfying assignment).
\end{theorem}
\begin{proof}
Adversary answers $0$ to every query; after $2^n-1$ queries the unqueried point $a$ is consistent both with $C\equiv0$ and with $C=\mathbf 1_{\{a\}}$ (computable with $n-1$ gates, Thm~\ref{thm:conn}).
\end{proof}
\textbf{Caveat (failed transfer as a P vs NP route):} the algorithm sees the description, not just an oracle; Thm~\ref{thm:needle} does not bound white-box algorithms.
\begin{theorem}[description-reading bound, PROVED/EXACT]\label{thm:read}
Any deterministic algorithm (any model with random-access read of the gate list) deciding $\mathrm{CSAT}$ on circuits with $s$ gates must read at least $s$ distinct gate records in the worst case.
\end{theorem}
\begin{proof}
Take the AND-tree $T$ (satisfiable). For each gate $i$, the circuit $T_i$ obtained by replacing gate $i$ with the constant-0 gate is unsatisfiable (gate $i$ lies on the path to the output of a monotone tree). If some record $i$ were never read on input $T$, the run is identical on $T_i$, giving a wrong answer on one of them.
\end{proof}
\begin{definition}[parameterized ledger]
For a solver $A$ with input $(C;n,s)$ write $T_A\ge \max\{\,s,\ \text{(structure-dependent search term)}\,\}$. Known: search term $\le 2^n\cdot\mathrm{poly}(s)$ (brute force); conditionally $2^{\delta n}$ for some $\delta>0$ under ETH (CONDITIONAL, known theorem: Impagliazzo--Paturi); strongly $2^{(1-o(1))n}$ under SETH for CNF-SAT, and Circuit-SAT inherits it with $s=\mathrm{poly}(n)$ (CONDITIONAL).
\end{definition}
