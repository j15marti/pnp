\section*{Status}
\textbf{P versus NP remains OPEN.} Nothing here is an unconditional lower bound on
the time of an arbitrary algorithm for Circuit-SAT. Labels: PROVED/EXACT, KNOWN THEOREM,
COMPUTATIONAL, CONDITIONAL, CANDIDATE, OPEN, FALSIFIED/RETIRED.

\section{Setup and parameters}
A circuit $C$ has $n$ inputs, $g$ fan-in-2 gates over the full binary basis $B_2$, and
$w=2g+1$ wires (two per gate, plus the output). CIRCUIT-SAT$(n,g)$: is $C^{-1}(1)\neq\emptyset$?
The lower bound is a function $T(n,g)$, so every statement is parameterized by $(n,g)$ (or $(n,w)$).

\section{Ladder of lower bounds}
\begin{prop}[Wire floor, PROVED/EXACT] If $f:\{0,1\}^n\to\{0,1\}$ depends on all $n$ variables,
every circuit for $f$ has $g\ge n-1$ gates and $w\ge 2n-1$ wires (the circuit graph must be connected
from the $n$ inputs to the output; each gate merges at most two components).
The single-point function $\mathrm{AND}_n$-with-literals attains $g=n-1$ in $B_2$.
\end{prop}
\begin{prop}[Reading floor, PROVED/EXACT, trivial] A deterministic algorithm that is correct on all
circuits with $g$ gates and no unread-wire promise must read $\Omega(w)$ description bits:
for the unique-satisfier family, changing one unread gate's type changes the answer.
\end{prop}
\begin{prop}[Black-box floor, PROVED/EXACT, model-restricted] In the evaluation-oracle model,
deciding satisfiability of $f:\{0,1\}^n\to\{0,1\}$ has deterministic query complexity exactly $2^n$
(adversary answers $0$ until one point remains; that point may or may not be satisfying). Since the
unique-satisfier circuit has only $g=n-1$ gates, the black-box bound is independent of $g$ once $g\ge n-1$.
\textbf{This does not bound white-box algorithms}: a white-box solver may inspect $C$ (e.g.\ unit propagation).
\end{prop}
\begin{prop}[Conditional white-box bound, CONDITIONAL] Under SETH-type hypotheses
Circuit-SAT$(n,g)$ requires $2^{(1-o(1))n}\cdot\mathrm{poly}(g)^{\Omega(1)}$ time; this is an assumption,
not a result. KNOWN THEOREM direction: time--space tradeoffs for SAT (Williams; Fortnow--Lipton--van Melkebeek--Viglas)
rule out time $n^{c}$, $c<2\cos(\pi/7)$, with $n^{o(1)}$ space; this is the repo's FLMV spine, not re-derived here.
\end{prop}

Observation: the single-point (unique-SAT) function meets the connectivity floor $g=n-1$ for $n\le 4$,
so the instance \emph{size} does not grow faster than $n$ on this family; any super-linear-in-$g$ hardness
must come from dense/hard instances ($g\gg n$), where exact minimal sizes are out of reach of enumeration.
