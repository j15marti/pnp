\paragraph{Status.} \textbf{P versus NP remains OPEN.} Nothing here is an unconditional superpolynomial lower bound for \CSAT. Labels: PROVED, KNOWN, COMPUTATIONAL, CANDIDATE, OPEN, RETIRED.

\section{Goal and method}
Build circuits step by step, tracking the number of gates $g$ and wires $w$ (fan-in 2, so $w=2g$), and the number of inputs $n$; record lower bounds as functions of the parameters $(n,m)$ where $m=w$ is the instance size. A lower bound may be parametrized: $T(n,m)$ rather than $T(N)$.

\begin{definition}
A \CSAT\ instance is a DAG of $g$ binary gates over $n$ inputs with a designated output; $m=2g$ wires. $\mathrm{UC}(n)$ is the minimum $g$ over circuits whose function has exactly one satisfying assignment (the ``planted'' hard-instance case).
\end{definition}

\section{Step ladder (wire-by-wire)}
\begin{lemma}[Connectivity, PROVED]\label{lem:conn}
If the output of a fan-in-2 circuit depends on all $n$ inputs, then $g\ge n-1$, so $w\ge 2(n-1)$.
\end{lemma}
\begin{proof}
Restrict to the cone of the output: $g'\le g$ gates and the $n$ inputs are all in it. This undirected graph is connected, has $n+g'$ vertices and $2g'$ edges (wires). Connectivity gives $2g'\ge n+g'-1$, i.e.\ $g'\ge n-1$.
\end{proof}
A function with exactly one satisfying point depends on every input, hence $\mathrm{UC}(n)\ge n-1$; the conjunction of $n$ literals gives $\mathrm{UC}(n)=n-1$ (PROVED).

\section{Lower bounds for solving, parametrized}
\begin{theorem}[Reading bound, PROVED]\label{thm:read}
For every $m\ge 2(n-1)$ there is a family of satisfiable instances with $m$ wires such that any deterministic algorithm that decides satisfiability by probing wire entries must probe all $m$ entries. Hence time $\Omega(m)$.
\end{theorem}
\begin{proof}
Chain $g_i=\mathrm{AND}(g_{i-1},x_{\sigma(i)})$ with $\sigma$ covering all inputs, $g$ gates, output $g_g$; it is satisfiable. Redirecting any single wire to the constant~0 makes the output identically 0 (every wire lies on the chain), so the instance becomes unsatisfiable. An algorithm skipping an entry cannot distinguish the two.
\end{proof}
\begin{theorem}[Oracle bound, KNOWN]
If the circuit is only a black box on $\{0,1\}^n$, deterministic search for a unique satisfying input needs $2^n$ queries (Grover: $\Omega(2^{n/2})$ quantum).
\end{theorem}
\paragraph{Ledger of the parametrized bound.}
\[
T_{\text{proved}}(n,m)=\Omega(m),\qquad T_{\text{upper}}(n,m)=O(2^n\,m),\qquad T_{\text{conditional}}(n,m)=2^{(1-o(1))n}\ \text{(SETH)}.
\]
The gap between $m$ and $2^n m$ \emph{is} the open problem. The black-box bound does not transfer: it relativizes, and a white-box bound must use the wiring. Known unconditional white-box progress is time--space: no algorithm solves SAT in time $n^{2\cos(\pi/7)-\epsilon}\approx n^{1.80}$ with $n^{o(1)}$ space (Williams; see repo ledger row L13, and \texttt{research/agents/claude/2026-10-28\_1326\_CLAUDE\_TPORT-0001.tex} for the alternation-trading barrier).
