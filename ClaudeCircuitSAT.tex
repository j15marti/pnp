\section*{Status}
\textbf{P versus NP remains OPEN.} Nothing here is an unconditional superpolynomial lower bound for Circuit-SAT. Status labels: PROVED, KNOWN THEOREM, COMPUTATIONAL, CONDITIONAL, CANDIDATE, OPEN.

\section{Goal and plan}
Find (possibly parameterized) lower bounds for Circuit-SAT by building circuits step by step, tracking the minimum number of gates/wires needed at each stage, and growing the number of inputs $n$.
Parameters: $n$ inputs, $m$ gates, $w=2m$ wires (fan-in 2), plus structural parameters (treewidth/pathwidth $k$, depth).

\section{Model}
\begin{definition}[AIG]
A circuit over $x_1,\dots,x_n$ is a sequence of gates $g_i=\ell\wedge\ell'$ where each literal is an input or earlier gate, possibly negated (negation is free, drawn as a wire bubble). $\mathrm{aig}(f)$ is the least number of gates computing $f$. Wires $=2\,\mathrm{aig}(f)$ (plus one output wire).
\end{definition}

\section{Proved: probe lower bound linear in wires}
\begin{definition}[Probe model]
The algorithm reads the circuit description through probes ``what are the two inputs of gate $i$ (with signs)?''.
\end{definition}
\begin{theorem}[PROVED]\label{thm:probe}
Any deterministic probe algorithm deciding satisfiability of circuits with $m$ gates needs at least $m-1$ probes (hence $\Omega(w)$).
\end{theorem}
\begin{proof}
Restrict to circuits $\bigwedge_{j=1}^{m'}\ell_j$ (a chain of $m'-1$ AND gates, $m'=m+1$) where $\ell_j$ are literals. It is satisfiable iff no variable appears with both signs. Adversary: answer every probe at position $j$ with a fresh variable, positive. After $q\le m'-2$ probes, two positions $a,b$ are unprobed; the instance is consistent with all answers if both are fresh distinct (satisfiable) and also if $\ell_b=\neg\ell_a$ (unsatisfiable). So $q\ge m'-1=m$ probes are needed. (Using $m'-2<q$ gives $q\ge m'-1$.) \qedhere
\end{proof}
\emph{Caveat.} This is just the cost of reading the input. It says nothing about the number of steps after reading, so it is not a P-vs-NP bound.

\section{Parameterized picture}
\begin{itemize}
\item \textbf{KNOWN THEOREM (upper bound).} Circuit-SAT with a width-$k$ path/tree decomposition is solvable in $2^{O(k)}\,m$ time by dynamic programming over bags.
\item \textbf{CONDITIONAL.} Under SETH, SAT on CNFs of pathwidth $k$ has no $(2-\epsilon)^k\mathrm{poly}(m)$ algorithm (Lampis and others, cited from memory; to verify). So the parameterized form $2^{\Theta(k)}\,m$ is tight under SETH.
\item \textbf{KNOWN, different object.} Gate-elimination gives circuit-size lower bounds of about $3n$ to $3.1n$ gates for explicit functions in the full binary basis. These bound the \emph{size of a circuit}, not the \emph{time to decide satisfiability}, so they do not transfer directly. The step ledger above is the $n\le3$ finite shadow of that quantity.
\end{itemize}
