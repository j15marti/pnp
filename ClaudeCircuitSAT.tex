\section*{Status (strict labels)}
\textbf{P versus NP remains OPEN.} Nothing here is an unconditional super-polynomial lower bound for Circuit-SAT. Labels: PROVED/EXACT, COMPUTATIONAL, CONDITIONAL, CANDIDATE, OPEN, FALSIFIED/RETIRED.

\section{Model}
A circuit $C$ over $n$ inputs is a DAG of fan-in-2 AND gates with free negation on wires (an AIG). $g$ = gates, $w=2g+1$ = wires counting the output edge. Circuit-SAT: decide $\exists x\in\{0,1\}^n: C(x)=1$. Parameters tracked: $n,g,w$, fan-out $\phi$, treewidth $\mathrm{tw}$, number of satisfying points $|C^{-1}(1)|$.

\section{Ladder, step 0--2: smallest instances (PROVED/EXACT)}
\textbf{n=1.} Non-constant functions $x,\bar x$ need $0$ gates; SAT is decided by reading one polarity bit.\\
\textbf{n=2 (COMPUTATIONAL, exhaustive BFS, \texttt{experiments/claude\_circuitsat/2026-10-25\_ccsat\_min\_circuits.py}).} Up to negation the 7 non-constant functions need: 2 with 0 gates ($x_1,x_2$), 4 with 1 gate (the AND/OR family), 1 with 3 gates (XOR/XNOR). The 4 unique-satisfying-point functions each need exactly 1 gate.

\begin{lemma}[dependence]\label{lem:dep}
If $|C^{-1}(1)|=1$ then $C$ depends on every input.
\end{lemma}
\begin{proof} If $C$ ignores $x_i$, flipping $x_i$ maps the unique satisfying point to a different satisfying point. \end{proof}

\begin{theorem}[wire lower bound for unique-SAT circuits]\label{thm:wire}
A fan-in-2 circuit with exactly one satisfying assignment on $n\ge1$ inputs has $g\ge n-1$ gates and $w\ge 2n-1$ wires; both are attained by a chain/tree AND of $n$ literals.
\end{theorem}
\begin{proof} By Lemma~\ref{lem:dep} every input has a directed path to the output. A fan-in-2 DAG whose output reaches $n$ distinct sources has $\ge n-1$ gates (induction: a gate merges at most two source-sets). Wires $=2g+1$. Tightness: the AND tree on literals $\ell_i=x_i\oplus s_i$ with sign vector $s$. \end{proof}

\section{Step 3: what the wires cost a solver}
\begin{theorem}[black-box evaluation, KNOWN folklore, PROVED by adversary]
Any algorithm accessing $C$ only through evaluation queries needs $2^n$ queries in the worst case to separate the family $\{\mathrm{AND}_s\}$ from the constant-0 circuit. (Adversary answers 0; each query kills one point.) Quantumly $\Theta(2^{n/2})$ (Grover/BBBV).
\end{theorem}
\begin{theorem}[white-box wire reading, PROVED/EXACT]
For the same family, with polarity bits on wires, deciding SAT requires reading $\ge n$ wire polarities (adversary hides one) and $n$ suffice. So the unique-SAT family has white-box complexity $\Theta(n)$ although black-box complexity is $2^n$.
\end{theorem}
\textbf{Lesson.} A parameterized lower bound must charge for \emph{structure that is not readable off wires cheaply}; sparse-satisfying-set size alone (Theorem~\ref{thm:wire}) gives only linear wire bounds. Hardness must come from circuits where the minimal wire count is large \emph{and} the local wire neighbourhood does not reveal the satisfying point (XOR-type mixing: XOR costs 3 gates at $n=2$ versus 1 for AND).

\section{Parameterized upper bounds (the target to beat), KNOWN}
Brute force $O(g2^n)$; treewidth DP $O(g\,2^{\mathrm{tw}})$ (Circuit-SAT is FPT in $\mathrm{tw}$); so any parameterized lower bound $T(n,g,\mathrm{tw})$ must respect $T\le O(g\,2^{\min(n,\mathrm{tw})})$. Conditional (SETH) lower bounds $2^{(1-\epsilon)n}$ are CONDITIONAL, not unconditional.
