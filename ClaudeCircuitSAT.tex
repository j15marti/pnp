\paragraph{Status.} \textbf{P versus NP remains OPEN.} Nothing here is an unconditional super-linear lower bound for Circuit-SAT. Labels: PROVED, KNOWN, COMPUTATIONAL, CANDIDATE, OPEN.

\section{Goal and model}
Find lower bounds, possibly parameterized, for Circuit-SAT by building circuits step by step (more inputs, more gates) and tracking the minimal number of wires needed at each step.
\begin{definition}[Circuit, wire cost]
A circuit over basis $B_2$ (all 2-input gates, negation absorbed) with $n$ inputs and $m$ gates has $2m$ wires (fan-in 2). For $f:\{0,1\}^n\to\{0,1\}$ let $G(f)$ be the minimum gate count and $W(f)=2G(f)$ the minimum wire count.
\end{definition}
\emph{Algorithmic model:} the instance $C$ is a read-only array of gate records (type, two predecessor indices); the algorithm has random access and a probe counts one record read.

\section{Rung 0: constants and literals ($G=0$)}
Functions of a single literal or constant need no wires. Deciding SAT for such a circuit requires inspecting the output designator: 1 probe.

\section{Rung 2: a probe lower bound for SAT decision}
\begin{theorem}[PROVED, trivial reading bound]\label{thm:read}
For every $m\ge1$ any deterministic algorithm that correctly decides satisfiability of $m$-gate circuits must probe at least $m$ gate records on some instance.
\end{theorem}
\begin{proof}
Let $C$ be the chain $g_1=x_1\wedge x_2$, $g_i=g_{i-1}\wedge x_{i+1}$ for $i=2..m$, output $g_m$; it is satisfied by the all-ones input. Let $C^{(j)}$ change only the type of $g_j$ to the constant $0$; then $C^{(j)}$ is unsatisfiable. If on input $C$ the algorithm never probes record $j$, its run on $C^{(j)}$ is identical, so it answers wrongly on one of them. Hence it probes all $m$ records on $C$.
\end{proof}
This is linear in the \emph{instance size}; it says nothing about time as a function of $n$. It is the base of the ladder: every rung's bound is at least the wire count of the cone of the output.

\section{Rung 3: counting and parameterized landscape}
\begin{itemize}
\item \textbf{KNOWN} (Shannon counting): there are at most $\big(16(n+m)^2\big)^m$ circuits, so some $f$ needs $G(f)=\Omega(2^n/n)$; this bounds instance size, not SAT time.
\item \textbf{KNOWN} Circuit-SAT of treewidth/pathwidth $k$ is solvable in $2^{O(k)}m$ time by dynamic programming over a path decomposition.
\item \textbf{CANDIDATE} (literature transfer, citation to be verified next run): under SETH, SAT parameterized by pathwidth $k$ admits no $(2-\varepsilon)^k\,\mathrm{poly}(m)$ algorithm (Iwata--Yoshida; Lokshtanov--Marx--Saurabh line). This would give a tight conditional parameterized bound $2^{k}$ with parameter = circuit width, the natural ``wire cut'' measure. The wire-by-wire ladder is intended to supply the exact width of the gadgets at each rung.
\item \textbf{OPEN} unconditional lower bound beyond $\Omega(m)$ time for Circuit-SAT on general circuits.
\end{itemize}
