\section{Goal and method}
Build Circuit-SAT bottom-up: for $n=1,2,3,\dots$ inputs, track the minimum number of wires (fan-in-2 gate inputs, so $\mathrm{wires}=2\cdot\mathrm{gates}$) needed to realize each function, then ask what each step costs a solver. The bound is allowed to be parameterized by $(n,m)$ with $n$ inputs and $m$ gates.

\section{Model}
Instance: a DAG of $m$ fan-in-2 gates over basis $B_2$ (all 16 binary gates), $n$ inputs, one output. Two access models: \emph{black-box} (evaluate $C(x)$) and \emph{white-box} (read gate list entries). $C(f)$ = minimal gate count of $f$; negation is free.

\section{Results so far}
\begin{lemma}[PROVED, white-box reading bound]\label{l:read}
In the gate-list query model, any deterministic Circuit-SAT algorithm makes $\Omega(m)$ queries, hence reads $\Omega(m)$ wires.
\end{lemma}
\emph{Proof.} Take $g_j=g_{j-1}\wedge x_{j}$ (chain, satisfiable by all-ones). If the algorithm leaves gate $j$ unread, the adversary answers consistently with the chain; replacing $g_j$ by constant $0$ gives an unsatisfiable circuit agreeing on every answered query. Hence all $\Omega(m)$ gates (for $m\le n$ chain length) must be read. \qed

\begin{lemma}[PROVED, black-box bound]
Any black-box algorithm needs $2^n$ evaluations in the worst case (unstructured search). This does not bound white-box algorithms.
\end{lemma}

\begin{lemma}[PROVED, step recursion]
$C_{n+1}\le 2C_n+3$ where $C_n=\max_f C(f)$ over $n$-input $f$ (Shannon expansion plus a 3-gate multiplexer).
\end{lemma}

\section{Open arrows / next steps}
\begin{enumerate}
\item Connect the ledger to hardness: is there a function family whose minimal circuits force Circuit-SAT instances of size $m$ with $\ge 2^{\Omega(n)}$ \emph{distinct} wire-level obstacles? Gap: wire-count lower bounds for single functions (Shannon counting gives $2^n/n$ existentially) say nothing about solver time.
\item Parameterize by treewidth/cutwidth of the circuit: SAT in $2^{O(\mathrm{tw})}m$; look for matching conditional bounds and any unconditional restricted-model (e.g.\ resolution, read-once) bounds.
\item Extend ledger to $n=4$ and track per-step wire increments $\Delta_n$.
\end{enumerate}
