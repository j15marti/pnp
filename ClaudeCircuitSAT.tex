\section*{0. Status (strict labels)}
\textbf{P versus NP remains OPEN.} Nothing here proves a superpolynomial lower bound
for \SAT{} on a general Turing/RAM model. Labels: PROVED/EXACT, KNOWN THEOREM,
COMPUTATIONAL, CONDITIONAL, CANDIDATE, OPEN, FALSIFIED/RETIRED.

\section{Program}
Build up from the simplest circuits, tracking for each step $(n,g)$ ($n$ inputs, $g$ fan-in-2
gates, $w=2g$ gate-input wires): (i) the minimal wire/gate count needed to realize the
function class, and (ii) a lower bound on the work needed to decide satisfiability.
Two cost models must be kept separate:
\begin{itemize}
\item \textbf{Black-box} (oracle access to $C$): adversary lower bound.
\item \textbf{White-box} (algorithm reads the netlist): only trivial bounds are known unconditionally.
\end{itemize}

\section{Ladder of exact results}
\begin{prop}[Read bound; PROVED/EXACT, trivial] If $C$ depends on $n$ inputs and is a
connected fan-in-2 circuit with a single output then $g\ge n-1$, $w\ge 2n-2$. Any
white-box SAT algorithm that is correct on all circuits must inspect, for the worst-case
instance, every gate lying on a path to the output that cannot be pruned, hence
$\Omega(g)$ time. This is linear and carries no hardness content.
\end{prop}
\begin{prop}[Black-box bound; PROVED/EXACT] Deterministic query complexity of \SAT{} with
evaluation-oracle access is exactly $2^n$ in the worst case: for the family
$\{C_a=[x=a]\}_{a\in\{0,1\}^n}$ each query $x$ eliminates at most one $a$, so an adversary
answers 0 until one candidate remains. Each $C_a$ has $g=n-1$ (an AND of literals; negations are free
on wires in the $B_2$ basis), so black-box hardness is already present at $g=n-1$.
\end{prop}
\begin{prop}[Relativization caveat; KNOWN THEOREM context] The black-box bound is
an oracle bound and is not a lower bound for white-box algorithms (cf.\ Baker--Gill--Solovay).
White-box algorithms exploit structure: the $C_a$ above are solved in $O(n)$ by reading
the netlist. Hence the real question is whether any \emph{small-size family} can force
$\omega(\mathrm{poly}(g))$ white-box work; this is exactly P vs NP and is OPEN.
\end{prop}

\section{Dynamic and reverse dynamic programming (run 3)}
\label{sec:dp}
\textbf{Question.} Can forward or reverse dynamic programming (DP) model a lower bound
for an NP-complete problem? Can comparing the 3-SAT and circuit-SAT versions of one
problem pick out the ideal algorithm?

\subsection{The two DPs on one instance}
Fix a 3-CNF $F$ (or a circuit $C$) on variables $x_1,\dots,x_n$ and an order $\pi$.
\begin{itemize}
\item \textbf{Forward DP:} after assigning prefix $S=\{\pi_1,\dots,\pi_i\}$ by $a$,
the state is the residual subproblem $F|_a$ (for a circuit: the netlist after constant
propagation, i.e.\ the values on the live wire frontier). Merge equal states and drop dead ones
(a falsified clause, or output forced to $0$). Width $W_\pi(i)=\#$distinct live states.
\item \textbf{Reverse DP:} start from ``output $=1$'' and eliminate variables from the back.
For CNF this is Davis--Putnam variable elimination (1960). It equals the forward DP on the reversed order.
\item \textbf{Meet in the middle:} $F$ is SAT iff a forward state at cut $S$ and a reverse
state at $\bar S$ are compatible. With the syntactic states below this check is set
disjointness on the clauses crossing the cut.
\end{itemize}

\subsection{Exact statements}
\begin{prop}[Semantic DP is representation-independent; PROVED/EXACT]
If states are residual \emph{functions} $f|_a$, then the width at cut $S$ is the number of
distinct nonzero rows of the $2^{|S|}\times2^{n-|S|}$ truth-table matrix $M_S$, which is
$2^{D^{\to}(M_S)}$ (one-way communication complexity). Reverse DP uses columns. This depends
only on $f$, so the 3-SAT and circuit-SAT versions of the same function give identical widths.
Over all orders, forward and reverse optima coincide (reverse the order).
\end{prop}
\begin{prop}[Semantic DP is clairvoyant; PROVED/EXACT] For UNSAT $f$ every residual is
$0$, so semantic width is $\le1$. Semantic DP therefore assumes the answer and gives no lower
bound. A lower-bound model must use \emph{syntactic} states, as real DP algorithms do.
\end{prop}
\begin{prop}[3-SAT vs.\ circuit-SAT under syntactic DP; PROVED/EXACT, easy]
\leavevmode
(i) For CNF, the state at cut $S$ is determined by the set of crossing clauses that $a$
leaves unsatisfied, so $W(S)\le 2^{\#\text{crossing clauses}}$. The circuit
$\bigwedge_j\bigvee_{\ell\in c_j}\ell$ with constant propagation has exactly the same
residuals.
(ii) For the Tseitin 3-CNF $T(C)$ of a general circuit, assigning inputs and gate variables in a
topological layout gives $W\le 2^{\mathrm{vs}}$, where $\mathrm{vs}$ is the number of
assigned wires with an unassigned consumer (the wire frontier). So DP on the 3-SAT
encoding \emph{simulates} the wire-by-wire circuit DP of \S2. Non-topological orders amount
to guessing gate values, which only adds freedom.
(iii) DP over the \emph{inputs only} (no gate variables) may need width up to
$2^{w_f2^{w_r}}$ for forward/reverse cutwidths $w_f,w_r$ (McMillan's OBDD bound; KNOWN THEOREM).
Encoding with auxiliary wire variables avoids this doubly exponential term.
\end{prop}
\textbf{Answer on the ``ideal algorithm'' (within DP).} The 3-SAT (Tseitin) form is the better
substrate: it contains the circuit DP and can beat input-only DP. The ideal DP is
tree/path-decomposition DP on the netlist, time $2^{O(\mathrm{tw}(C))}\mathrm{poly}$
(KNOWN THEOREM; cf.\ Alekhnovich--Razborov, branch-width SAT algorithms, FOCS 2002).

\begin{prop}[DP lower bounds exist but only for the DP class; KNOWN THEOREMS $\Rightarrow$ EXACT TRANSFER]
On an UNSAT $F$ the forward syntactic DP has no early exit. Its merged state graph is an
ordered read-once branching program for the search problem ``find a falsified clause''.
Such programs correspond to ordered/regular resolution refutations of polynomially related
size (Krajíček; Lovász--Naor--Newman--Wigderson). The reverse DP \emph{is} ordered
resolution (Davis--Putnam). Hence explicit 3-CNFs force total DP work $2^{\Omega(n)}$:
pigeonhole (Haken 1985, after 3-CNF encoding), Tseitin formulas on expanders (Urquhart 1987),
and random 3-CNF at constant ratio (Chvátal--Szemerédi 1988).
\end{prop}

\subsection{Why this does not reach P vs NP (barrier audit)}
\begin{itemize}
\item \textbf{FAILED TRANSFER to all algorithms.} A lower bound for a DP class is a lower
bound for a proof system (resolution-type). Superpolynomial lower bounds for \emph{every}
proof system is equivalent to NP $\ne$ coNP (Cook--Reckhow 1979), which is stronger than P $\ne$ NP.
\item \textbf{No single ideal algorithm.} Tseitin formulas are exponential for every DP above,
yet Gaussian elimination solves them in polynomial time. Pigeonhole is exponential for DP
but polynomial for cutting planes. Comparing the 3-SAT and circuit versions therefore picks the best \emph{DP}, not the best algorithm.
\item \textbf{Communication-style caps.} Any lower bound obtained by cutting the
\emph{instance description} (streaming, one-way protocol) is at most the input length.
Exponential DP widths come from cutting the \emph{assignment space}, which arbitrary
algorithms need not enumerate.
\item \textbf{Richer states = stronger proof systems.} DP with affine/parity states
corresponds to Res($\oplus$)-type systems, and DP with linear-inequality states to cutting
planes. Each strengthening is a known open or partially solved lower-bound frontier.
Recent exponential bounds for regular Res($\oplus$) exist (to be verified next run, so
treated as CANDIDATE citations).
\end{itemize}

\subsection{State-parameterized (adaptive) forward/reverse DP (run 3)}
\label{sec:adaptive}
\textbf{Question.} Suppose each step's parameters depend on the current state: which
variable to branch on next, which branch to try first, whether to step forward (branch) or
backward (eliminate), and possibly how the state is represented. Does this change the conclusion above?

Write the DP as a \emph{policy} $\sigma$ mapping each state $s$ to a parameter $\sigma(s)$.
Cost is the number of states visited. Two quantities must be kept apart:
\[
\mathrm{OPT}(F)=\min_{\sigma}\mathrm{cost}_\sigma(F)\quad\text{(best policy, chosen with hindsight)},\qquad
\mathrm{ALG}_\sigma(F)\ \text{for a fixed policy }\sigma\text{ computable in poly time}.
\]

\begin{prop}[Optimal adaptivity on SAT instances is a certificate; PROVED/EXACT]
If $F$ is satisfiable, $\mathrm{OPT}(F)\le n+1$. The policy that always branches toward a fixed
satisfying assignment visits one path. Hence an adaptive DP lower bound \emph{must} charge for
computing $\sigma$. Choosing the parameters optimally is as hard as SAT itself: an optimal parameter sequence \emph{is} an NP witness.
\end{prop}

\begin{prop}[Syntactic adaptive DP stays inside resolution; KNOWN THEOREMS]
On UNSAT $F$, with states being residual formulas:
(i) per-state branching variable plus merging of equal states is a read-once branching
program, i.e.\ regular resolution;
(ii) mixing branching with backward elimination steps, and storing derived (learned) clauses,
is CDCL-style search. Its runs are simulated by general resolution, and CDCL with restarts
p-simulates resolution (Pipatsrisawat--Darwiche 2011; Atserias--Fichte--Thurley 2011).
Haken, Urquhart and Chvátal--Szemerédi are lower bounds for \emph{general} resolution.
So $\mathrm{OPT}(F)\ge 2^{\Omega(n)}$ still holds for explicit 3-CNFs, and adaptivity does not
remove the exponential bound for this class.
\end{prop}

\begin{prop}[Optimal vs.\ achievable adaptivity; KNOWN THEOREM]
Even inside resolution, finding a near-optimal policy is hard: resolution is not
automatable unless P $=$ NP (Atserias--Müller 2019). So $\mathrm{ALG}_\sigma$ can be far above
$\mathrm{OPT}$. That gap shows how hard \emph{choosing} the parameters is. It is not a lower bound on SAT.
\end{prop}

\paragraph{Where the conclusion changes: adaptive state representation.}
If the parameter may also switch the state \emph{type}, the DP climbs the proof-system hierarchy:
\begin{center}\begin{tabular}{@{}lll@{}}\toprule
state type chosen per step & proof system & superpoly.\ lower bound\\\midrule
residual clauses & resolution / CDCL & KNOWN (Haken 1985)\\
parities / affine spaces & Res($\oplus$) & regular/tree-like versions (CANDIDATE citations, verify); general: OPEN\\
linear inequalities & cutting planes & KNOWN (Pudl\'ak 1997, via interpolation)\\
polynomials & polynomial calculus & KNOWN (Razborov 1998, PHP)\\
bounded-depth formulas & $AC^0$-Frege & KNOWN (Ajtai; Beame--Impagliazzo--Pitassi; PHP)\\
arbitrary formulas / circuits & Frege / extended Frege & OPEN\\
arbitrary poly-time state map & every algorithm & OPEN $=$ P vs NP\\\bottomrule
\end{tabular}\end{center}
Every deterministic algorithm is an adaptive DP whose state is its full machine configuration.
So the fully parameterized model is universal: a lower bound for it is exactly P $\ne$ NP.
Lower bounds for the nondeterministic optimum $\mathrm{OPT}$ over all state types would be
NP $\ne$ coNP (Cook--Reckhow). \textbf{Answer:} parameterization leaves the conclusion unchanged
while states are syntactic restrictions, possibly with learned clauses. It turns into the full open problem once the
parameter may choose the state representation freely. The open frontier is the
Frege row.

\subsection{Wire ledger for the unique-SAT family (PROVED/EXACT)}
For $C_a=[x=a]$ the minimum is exactly $n-1$ gates, $2(n-1)$ gate-input wires: $0,1,2,3$ for $n=1..4$.
Lower bound: relevance (all $n$ inputs matter, connected cone); upper bound: AND of literals. So the
incremental cost of one more input is exactly \emph{one gate, two wires}. For the hardest functions the
increments are $0,1,3,\ge2$, noisy, and they grow as $\Theta(2^n/n)$ by counting (Shannon/Lupanov, KNOWN THEOREM).
Parametrised reading: write $s=$ gates, $n=$ inputs. The black-box bound is $2^n$ queries for every $s\ge n-1$;
the white-box cost of the same instances is $O(s)$. The exponent in $n$ is therefore a property of the access model,
not of the wiring, which is the Baker--Gill--Solovay gap again.

\subsection{AIG ledger and reading bound (run 1, merged)}
In the AIG model (2-input ANDs, free edge negation) the exact sizes are $C_{\mathrm{AIG}}(1,2,3)=0,3,6$
(XOR costs 3; parity costs $3(n-1)$ for $n\le3$). Script \texttt{research/claude\_circuitsat/aig\_exact.py},
COMPUTATIONAL. The $B_2$ and AIG ladders differ by the XOR gadget, so sizes are only comparable within one model.
\begin{prop}[Reading bound; PROVED/EXACT] Any algorithm deciding \SAT{} on $m$-gate instances must read $\Omega(m)$ gates.
Take $\varphi=x_1\wedge\bigwedge_i s_i$ with each slot $s_i$ a tautology; replacing one slot $s_j$ by $\neg x_1$ makes it UNSAT, so
an algorithm skipping $s_j$ errs on one of the two. This is linear and carries no hardness content.
\end{prop}

\subsection{Repair of a claim in the run-1 text}
Run 1 stated that parity costs $3(n-1)$ in AIG for all $n$; only $n\le3$ is verified here. The general statement is
a known upper bound, and its optimality is unproved here (CANDIDATE).
