\section*{Status}
\textbf{P versus NP remains \OPEN.} Nothing here proves a super-linear lower bound
for Circuit-SAT. This document tracks a \emph{ladder}: start from the smallest
circuit-SAT instances, count gates and wires exactly, grow the number of inputs,
and record at each rung what is proved, what is only known from the literature,
and what is open. Statuses: \PROVED, \KNOWN, \COMP, \CAND, \OPEN, \RET.

\section{Model}
A circuit over basis $B_2$ (all 2-input gates, negation free) has gates of fan-in
2, so $\mathrm{wires}=2\cdot\mathrm{gates}$ (counting each gate input, with fan-out
counted at the source). $C_{B_2}(f)$ is the minimum number of gates.
\textsc{Circuit-SAT}$_{n}$: given a circuit $C$ on $n$ inputs, decide $\exists x\,C(x)=1$.
Two distinct objects are tracked, and must not be confused:
\begin{enumerate}[nosep]
\item[(A)] the \emph{solver}: a circuit family computing the decision function of Circuit-SAT on encoded instances (the object whose lower bound would separate P from NP/poly);
\item[(B)] the \emph{instance}: the circuit $C$ itself, whose wires we count in the ladder.
\end{enumerate}

\section{Proved structural lower bounds}
\begin{lemma}[connectivity]\PROVED. If $f:\{0,1\}^n\to\{0,1\}$ depends on all $n$ inputs, any fan-in-2 circuit has $\ge n-1$ gates and $\ge 2(n-1)$ wires.
\end{lemma}
\emph{Proof.} Every input must have a directed path to the output, so the underlying graph restricted to the cone is connected on $\ge n+g$ vertices with $\le 2g$ edges, and a connected graph on $n+g$ vertices needs $\ge n+g-1$ edges, so $2g\ge n+g-1$, i.e. $g\ge n-1$. \qed

\begin{proposition}[tightness for symmetric prefixes]\PROVED.
$C_{B_2}(\mathrm{AND}_n)=C_{B_2}(\mathrm{OR}_n)=C_{B_2}(\mathrm{XOR}_n)=n-1$.
\end{proposition}
(Upper bound by a tree; lower bound by the Lemma.) Hence the first rung family
(``is a single satisfying assignment fixed'', i.e.\ SAT of a read-once AND/OR formula) has
linear exact complexity; no information is lost between $n=1,2,3$ and general $n$.

\begin{lemma}[reading the instance]\PROVED\ (conditional on encoding).
Let $E$ be an encoding of $N$ bits of instance description in which every bit can flip
the SAT answer for some setting of the others. Then any solver circuit has $\ge N-1$ gates.
\end{lemma}
This is the only lower bound on side (A) that is proved here; it is linear and generic.

\section{Known frontier (side A)}
\begin{itemize}[nosep]
\item Counting (Shannon): almost every $f$ on $n$ inputs needs $\Omega(2^n/n)$ gates. \KNOWN. Non-explicit.
\item Gate elimination gives explicit $3n-o(n)$ (Blum 1984) and about $3.1n-o(n)$ for $B_2$ (Find--Golovnev--Hirsch--Kulikov 2016; Li--Yang 2022 improved constants); $5n-o(n)$ over $U_2$ (Iwama--Lachish--Morizumi--Raz). \KNOWN\ (recalled; citations to be verified against primary sources next run).
\item No super-linear lower bound for any explicit function in NP over general circuits. \KNOWN\ open. Barriers: relativization, natural proofs, algebrization.
\end{itemize}
