\section*{Status}
\textbf{P versus NP remains OPEN.} No superpolynomial lower bound for Circuit-SAT is claimed. This document records an incremental ladder: for $n=1,2,3,\dots$ inputs, the minimal number of gates/wires needed to build circuits, and the lower bounds that follow for solving Circuit-SAT as a function of the parameters $(n,s)$ ($n$ inputs, $s$ wires). Labels: PROVED/EXACT, KNOWN THEOREM, COMPUTATIONAL, CONDITIONAL, CANDIDATE, OPEN, FALSIFIED/RETIRED.

\section{Model}
Circuit over basis $B_2$ (all 2-input gates), fan-in 2, unbounded fan-out. Size $=$ number of gates $g$; wires $=2g$ (gate in-edges), plus one output designation. Circuit-SAT: given $C$, does some $x\in\{0,1\}^n$ have $C(x)=1$? Parameters: $n$, $s=|{\rm wires}|$. Two access models: (E) \emph{evaluation black box} (only queries $C(x)$), (S) \emph{structural} (full netlist).

\section{Proved lower bounds on structure}
\begin{prop}[PROVED/EXACT, wire floor]
If $f$ depends on all $n$ variables then every circuit for $f$ has $\ge n-1$ gates and $\ge 2(n-1)$ wires; AND$_n$ attains it. (Each input must lie on a path to the output; a fan-in-2 DAG with a single sink and $n$ connected sources has $\ge n-1$ gates.)
\end{prop}
\begin{prop}[PROVED/EXACT, query bound in model (E)]
Any (deterministic or randomized with error $<1/2$ success) algorithm for Circuit-SAT in model (E) needs $2^n$ (resp.\ $\Omega(2^n)$) evaluation queries on the family $C_a=[x=a]$, $a\in\{0,1\}^n$, each of size $g=n-1+\#\text{negations}$, i.e.\ $O(n)$ wires. Adversary: answer 0 to every query until one point remains.
\end{prop}
\begin{prop}[PROVED/EXACT, parameterized reading]
For model (E) the bound is $2^{n}$ regardless of $s$, while structure (model (S)) lets AND-of-literals instances be solved in $O(s)$. Hence any super-linear lower bound must exploit the \emph{gap between} (E) and (S): it cannot come from query counting. This is the circuit-level form of the repository's ``free semantic SAT answers'' restriction.
\end{prop}

\section{Known gate-elimination bounds (KNOWN THEOREM, cited from memory; verify before relying)}
Explicit functions need $\ge 3n-o(n)$ gates over $B_2$ (Blum 1984: $3n-3$; Li--Yang, STOC 2022: $3.1n-o(n)$). These are lower bounds on \emph{computing} a function, not on deciding SAT of a given circuit; they supply the wire-count recursion used in the ladder: setting one input to a constant kills its fan-out gates, giving $\ge 2$--$3$ eliminated gates per variable. They give the floor $\Omega(n)$ and cannot exceed $O(n)$ by current technology (no explicit $\omega(n)$ circuit lower bound for $B_2$ is known).

\section{Parameterized lower-bound targets}
\begin{enumerate}
\item (CANDIDATE) $T_{\mathrm{SAT}}(n,s)\ge 2^{\Omega(n/\tau)}$ where $\tau$ is a structural parameter (treewidth/pathwidth of the circuit); the matching upper bound $2^{O(\tau)}\mathrm{poly}(s)$ is KNOWN for small treewidth, and ETH/SETH would give the converse CONDITIONALly.
\item (CANDIDATE) Time-space: for the bound in the repository's ARC/FLMV spine, see \texttt{main.tex}; Buss--Williams (ledger row L13) bars alternation trading below $2\cos(\pi/7)\approx1.80$ for SAT, and the repo notes in \texttt{AGENTS.md} (L34) give a barrier for portfolio versions.
\item (OPEN) Wire-by-wire monotone ladder: define $W(n)=\max$ over unsatisfiable-or-unique-SAT instances of the minimal wires to certify non-membership; the data above gives $W(1),W(2),W(3)\le 0,2,8$. Proving $W(n)$ superpolynomial would be a circuit lower bound implying $\mathrm{NP}\not\subseteq \mathrm{P/poly}$, hence is OPEN.
\end{enumerate}
