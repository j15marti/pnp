\section{Goal and model}
Build \CSAT{} instances from the simplest case upward, tracking the wires (connections) needed per step and the cost of deciding satisfiability, with the aim of a (possibly parameterized) lower bound.
\begin{definition}[Instance]
A circuit $C$: $n$ input nodes, $g$ gates of fan-in 2 over the full binary basis $B_2$ (negation is free on wires), one output, every node reaches the output, underlying undirected graph connected. Size $m=n+g$; wires $w=2g$.
\end{definition}
\begin{definition}[Excess]
$\mu(C)=w-(n+g)+1=g-(n-1)$, the cyclomatic number of the underlying graph.
\end{definition}

\section{Wire-level lemmas (PROVED/EXACT)}
\begin{lemma}[Connection floor]\label{l:floor}
$g\ge n-1$, so $w\ge 2(n-1)$, i.e.\ $\mu\ge0$, with $\mu=0$ iff the underlying graph is a tree (a read-once formula).
\end{lemma}
\begin{proof} A connected graph on $n+g$ nodes has $\ge n+g-1$ edges; $w=2g$. \end{proof}

\begin{lemma}[Reading lower bound]\label{l:read}
Let $\Phi_n$ be the family of $n$-literal conjunctions $\ell_1\wedge\dots\wedge\ell_n$ (each $\ell_i\in\{x_j,\neg x_j\}$ over $n$ variables), presented as an AND-chain whose $i$-th wire label is $\ell_i$. Any deterministic algorithm with random access to the encoding that decides satisfiability must read all $n$ labels (worst case).
\end{lemma}
\begin{proof} Suppose label $i$ is unread on some run. The adversary answers the other labels so that they are pairwise consistent and, for $n\ge2$, one other label is $x_j$. Choosing $\ell_i=x_j$ gives SAT, choosing $\ell_i=\neg x_j$ gives UNSAT, with identical transcripts. \end{proof}
Hence time $\Omega(m)$ for \CSAT{} in general, and $\Theta(m)$ on trees ($\mu=0$; upper bound by the tree DP below).

\begin{proposition}[Excess-parameterized upper bound]\label{p:ub}
\CSAT{} is decidable in time $2^{\mu}\cdot O(m)$ (deterministic).
\end{proposition}
\begin{proof}
Take a spanning tree $T$ of the underlying graph; the $\mu$ non-tree edges each get one endpoint put in a set $F$, $|F|\le\mu$. Enumerate the $2^{|F|}$ value assignments to nodes of $F$. $G-F\subseteq T$ is a forest; each gate constraint $(h;a,b)$ lives in the star of $h$, so the standard rooted-tree DP with tables of size $\le4$ (value of node, value of parent) checks consistency and the output requirement in $O(m)$.
\end{proof}
Consequence: \CSAT{} with $\mu=O(\log m)$ is in P; any super-polynomial hardness must live at $\mu=\omega(\log m)$, i.e.\ $g\ge n+\omega(\log n)$.

\section{CNF embedding and the conditional matching bound}
A 3CNF with $n$ variables, $m_c$ clauses becomes a circuit with $g=3m_c-1$, $w=6m_c-2$, hence $\mu=3m_c-n$.
\begin{proposition}[CONDITIONAL]
Assume ETH. Then \CSAT{} has no $2^{o(\mu)}\mathrm{poly}(m)$-time algorithm.
\end{proposition}
\begin{proof} By sparsification, ETH gives instances with $m_c=O(n)$, so $\mu=O(n)$; a $2^{o(\mu)}$ algorithm would give $2^{o(n)}$ 3SAT. \end{proof}
Together with Proposition~\ref{p:ub}: under ETH, the right parameter is the excess and $T(\mu)=2^{\Theta(\mu)}\mathrm{poly}$. The unconditional gap is: proved $\Omega(m)$, upper $2^{\mu}m$. \textbf{OPEN:} any unconditional bound $\omega(m)$ on a deterministic RAM/TM for this family.
