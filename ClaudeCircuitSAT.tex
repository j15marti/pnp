\section{Goal and method}
Start from the smallest Circuit-SAT instances, grow the number of inputs $n$ one at a
time, and record for each rung (i) the minimum number of gates $g$ and wires $w=2g$
(fan-in-2 gates, all 16 binary gate types, free constants and negation),
(ii) the \emph{structure} of the wiring, summarised by a parameter, and
(iii) the best known upper and lower bound on the cost of \emph{deciding} satisfiability.
The target is a parametrised bound $T(n,e,s)$, where $s$ is circuit size and
$e$ is the \emph{reconvergence number} defined below.

\section{Definitions}
A circuit $C$ has $n$ inputs, $g$ fan-in-2 gates, one output. Wires $=$ gate-input edges, $w=2g$.
Let $G(C)$ be the underlying undirected graph on inputs and gates (inputs that reach
the output only). Its cyclomatic number is
\[ e(C)=w-(n+g)+1 = g-n+1 \quad(\text{if connected}).\]
For a function $f$ depending on all $n$ inputs, $C(f)$ is its minimum gate count and
$r(f)=C(f)-(n-1)$ its \emph{minimum reconvergence number}.

\section{Proved structural facts}
\begin{thm}[PROVED, connectivity bound]\label{thm:conn}
If $C$ computes $f$ depending on all $n$ inputs then $g\ge n-1$, $w\ge 2(n-1)$, and equality holds iff $G(C)$ is a tree, i.e.\ $C$ is a read-once formula (every input has fan-out one and no gate has fan-out $>1$).
\end{thm}
\begin{proof}
Each essential input has a directed path to the output, so $G(C)$ is connected on $n+g$ vertices with $2g$ edges; connectivity gives $2g\ge n+g-1$. Equality iff acyclic. A tree with fan-in 2 and a single sink has all fan-outs 1.
\end{proof}
\begin{prop}[KNOWN, acyclic CSP]\label{prop:tree}
Circuit-SAT is decidable in $O(s)$ time when $e(C)=0$, and in $O(2^{e}\,s)$ time in general.
\end{prop}
\begin{proof}[Sketch]
Circuit-SAT is a CSP with one constraint per gate on $(\text{out},\text{in}_1,\text{in}_2)$. If $G(C)$ is a tree the constraint hypergraph is acyclic and Yannakakis' semijoin reduction decides it in linear time. For general $C$ choose a spanning tree, cut each of the $e$ non-tree wires by a fresh leaf, guess its value ($2^e$ choices), add the equality constraint at its source, and solve each acyclic instance.
\end{proof}
Also $T\le O(2^{n}s)$ by enumeration, so $T(n,e,s)\le O(2^{\min(n,e)}s)$ (KNOWN).

\section{Decision-cost ladder}
\begin{center}\begin{tabular}{@{}lll@{}}\toprule
regime & upper bound & lower bound\\\midrule
$e=0$ (read-once) & $O(s)$ (KNOWN) & $\Omega(s)$ reading (PROVED trivially: unread gate can flip the answer)\\
$e$ fixed & $O(2^e s)$ (KNOWN) & $\Omega(s)$ only; $2^{\Omega(e)}$ OPEN\\
$e\approx n$ general & $O(2^n s)$ & $2^{(1-\epsilon)n}$ excluded under SETH (CONDITIONAL, since CNF-SAT embeds with $s=O(m)$)\\
black-box access only & $2^n$ queries & $2^n$ queries (PROVED by adversary: unique satisfying point $=$ AND of literals, which has $r=0$, so white-box access is what makes it easy)\\\bottomrule
\end{tabular}\end{center}
The last row isolates the point of the programme: hardness must come from \emph{white-box wiring} information, and the parameter $e$ measures exactly how much wiring the algorithm can exploit.

\section{Open arrows (nothing here is a lower bound for P vs NP)}
\begin{enumerate}
\item \textbf{OPEN} Unconditional $2^{\Omega(e)}$ lower bound (any model with unrestricted RAM would imply $\mathrm{P}\neq\mathrm{NP}$ for $e=\omega(\log s)$); identify a restricted but natural model (DPLL/resolution-style, monotone, bounded depth) where it is provable.
\item \textbf{OPEN} Compute $r(f)$ and formula-vs-circuit gap for $n=4,5$ with smarter enumeration (canonical forms under input permutation, SAT-based exact synthesis).
\item \textbf{CANDIDATE} Relate $e$ to treewidth and to proof-size lower bounds for resolution on CNF encodings of circuits with $r(C)=e$: small $e$ gives resolution proofs of size $2^{O(e)}\mathrm{poly}(s)$; a matching bound for explicit families is a model-matched target.
\item \textbf{CANDIDATE} Gate-elimination view: Theorem~\ref{thm:conn} is the base case of the $n-1$ gate bound; the $3n-o(n)$ and $3.1n-o(n)$ bounds (Li--Yang) refine it for explicit functions, tracking how many wires each eliminated input removes.
\end{enumerate}
