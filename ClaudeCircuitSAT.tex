\paragraph{Status.} \textbf{P versus NP remains OPEN.} No unconditional superpolynomial lower bound
for \CSAT{} is claimed. Labels: PROVED, KNOWN, COMPUTATIONAL, CONDITIONAL, CANDIDATE, OPEN, RETIRED.

\section{Goal and model}
\CSAT: given a fan-in-2 circuit $C$ over basis $B_2$ (all 16 binary gates) with $n$ inputs and $m$ gates
($2m$ wires), is there $x\in\{0,1\}^n$ with $C(x)=1$? We seek lower bounds $T(n,m)$ that may be
\emph{parametrized} in $n$, $m$, and structural parameters (width, treewidth, cutwidth, depth).
Two quantities are tracked at every rung: (i) $\mathrm{wires}(f)=2\,C_{B_2}(f)$, the minimum connections
needed to \emph{build} a circuit for a constraint $f$; (ii) the cost of \emph{deciding} satisfiability of
circuits of that shape. Rung $k$ increases the number of inputs from $k-1$ to $k$.

\section{Proved baseline (rungs: description size)}
\begin{prop}[Connectivity, PROVED]\label{p:conn}
A circuit with $n$ inputs computing a function that depends on all $n$ inputs has $\ge n-1$ gates, hence $\ge 2(n-1)$ wires; AND of $n$ literals meets this with equality.
\end{prop}
\begin{proof} Every input has a directed path to the output, so the underlying undirected graph on inputs and gates is connected. Adding gates one at a time in topological order, a gate with two operands joins at most two existing components into one, so it lowers the component count by at most 1. Starting from $n$ components (the inputs), connectivity needs at least $n-1$ gates. \end{proof}

\begin{prop}[Reading lower bound, PROVED]\label{p:read}
Any deterministic algorithm that decides \CSAT{} on instances of $m$ gates, given by random access to the gate list, makes $\Omega(m)$ probes.
\end{prop}
\begin{proof} Take $m/2$ variables $y_1,\dots$; instance family: $C=\bigwedge_k (\ell_{k}\wedge\ell'_k)$ where $\ell_k=y_k$ and $\ell'_k\in\{y_k,\neg y_k\}$ (AND-tree, $m=\Theta$ of the number of literals). It is satisfiable iff no $\ell'_k$ is negated. An adversary answers every probe of a position $\ell'_k$ with $y_k$; unless all $m/2$ such positions are probed, the unprobed one could be $\neg y_k$ and the answer flips. \end{proof}
So $T(n,m)=\Omega(m)$ unconditionally (tight for this family). This is the only unconditional bound in terms of $m$ currently proved here; it is independent of $n$.

\section{Counting regime: where hardness can live}
\begin{prop}[Shannon, KNOWN; brute-force crossover, PROVED]\label{p:count}
Most $n$-input functions need $\Omega(2^n/n)$ gates. Exhaustive search decides \CSAT{} in $O(2^n m)$. Hence if $m\ge 2^n/n$, the truth-table/brute-force algorithm runs in $\mathrm{poly}(m)$; any superpolynomial-in-$m$ lower bound must live in the regime $m\le 2^{n}/n$, i.e.\ $n\ge \log m+\log\log m$. 
\end{prop}
This fixes the two natural parametrizations: $T(n,m)\le \min(2^n m,\ \cdot)$, and the open question is whether $2^{\delta n}$ for fixed $\delta>0$ is forced when $m=\mathrm{poly}(n)$.

\section{Parametrized upper bounds and what hardness must beat}
\begin{itemize}
\item $O(2^n m)$ brute force (n-parametrized). $O(2^{w} m)$ for circuits of cutwidth/pathwidth $w$ by dynamic programming over a layout (KNOWN technique; each rung of the ladder with width $\le w$ is easy).
\item CONDITIONAL: under SETH, no $O((2-\varepsilon)^n\mathrm{poly}(m))$ algorithm for \CSAT; since pathwidth of the formula graph is $\le n$, also no $(2-\varepsilon)^{\mathrm{pw}}\mathrm{poly}$. Hence a lower bound of the form $2^{\Omega(w)}$ in the width parameter is as strong as SETH-type conjectures and unconditional proof is OPEN.
\end{itemize}

\section{Known unconditional bounds this must not contradict}
\begin{itemize}
\item KNOWN: explicit functions need $\ge 3.1n-o(n)$ gates over $B_2$ (Li--Yang, STOC 2022; improving Blum's $3n-o(n)$) and no superlinear bound is known for any explicit function in NP; so no wire-counting argument over $B_2$ can currently give $\omega(n)$ \emph{size} bounds. This is the central barrier for the ``minimum connections'' approach.
\item KNOWN: SAT time--space: $n^{2\cos(\pi/7)}\approx n^{1.80}$ for $n^{o(1)}$ space (Williams); alternation trading cannot go further (Buss--Williams) -- see repository ledger row L13.
\item Barriers: relativization/algebrization for diagonal arguments; natural proofs for ``large, constructive'' circuit properties (conditional on one-way functions); a wire-counting proof of a superpolynomial \emph{size} bound would have to be non-natural.
\end{itemize}
