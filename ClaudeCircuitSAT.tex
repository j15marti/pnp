\section*{Status}
\textbf{P versus NP remains OPEN.} Nothing here is an unconditional super-polynomial lower bound for Circuit-SAT. Labels: PROVED/EXACT, KNOWN THEOREM, COMPUTATIONAL, CONDITIONAL, CANDIDATE, OPEN, FALSIFIED/RETIRED.

\section{Setup}
A circuit $C$ over $n$ inputs has $s$ fan-in-2 gates and $w=2s$ internal wires (plus output). Circuit-SAT: given $C$ (as a read-only string), decide $\exists x\; C(x)=1$. A parameterized bound has the form $T(n,s,\dots)$; the interesting regimes are $s=\mathrm{poly}(n)$ vs.\ $T=2^{\Omega(n)}$ (ETH/SETH-type, CONDITIONAL) and unconditional bounds in restricted models.

\section{Ladder of rungs}
Rung $n$: the hardest instances are \emph{point circuits} (unique satisfying assignment $a\in\{0,1\}^n$), and the question is the minimal wiring needed to build them and what any solver must do with it.

\begin{lem}[Connectivity, PROVED/EXACT]\label{lem:conn}
If $f$ depends on all $n$ inputs, every fan-in-2 circuit for $f$ has at least $n-1$ gates and at least $2(n-1)$ gate-input wires. 
\end{lem}
\emph{Proof.} The DAG restricted to ancestors of the output reaches all $n$ inputs; a fan-in-2 tree/DAG with $n$ distinct leaves-reachable has $\ge n-1$ internal nodes (merging $n$ components into one needs $\ge n-1$ binary joins). \qed

\begin{prop}[Point-function upper bound, PROVED/EXACT]
Every point function $[x=a]$ has a circuit with $n-1$ gates (AND tree over literals; negations free in the full binary basis). So Lemma~\ref{lem:conn} is tight for the full binary basis, and each rung's wire count is exactly $2(n-1)$.
\end{prop}

\begin{thm}[Black-box rung bound, PROVED/EXACT]\label{thm:bb}
Any deterministic algorithm that only evaluates $C$ on chosen inputs needs $2^n$ evaluations in the worst case on point circuits (adversary answers 0 until one candidate remains).
\end{thm}
This is not a lower bound for Circuit-SAT proper: a white-box solver reads the wiring. Bridging black-box to white-box is the OPEN core (relativization barrier: Theorem~\ref{thm:bb} relativizes).

\begin{prop}[Read bound, PROVED/EXACT]
Any correct solver must read $\Omega(s)$ bits of the description in the worst case when $s$ gates may affect satisfiability (adversary flips an unread gate). Hence $T=\Omega(s)$; this is the only unconditional parameterized bound in this note.
\end{prop}

\section{Parameterized targets (CANDIDATE / OPEN)}
\begin{itemize}
\item Fan-out/width-parameterized: SAT of circuits of treewidth $k$ is solvable in $2^{O(k)}\mathrm{poly}$; lower bound $2^{\Omega(k)}$ under SETH (KNOWN THEOREM direction for CNF-treewidth; transfer to circuit-SAT is CONDITIONAL).
\item Rung-wise wire budget: define $W(n)$ = min wires to encode a hard family; need a theorem that a solver must ``touch'' superlinear many wire-pairs. OPEN.
\item Restricted models (monotone, bounded-depth, read-once): candidate for unconditional rungs; to be examined next.
\end{itemize}
