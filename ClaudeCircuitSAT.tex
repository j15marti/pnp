\section*{Status}
\textbf{P versus NP remains OPEN.} Nothing here is an unconditional super-polynomial lower bound for Circuit-SAT. Labels: PROVED/EXACT, KNOWN THEOREM, COMPUTATIONAL, CANDIDATE, OPEN.

\section{Setup}
Circuit: DAG, fan-in-2 gates over the full basis $B_2$, $n$ inputs, $m$ gates, $w=2m$ wires (gate-input edges; fan-out counted per edge). Circuit-SAT$(n,m)$: given $C$, decide $\exists x\; C(x)=1$. Parameters: $n$, $m$ (or $w$), fan-out, treewidth $\mathrm{tw}$.

\section{Proved step lemmas}
\begin{lem}[Relevance, PROVED/EXACT]\label{lem:rel} If $C$ depends on all $n$ inputs then $C$ has $\ge n-1$ gates and $\ge 2(n-1)$ wires. (Connectivity: the output must reach $n$ sources through fan-in-2 gates.)\end{lem}
\begin{lem}[Point-function tightness, PROVED/EXACT]\label{lem:pt} For $a\in\{0,1\}^n$ the function $[x=a]$ has $\mathrm C=n-1$ (conjunction of literals; negations absorbed in $B_2$). Matches $n=2,3$ data: $1,2$.\end{lem}
\begin{thm}[Black-box floor, KNOWN/PROVED]\label{thm:bb} Any algorithm that accesses $C$ only through evaluation needs $2^n$ evaluations worst case to decide SAT for the family $\{[x=a]\}\cup\{0\}$ (adversary answers 0 until one point is left). Hence with $m=n-1$ gates, black-box cost is $2^{\,m+1}$.\end{thm}
\textbf{Consequence / parametrization.} A lower bound for white-box algorithms must exploit the wires. Trivial white-box bound: $\Omega(w)$ (read the circuit). Gap: $\Omega(w)$ vs.\ best known $2^{n}\cdot\mathrm{poly}$ upper bound is the whole problem. Ladder question per step $(n,m)$: the least $T(n,m)$ such that some algorithm decides all circuits with $\le m$ gates; computationally $T(n,m)$ is bounded by the number of distinct functions with $\mathrm C\le m$ (table above) only for a lookup-table algorithm, which is not a lower bound.

\section{Parametrized targets (CANDIDATE)}
\begin{enumerate}
\item $\mathrm{tw}$: SAT is $2^{O(\mathrm{tw})}m$; tight under SETH (KNOWN, conditional). Unconditional analogue: OPEN.
\item $m-n$ (excess gates over the relevance floor, Lemma~\ref{lem:rel}): CANDIDATE conjecture $T(n,m)\ge 2^{\,n-O(1)}$ for $m\le n+c$ is \emph{false-looking} because circuits with $m<n+c$ are near-trees and solvable by DP in $\mathrm{poly}$; so the hardness must come from fan-out. Record: parameter ``number of fan-out-$\ge2$ nodes'' $k$ gives FPT-style $2^{O(k)}\mathrm{poly}$ (decide by branching on those nodes; PROVED/EXACT). Lower bound in $k$: OPEN.
\item Gate elimination (KNOWN THEOREM, Li--Yang $\approx3.1n$ circuit lower bound for explicit functions) bounds \emph{size of a function}, not time to solve SAT; transfer needs a new bridge.
\end{enumerate}
