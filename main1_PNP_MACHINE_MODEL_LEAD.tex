\section{TMR-1/093: exact full-rank path transfers, inverse and singular spectrum}
\noindent\textbf{Status: PROVED finite-template matrix identity and exact
arbitrary-length theorem; finite COMPUTATIONAL verification; independent
review, full LaTeX compilation and kernel reconstruction OPEN. P versus NP OPEN.}

Let $S$ be the $5\times5$ cyclic permutation matrix with
$S_{a,b}=1$ exactly when $b-a\equiv1\pmod5$.
The beat adjacency matrix is
$R=S+S^3=S(I+S^2)$, and a directed path of $\ell$ arcs
traversed with $f$ forward and $g$ backward edges
($f+g=\ell$) has two-terminal extension matrix
$M=R^f(R^{\mathsf T})^g$.

\begin{theorem}[C1-T5-FULLRANK-PATH-TRANSFER093]
For every finite oriented path, regardless of its arc ordering,
\[
 \boxed{\det M=2^\ell,\qquad
       R^{-1}=\frac{-I-S+S^2+S^3+S^4}{2}.}
\]
Each row and column sum of $M$ is $2^\ell$.
For $\varphi=(1+\sqrt5)/2$, the singular values of $M$ are
\[
 2^\ell,\ \varphi^\ell,\ \varphi^\ell,\
 \varphi^{-\ell},\ \varphi^{-\ell},
 \qquad
 \kappa_2(M)=(1+\sqrt5)^\ell.
\]
Every entry of $M$ is strictly positive whenever $\ell\ge4$,
although $M$ always has full rank five.
Four is the smallest length guaranteeing this for all orientations.
No finite oriented path implements an exactly independent two-terminal
relation $c\mathbf1\mathbf1^{\mathsf T}$ with $c>0$.
\end{theorem}
\begin{proof}
Every traversal-forward edge contributes $R$ and each backward edge
contributes $R^{\mathsf T}$; both are polynomials in $S$ and commute.
The permutation $S$ and its square are five-cycles, so
$\det S=1$ and $\det(I+S^2)=2$.
Thus $\det R=\det R^{\mathsf T}=2$, giving the determinant formula.
The polynomial identity
$(S+S^3)(-I-S+S^2+S^3+S^4)=2I$ follows from $S^5=I$,
proving the inverse. Each factor has row and column sum two.

Finite Fourier diagonalization yields the eigenvalues
$\zeta^k+\zeta^{3k}$ of $R$, where $\zeta$ is a primitive
fifth root of unity. Their magnitudes are $2$ once,
$\varphi$ twice and $\varphi^{-1}$ twice. Reversing arcs conjugates
the Fourier eigenvalues, hence all path singular values depend
only on $\ell$, proving the condition number formula.
For length four, the five first-row vectors, indexed by
$f=0,1,2,3,4$, are respectively
\[
(4,1,6,1,4),\ (1,6,1,4,4),\ (6,1,4,4,1),\
(1,4,4,1,6),\ (4,4,1,6,1).
\]
They are all positive. Multiplication by either beat matrix
preserves strict positivity, so this holds for all longer paths.
For $\ell=3$, the all-forward first row is $(3,0,3,1,1)$,
which proves sharpness. A matrix $c\mathbf1\mathbf1^{\mathsf T}$
has rank one, whereas $\det M\ne0$.
\end{proof}

\noindent\textbf{Machine and obstruction audit.}

has at most $\ell+1$ bits. Sequential exact multiplication of
fixed $5\times5$ integer matrices and of inverse numerators

\section{TMR-1/092: exact Fourier-flow duality and polynomial-space cycle-rank counting}
\noindent\textbf{Status: PROVED explicit finite Fourier identity and exact parameterized
algorithms; finite COMPUTATIONAL verification; independent review and kernel
replay OPEN. P versus NP OPEN.}

Let $G=(V,E)$ be an arbitrary authenticated finite directed constraint graph
with $n=|V|$, $m=|E|$, and $c$ weak components (isolates included).
Loops and opposed arcs are allowed. Define $\rho=m-n+c$, the directed
incidence matrix $\partial$ over $\mathbb F_5$ (outflow minus inflow),
$\zeta=\exp(2\pi i/5)$, and
$\lambda_k=\zeta^k+\zeta^{3k}$ for $k\in\mathbb F_5$.

\begin{theorem}[C1-T5-FOURIER-FLOW-DUAL092]
For the fixed beat relation $R(a,b)=1$ exactly when
$b-a\bmod 5\in\{1,3\}$, the exact count on every such $G$ is
\[
\boxed{\#\operatorname{Hom}(G,T_5)
 =5^{n-m}\sum_{f\in\ker_{\mathbb F_5}\partial}
       \prod_{e\in E}\lambda_{f_e}.}
\]
The circulation kernel has dimension $\rho$, including when $G$ has
self-loops. The expression is always a nonnegative integer and may be
evaluated with exact rational-integer arithmetic in the cyclotomic ring
$\mathbb Z[\zeta]$, using
$\zeta^4=-1-\zeta-\zeta^2-\zeta^3$.
\end{theorem}
\begin{proof}
Finite Fourier inversion gives
\[
R(d)=\frac{1}{5}\sum_{k\in\mathbb F_5}
       (\zeta^k+\zeta^{3k})\zeta^{-kd}.
\]
Expand the product of these formulas over all arcs and interchange the
finite sums. For a fixed vector $f$, summation over $x_v\in\mathbb F_5$
is five if its signed exponent vanishes and zero otherwise. All exponents
vanish precisely when $\partial f=0$. This gives $5^n$ times the
surviving flow products, times the prefactor $5^{-m}$. A spanning forest
provides an $(n-c)$-rank minor of the incidence matrix. Its rows also
have one dependence per component, hence
$\dim\ker\partial=m-(n-c)=\rho$. Loop columns are zero and cause no
exception. Equality with the integer count proves exact cyclotomic
coefficient cancellation and necessary divisibility.
\end{proof}

\begin{theorem}[C1-T5-DUAL-ENUMERATION-UPPER092]
There is a uniform deterministic sequential binary-machine algorithm
for exact T5 counting, decision and witness construction with
\[
T(L)=\operatorname{poly}(L)\min\{5^{\rho},\,2^{n-c}\},
\qquad
S_{\mathrm{peak}}(L)=\operatorname{poly}(L),
\]
where $L$ is the authenticated source bit length. In particular,
T5 counting is fixed-parameter tractable in cycle rank $\rho$.
These expressions are upper bounds, not resource lower bounds.
\end{theorem}
\begin{proof}
Select a spanning forest of the underlying undirected multigraph,
retaining distinct directed arc identities. Assign arbitrary
$\mathbb F_5$ values to the $\rho$ cotree arcs. The conservation
equations uniquely determine all tree-edge flows by eliminating leaves
bottom-up. Thus the Fourier expression has exactly $5^\rho$ summands.
Each is multiplied in the degree-four integer basis for $\mathbb Z[\zeta]$,
so all intermediate coefficients have polynomial bit length: a product
of $m$ sums of two roots has coefficient magnitudes at most
$O(8^m)$, and summing $5^\rho$ such products adds $O(\rho)$ bits.
Exact division by powers of five finishes the computation. The graph,
flow vector, four coefficients, and integer work registers use
polynomial live space.

Alternatively, assign each of the $n-c$ tree arcs one of the two
permitted modular differences $1$ or $3$. Fix one root color per
component to zero, propagate its colors uniquely along the forest,
and test all remaining arcs. Exactly $5^c$ homomorphisms correspond
to each accepted increment choice, by independent componentwise
translations in $\mathbb F_5$. This evaluates the exact count with
$2^{n-c}$ choices and polynomial work per choice. Choosing the smaller
enumeration proves the bound. Once the count is positive, the primal
search or a standard pinned-variable version of the same enumeration
constructs a witness without increasing the exponential parameter.
\end{proof}

\section{TMR-1/090: parsimonious bounded-width Boolean compiler and bit-linear decision normalization}
\noindent\textbf{Status: PROVED explicit finite-template compiler and encoded-bit bound;
finite COMPUTATIONAL verification; independent review OPEN. P versus NP OPEN.}

For any directed source graph $G$ on $n$ vertices and $m$ arcs, including
self-loops and opposed arcs, create five one-hot bits $y_{v,a}$
and two auxiliary bits $p_v,q_v$ per source vertex.
The ten clauses $\neg y_{v,a}\vee\neg y_{v,b}$ for $a<b$
enforce at most one color.
Encode $p_v\leftrightarrow(y_{v,0}\vee y_{v,1})$ and
$q_v\leftrightarrow(y_{v,2}\vee y_{v,3})$ with three
clauses each, of widths $(3,2,2)$, and require
$p_v\vee q_v\vee y_{v,4}$.
For each directed arc $u\to v$ and $a\in\mathbb Z_5$ add
\[
\neg y_{u,a}\ \vee\ y_{v,a+1}\ \vee\ y_{v,a+3},
\qquad \text{all color subscripts modulo }5.
\]

\begin{theorem}[C1-T5-3CNF-PARSIMONIOUS-ISOLATE-OPT090]
The resulting Boolean formula $F(G)$ has exactly $7n$ Boolean
variables, $17n+5m$ clauses of width at most three,
including $14n$ binary and $3n+5m$ ternary clauses,
and $37n+15m$ literal occurrences. Its satisfying Boolean
assignments are in \emph{bijection} with the $T_5$ homomorphisms:
\[
\boxed{\#\mathrm{SAT}(F(G))=\#\operatorname{Hom}(G,T_5).}
\]
For canonical unary-count, fixed-width endpoint source encoding
of length $L_G=n+m+2+2m\,w(n)$, where
$w(t)=\max(1,\lceil\log_2\max(2,t)\rceil)$,
a canonical unary-count and binary-index target encoding has
exact length
\[
L_F=58n+15m+2+(37n+15m)(w(7n)+1).
\]
After removing the $i$ isolated source vertices and relabeling
the remaining $n'=n-i$ vertices, the target formula $F(G')$
satisfies the exact count relation
$\#\operatorname{Hom}(G,T_5)=5^i\#\mathrm{SAT}(F(G'))$
and the \emph{decision} output has
\[
\boxed{L_{F(G')}\le 200L_G}.
\]
The full compiler is parsimonious; the isolate-normalized
constant-factor compiler is not parsimonious unless $i=0$.
\end{theorem}
\begin{proof}
The ten binary exclusions and one final ternary inclusion
ensure exactly one of the five color bits is true. The two
auxiliary variables are uniquely determined by their
definitional equivalences. For a source arc $u\to v$,
only the clause indexed by the unique true color $a=x_u$
has a false first literal; it forces the unique color
$x_v$ into $\{a+1,a+3\}$, exactly the beat relation.
This also rejects every self-loop. Consequently the witness
map is bijective.

The ten exclusions and four binary gate implications
give $14n$ binary clauses; the two gate implications
in the reverse direction, final inclusion, and five
per arc give $3n+5m$ ternary clauses.
The canonical binary-index encoding uses
$7n$ unary variable-header ones, $17n+5m$
unary clause-header ones, two header terminators,
two clause-width bits per clause, and
$37n+15m$ signed literals of width $w(7n)+1$;
substitution proves the displayed exact length.

Each isolated vertex contributes five unconstrained colors,
so deleting $i$ such vertices introduces the factor $5^i$.
If $m\ge1$, every remaining vertex belongs to some arc,
thus $n'\le2m$. Since $w(7n')\le w(n)+3$,
\[
\begin{aligned}
L_{F(G')}&\le 58(2m)+15m+2+
 (37(2m)+15m)(w(n)+4)\\
 &=487m+89m\,w(n)+2\\
 &\le489m+89m\,w(n)
 <200(m+2m\,w(n))\le200L_G.
\end{aligned}
\]
For $m=0$, emit the two-bit empty CNF. Valid input
parsing, source validation, vertex relabeling, clause
emission and witness reconstruction can be performed
with conservative simultaneous sequential-bit-machine
bounds $O((L_G+1)^4)$ time and
$O((L_G+1)^2)$ peak writable space. No unrestricted
time lower bound follows.
\end{proof}

\noindent\textbf{Counting classification qualification.}
Dyer and Greenhill (2000) prove $\#\mathrm{Hom}(-,C_5)$
is $\#P$-complete under their counting-reduction convention;
the previously established parsimonious five-arc

under at least the same convention. This theorem supplies
a parsimonious reverse compiler to width-at-most-three CNF.
Do not conflate this with exactly-three-literals-per-clause
or infer parsimonious $\#3SAT$-completeness without a
separate reduction. The full local executable checked

\section{TMR-1/089: strongly connected Eulerian bipartite four-cycle hardness}
\noindent\textbf{Status: PROVED explicit finite-template reduction,
finite COMPUTATIONAL checks; independent review OPEN; P versus NP OPEN.}

Fix $R(x,y)\iff y-x\bmod5\in\{1,3\}$. For an undirected simple
source graph $H$ with $n$ vertices and $m$ edges, replace each
$\{u,v\}$ ($u<v$) with two vertex-disjoint directed two-edge paths
$u\to a_{uv}\to v$ and $v\to b_{uv}\to u$.

\begin{theorem}[C1-T5-SYMMETRIC-C4-EULERIAN-HARDNESS089]
The resulting digraph $D(H)$ has $N=n+2m$ vertices and $M=4m$
arcs. It is loopless, bipartite, balanced
($\deg^+(z)=\deg^-(z)$ for every vertex), has no antiparallel pair,
and all arcs partition into directed four-cycles. If $H$ is
connected and $m\ge1$, then $D(H)$ is strongly connected.
Moreover
\[
\boxed{\#\operatorname{Hom}(D(H),T_5)
=2^m\,\#\operatorname{Hom}(H,C_5).}
\]
Consequently, $T_5$-homomorphism remains NP-complete when its input
is restricted to strongly connected, bipartite, balanced directed
simple graphs whose arcs partition into directed four-cycles.
\end{theorem}
\begin{proof}
For prescribed endpoint difference $d$ the number $f(d)$ of
assignments to the internal vertex of a forward two-edge path
is the coefficient of residue $d$ in $(z+z^3)^2\bmod(z^5-1)$:
\[
(f(0),f(1),f(2),f(3),f(4))=(0,1,1,0,2).
\]
Oppositely directed paths give
\[
f(d)f(-d)=(0,2,0,0,2)_d
=2\,\mathbf 1_{\{d\in\{1,4\}\}},
\]
exactly twice the adjacency indicator of the undirected cycle
$C_5$. Each auxiliary interior is disjoint from every other
edge's interior, so all extension weights multiply, proving
the exact counting identity. Each source edge's four arcs form
a directed four-cycle; it can be traversed from either endpoint
to the other, proving strong connectivity when the skeleton
is connected. Each original vertex has one incoming and one
outgoing arc per incident source edge; each internal vertex
has one of each. Bipartiteness follows by putting originals
and interiors in separate classes.

To obtain NP-hardness even with strong connectivity from
unrestricted $C_5$-coloring, for nonempty $H$ identify one
root from every connected component into one common root;
call the resulting connected simple graph $H^\wedge$.
If $c$ is the source component count, then
\[
n^\wedge=n-c+1,\quad m^\wedge=m,\qquad
\#\operatorname{Hom}(H^\wedge,C_5)
=5^{1-c}\#\operatorname{Hom}(H,C_5).
\]
The counting formula follows because the five cyclic
translations of $C_5$ act freely on the color of a
specified component root. Thus original coloring is
equivalent to wedge coloring. For $m\ge1$ the target
$D(H^\wedge)$ is strongly connected. For $m=0$ output a
single-vertex YES digraph, which is strongly connected
under length-zero reachability. Invalid canonical source
strings map to the constant NO instance $D(K_6)$,
which also belongs to the required restricted class.
The source $K_6$ is not $C_5$-colorable, since the
target $C_5$ is loopless on only five vertices.
Hell--Ne\v{s}et\v{r}il's fixed-target theorem
makes $C_5$-colorability NP-complete, completing the
many-one decision reduction.
\end{proof}

The authenticated undirected source encoding uses
$1^n0\,1^m0$ and a sorted, duplicate-free list of
low-bit-first pairs $u<v<n$, with
$w(n)=\max(1,\lceil\log_2\max(2,n)\rceil)$ and
$L=n+m+2+2mw(n)$. The direct four-cycle target has
\[
L_D=n+6m+2+8m\,w(n+2m)\le8L
\quad\text{for simple source graphs }n\ge2.
\]
The connected wedge has no more vertices or edges
than its original source, so the same bound applies
to the full construction when $m\ge1$.
The two exceptional outputs are constant size.
Canonical source and target parsing, graph traversal,
sorting, serialization, witnesses, malformed-input
rejection, and peak writable storage are all charged
on the same sequential bit-machine realization;
conservative simultaneous upper bounds are
$O((L+1)^4)$ bit-time and $O((L+1)^2)$ writable bits.
The reduction has exact factor $2^m$ (and
$5^{1-c}$ after wedging), hence is \emph{not}
parsimonious, although it preserves SAT/UNSAT
and allows witness recovery.

\section{TMR-1/088: exact path transfer and certified spectral counting for long subdivisions}
\noindent\textbf{Status: PROVED restricted constructive algebraic theorem;
finite COMPUTATIONAL checks; independent review OPEN; P versus NP OPEN.}

\begin{theorem}[C1-T5-LONG-PATH-SPECTRAL-DECORRELATION088]
Fix the regular tournament on $\mathbb{Z}_5$ with
$R(a,b)\iff b-a\pmod 5\in\{1,3\}$.  For a traversal of an
oriented path of length $\ell\ge1$ with $p$ forward arcs and
$q=\ell-p$ backward arcs in any order, let $T_{p,q}(a,b)$
denote its number of internal extensions for endpoint colors $a,b$.
Then
\[
T_{p,q}(a,b)=\sum_{k=0}^{\ell}\binom{\ell}{k}
 {\bf1}\{p-3q+2k\equiv b-a\pmod5\},
\qquad
\left|\frac{5T_{p,q}(a,b)}{2^\ell}-1\right|
 \le4(13/16)^\ell.
\]
Let $H$ be any undirected multigraph with $n$ vertices including
isolates and $m$ edges. Replace each source edge $e$ by an internally
vertex-disjoint oriented path of length $\ell_e$, with arbitrary
orientations. Put $S=\sum_e\ell_e$,
$Q_0=5^{n-m}2^S$, and
$\delta_e=4(13/16)^{\ell_e}$. The exact number $Z$ of
$T_5$-homomorphisms of the resulting directed graph obeys
\[
Q_0\prod_e\max\{0,1-\delta_e\}\ \le Z\ \le\
Q_0\prod_e(1+\delta_e).
\]
When $m=0$ equality $Z=Q_0=5^n$ holds.
When $m\ge1$ and $0<\epsilon\le1$,
if $8m(13/16)^{\ell_{\min}}\le\epsilon$ then
\[
(1-\epsilon)Q_0\le Z\le(1+\epsilon)Q_0.
\]
The approximator is deterministic, with exact rational output and
exact integer certification of the length condition.
\end{theorem}
\begin{proof}
For a path edge followed in its orientation the increment
polynomial is $x+x^3=x(1+x^2)$; against its orientation it is
$x^{-1}+x^{-3}=x^{-3}(1+x^2)$.
Modulo $x^5-1$, the entire product is
$x^{p-3q}(1+x^2)^\ell$, so its residue coefficient gives
the displayed exact binomial transfer, depending only on $p,q$
and not on their order.
For a primitive fifth root $\omega$, a root-of-unity filter gives
the transfer coefficient as one-fifth the sum
\[
\sum_{j=0}^4\omega^{j(p-3q-(b-a))}
 (1+\omega^{2j})^\ell.
\]
The $j=0$ term equals $2^\ell$.
For $j\ne0$, the magnitude $|1+\omega^{2j}|$ is at most
$\varphi=(1+\sqrt5)/2$, and $\varphi/2\le13/16$.
Taking absolute values yields the uniform spectral inequality.
For each fixed coloring of the skeleton vertices, extension counts
multiply over the internally disjoint paths, and every transfer lies
in the stated multiplicative interval. Summing the same bounds over
all $5^n$ skeleton assignments gives the global rational interval.
If $\delta_e\le\epsilon/(2m)$, the lower factor is at least
$(1-\epsilon/(2m))^m\ge1-\epsilon/2$, and the upper is at most
$\exp(\epsilon/2)\le1+\epsilon$ for $0<\epsilon\le1$.
\end{proof}

$L=N+M+2+2Mw(N)$, where
$w(N)=\max\{1,\lceil\log_2\max\{2,N\}\rceil\}$.

The estimate is output as a numerator-denominator pair

The source, decomposition, epsilon, arithmetic and output admit

\section{TMR-1/087: sharp three-subdivision saturation for the fixed five-symbol CSP}
\noindent\textbf{Status: PROVED exact path theorem; finite COMPUTATIONAL audit; independent review OPEN; P versus NP OPEN.}

\begin{theorem}[C1-T5-THREE-SUBDIVISION-SATURATION087]
For the fixed relation \(R(x,y)\iff y-x\pmod 5\in\{1,3\}\),
let \(H\) be any finite undirected multigraph. Replace each edge by
an internally vertex-disjoint path of length at least three,
orienting the constituent edges in arbitrary directions. The resulting
digraph always has a homomorphism into \(T_5\).
For any oriented path of length three, the exact number of internal
extensions of equal-colored endpoints is \(3\) when its three arcs
are coherently oriented and \(1\) otherwise. For any oriented
path of length at least four, \emph{all} endpoint color pairs have
positive numbers of internal extensions.
\end{theorem}

\begin{proof}
Fix a path of length \(\ell\) and index each edge in path order.
Write \(\sigma_i=+1\) for a forward directed edge and \(-1\) for
a backward one. Every allowed signed edge color difference is

endpoint set of the nonforest quotient edges, and $c_*$ the

 T=O((L+1)^8(1+5^b)),\qquad

\]

\begin{lemma}[J56: charged uniform recovery of a fixed-size junta]

$f:\{0,1\}^{\ell}\to\{0,1\}$ depending on at most $w$ coordinates permits a

 2\ell\sum_{j=0}^{w-1}\binom{\ell-1}{j}+2^w

discovered support by at most $2^w$ extra paid evaluations, then build its
Boolean Shannon expansion. Each call actually computes $f$; there is no
free support-location oracle.
\end{proof}

\section{C1-T5-UNIFIED-SHORTCUT-QUOTIENT /082: exact two- and three-path rigidity}

\noindent\textbf{Status and source authentication.}
Fix $S=\{1,3\}\subset\mathbb Z_5$ and $R(x,y)\iff y-x\in S$.
Retain the authenticated, sorted, low-first input word of length
$L=n+m+2+2mw(n)$, with $w(n)=\max(1,\lceil\log_2\max(2,n)\rceil)$.
Each witness has precisely $3n$ bits with all decoded labels in

and opposing arcs have count zero. This section is an exact
source-to-source preprocessing theorem, not a universal SAT solver.

\begin{lemma}[Shortcut rigidity for the exact $T_5$ relation /082]
Suppose $G$ has a path $x\to a\to b$ and the shortcut $x\to b$.
Every satisfying assignment obeys
$a=x+3,\ b=x+1\pmod5$.
If instead $G$ has the path
$x\to a\to b\to c$ and shortcut $x\to c$, then every satisfying
assignment obeys
$a=x+1,\ b=x+2,\ c=x+3\pmod5$.
The vertices in each motif are distinct.
\end{lemma}
\begin{proof}
Every directed arc has modular increment $1$ or $3$.
For a two-arc path, the sums are $2,4,6$; only $6=1\bmod5$
is in $S$, so both increments equal $3$.
For a three-arc path, the sums are $3,5,7,9$; only $3\bmod5$
is in $S$, so every increment equals $1$.
The shortcut itself is an original source arc, so all equations
are logical consequences of the original CSP.
Length four already admits ten allowed increment sequences and
does not yield analogous uniqueness from path length alone.
\end{proof}

\begin{theorem}[Uniform authenticated shortcut quotient /082]
On an arbitrary valid directed $T_5$ source $G$, enumerate every
two- and three-arc path with the corresponding start-to-end shortcut.
Insert all forced differences from the previous lemma into a
source-derived modular-equation system. If inconsistent, there are
no original satisfying assignments. Otherwise let $k$ be the number
of resulting components, and choose potentials $p_v\in\mathbb Z_5$
so every constrained original label has the unique expression
$x_v=t_{[v]}+p_v\pmod5$ for one free label
$t_{[v]}\in\mathbb Z_5$ per component.

For every original arc $u\to v$, if $[u]=[v]$, require
$p_v-p_u\in S$. If the two classes differ, write
$i<j$, $\delta=t_j-t_i$, and impose
\[
 \delta\in
 \begin{cases}
 S+p_u-p_v,&[u]=i,\\
 p_v-p_u-S,&[u]=j.
 \end{cases}
\]
Intersect these masks on each unordered pair of quotient classes.
An empty mask means count zero. Otherwise the resulting quotient
has at most $n$ variables and each pair-mask cardinality is at most
two. Its satisfying assignments are in explicit bijection with
those of $G$, so
\[
 \boxed{\#\operatorname{Hom}(G,T_5)=
       \#\operatorname{Solutions}(Q(G)).}
\]
There is a uniform sequential bit-machine compiler with
simultaneous conservative
$T=O((L+1)^7)$ time and $S_{\rm peak}=O((L+1)^3)$ writable bits
for the \emph{entire preprocessing and quotient output}.
The quotient may remain NP-complete to solve.
\end{theorem}
\begin{proof}
Scan actual source arcs and ordered tuples; each derived equation
holds in every original solution. A weighted modular component
data structure detects contradiction or gives a unique
potential for each vertex relative to its component's free root.
For each original arc, the displayed mask follows by substituting
$x_v=t_{[v]}+p_v$ in its original beat predicate. Conversely
every quotient solution satisfies every source arc under the
unique lift. Hence the maps are mutually inverse even in
disconnected and zero-solution cases. A sequential realization
can stream $O(n^4)$ motif tests into a polynomial-size potential
structure, then form pair-mask intersections and output them.
It does not have to store all motif tuples. Addresses, input
authentication, contradictions, arithmetic, output and retained
work are charged in the same physical time-space ledger.
\end{proof}

\begin{lemma}[Exact voltage transfer of a triangle-free bundle /085]

\section{Strict three-distinct-variable CNF for the five-symbol beat CSP /091}

$b-a\bmod5\in\{1,3\}$.  A canonical source is a loop-permitted

\begin{lemma}[Unique global Boolean constant /091]
Let $z,s,t$ be three fresh Boolean variables and put one
three-literal clause on the three distinct variables for each of the
seven assignments in $\{0,1\}^3\setminus\{(1,0,0)\}$, falsified only
by its associated assignment. Then this seven-clause CNF has
exactly one satisfying assignment, namely $(z,s,t)=(1,0,0)$.
\end{lemma}
\begin{proof}
A clause with literals $x$ for a forbidden zero coordinate and
$\neg x$ for a forbidden one coordinate is falsified exactly
at that forbidden triple. The conjunction eliminates seven distinct
triples and retains precisely the eighth.
\end{proof}

\begin{theorem}[Parsimonious strict 3-CNF /091]
For every valid directed $T_5$ source $G$, with $n$ vertices and $m$
arcs, there exists a canonical strict $3$-CNF $F^*(G)$ having exactly
\[
 B=7n+3\quad\text{variables},\qquad
 K=17n+5m+7\quad\text{clauses},
\]
where \emph{every clause has exactly three literals on three
distinct Boolean variables}. There is a bijection between the
satisfying Boolean assignments and the homomorphisms $G\to T_5$:
\[
 \boxed{\#\mathrm{SAT}(F^*(G))=\#\mathrm{Hom}(G,T_5).}
\]
This includes self-loops, opposed arcs, disconnected sources, and the
empty source. The exact canonical target bit length is
\[
 \boxed{L_{F^*}=7n+5+
    (17n+5m+7)\bigl(6+3w(7n+3)\bigr).}
\]
\end{theorem}
\begin{proof}
Start from the unique-extension one-hot compiler /090, having five
color bits and two functionally determined gate bits per source vertex,
fourteen two-literal clauses and three three-literal clauses per
vertex, and five three-literal clauses for each directed source arc.
Use the preceding global gadget, and replace every binary clause
$(\ell\lor r)$ by $(\ell\lor r\lor\neg z)$.
The seven gadget clauses force $z=1$ uniquely, so each padded
clause has exactly the previous truth value. All old binary
clauses have two distinct variables, the fresh $z$ differs from both,
and old ternary clauses also have three distinct variables, even
for a loop, because the indices $a,a+1,a+3$ differ modulo five.
Every satisfying source witness extends uniquely to the previous
gate bits and then to $(z,s,t)=(1,0,0)$; conversely any satisfying
Boolean assignment determines exactly one source witness.
The headers $1^B0\,1^K0$, the two-bit clause-width fields, and
three signed $w(B)$-address literals per clause give
$L=B+K+2+K(2+3(w(B)+1))$, yielding the displayed formula.
\end{proof}

\]

L_{F^*(G')}&\le 14m+5+(39m+7)(18+3w)\\
