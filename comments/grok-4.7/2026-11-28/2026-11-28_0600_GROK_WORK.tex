\subsection*{Setting}
Notation is as in \path{comments/grok-4.7/2026-11-28/2026-11-28_0455_GROK_WORK.tex}. $F$ is a CNF with clauses of 1--3 literals. $N=N(F)$ is the normalized native factor system of pinned \texttt{compile()} (\texttt{21b9d53}, blob \texttt{7cb0303}): carriers $y$ (namely $x_v,z_i,w_{i,0},w_{i,1},f$) and B factors ``exactly two of four signed ports''. $N_g:=N\wedge(g=0)$ for a gauge carrier $g$. Axioms are polynomial \emph{equalities}: $P_C=\prod_{\ell\in C}(1-\hat\ell)$ for clauses and the forbidden-word indicators $\delta_a$ for factors. $I_x=(x_i^2-x_i)_i$ is the Boolean ideal. The substitution is $\rho:\mathbb R[y]\to\mathbb R[x]$, $y\mapsto p_y$, where $p_y$ is the multilinear extension of the canonical completion $\sigma_y$. It satisfies $\deg p_y\le3$, $p_f=0$, and $p_y$ depends on $\le3$ source variables.

\begin{definition}[degree conventions; standard]
A \emph{degree-$d$ SOS refutation} of equalities $A$ is an identity in $\mathbb R[x]$
\[\textstyle\sum_{a\in A}q_a\,a+\sum_i t_i(x_i^2-x_i)+\sum_j s_j^2=-1,\]
with every summand of degree $\le d$. A \emph{degree-$d$ SA refutation} replaces $\sum_j s_j^2$ by $\sum_J c_J\,J$, where $c_J\ge0$ and each $J=\prod_{i\in S}x_i\prod_{i\in T}(1-x_i)$ is a junta with $|S|+|T|\le d$. $\SOS(A)$ and $\SA(A)$ denote the minimum degrees ($\infty$ if none).
\end{definition}

\begin{lemma}[two substitution facts; PROVED WITH SCOPE, checked exactly as (A),(J) below]\label{lem:facts}
(a) For every forbidden word $a$ of a B factor of gadget $i$, $\rho(\delta_a)=c_a P_{C_i}+u_a$ with $c_a\in\{0,1\}$ and $u_a\in I_x$, and every term of $u_a$ has degree $\le3s_a$, where $s_a\le4$ is the factor's carrier count. (b) $\rho(y^2-y)=p_y^2-p_y\in I_x$ with a representation of degree $\le6$. Moreover $p_y$ and $1-p_y$ are each congruent mod $I_x$, in degree $\le3$, to a $0/1$ sum of point juntas $\delta_b(x)$ on $\le3$ variables.
\end{lemma}
\begin{proof}
(a) is the 04:55 identity $\ml(\rho(\delta_a))=c_aP_{C_i}$. That identity holds over $\mathbb Z$ and was verified as a function on the $\le3$ clause variables. Since $\ml(q)-q$ is generated by $I_x$ within degree $\deg q\le 3s_a$, (a) follows. (b) $\sigma_y$ is $0/1$-valued, so $p_y^2-p_y$ vanishes on the cube of its $\le3$ variables. Its multilinearization is therefore $0$, and the reduction uses multiples of $x_i^2-x_i$ of degree $\le6$. Any $0/1$ function $h$ on $k\le3$ variables equals $\sum_{b:h(b)=1}\delta_b$ as a multilinear polynomial.
\end{proof}

\begin{theorem}[SA/SOS degree transfer through the pinned compiler; PROVED WITH SCOPE, author proof, no independent review]\label{thm:sos}
For every CNF $F$ with clauses of 1--3 literals and every gauge carrier $g$:
\[\SOS(F)\le 3\,\SOS(N(F))\le 3\,\SOS(N_g(F))+6,\qquad \SA(F)\le 3\,\SA(N(F))\le3\,\SA(N_g(F))+3.\]
\end{theorem}
\begin{proof}
\emph{Substitution (SOS).} Apply the ring homomorphism $\rho$ to a degree-$d$ SOS refutation of $N$:
$\sum_a\rho(q_a)\rho(\delta_a)+\sum_y\rho(t_y)\rho(y^2-y)+\sum_j\rho(s_j)^2=-1$. By Lemma~\ref{lem:facts}(a), $\rho(q_a)\rho(\delta_a)=c_a\rho(q_a)P_{C_i}+\rho(q_a)u_a$. Here $\deg q_a\le d-s_a$, so $\deg(\rho(q_a)P_{C_i})\le3(d-s_a)+3\le3d$, and $\rho(q_a)u_a$ is a Boolean-ideal combination of degree $\le3(d-s_a)+3s_a=3d$. If $s_a=0$, the factor is a constant and $c_aP_{C_i}$ is trivial. Next, $\rho(t_y)(p_y^2-p_y)$ is an ideal combination of degree $\le3(d-2)+6=3d$. Finally $\deg\rho(s_j)^2\le3d$. This is a degree-$3d$ SOS refutation of $F$.

\emph{Substitution (SA).} The axiom and ideal parts are as above. For a junta $J$ of degree $\le d$, $\rho(J)$ is a product of $\le d$ factors $p_y$ or $1-p_y$. By Lemma~\ref{lem:facts}(b), each factor is congruent in degree $\le3$ to a nonnegative sum of juntas. The product of juntas is a junta: repeated literals reduce by $x^2\equiv x$, and $x(1-x)\in I_x$ kills a term. So $\rho(J)\equiv\sum_K e_K K$ with integers $e_K\ge0$, juntas $K$ of degree $\le3d$, and all reduction terms of degree $\le3d$. Nonnegativity of the $c_J$ is preserved.

\emph{Gauge (SOS).} Let $(*)$ be a degree-$d$ refutation of $N_g$, with the extra term $q_g\,g$. Multiply $(*)$ by $(1-g)^2$. This uses $g(1-g)^2=(g-1)(g^2-g)\in I$ and $(1-g)^2=(1-g)+(g^2-g)$, and turns each $s_j^2$ into $((1-g)s_j)^2$. The result, of degree $\le d+2$, is
$\Pi_1:\ \sum_a(1-g)^2q_a\delta_a+(\text{ideal})+\sum_j((1-g)s_j)^2=-(1-g)$.
Next, $\tau:y\mapsto1-y$ maps every forbidden set of $N$ onto itself. Flipping a carrier flips all of its ports, so the weight $w$ goes to $4-w$; this is check (T). Also $\tau(g)=1-g$ and $\tau(y^2-y)=y^2-y$. Applying $\tau$ to $(*)$ and multiplying by $g^2$, using $(1-g)g^2=-g(g^2-g)$ and $g^2=g+(g^2-g)$, gives
$\Pi_2:\ \sum_a g^2\tau(q_a)\delta_{\bar a}+(\text{ideal})+\sum_j(g\,\tau(s_j))^2=-g$.
$\Pi_1+\Pi_2$ is a degree-$(d+2)$ SOS refutation of $N$. So $\SOS(N)\le\SOS(N_g)+2$.

\emph{Gauge (SA).} Use the multipliers $(1-g)$ and $g$ in place of their squares. These are degree-1 juntas, and a junta times a literal is a junta, so $\SA(N)\le\SA(N_g)+1$.
\end{proof}

\begin{corollary}[semi-algebraic lanes stall on the pinned-compiler random family; PROVED WITH SCOPE relative to KNOWN THEOREMS]\label{cor:sos}
Fix $\Delta\ge5.2$, and let $F$ be a uniformly random 3-CNF with $n$ variables and $\Delta n$ clauses. With probability $1-o(1)$:
(1) $F$ and $N(F)$ are UNSAT (first moment: $2^n(7/8)^{\Delta n}\to0$).
(2) $\SOS(F)\ge\varepsilon n$ for some $\varepsilon=\varepsilon(\Delta)>0$. This is a KNOWN THEOREM: Grigoriev, Theoret.\ Comput.\ Sci.\ 259 (2001), for random 3-XOR, and Schoenebeck, FOCS 2008, for random 3-SAT. The lower bound is witnessed by a degree-$\varepsilon n$ pseudo-expectation satisfying the clause equalities. Weak duality, i.e.\ applying $\tilde E\circ\ml$ to the identity, excludes refutations in the convention above.
(3) $\SA(F)\ge\varepsilon n/2$, since every junta $J$ satisfies $J\equiv J^2$ mod $I$ in degree $2\deg J$, so $\SOS\le2\,\SA$.
(4) For every gauge $g$: $\SOS(N_g(F))\ge\varepsilon n/3-2$ and $\SA(N_g(F))\ge\varepsilon n/6-1$. In particular, no level-$k$ Sherali--Adams LP and no degree-$k$ Lasserre/SOS SDP refutes $N_g(F)$ for $k=o(n)$, which includes every $k=O(\log N_{\rm bits})$.
\end{corollary}

\begin{remark}[what this does and does not cover; scope]
\emph{Covered:} any lane whose refutation, if it succeeds, converts to a bounded-degree SA or SOS refutation of $N_g$. By LP/SDP duality this includes the level-$k$ Sherali--Adams LP on carriers (local factor tables on $\le k$ carriers plus arbitrary linear reasoning) and degree-$k$ Lasserre. Together with the 04:55 PC result, PC over $\mathbb R$ is also covered (it is simulated by SOS with degree doubling; Berkholz, STACS 2018), but that adds nothing new. Degree-1 linear reasoning over the native linear constraints is vacuous on \emph{every} instance (the all-$\tfrac12$ point; 04:55 remark).
\emph{Not covered:} Cutting Planes, i.e.\ integer rounding (Chv\'atal--Gomory). A T1 Hall, matching or flow lane that uses integrality \emph{globally}, rather than through bounded-width exports, is covered only under the hypothesis HALL-SIM: ``the lane's terminal yields a degree-$O(k)$ SA/SOS refutation''. HALL-SIM is \textbf{OPEN} and is not proved here. Also not covered: extension variables, and DAG-like branching beyond bounded depth. The constants $3$ and $6$ are not optimized. As at 04:55, these are proof-system lower bounds, \textbf{not} algorithmic lower bounds and \textbf{not} statements about P vs NP. At the pinned corpus sizes ($n=24,40$) the bound is uninformative.
\end{remark}
