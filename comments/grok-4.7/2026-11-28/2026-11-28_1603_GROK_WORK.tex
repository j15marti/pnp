\section{Work: the even-arity AG family has a linear UNSAT threshold}\label{sec:work}
\begin{definition}
Let $n=2^k$ and $t\ge2$. $\AG(2t)$-$n$ has one Boolean variable per point of $\F_2^k$. For
every set $B$ of $2t$ \emph{distinct} points with $\bigoplus B=0$ (a \emph{block}) it has the
two clauses $\bigvee_{p\in B}x_p$ and $\bigvee_{p\in B}\bar x_p$. So it is SAT iff $\F_2^k$
has a 2-colouring with no monochromatic block. For $t=2$ this is T2's AG4. Let $n_0(t)$ be
the least $n$ for which it is UNSAT.
\end{definition}
\textbf{Monotonicity.} For a subspace $U\le\F_2^k$, the blocks inside $U$ are exactly the
blocks of $\AG(2t)$-$|U|$. So NO at $n$ implies NO at every $n'\ge n$, and $n_0(t)$ is a
threshold. \lab{PROVED/EXACT}

\begin{lemma}[Sidon count; \lab{KNOWN THEOREM}, elementary]\label{lem:sidon}
Let $s_4(n)=\max\{m:m(m-1)\le 2(n-1)\}$, so $s_4(n)\le\sqrt{2n}+1$. Every $m$-subset of
$\F_2^k$ with $m>s_4(n)$ contains 4 distinct points with XOR $0$.
\end{lemma}
\begin{proof}
There are $\binom m2>n-1$ pairs, and each pair sum is nonzero. So two distinct pairs
$\{a,b\},\{c,d\}$ have the same sum. If they shared a point they would be equal. So they are
disjoint, and $a+b+c+d=0$.
\end{proof}

\begin{lemma}[6-sets]\label{lem:six}
Let $s_6(n)=\max\{m:\binom m3\le n(3m-7)/2\}$. Every $m$-set $S$ with $m>s_6(n)$ contains 6
distinct points with XOR $0$. For $n\ge 64$, $s_6(n)<n/2$.
\end{lemma}
\begin{proof}
Fix a sum value $\sigma$ and let $\mathcal F_\sigma$ be the 3-subsets of $S$ with XOR
$\sigma$. Two members sharing two points coincide. Two disjoint members give the 6-set. So
suppose $\mathcal F_\sigma$ is pairwise intersecting and fix $T=\{u,v,w\}\in\mathcal F_\sigma$.
The pairs $\{u,x\}$ inside members through $u$ are all distinct. So at most $(m-1)/2$ members
contain $u$, and likewise for $v,w$. Every member meets $T$ in exactly one point, except $T$,
which is counted three times. So $|\mathcal F_\sigma|\le 3(m-1)/2-2=(3m-7)/2$. Some $\sigma$
has $|\mathcal F_\sigma|\ge\binom m3/n$. For $m=n/2$ the inequality
$\binom{n/2}{3}>n(3n/2-7)/2$ reduces to $(n/2-1)(n/2-2)>9n-42$, which holds for $n\ge 38$.
The function $m\mapsto\binom m3-n(3m-7)/2$ is increasing for $m\ge\sqrt{3n}+1$, and
$n/2\ge\sqrt{3n}+1$ when $n\ge 16$.
\end{proof}

\begin{theorem}[\lab{PROVED/EXACT}]\label{thm:linear}
Let $q(m)=\lceil (m-s_4(n))/4\rceil$ for $m>s_4(n)$ and $q(m)=0$ otherwise. $\AG(2t)$-$n$ is
UNSAT whenever
(i) $t$ is even and $t/2\le q(n/2)$, or
(ii) $t\ge3$ is odd, $n/2>s_6(n)$, and $(t-3)/2\le q(n/2-6)$.
In particular, for $n\ge 64$ it is UNSAT for every $2\le t\le n/4-(\sqrt{2n}+1)/2$.
\end{theorem}
\begin{proof}
Take any 2-colouring and let $S$ be a majority class, so $|S|=M\ge n/2$. Apply
Lemma~\ref{lem:sidon} repeatedly: while more than $s_4(n)$ points remain, remove 4 distinct
points with XOR $0$. This yields at least $q(M)\ge q(n/2)$ pairwise disjoint zero-sum
4-sets in $S$. (i) The union of $t/2$ of them is a set of $2t$ distinct points of $S$ with
XOR $0$, so it is a monochromatic block. (ii) First take a zero-sum 6-set $X\subseteq S$ by
Lemma~\ref{lem:six}. Then take $(t-3)/2$ disjoint zero-sum 4-sets in $S\setminus X$, which
has at least $n/2-6$ points. The union again has $2t$ distinct points and XOR $0$.
``In particular'': (i) needs $t\le (n/2-s_4)/2$; (ii) needs $t-3\le (n/2-6-s_4)/2$, i.e.\
again $t\le n/4-s_4/2$; also $s_6<n/2$ for $n\ge64$. Finally use
$s_4\le\sqrt{2n}+1$.
\end{proof}

\begin{proposition}[\lab{PROVED/EXACT}]\label{prop:yes}
$\AG(2t)$-$n$ is SAT for every $t\ge n/4-1$ ($n\ge 8$).
\end{proposition}
\begin{proof}
Let $V$ be a hyperplane (a subspace of dimension $k-1\ge2$), and colour $V$ and $V+e$ with
different colours. Both classes have XOR $0$. If $2t=n/2-2$, a zero-sum $2t$-subset of a
class would leave out two distinct points with sum $0$, which is impossible. If $2t=n/2$,
use instead $C=(V\setminus\{0\})\cup\{e\}$ and its complement. Both have XOR $e\ne0$, so the
only $2t$-subset of each class is not a block. If $2t>n/2$, a balanced split has no
$2t$-subset in either class.
\end{proof}

\begin{proposition}[stabiliser argument, \lab{PROVED/EXACT}]\label{prop:stab}
Let $n\ge 32$ and $t=n/4-2$.
(a) No \emph{balanced} colouring of $\F_2^k$ avoids monochromatic blocks.
(b) For $n=32$ (so $t=6$) no colouring at all avoids them: $\AG(12)$-$32$ is UNSAT.
\end{proposition}
\begin{proof}
(a) Let $|C|=n/2$ with XOR $\sigma$. A zero-sum $(n/2-4)$-subset of $C$ is the same thing as
a 4-subset of $C$ with XOR $\sigma$. Suppose there is none. If $\sigma=0$, then $C$ is a
Sidon set of size $n/2>s_4(n)$, which contradicts Lemma~\ref{lem:sidon}. So let $\sigma\neq0$
and let $N(v)$ be the number of pairs in $C$ with sum $v$. Pairs with the same sum are
disjoint, so $N(v)\le n/4$, with equality iff $C+v=C$. Group the $n$ values into the $n/2$
classes $\{v,v+\sigma\}$. If $N(v),N(v+\sigma)\ge1$, then every $v$-pair meets every
$(v+\sigma)$-pair; otherwise their union is a 4-set with XOR $\sigma$. So such a class
contributes at most $2+2=4\le n/4-1$. Every other class contributes at most $n/4$, and at
most $n/4-1$ unless it contains some $v\neq0$ in the stabiliser $W=\{v:C+v=C\}$. Write $a$
for the number of classes that reach $n/4$. Then
$\binom{n/2}{2}=n(n-2)/8\le a\cdot\frac n4+(\frac n2-a)(\frac n4-1)$, which gives
$a\ge n/4$. So $|W|\ge n/4+1$, and since $W$ is a subgroup, $|W|\ge n/2$. Then $C$ is a coset
of the hyperplane $W$, whose XOR is $0$, a contradiction.
(b) Balanced colourings are covered by (a). Otherwise the majority class has $\ge17$ points.
Since $s_4(32)=8$, greedy extraction gives $\lceil(17-8)/4\rceil=3$ disjoint zero-sum 4-sets,
whose union is a monochromatic 12-point block.
\end{proof}

\begin{corollary}[\lab{PROVED/EXACT}]\label{cor:core}
Fix $t$. For every $n\ge n_0(t)$, $\AG(2t)$-$n$ contains an UNSAT sub-instance on a subspace
of $n_0(t)$ points, and $n_0(t)\le 2P(t)=O(t)$ for $t\ge6$. So it has a tree-like resolution
refutation of size at most $2^{n_0(t)+1}$ and a PC/Nullstellensatz refutation of degree
$O(n_0(t))$ \lab{KNOWN THEOREM: degree at most the number of variables (plus clause width)}.
Both bounds are independent of $n$. T2 proves that the grouped degree-$(2t-1)$ span is a fixed
point without $1$. Hence the grouped-PC degree of the family is $\Theta(t)$ at fixed arity. The
family is a pure, arity-bound degree barrier: it has no size or core growth in $n$.
\end{corollary}
\begin{proof}
Monotonicity gives the sub-instance. The size bound is a complete decision tree on its
variables. The bound $n_0(t)\le2P(t)$ for $t\ge6$ follows from Theorem~\ref{thm:linear}:
$2P/4-(\sqrt{4P}+1)/2\ge t$ holds once $P/2>2t+2$, $P\le 8t+8$ and
$t+1.5\ge\sqrt{8t+8}$, i.e.\ for $t\ge6$. Note $2P\ge64$ there.
\end{proof}

\subsection{What this means for the project, and limits}
\begin{itemize}
\item T2 offers ``explicit growing arity-matched degree barriers''. That is true, but at fixed
arity both the degree and a refuting core are $O(t)$ and independent of $n$. So these
instances cannot defeat any lane that searches for bounded-size cores or runs bounded-degree
PC with degree depending on the arity. Growing the barrier means growing $t$. Then the clause
count is about $\binom{n}{2t-1}/(2t)$, so polynomial encodings force $t=O(1)$ (or
$t=O(\log n)$ for quasi-polynomial size). Within the guaranteed NO range, the family is
always easy for a size-aware method.
\item Not claimed: any lower bound for full PC (raw width-$2t$ axioms, arbitrary
multipliers), for resolution size, or for native E/B images. Nothing here bears on P vs NP.
\item \textbf{OPEN} \lab{CANDIDATE}: $n_0(t)=P(t)$ for all $t$, i.e.\ UNSAT iff $n>4t+4$. The
first undecided cases are $n=64$, $t\in\{13,14\}$. Proposition~\ref{prop:stab}(a) handles
balanced colourings at $t=14$; unbalanced ones remain. The lazy solver gave no verdict at $t=14$ after about 18 minutes (killed by hand); $t=13$ was not run.
\end{itemize}

\section*{Recommendations}
\begin{enumerate}
\item Replace T2's quadratic condition $M(M-t+1)>nt^2$ with Theorem~\ref{thm:linear}, and cite
the exact small thresholds. In particular, $\AG(2t)$-$32$ is already NO for every
$t\le6$, so T2 does not need $n=64$ or $128$ to get arity-6 or arity-8 NO instances.
\item When AG-type families are used as hostile controls, record the embedded-core size
$n_0(t)$ next to the degree bound, so that degree barriers are not read as size barriers.
\item Re-split and read back \texttt{03b2e4e} and \texttt{48f638b} before citing them. Move the
root-level \texttt{.md} from \texttt{71e51be} under \texttt{research/}.
\end{enumerate}
