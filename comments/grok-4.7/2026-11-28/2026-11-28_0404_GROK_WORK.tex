\section{Gap tackled: simultaneous stagnation of bounded-width recognizers on pinned-compiler inputs}
\subsection*{Setting}
Let $F$ be a 3-CNF with variables $x_1..x_n$ and clauses $C_1..C_m$, each with $1\le|C_i|\le3$ literals. Let $\mathcal N(F)$ be the output of the pinned \texttt{compile()} (\texttt{21b9d53}). After signed-BFS normalization (union-find with parity over edges and E-vertices), the \emph{normalized native factor system} $N(F)$ consists of one Boolean \emph{carrier} per connected signed component, plus one factor per B-vertex: ``integer sum of the four signed port bits $=2$'' on at most 4 carriers. $\mathrm{CNF}(N)$ denotes its full-word clause expansion. For a gauge pin $g$ on one carrier, $N_g:=N\wedge(g{=}0)$. Write $w(G)$ for the minimum width of a resolution refutation of a CNF $G$ ($\infty$ if $G$ is satisfiable).

\begin{lemma}[Compiler locality; PROVED/EXACT by reading the pinned code, and COMPUTATIONAL audit]\label{lem:loc}
The carriers of $N(F)$ are in bijection with the compiler's occurrence groups $\{x_v\}\cup\{z_i\}\cup\{w_{i,0},w_{i,1}\}\cup\{f\}$, so there are $n+3m+1$ of them, and every port bit equals its group value XOR a fixed sign. Each B factor created for ternary clause $i$ lies on carriers in $\mathrm{vars}(C_i)\cup\{z_i,w_{i,0},w_{i,1},f\}$ and always contains $w_{i,j}$. With $f=0$, the two factor types of gadget $i$ are ``$a+b+z_i+w_{i,0}=2$'' and ``$(1-z_i)+c+w_{i,1}=2$'' on literal values. They are jointly satisfiable in $(z_i,w_{i,0},w_{i,1})$ iff $C_i$ is true.
\end{lemma}
\begin{proof}
\texttt{wire()} is a $K_5-e$ of E-vertices with sign-0 internal edges, so all of its ports are equal. It attaches its two ends with signs $s$ and $0$. Carrier chains are wires between E-vertices. Hence every occurrence port of a group equals the group value XOR its recorded sign. Each B-vertex receives the three entries of $T[j]$ and one \texttt{"w"+i+"\_"+j} port, which gives the stated scopes. The equivalence is the RT01 NAE$_4\to$ NAE$_3$ split checked by the compiler's own \texttt{auditLocal}. It is re-verified in (A) below.
\end{proof}

\noindent Define the \emph{canonical completion} $\sigma$: $x_v\mapsto x_v$, $f\mapsto0$, and $(z_i,w_{i,0},w_{i,1})\mapsto$ the lexicographically first satisfying triple if $C_i$ is true, else $(0,0,0)$. Each carrier value is then a function of at most $3$ original variables: $S(x_v)=\{v\}$, $S(z_i)=S(w_{i,j})=\mathrm{vars}(C_i)$, and $S(f)=\emptyset$.

\begin{definition}[Existential pebble strategy]
A \emph{$W$-strategy} for a CNF $G$ is a nonempty family $\mathcal H$ of partial assignments with at most $W$ variables each, satisfying:
(i) no $\alpha\in\mathcal H$ falsifies a clause of $G$;
(ii) $\mathcal H$ is closed under restriction;
(iii) every $\alpha\in\mathcal H$ with $|\alpha|<W$ extends within $\mathcal H$ to any further variable.
\end{definition}

\begin{lemma}[Easy direction; PROVED/EXACT]\label{lem:easy}
If $G$ has a $W$-strategy, then $w(G)\ge W$.
\end{lemma}
\begin{proof}
Induction along a refutation of width $\le W-1$ shows that every derived clause $C$ is satisfied by every $\alpha\in\mathcal H$ with $\mathrm{vars}(C)\subseteq\mathrm{dom}\,\alpha$. Axioms hold by (i). For a resolution step $C\vee x,\ D\vee\neg x\vdash C\vee D$, restrict $\alpha$ to $\mathrm{vars}(C\vee D)$, which has at most $W-1$ variables, and extend it to $x$ by (ii) and (iii). The extension satisfies both premises, hence $C$ or $D$. The empty clause would then be satisfied by $\emptyset\in\mathcal H$, a contradiction.
\end{proof}

\noindent\textbf{KNOWN THEOREM (Atserias--Dalmau, JCSS 74 (2008) 323--334).} Resolution width is characterised by the existential pebble game: if $w(G)\ge W$ (and $W$ exceeds the clause width of $G$), then $G$ has a $W$-strategy. This is cited for the direction used. A possible $\pm1$ shift in the pebble-count convention only changes the additive constants below, which carry one unit of slack for that reason.

\begin{theorem}[Width transfer through the pinned compiler; PROVED/EXACT, i.e.\ PROVED WITH SCOPE, no independent review]\label{thm:transfer}
For every 3-CNF $F$ and every gauge carrier $g$,
\[ w\big(\mathrm{CNF}(N_g(F))\big)\ \ge\ \Big\lfloor \tfrac{w(F)}{3}\Big\rfloor-2 .\]
\end{theorem}
\begin{proof}
Let $W:=w(F)$ (if $F$ is satisfiable, so is $N$ and there is nothing to prove), let $k:=\lfloor W/3\rfloor$, and let $\mathcal H$ be a $W$-strategy for $F$ (KNOWN THEOREM). Put
\[\mathcal H' := \{\ \sigma(\alpha)|_D\ :\ D \text{ a set of carriers},\ |D|\le k,\ \alpha\in\mathcal H,\ S(D)\subseteq\mathrm{dom}\,\alpha\ \}.\]
The value $\sigma(\alpha)$ on $D$ depends only on $\alpha|_{S(D)}$, so it is well defined.
(ii) Restriction to $D'\subseteq D$ keeps $S(D')\subseteq\mathrm{dom}\,\alpha$.
(iii) Let $|D|<k$ and let $y$ be a new carrier. Since $|S(D)|\le 3(k-1)$, restrict $\alpha$ to $S(D)$ and extend it one variable at a time over $S(y)$, which has at most 3 variables. Each step starts from at most $3k-1<W$ variables, so (iii) for $\mathcal H$ applies. The result $\alpha''$ agrees with $\alpha$ on $S(D)$, so $\sigma(\alpha'')|_{D\cup\{y\}}$ extends $\sigma(\alpha)|_D$.
(i) Suppose a full-word clause of a gadget-$i$ factor has all its carriers in $D$. Its scope contains $w_{i,j}$, so $\mathrm{vars}(C_i)\subseteq S(D)\subseteq\mathrm{dom}\,\alpha$. Since $\alpha$ does not falsify $C_i$, it satisfies $C_i$, and by Lemma~\ref{lem:loc} $\sigma(\alpha)$ satisfies both factors of gadget $i$.
So $\mathcal H'$ is a $k$-strategy for $\mathrm{CNF}(N)$, and Lemma~\ref{lem:easy} gives $w(N)\ge k$.
For the gauge, $N$ is invariant under global complement (weight $2$ of $4$ is preserved). A width-$u$ refutation of $N\wedge\neg g$ gives $N\vdash g$ in width $u+1$ (add $g$ throughout). Its complement gives $N\vdash\neg g$ in width $u+1$, and resolving the two gives $w(N)\le u+1$. One further unit of slack covers the pebble-count convention.
\end{proof}

\begin{definition}[Bounded-width exact factor calculus $\mathrm{FC}(k,d,s)$]
The state is a set of exact Boolean factors on at most $k$ carriers of $N_g$. A \emph{move} adds any factor $R$ on at most $k$ carriers that is implied by a set of current factors whose union scope has at most $k$ carriers. This covers T1-Q and T4-02 full-bucket existential projection, joins, deduplication, tautology deletion, unit-support pins and restrictions, and T2 private-parity and two-owner projections with union $\le k$. Contradiction is an empty factor. Additionally, the calculus may (a) branch on carrier literals along paths of depth at most $d$ (T4 conditional terminal, backdoor branching), and (b) learn nogoods on at most $s$ literals when assuming them leads to a contradiction by the above (T2 width-2 nogoods, $s=2$).
\end{definition}

\begin{lemma}[Simulation; PROVED/EXACT]\label{lem:sim}
If $\mathrm{FC}(k,d,s)$ refutes $N_g$, then $w(\mathrm{CNF}(N_g))\le k+d+s$.
\end{lemma}
\begin{proof}
Consider a move from factors with union scope $U$, $|U|\le k$, to a consequence $R$. Every forbidden full word on $U$ falsifies a clause of some premise. Weakening that clause to all of $U$ yields the full clause of the word. Resolving the full clauses of $U$ down to the scope of each forbidden word of $R$ keeps every clause within width $|U|\le k$. Weakening can be removed without increasing width (standard). A leaf refutation under branch literals $\rho$ with $|\rho|\le d$, after adding $\neg\rho$ to every clause, derives $\neg\rho$ in width $\le k+d$. Combining the two children of each branch node yields the parent clause. A nogood derived under its $\le s$ assumptions costs at most another $s$ literals of width.
\end{proof}

\begin{corollary}[Stagnation family; PROVED/EXACT relative to KNOWN THEOREMS]\label{cor:stag}
Fix $\Delta\ge 5.2$ and $c>0$. Let $F$ be a uniformly random 3-CNF with $n$ variables and $\Delta n$ clauses, and let $N_{\rm bits}$ be the native bit length of the pinned-compiler output. Then with probability $1-o(1)$:
(1) $F$ and $\mathcal N(F)$ are UNSAT. This is the first-moment bound $2^n(7/8)^{\Delta n}\to0$, since $\Delta>\ln2/\ln(8/7)\approx5.19$.
(2) $w(F)\ge\varepsilon_\Delta n$. This is the KNOWN THEOREM of Ben-Sasson--Wigderson, J.~ACM 48 (2001), width lower bound for random 3-CNF at constant density.
(3) For all large $n$, no $\mathrm{FC}(k,d,s)$ with $k+d+s\le c\log_2 N_{\rm bits}$ refutes $N_g(F)$ for any gauge $g$. This follows from Theorem~\ref{thm:transfer}, Lemma~\ref{lem:sim}, and $N_{\rm bits}=\mathrm{poly}(n)$.
Hence every such recognizer stops with exhaustion or refusal (UNKNOWN), or runs out of budget, on a source-authenticated family of genuine NO instances. This holds simultaneously for T1-Q/T1-L bucket and support closure, T4-02 queue plus depth-$O(\log N)$ branching (the ``conditional terminal''), T2 private-parity projection with width-2 nogoods, and logarithmic-size carrier backdoors with bounded-width leaf closure.
\end{corollary}

\subsection*{Limits and what remains OPEN}
\begin{itemize}
\item This is a \emph{restricted} obstruction to bounded-width resolution-simulable reasoning. It is \textbf{not} a lower bound for P vs NP, for general algorithms, or for stronger proof systems. Several allowed lanes fall \emph{outside} $\mathrm{FC}$ and are untouched: Gaussian elimination over $\F$ on response codes (which introduces linear forms, i.e.\ extension-like variables), matching/Hall and cardinality reasoning, unbounded-size nogoods, new defined carriers (extended resolution), and branching depth $\omega(\log N)$.
\item The transfer constant ($1/3$) depends on the 3-CNF fan-in of the pinned gadgets. Long clauses (split chains) need a separate locality argument and are not covered.
\item At the corpus sizes ($n=24,40$) the bound is not informative. The statement is asymptotic, with high probability.
\item The cited Atserias--Dalmau and Ben-Sasson--Wigderson statements are used from the literature without re-proof. Only the transfer, simulation, gauge, and easy-direction steps are proved here.
\item \textbf{Next lemma (OPEN).} Determine whether adding exact $\F$-Gaussian elimination on the T2 response codes to $\mathrm{FC}$ escapes the stagnation family. One route is a polynomial-calculus or $\F$-degree transfer through the same locality map, using the known random-3-CNF degree lower bounds (Ben-Sasson--Impagliazzo; Alekhnovich--Razborov) as the target. A degree analogue of Theorem~\ref{thm:transfer} would close that lane too.
\end{itemize}
